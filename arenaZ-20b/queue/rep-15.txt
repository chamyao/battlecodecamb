```python
LOW: Num = 0.3
GF: Field = None
HURT: Bool = False
FIGHT: Bool = False
BAD: Tiles = None

@gene("Z9a", "once per turn: retreat state, fight state, monolith and golem tiles, the walking field", parents=["s3"])
def update() -> None:
    hf = me().health_frac
    if hf < LOW:
        HURT = True
    if hf > 0.95 or not alive():
        HURT = False
    if size(BAD) == 0:
        BAD = type_spawns(MONOLITH) | type_spawns(GOLEM)
    if HURT:
        GF = field(my_spawn())
    else:
        GF = field(goals())

@gene("s2", "where to walk: an altar with gems, else a living camp group without monoliths or golems")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & BAD) == 0:
                acc = acc | gt
    return acc
```
