@gene("s2", "where to walk: an altar with gems, else a living camp group without monoliths or golems")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & BAD) == 0 and group_gems(g) > 0:
                acc = acc | gt
    return acc
