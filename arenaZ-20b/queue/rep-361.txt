@gene("Z20c", 'once per turn: retreat state, fight state, monolith and golem tiles, the walking field')
def update() -> None:
    hf = me().health_frac
    if hf < LOW:
        HURT = True
    if hf > 0.76 or not alive():
        HURT = False
    if size(BAD) == 0:
        BAD = type_spawns(MONOLITH) | type_spawns(GOLEM)
    FIGHT = count(e.is_champion for e in enemies()) > 0 or any(is_awake(c) and dist(c.pos, me().pos) <= 6 for c in camps())
    if (gold() >= 469 and gems() == 0) or HURT:
        GF = field(my_spawn())
    else:
        GF = field(goals())
