```python
@gene("Z26a", 'value of casting an ability: damage on fights, heal and shield when hurt, dash toward goals', parents=['Z4a', 's1', 's2', 's4'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion:
                return 85
            if o.unit.is_camp and gems() == 0 and not ((o.unit.type == MONOLITH or o.unit.type == GOLEM) and not is_awake(o.unit)):
                return 52
        return 0
    if k == AB_HEAL:
        return 75 if me().health_frac < 0.7 else 0
    if k == AB_SHIELD:
        return 75 if me().health_frac < 0.8 and FIGHT else 0
    if k == AB_RESET:
        return 50 if not can_attack() else 0
    if k == AB_DASH:
        if valid(o.tile):
            return 55 if progress(o.tile) > 0 else 0
    return 0
```
