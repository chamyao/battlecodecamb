```python
BAD: Tiles = None

@gene("Z6a", "once per turn: retreat state, fight state, monolith and golem tiles, the walking field", parents=["s3"])
def update() -> None:
    hf = me().health_frac
    if hf < 0.45:
        HURT = True
    if hf > 0.95 or not alive():
        HURT = False
    if size(BAD) == 0:
        BAD = type_spawns(MONOLITH) | type_spawns(GOLEM)
    if HURT:
        GF = field(my_spawn())
    else:
        GF = field(safe_goals())

@gene("new", "where to walk: an altar with gems, else a living camp group without monoliths or golems", parents=["s2"])
def safe_goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & BAD) == 0:
                acc = acc | gt
    return acc

@gene("s3", "distance field to the goals (cached per turn)")
def goal_field() -> Field:
    return GF

@gene("s4", "how much closer a step brings me to the goals (or home when hurt)")
def progress(t: Tile) -> Num:
    return at(GF, me().pos) - at(GF, t)
```
