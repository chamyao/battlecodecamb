@gene("s2", 'where to walk: home when hurt, an altar with gems, else a living camp group that is safe to fight')
def goals() -> Tiles:
    if (gold() >= 300 and gems() == 0) or HURT:
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            k = max((spawn_type(t) for t in gt), default=-1)
            if k != MONOLITH and k != GOLEM:
                acc = acc | gt
    return acc
