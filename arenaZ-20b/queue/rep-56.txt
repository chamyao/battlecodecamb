TGT: Tiles = None
@gene("Z4a", 'once per turn: retreat state, fight state, monolith and golem tiles, the walking field', parents=['s3'])
def update() -> None:
    hf = me().health_frac
    if hf < LOW:
        HURT = True
    if hf > 0.95 or not alive():
        HURT = False
    if size(BAD) == 0:
        BAD = type_spawns(MONOLITH) | type_spawns(GOLEM)
    FIGHT = count(e.is_champion for e in enemies()) > 0 or any(is_awake(c) and dist(c.pos, me().pos) <= 5 for c in camps())
    TGT = enemy_tiles() | (camp_tiles() - BAD) if gems() == 0 else enemy_tiles()
    if HURT or (gold() >= 300 and gems() == 0):
        GF = field(my_spawn())
    else:
        GF = field(goals())
@gene("Z26a", 'value of casting an ability: damage on fights, heal and shield when hurt, dash toward goals', parents=['Z4a', 's1', 's2', 's4'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion:
                return 85
            if o.unit.is_camp and gems() == 0 and not ((o.unit.type == MONOLITH or o.unit.type == GOLEM) and not is_awake(o.unit)):
                return 65
        if valid(o.tile) and o.tile in TGT:
            return 66
        return 0
    if k == AB_HEAL:
        return 75 if me().health_frac < 0.7 else 0
    if k == AB_SHIELD:
        return 75 if me().health_frac < 0.8 and FIGHT else 0
    if k == AB_RESET:
        return 62 if not can_attack() else 0
    if k == AB_DASH:
        if valid(o.tile):
            return 55 if progress(o.tile) > 0 else 0
    return 0
