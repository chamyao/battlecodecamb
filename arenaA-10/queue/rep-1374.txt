```python
@gene('s6', 'visible enemy champions within r tiles (nearly dead ones count only if carrying gems; recalling or tributing ones are busy channeling and do not count)')
def enemies_within(r: Num) -> Num:
    return count(
        dist(here(), u.pos) <= r and (u.health_frac >= 0.15 or unit_gems(u) > 0) and not is_recalling(u) and not is_tributing(u)
        for u in enemies()
    )
```
