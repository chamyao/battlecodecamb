```python
@gene('r10', 'bank now: carrying gems and the bag is ready, or next to an altar with 3+ gems, or the bag alone would take the team to 1000 (a tribute wins the game)', parents=['r10'])
def bank_on() -> Bool:
    g = gems()
    if g <= 0:
        return False
    return bag_ready() or (g >= 3 and valid(adj_altar())) or team_gems() + g >= 1000
```
