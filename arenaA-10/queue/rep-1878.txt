@gene('v2', 'start of turn: choose the rule, its goal and target; placing the team lock clears the lock byte so the team buys and places another; shared-memory and priority writes (fighters, retreaters and pre-fight bankers act first next round, bankers at an altar or walking to bank next)')
def update() -> None:
    if not MAPPED:
        ALTAR_RING = altar_ring()
        SAFE_RING = safe_altar_ring()
        MONO_GROUPS = mono_groups()
        TERRAIN = walls() | holes() | altars() | enemy_spawn()
        BIG_CAMPS = type_spawns(MONOLITH) | type_spawns(GOLEM)
        CAMP_BASE = camp_spawns() - BIG_CAMPS - MONO_GROUPS
        SK0 = slot_kind(0)
        SK1 = slot_kind(1)
        MAPPED = True
    RULE = rule()
    GOAL = rule_goal()
    TGT = rule_target()
    OPEN = False
    if walks() and valid(GOAL) and GOAL != here():
        BLOCK = blocked()
        OPEN = walk_open()
        FIELD = walk_field()
    LOCK_BUY = RULE >= 2 and at_spawn() and wants_lock()
    if LOCK_BUY:
        write_shared(7, read_shared(7) + 2)
    SHOP_GOLD = gold() - (10 if LOCK_BUY else 0)
    if RULE == 8:
        write_shared(7, 0)
    if RULE == 7 or RULE == 3 or RULE == 9:
        set_priority(150 + rand(20))
    elif RULE == 10 or RULE == 6 or RULE == 17 or RULE == 12:
        set_priority(120 + rand(20))
    elif RULE == 20:
        set_priority(1 + rand(100))
