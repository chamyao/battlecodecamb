@gene('w7', 'a dash may land on p: a free middle tile next to both ends that no visible enemy champion is adjacent to, unless p is adjacent to me; below half health it never lands within 2 of an enemy champion', parents=['w7'])
def dash_path_ok(p: Tile) -> Bool:
    if dist(here(), p) <= 1:
        return True
    ev = enemies()
    if hurt_below(0.5) and any(dist(e.pos, p) <= 2 for e in ev):
        return False
    return any(dist(q, p) <= 1 and not (q in BLOCK) and not any(dist(e.pos, q) <= 1 for e in ev) for q in dash_mids())
