@gene("r18", "on a spawn tile of a group that is neither cleared nor respawning: walk to the nearest living camp (not monolith or golem); reads the tile's group directly instead of recomputing the camp walk's goal")
def r_live() -> Bool:
    g = spawn_group(here())
    if g < 0 or group_reward(g) or group_respawning(g):
        return False
    c = live_camp()
    return exists(c) and not (c.pos in BIG_CAMPS)
