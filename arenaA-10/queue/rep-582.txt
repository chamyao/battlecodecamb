@gene("r11", "bag ready: 15+, or 10+ with an altar 3 turns away or a threat near, or 6+ with an altar 2 turns away, or late, or hurt", parents=['r11'])
def bag_ready() -> Bool:
    g = gems()
    if g >= 15 or round() > 1500 or hurt_below(0.3):
        return True
    a = altar_eta()
    return (g >= 10 and (a <= 3 or threat_near())) or (g >= 6 and a <= 2)
