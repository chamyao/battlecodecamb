@gene("r14", "fight the first camp in reach, but do not wake a sleeping monolith or golem while below half health")
def r_camp() -> Bool:
    c = reach_camp()
    if not exists(c) or not in_reach(c):
        return False
    return is_awake(c) or not (c.pos in BIG_CAMPS) or not hurt_below(0.5)
