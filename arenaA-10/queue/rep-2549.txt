```python
@gene("r18", "on a dead camp of an uncleared group: walk to the nearest living camp of the same group (not monolith or golem)")
def r_live() -> Bool:
    c = live_camp()
    h = here()
    return dist(camp_goal(), h) == 0 and exists(c) and not (c.pos in BIG_CAMPS) and spawn_group(camp_spawn(c)) == spawn_group(h)
```
