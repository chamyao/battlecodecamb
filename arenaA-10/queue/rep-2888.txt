```python
@gene('s6', 'visible enemy champions within r tiles (nearly dead ones count only if carrying gems or adjacent)')
def enemies_within(r: Num) -> Num:
    h = here()
    return count(
        dist(h, u.pos) <= r and (u.health_frac >= 0.1522 or unit_gems(u) > 0 or dist(h, u.pos) <= 1)
        for u in enemies()
    )
```
