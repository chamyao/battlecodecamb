```python
@gene('a9', "shopping on spawn: the team's altar lock (only before round 1200 and with 40+ gold, so a shard still follows: a late or starving lock rarely pays back), then shards in the policy's split, with a durability health emblem (10% less damage) after the treasure and jungle emblems")
def shop_value(o: Option) -> Num:
    if o.kind == BUY_LOCK:
        return 200 if LOCK_BUY and RULE >= 2 and at_spawn() and round() < 1200 and gold() >= 40 else 0
    want = next_shard()
    if want == 36 and items_owned(17) == 0:
        want = 16
    if o.item != want:
        return 0
    if o.count > shard_left() or RULE < 2 or not at_spawn():
        return 0
    return 190 + o.count / 100
```
