@gene("r11", "bag ready: 18+, or 15+ when below 75% health, or 10+ with an altar 3 turns away or a threat near, or late, or hurt", parents=['r11'])
def bag_ready() -> Bool:
    g = gems()
    return g >= 18 or (g >= 15 and hurt_below(0.75)) or (g >= 10 and altar_eta() <= 3) or (g >= 10 and threat_near()) or round() > 1500 or hurt_below(0.3)
