@gene('m11', "the camp walk's goal (camps within 3 of a visible enemy champion only after the uncontested ones): weak gem-paying groups first once I have bought 6 times (skipping groups an ally is already capturing), else the nearest non-big spawn of a live group that no ally is already capturing, falling back to any non-big camp spawn, then any camp spawn")
def camp_goal() -> Tile:
    held = held_groups()
    if BUYS >= 6 and size(GEM_BASE) > 0:
        t = closest(GEM_BASE - respawning() - held)
        if valid(t):
            return t
    free = CAMP_BASE - respawning() - BIG_CAMPS - held
    t = closest(free - contested())
    if valid(t):
        return t
    t = closest(free)
    if valid(t):
        return t
    t = closest(CAMP_BASE - BIG_CAMPS)
    if valid(t):
        return t
    return closest(CAMP_BASE)

@gene("new", "tiles within 3 of a visible enemy champion (a camp there is contested)", parents=["m11"])
def contested() -> Tiles:
    acc = CAMP_BASE - CAMP_BASE
    for u in enemies():
        acc = acc | near(u.pos, 3)
    return acc
