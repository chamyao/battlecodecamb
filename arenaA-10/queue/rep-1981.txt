```python
@gene('a8', 'a step downhill on the walking field to GOAL; among equal steps prefer tiles farther from enemies and from awake camps that could hit them, in bushes, and next to allies (stay grouped)')
def move_value(o: Option) -> Num:
    g = step_gain(o.tile)
    if g <= 0:
        return 0
    v = 100 + g
    ft = foe_tiles()
    if size(ft) > 0:
        v -= 0.3 * count(dist(t, o.tile) <= 2 for t in ft)
        v -= 0.1 * count(dist(t, o.tile) <= 3 for t in ft)
        if o.tile in bushes():
            v += 0.05
    v -= 0.2 * count(c.attack_range > 0 and is_awake(c) and dist(c.pos, o.tile) <= c.attack_range for c in camps())
    v += 0.04 * size(ally_tiles() & near(o.tile, 2))
    return v
```
