@gene('w9', 'the dash landing tile closest to GOAL by walking, ties broken by fewest enemy champions within 3 of it (weighted 15 each), invalid if none; my tile and enemy tiles read once', parents=['w9'])
def dash_spot() -> Tile:
    h = here()
    if not valid(GOAL) or GOAL == h:
        return tile(-1, -1)
    mine = at(FIELD, h)
    bk = INF
    s = tile(-1, -1)
    ft = foe_tiles()
    for t in near(h, 2):
        d = at(FIELD, t)
        if d < mine and t != h and not (t in BLOCK):
            if dash_path_ok(t):
                k = d * 10 + 15 * count(dist(e, t) <= 3 for e in ft)
                if k < bk:
                    bk = k
                    s = t
    return s

@gene('a5', 'cast-ahead applies: the chosen enemy is off its spawn and within 5, or within 7 when below 30% health (finish it)')
def ahead_on() -> Bool:
    f = ahead_foe()
    if not exists(f) or f.pos in enemy_spawn():
        return False
    d = dist(here(), f.pos)
    return d <= 5 or (d <= 7 and f.health_frac < 0.3)
