```python
@gene('r16', 'standing on a cleared camp spawn: hold so the capture timer runs, unless a healthy ally (30%+) holds an earlier tile (row-major) of the same group (one holder is enough; the rest move on; a nearly dead holder may leave)')
def r_hold() -> Bool:
    h = here()
    g = spawn_group(h)
    if g < 0 or not group_reward(g):
        return False
    k = h.y * 1000 + h.x
    gt = group_tiles(g)
    return not any(a.pos.y * 1000 + a.pos.x < k and a.health_frac >= 0.3 for a in allies() if a.pos in gt)
```
