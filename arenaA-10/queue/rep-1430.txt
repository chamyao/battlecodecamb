@gene('m11', "the camp walk's goal: nearest non-big spawn of a live group, unless the next-nearest live group is at most 5 farther and holds twice the gems; falling back to any non-big camp spawn, then any camp spawn")
def camp_goal() -> Tile:
    live = CAMP_BASE - respawning() - BIG_CAMPS
    t = closest(live)
    if valid(t):
        g1 = spawn_group(t)
        if g1 >= 0:
            t2 = closest(live - group_tiles(g1))
            if valid(t2):
                g2 = spawn_group(t2)
                if g2 >= 0 and group_gems(g2) > 2 * group_gems(g1) and dist(here(), t2) <= dist(here(), t) + 5:
                    return t2
        return t
    t = closest(CAMP_BASE - BIG_CAMPS)
    if valid(t):
        return t
    return closest(CAMP_BASE)
