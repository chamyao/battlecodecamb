```python
@gene('a5', 'cast-ahead applies: the chosen enemy is off its spawn and within 5, or within 7 when below 30% health (finish it)')
def ahead_on() -> Bool:
    f = ahead_foe()
    if not exists(f) or f.pos in enemy_spawn():
        return False
    d = dist(here(), f.pos)
    return d <= 5 or (d <= 7 and f.health_frac < 0.3)

@gene('a4', 'cast-ahead target: enemies off their spawn and within 7 (the cast-ahead range), carrying the most gems, then nearest, with a strong pull toward wounded ones (finishable)')
def ahead_foe() -> Unit:
    es = enemy_spawn()
    h = here()
    return best(enemies(), lambda u: -INF if (u.pos in es or dist(h, u.pos) > 7) else unit_gems(u) * 10 - dist(h, u.pos) - 15 * u.health_frac)
```
