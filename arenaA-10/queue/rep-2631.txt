```python
@gene("c2", "the altar or unit tile this turn's rule acts on (reads this turn's RULE set by update() instead of re-running every rule condition)")
def rule_target() -> Tile:
    r = RULE
    if r == 6 or r == 9 or r == 10 or r == 17:
        return adj_altar()
    if r == 7:
        return fight_foe().pos
    if r == 8:
        return lock_altar()
    if r == 13 or r == 14:
        return reach_camp().pos
    return tile(-1, -1)

@gene('s13', 'in-reach enemy champion: one my next basic attack kills first, then carried gems, then lowest health')
def fight_foe() -> Unit:
    return best(enemies(), lambda u: (1000 if attack_damage(me(), u) >= u.health + shield(u) else 0) + unit_gems(u) * 10 - u.health if in_reach(u) else -INF)
```
