@gene("r7", "fight the weakest enemy champion in reach (not on its own spawn); a hurt carrier of 10+ gems does not trade with a healthy gemless foe (it banks or runs instead)")
def r_fight() -> Bool:
    f = fight_foe()
    if not exists(f):
        return False
    h = here()
    if dist(h, f.pos) > me().attack_range or f.pos in enemy_spawn():
        return False
    return not (gems() >= 10 and hurt_below(0.35) and f.health_frac > 0.7 and unit_gems(f) == 0)
