@gene('w4', 'nearest tile of s by walking around terrain and camps (units are ignored as walls: they move); if none is walk-reachable, fall back to the straight-line nearest free tile of s (any tile of s if all are held); invalid if s empty', parents=['w4'])
def closest(s: Tiles) -> Tile:
    if here() in s:
        return here()
    cand = s - taken()
    w = walk_nearest(cand, here(), (TERRAIN | camp_block()) - cand - near(here(), 0))
    if valid(w):
        return w
    if size(cand) > 0:
        return nearest(cand, here())
    return nearest(s, here())
