@gene('w9', 'the dash landing tile closest to GOAL by walking, ties broken by fewest enemy champions within 3 of it (weighted 15 each, 40 on a retreat, where landing near them is the danger), invalid if none; enemy tiles read once', parents=['w9'])
def dash_spot() -> Tile:
    h = here()
    if not valid(GOAL) or GOAL == h:
        return tile(-1, -1)
    mine = at(FIELD, h)
    bk = INF
    s = tile(-1, -1)
    ft = foe_tiles()
    wgt = 40 if RULE == 3 else 15
    for t in near(h, 2):
        d = at(FIELD, t)
        if d < mine and t != h and not (t in BLOCK):
            if dash_path_ok(t):
                k = d * 10 + wgt * count(dist(e, t) <= 3 for e in ft)
                if k < bk:
                    bk = k
                    s = t
    return s
