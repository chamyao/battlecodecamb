@gene('a4', 'cast-ahead target: enemies off their spawn, ones inside my longest cast range strongly first (a cast can land), then carrying the most gems, then nearest, with a strong pull toward wounded ones (finishable)')
def ahead_foe() -> Unit:
    es = enemy_spawn()
    h = here()
    cr = max(slot_range(0), slot_range(1))
    return best(enemies(), lambda u: -INF if u.pos in es else unit_gems(u) * 10 - dist(h, u.pos) - 16 * u.health_frac + (60 if dist(h, u.pos) <= cr else 0))
