```python
@gene('w9', 'the dash landing tile closest to GOAL by walking, only if it gains 2+ tiles over standing (a 1-tile dash wastes the cooldown a step would match), ties broken by fewest enemy champions within 3 of it (weighted 15 each), invalid if none; enemy tiles read once', parents=['w9'])
def dash_spot() -> Tile:
    h = here()
    if not valid(GOAL) or GOAL == h:
        return tile(-1, -1)
    mine = at(FIELD, h)
    bk = INF
    s = tile(-1, -1)
    ft = foe_tiles()
    for t in near(h, 2):
        d = at(FIELD, t)
        if d <= mine - 2 and not (t in BLOCK):
            if dash_path_ok(t):
                k = d * 10 + 15 * count(dist(e, t) <= 3 for e in ft)
                if k < bk:
                    bk = k
                    s = t
    return s
```
