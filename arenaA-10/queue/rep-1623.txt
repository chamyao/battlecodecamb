```python
@gene("r11", "bag ready: 15+, or 10+ with an altar 3 turns away or a threat near, or 5+ after round 1500, or any after round 1800, or hurt", parents=['r11'])
def bag_ready() -> Bool:
    g = gems()
    r = round()
    return g >= 15 or (g >= 10 and altar_eta() <= 3) or (g >= 10 and threat_near()) or (g >= 5 and r > 1500) or r > 1800 or hurt_below(0.3)
```
