```python
@gene("r9", "pre-fight bank: altar next to us and 5+ gems with an enemy able to reach us within 2 turns, or 3+ gems with one able to hit us next turn")
def r_prefight() -> Bool:
    g = gems()
    if g < 3 or not valid(adj_altar()):
        return False
    t = threat_eta()
    return (g >= 5 and t <= 2) or t <= 1
```
