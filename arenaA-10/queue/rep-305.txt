@gene('r1', 'nothing to do: dead, channeling, cannot act, or recalling with nothing near (stay put so the recall completes)', parents=['r1'])
def r_idle() -> Bool:
    if not alive() or channeling():
        return True
    if not can_move() and not can_attack():
        return True
    return is_recalling(me()) and not under_fire()
