```python
@gene('r7', 'fight the weakest enemy champion in reach (not on its own spawn, nor next to it while I am below half health unless one hit kills it: it heals there and its respawning allies join)')
def r_fight() -> Bool:
    f = fight_foe()
    if not exists(f) or dist(here(), f.pos) > me().attack_range or f.pos in enemy_spawn():
        return False
    if hurt_below(0.5) and size(near(f.pos, 1) & enemy_spawn()) > 0:
        return attack_damage(me(), f) >= f.health + shield(f)
    return True
```
