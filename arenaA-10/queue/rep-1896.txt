@gene("r14", "fight the first camp in reach, but do not wake a sleeping camp while an enemy champion is within 6 (we would be jumped mid-fight)")
def r_camp() -> Bool:
    c = reach_camp()
    if not exists(c) or not in_reach(c):
        return False
    return is_awake(c) or enemies_within(6) == 0
