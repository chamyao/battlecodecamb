@gene('w10', 'GOAL can be walked to: not terrain and not held by an enemy champion unless it is over 3 away (it will likely have moved) or recalling (it is leaving); allies move, so an ally on GOAL does not block the walk', parents=['w10'])
def walk_open() -> Bool:
    if not valid(GOAL) or GOAL == here():
        return False
    if GOAL in TERRAIN:
        return False
    if dist(here(), GOAL) > 3:
        return True
    return not any(u.pos == GOAL and not is_recalling(u) for u in enemies())
