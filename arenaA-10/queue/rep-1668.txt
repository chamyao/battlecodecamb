```python
@gene('r15', 'go for monoliths too: nothing tributed by round 800, or behind after round 1000 with 3+ allies within 6 (monoliths need a group); cheap checks first')
def r_mono() -> Bool:
    r = round()
    if r <= 800:
        return False
    tg = team_gems()
    if tg == 0:
        return True
    return r > 1000 and tg < enemy_gems() and size(ally_tiles() & near(here(), 6)) >= 4
```
