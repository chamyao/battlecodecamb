```python
@gene("a15", "shards of the first k kinds in the turn's split (of the shards SHOP_GOLD pays for); before round 300 every shard goes to the first kind (Treasure) so its emblem completes early")
def shard_until(k: Num) -> Num:
    n = floor(SHOP_GOLD / 30)
    if round() < 300:
        return n
    t = floor(n / 2)
    j = floor((n - t) / 2)
    s = floor((n - t - j) / 2)
    if k == 1:
        return t
    if k == 2:
        return t + j
    if k == 3:
        return t + j + s
    return n
```
