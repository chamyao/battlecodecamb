```python
@gene('r15', 'go for monoliths too: nothing tributed by round 800 (or a monolith-only map), or after round 1000 with 3+ allies within 7 when behind, or after round 1400 with them regardless, or after round 1600 with 2+ allies (monolith captures decide late games)')
def r_mono() -> Bool:
    if (team_gems() == 0 or MONO_ONLY) and round() > 800:
        return True
    if round() <= 1000:
        return False
    n = size(ally_tiles() & near(here(), 7))
    if round() > 1600 and n >= 3:
        return True
    if n < 4:
        return False
    return team_gems() < enemy_gems() or round() > 1400
```
