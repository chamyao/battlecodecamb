@gene("r16", "standing on a cleared camp spawn: hold so the capture timer runs, unless hurt below 40% with an enemy champion within 3")
def r_hold() -> Bool:
    g = spawn_group(here())
    if g < 0 or not group_reward(g):
        return False
    return not (hurt_below(0.4) and enemies_within(3) > 0)
