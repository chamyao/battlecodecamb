```python
@gene('r12', 'threat near: an enemy champion that out-trades us (it needs at most 1.5x the hits we need) within 5, or within 7 when we are hurt, or able to hit us within 3 turns')
def threat_near() -> Bool:
    h = here()
    m = me()
    hurt = hurt_below(0.5)
    for u in enemies():
        if u.health_frac >= 0.15 or unit_gems(u) > 0:
            d = dist(h, u.pos)
            if d <= 5 or (hurt and d <= 7) or eta(d - u.attack_range, u.move_speed) <= 3:
                mine = u.health / max(1, attack_damage(m, u))
                theirs = m.health / max(1, attack_damage(u, m))
                if theirs <= 1.5 * mine:
                    return True
    return False
```
