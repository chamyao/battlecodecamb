@gene('w1', 'camp spawn tiles out of vision within 12 of me, not respawning and not in a cleared group waiting for capture: assumed occupied (farther ones are re-read when they come into view)')
def unseen_camps() -> Tiles:
    return (camp_spawns() & near(here(), 12)) - vision() - near(here(), 0) - respawning() - cleared_groups()
