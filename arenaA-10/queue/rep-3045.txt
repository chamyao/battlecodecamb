@gene('r11', 'bag ready: 12+, or 10+ with an altar 3 turns away or a threat near, or 5+ below 40% health (bank before the next fight takes it), or late, or hurt', parents=['r11'])
def bag_ready() -> Bool:
    g = gems()
    return g >= 12 or (g >= 10 and altar_eta() <= 3) or (g >= 10 and threat_near()) or (g >= 5 and hurt_below(0.4)) or round() > 1500 or hurt_below(0.3)
