```python
@gene('r16', 'standing on a cleared camp spawn: hold so the capture timer runs, unless an ally holds an earlier tile (row-major) of the same group, or an ally is on the group and I carry 10+ gems (it holds, I go bank)')
def r_hold() -> Bool:
    h = here()
    g = spawn_group(h)
    if g < 0 or not group_reward(g):
        return False
    others = (ally_tiles() - near(h, 0)) & group_tiles(g)
    if gems() >= 10 and size(others) > 0:
        return False
    k = h.y * 1000 + h.x
    return not any(t.y * 1000 + t.x < k for t in ally_tiles() & group_tiles(g))
```
