```python
@gene('s6', 'visible enemy champions within r tiles (ones below 25% health count only if carrying gems; recalling ones are leaving and do not count)')
def enemies_within(r: Num) -> Num:
    return count(
        dist(here(), u.pos) <= r and (u.health_frac >= 0.25 or unit_gems(u) > 0) and not is_recalling(u)
        for u in enemies()
    )
```
