@gene('w3', 'tiles I cannot step onto: terrain, altars, enemy spawn, units (except recalling enemy champions, which are leaving, and enemy champions over 5 away, which will have moved), unseen camp spawns')
def blocked() -> Tiles:
    et = enemy_tiles()
    h = here()
    for u in enemies():
        if is_recalling(u) or dist(h, u.pos) > 5:
            et = et - near(u.pos, 0)
    return (TERRAIN | ally_tiles() | et | camp_block()) - near(h, 0)
