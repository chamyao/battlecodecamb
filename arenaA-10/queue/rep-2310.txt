@gene('m12', 'bank walk: open safe altar unless a safe altar is much nearer; else any safe altar, else any altar; with enemies near, the candidate with fewer enemy champions within 6 of it', parents=['m12'])
def bank_goal() -> Tile:
    a = bank_t1()
    b = bank_t2()
    if valid(a) and valid(b):
        if a != b and enemies_within(8) > 0:
            na = count(dist(u.pos, a) <= 6 for u in enemies())
            nb = count(dist(u.pos, b) <= 6 for u in enemies())
            if na != nb:
                return a if na < nb else b
        if dist(here(), b) + 4 < dist(here(), a):
            return b
        return a
    if valid(a):
        return a
    if valid(b):
        return b
    return bank_t3()
