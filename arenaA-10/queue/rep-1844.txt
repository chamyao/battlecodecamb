@gene('c1', "where this turn's rule walks; banking under threat goes to an altar tile within 2 if there is one (speed), else the bank walk like unthreatened banking (safe, unlocked altar first)")
def rule_goal() -> Tile:
    r = rule()
    if r == 3:
        return closest(my_spawn())
    if r == 11:
        n = closest(ALTAR_RING & near(here(), 2))
        if valid(n):
            return n
        return bank_goal()
    if r == 12:
        return bank_goal()
    if r == 15:
        return closest(camp_spawns() - respawning() - type_spawns(GOLEM))
    if r == 18:
        return closest(SAFE_RING)
    if r == 19:
        return live_camp().pos
    if r == 20:
        return camp_goal()
    return tile(-1, -1)
