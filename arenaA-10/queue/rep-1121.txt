```python
@gene('r5', 'recall with 5+ gems when hurt (threshold rises faster with carried gems: 0.67 at 10, 0.83 at 20, up to 0.9) and nothing near, unless an altar is close enough to bank first; cheapest checks first')
def r_cashout() -> Bool:
    g = gems()
    if g < 5:
        return False
    f = min(0.9, 0.5 + g / 60)
    if not hurt_below(f) or at_spawn():
        return False
    if under_fire():
        return False
    return altar_eta() > 4
```
