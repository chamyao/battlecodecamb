@gene("s10", "fewest turns until a visible enemy champion could hit us (1000 if none); a stunned enemy is that many turns later")
def threat_eta() -> Num:
    h = here()
    return min((eta(dist(h, u.pos) - u.attack_range, u.move_speed) + buff_turns(u, 1) for u in enemies()), default=1000)
