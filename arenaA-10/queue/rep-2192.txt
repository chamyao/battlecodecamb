@gene('s15', 'the nearest visible camp, big camps (monoliths, golems) ranked last; camps of the group I stand on count 6 tiles nearer (finish the group being cleared); wounded camps preferred among near-equal distances')
def live_camp() -> Unit:
    h = here()
    g = spawn_group(h)
    return best(camps(), lambda c: -dist(h, c.pos) - 2 * c.health_frac - (1000 if c.pos in BIG_CAMPS else 0) + (6 if g >= 0 and spawn_group(camp_spawn(c)) == g else 0))
