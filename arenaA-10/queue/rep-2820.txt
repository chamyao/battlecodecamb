```python
@gene("r18", "on a dead camp of an uncleared group: walk to the nearest living camp (not monolith or golem, and not in a monolith group: walking up wakes nothing, but the fight there would)")
def r_live() -> Bool:
    c = live_camp()
    return dist(camp_goal(), here()) == 0 and exists(c) and not (c.pos in BIG_CAMPS) and not (camp_spawn(c) in MONO_GROUPS)
```
