@gene("a13", "a purchase is still due this turn (other options wait, unscored); after round 1200 a missing lock no longer holds up the turn")
def shop_pending() -> Bool:
    return RULE >= 2 and at_spawn() and can_shop() and ((LOCK_BUY and not has_lock() and gold() >= 10 and round() <= 1200) or (next_shard() >= 0 and gold() >= 30))
