@gene("c2", "the altar or unit tile this turn's rule acts on (reads the turn's rule keep)")
def rule_target() -> Tile:
    r = RULE
    if r == 6 or r == 9 or r == 10 or r == 17:
        return adj_altar()
    if r == 7:
        return fight_foe().pos
    if r == 8:
        return lock_altar()
    if r == 13 or r == 14:
        return reach_camp().pos
    return tile(-1, -1)

@gene('m2', 'an altar next to us: one free of enemy locks first (our own lock does not block our tribute), else an enemy-locked one (a tribute there hits the lock); the lock list is read only when an altar is adjacent')
def adj_altar() -> Tile:
    adj = altars() & near(here(), 1)
    if size(adj) == 0:
        return tile(-1, -1)
    la = adj & locked_altars()
    if size(la) == 0:
        return best(adj, lambda t: 0)
    el = la - la
    for t in la:
        if lock_enemy(t):
            el = el | near(t, 0)
    open_adj = adj - el
    if size(open_adj) > 0:
        return best(open_adj, lambda t: 0 if t in la else 1)
    return first_tile(adj)
