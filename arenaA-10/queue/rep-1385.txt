@gene('m11', "the camp walk's goal: nearest non-big spawn of a live group not within 2 of a visible enemy champion, then any such spawn, falling back to any non-big camp spawn, then any camp spawn")
def camp_goal() -> Tile:
    live = CAMP_BASE - respawning() - BIG_CAMPS
    et = enemy_tiles()
    if size(et) > 0:
        t = closest(live - ring(et, 2))
        if valid(t):
            return t
    t = closest(live)
    if valid(t):
        return t
    t = closest(CAMP_BASE - BIG_CAMPS)
    if valid(t):
        return t
    return closest(CAMP_BASE)
