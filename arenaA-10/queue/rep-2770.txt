@gene('r12', 'threat near: enemy within 5, hurt with an enemy within 7, or able to hit us within 3 turns; recalling enemies are leaving and never count; while I am above 80% health, enemies below 40% within 5 do not count (they will not engage)')
def threat_near() -> Bool:
    if threat_eta() <= 3:
        return True
    h = here()
    if me().health_frac > 0.8:
        return any(dist(h, u.pos) <= 5 and u.health_frac >= 0.4 and not is_recalling(u) for u in enemies())
    r = 7 if hurt_below(0.5) else 5
    return any(dist(h, u.pos) <= r and (u.health_frac >= 0.15 or unit_gems(u) > 0) and not is_recalling(u) for u in enemies())
