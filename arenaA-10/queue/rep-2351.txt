```python
@gene('r1', 'nothing to do: dead, channeling, cannot act, or recalling with nothing near, or recalling a bag of 10+ gems with no enemy within 3 (stay put so the recall completes)', parents=['r1'])
def r_idle() -> Bool:
    if not alive() or channeling() or (not can_move() and not can_attack()):
        return True
    if not is_recalling(me()):
        return False
    return not under_fire() or (gems() >= 10 and enemies_within(3) == 0)
```
