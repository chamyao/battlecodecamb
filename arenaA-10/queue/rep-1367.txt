@gene("v1", "value of doing this option now; casts are not held back by a pending purchase (they spend no gold)")
def score(o: Option) -> Num:
    k = o.kind
    if k == PICK:
        return pick_value(o.champ)
    if k == BUY or k == BUY_LOCK:
        return shop_value(o)
    if k == CAST:
        return cast_value(o)
    if shop_pending():
        return 0
    return act_value(o)
@gene('a13', 'a purchase is still due this turn (other options wait, unscored); cheap checks first, gold read once')
def shop_pending() -> Bool:
    if RULE < 2:
        return False
    g = gold()
    if g < 10 or not can_shop() or not at_spawn():
        return False
    return (LOCK_BUY and not has_lock()) or (g >= 30 and next_shard() >= 0)
