@gene('m2', 'an altar next to us: one free of enemy locks first (our own lock does not block our tribute), else an enemy-locked one (a tribute there hits the lock); off the altar ring there is none (no engine read)')
def adj_altar() -> Tile:
    if not (here() in ALTAR_RING):
        return tile(-1, -1)
    adj = altars() & near(here(), 1)
    if size(adj) == 0:
        return tile(-1, -1)
    la = adj & locked_altars()
    el = la - la
    for t in la:
        if lock_enemy(t):
            el = el | near(t, 0)
    open_adj = adj - el
    if size(open_adj) > 0:
        return best(open_adj, lambda t: 0 if t in la else 1)
    return first_tile(adj)
@gene("s14", "the first visible camp in reach (my tile and range read once)")
def reach_camp() -> Unit:
    m = me()
    h = m.pos
    r = m.attack_range
    return best(camps(), lambda c: 1 if dist(h, c.pos) <= r else 0)
