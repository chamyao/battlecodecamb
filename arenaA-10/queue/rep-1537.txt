```python
@gene('r10', 'bank now: carrying gems and the bag is ready, or next to an altar with 3+ gems, or 5+ gems after round 1200 (late gems decide the game)', parents=['r10'])
def bank_on() -> Bool:
    g = gems()
    return g > 0 and (bag_ready() or (g >= 3 and valid(adj_altar())) or (g >= 5 and round() > 1200))
```
