@gene("c2", "the altar or unit tile this turn's rule acts on; a fight targets an in-reach enemy our attack kills (off its spawn) before the usual pick")
def rule_target() -> Tile:
    r = rule()
    if r == 6 or r == 9 or r == 10 or r == 17:
        return adj_altar()
    if r == 7:
        es = enemy_spawn()
        k = best(enemies(), lambda u: unit_gems(u) - u.health if in_reach(u) and not (u.pos in es) and attack_damage(me(), u) >= u.health + shield(u) else -INF)
        if exists(k) and in_reach(k) and not (k.pos in es) and attack_damage(me(), k) >= k.health + shield(k):
            return k.pos
        return fight_foe().pos
    if r == 8:
        return lock_altar()
    if r == 13 or r == 14:
        return reach_camp().pos
    return tile(-1, -1)
