@gene('r13', 'every altar-adjacent tile is a camp spawn and the camp in reach stands on the nearest one: fight the blocker, only if it is farmable (v33 monolith rule: never wake a monolith group for it)')
def bank_blocked() -> Bool:
    c = reach_camp()
    if not exists(c) or not farmable(c) or not (c.pos in camp_spawns()):
        return False
    return not valid(bank_t1()) and not valid(bank_t2()) and c.pos == bank_t3()

@gene("new", "camp in reach worth hitting: not in a monolith group (a hit wakes the group), unless the team goes for monoliths, or it stands on an altar-ring tile and I carry gems (boss-battle) (v33 i1)", parents=["r13"])
def farmable(c: Unit) -> Bool:
    if not in_reach(c):
        return False
    if not (camp_spawn(c) in MONO_GROUPS):
        return True
    if gems() > 0 and camp_spawn(c) in ALTAR_RING:
        return True
    return r_mono()
