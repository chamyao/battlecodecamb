@gene("new", "the in-reach enemy champion the fight targets dies to one basic attack now (health plus shield) (v33 B2938a)", parents=["c0"])
def r_killshot() -> Bool:
    f = fight_foe()
    return exists(f) and attack_damage(me(), f) >= f.health + shield(f)

@gene("m13", "nearest tile next to a safe altar that no enemy lock blocks (our own lock does not block our tribute)")
def bank_t1() -> Tile:
    la = locked_altars()
    el = la - la
    for t in la:
        if lock_enemy(t):
            el = el | near(t, 0)
    return closest(SAFE_RING - ring(el, 1))
