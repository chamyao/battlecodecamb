```python
@gene("s10", "fewest turns until a visible enemy champion could hit us (1000 if none); a champion channelling a recall is leaving and does not count")
def threat_eta() -> Num:
    return min((eta(dist(here(), u.pos) - u.attack_range, u.move_speed) for u in enemies() if channel_left(u) == 0), default=1000)
```
