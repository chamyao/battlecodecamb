@gene('m11', "the camp walk's goal: nearest non-big spawn of a live group that no ally is already capturing (below 60% health, skip a group of strength 400+ when another is free), falling back to any non-big camp spawn, then any camp spawn")
def camp_goal() -> Tile:
    live = CAMP_BASE - respawning() - BIG_CAMPS - held_groups()
    t = closest(live)
    if valid(t):
        if hurt_below(0.6) and group_threat(t) >= 400:
            g = spawn_group(t)
            if g >= 0:
                t2 = closest(live - group_tiles(g))
                if valid(t2):
                    return t2
        return t
    t = closest(CAMP_BASE - BIG_CAMPS)
    if valid(t):
        return t
    return closest(CAMP_BASE)
