@gene("c2", "the altar or unit tile this turn's rule acts on; a camp fight hits an awake camp in reach before waking a sleeping one")
def rule_target() -> Tile:
    r = rule()
    if r == 6 or r == 9 or r == 10 or r == 17:
        return adj_altar()
    if r == 7:
        return fight_foe().pos
    if r == 8:
        return lock_altar()
    if r == 13:
        return reach_camp().pos
    if r == 14:
        c = best(camps(), lambda c: (2 if is_awake(c) else 1) if in_reach(c) else 0)
        return c.pos if exists(c) else reach_camp().pos
    return tile(-1, -1)
