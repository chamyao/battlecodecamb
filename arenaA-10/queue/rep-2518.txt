```python
@gene("s10", "fewest turns until a visible enemy champion could hit us (1000 if none); ones recalling home are leaving and do not count")
def threat_eta() -> Num:
    return min((eta(dist(here(), u.pos) - u.attack_range, u.move_speed) for u in enemies() if not is_recalling(u)), default=1000)
```
