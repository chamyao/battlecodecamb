@gene('r15', 'go for monoliths too: nothing tributed by round 800, or behind after round 1000 with 3+ allies within 6 (monoliths need a group)')
def r_mono() -> Bool:
    if team_gems() == 0 and round() > 800:
        return True
    return round() > 1000 and team_gems() < enemy_gems() and size(ally_tiles() & near(here(), 6)) >= 4
@gene('r12', 'threat near: enemy within 5, two or more enemies within 9, hurt with an enemy within 7, or able to hit us within 3 turns')
def threat_near() -> Bool:
    return enemies_within(5) > 0 or enemies_within(9) >= 2 or (hurt_below(0.5) and enemies_within(7) > 0) or threat_eta() <= 3
