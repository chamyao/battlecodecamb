@gene('a0', "value of a move, attack, tribute, recall, lock or ward option under this turn's rule; default rule also hits enemy champions in reach (not on their own spawn, gem carriers and wounded ones first)")
def act_value(o: Option) -> Num:
    k = o.kind
    r = RULE
    if k == MOVE:
        return move_value(o) if walks() else 0
    if k == ATTACK:
        if r == 7 or r == 14:
            return 178 if o.unit.pos == TGT else 0
        if r == 13:
            return 150 if o.unit.pos == TGT else 0
        if r == 20:
            if o.unit.is_enemy and o.unit.is_champion:
                if o.unit.pos in enemy_spawn():
                    return 0
                return 120 + min(10, unit_gems(o.unit) / 5) + 5 * (1 - o.unit.health_frac)
            return 90 if o.unit.is_camp and o.unit.pos == reach_camp().pos else 0
        return 0
    if k == TRIBUTE:
        return 150 if (r == 6 or r == 9 or r == 10 or r == 17) and o.tile == TGT else 0
    if k == RECALL:
        return 150 if r == 4 or r == 5 else 0
    if k == LOCK:
        return 150 if r == 8 and o.tile == TGT else 0
    if k == WARD:
        return 130 if r == 20 and o.tile == here() and here() in bushes() and count_units(enemies()) == 0 else 0
    return 0
@gene('a12', 'dash this turn: retreat with fuel > 40k; walking rules with GOAL over 2 away and fuel > 45k, but the camp walk only for goals over 6 away (keep the dash for fights and escapes)')
def dash_gate() -> Bool:
    if RULE == 3:
        return fuel() > 40000
    if RULE == 20 and valid(GOAL) and dist(here(), GOAL) <= 6:
        return False
    if walks() and valid(GOAL) and dist(here(), GOAL) > 2 and fuel() > 45000:
        return True
    return False
