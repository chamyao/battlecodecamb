```python
@gene("s11", "turns to stand next to an altar we can tribute at (not enemy-locked, no visible enemy champion within 2 of it; 1000 if none)")
def altar_eta() -> Num:
    h = here()
    ft = enemy_tiles()
    d = min((max(0, dist(h, a) - 1) for a in altars() - locked_altars() if not any(dist(e, a) <= 2 for e in ft)), default=-1)
    if d < 0:
        return 1000
    return eta(d, me().move_speed)
```
