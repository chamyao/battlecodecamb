@gene("r8", "place the team's altar lock: holder with an empty bag next to an altar, lock not yet placed, no enemy champion within 5 (placing it then wastes the turn)")
def r_lock() -> Bool:
    v = read_shared(7)
    return gems() == 0 and has_lock() and fuel() > 60000 and v - 2 * floor(v / 2) == 0 and enemies_within(5) == 0 and valid(lock_altar())
