@gene('a3', 'cast-ahead: damage, area or stun (slot kind k) on a close enemy champion before it reaches us; big bonus when the cast kills', parents=['a3'])
def ahead_value(o: Option, k: Num) -> Num:
    f = ahead_foe()
    if not exists(f) or RULE < 6 or k == AB_SHIELD or k == AB_HEAL or k == AB_RESET or not ahead_on():
        return 0
    near_foe = dist(o.tile, f.pos) <= 1
    if k == AB_AOE:
        if not near_foe:
            return 0
    elif o.tile != f.pos:
        return 0
    base = 180 - o.slot
    if k == AB_CC and threat_eta() <= 2:
        base += 20
    if k == AB_DAMAGE and f.health_frac < 0.3:
        base += 30
    if (k == AB_DAMAGE or k == AB_AOE) and cast_damage(me(), o.slot, f) >= f.health + shield(f):
        base += 40
    return base

@gene('a9', "shopping on spawn: the team's altar lock (only before round 1200: a late lock is rarely placed in time), then shards in the policy's split; cheap checks first")
def shop_value(o: Option) -> Num:
    if RULE < 2 or not at_spawn():
        return 0
    if o.kind == BUY_LOCK:
        return 200 if LOCK_BUY and round() < 1200 else 0
    if o.item != next_shard() or o.count > shard_left():
        return 0
    return 190 + o.count / 100
