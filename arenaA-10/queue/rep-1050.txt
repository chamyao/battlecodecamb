@gene("c2", "the altar or unit tile this turn's rule acts on (reads the turn's rule keep)")
def rule_target() -> Tile:
    r = RULE
    if r == 6 or r == 9 or r == 10 or r == 17:
        return adj_altar()
    if r == 7:
        return fight_foe().pos
    if r == 8:
        return lock_altar()
    if r == 13 or r == 14:
        return reach_camp().pos
    return tile(-1, -1)

@gene("w5", "walking field to GOAL, searched out to my own tile; reuses the turn's block set")
def walk_field() -> Field:
    g = near(GOAL, 0)
    return field_avoid(g, BLOCK - g, near(here(), 0))
