```python
@gene("a15", "shards of the first k kinds in the turn's split (of the shards SHOP_GOLD pays for)")
def shard_until(k: Num) -> Num:
    n = floor(SHOP_GOLD / 30)
    if k >= 4:
        return n
    t = floor(n / 2)
    if k == 1:
        return t
    j = floor((n - t) / 2)
    if k == 2:
        return t + j
    return t + j + floor((n - t - j) / 2)
```
