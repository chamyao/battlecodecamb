```python
@gene('r4', 'recall: below 30% health with no enemy champion able to hit us within 2 turns (damage cancels the channel), or below 45% carrying 10+ gems with no enemy champion within 6 (bring the bag home)')
def r_recall() -> Bool:
    if at_spawn():
        return False
    return (hurt_below(0.3) and threat_eta() > 2) or (gems() >= 10 and hurt_below(0.45) and enemies_within(6) == 0)
```
