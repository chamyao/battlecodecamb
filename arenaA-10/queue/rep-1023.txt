```python
@gene("a7", "a dash onto the landing tile that brings us closest to GOAL (slot 0 first); on a retreat the dash outranks every other cast")
def dash_value(o: Option) -> Num:
    if not dash_gate():
        return 0
    s = dash_spot()
    if not valid(s) or o.tile != s:
        return 0
    return (176 if RULE == 3 else 140) - o.slot
```
