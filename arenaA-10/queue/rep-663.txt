@gene("s11", "turns to stand next to an altar we can tribute at (not enemy-locked: our own visible lock does not block; 1000 if none)")
def altar_eta() -> Num:
    la = locked_altars()
    bad = la - la
    for t in la:
        if not (lock_hits(t) > 0 and not lock_enemy(t)):
            bad = bad | near(t, 0)
    h = here()
    d = min((max(0, dist(h, a) - 1) for a in altars() - bad), default=-1)
    if d < 0:
        return 1000
    return eta(d, me().move_speed)
