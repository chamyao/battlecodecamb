@gene("r18", "on a dead camp of an uncleared group: walk to the nearest living camp (not monolith or golem); cheap camp checks before the camp walk's goal")
def r_live() -> Bool:
    c = live_camp()
    if not exists(c) or c.pos in BIG_CAMPS:
        return False
    return dist(camp_goal(), here()) == 0
