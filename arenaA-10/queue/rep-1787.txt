@gene('r5', 'recall with 5+ gems when hurt (threshold 0.53 at 5, 0.62 at 10, 0.78 at 20, capped 0.8) and nothing near (no enemy able to hit us within the 5-turn channel), unless an altar is close enough to bank first')
def r_cashout() -> Bool:
    if at_spawn():
        return False
    g = gems()
    if g < 5:
        return False
    f = min(0.8, 0.45 + g / 60)
    if not hurt_below(f):
        return False
    if under_fire() or threat_eta() <= 5:
        return False
    return altar_eta() > 4
