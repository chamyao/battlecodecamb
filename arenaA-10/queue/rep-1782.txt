@gene("r3", "retreat on foot: below 30% health and under fire, or below 45% with two or more enemy champions within 4 (outnumbered fights are lost)")
def r_run() -> Bool:
    if at_spawn():
        return False
    if hurt_below(0.3):
        return under_fire()
    return hurt_below(0.45) and enemies_within(4) >= 2
