@gene('Z124a', 'start of turn: refresh the goal field only when the goal set changed; home field once; retreat flag (also with a big bag near a stronger foe)')
def update() -> None:
    if not HOMED:
        HF = field(my_spawn())
        HOMED = True
    live = camp_tiles()
    DEAD = (DEAD | ((camp_spawns() & vision()) - live)) - live
    if size(DEAD) > 0:
        for g in range(num_groups()):
            if group_respawning(g):
                DEAD = DEAD - group_tiles(g)
    k = goal_key()
    if k != GKEY:
        GS = goals()
        GF = field(GS)
        GKEY = k
    RETREAT = (gems() >= 15 and any(dist(u.pos, me().pos) <= 4 and u.health > me().health for u in enemies())) or ((me().health_frac < 0.35 or (me().health_frac < 0.5 and gems() >= 10)) and under_threat())
    WF = HF if RETREAT else GF
    here = me().pos
    blk = (walls() | holes() | altars() | enemy_spawn() | ally_tiles() | enemy_tiles() | camp_tiles() | (camp_spawns() - vision() - DEAD)) - near(here, 0)
    mine = at(WF, here)
    if mine > 0 and mine < 100000 and not any(at(WF, t) < mine for t in near(here, 1) - blk):
        tgt = my_spawn() if RETREAT else GS
        WF = field_avoid(tgt, blk - tgt, near(here, 0))
