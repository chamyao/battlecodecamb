@gene("s2", "where to walk: an altar with gems, else a living camp group or a weaker visible enemy carrying gems")
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    for u in enemies():
        if unit_gems(u) >= 5 and u.health < me().health and not (u.pos in enemy_spawn()):
            acc = acc | near(u.pos, 1)
    return acc
