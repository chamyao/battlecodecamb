@gene("s1", "value of doing this option now: draft strong champions, farm until the bag is ready")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if not bag_ready() else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "draft preference: Vendetta, Vesper Vale, Tavi, Ryth, Ren, then anything", parents=["s1"])
def pick_value(c: Num) -> Num:
    if c == VENDETTA:
        return 6
    if c == VESPER_VALE:
        return 5
    if c == TAVI:
        return 4
    if c == RYTH:
        return 3
    if c == REN:
        return 2
    return 1

@gene("new", "bag ready to bank: 12+ gems, or any gems late or when hurt", parents=["s2"])
def bag_ready() -> Bool:
    g = gems()
    return g >= 12 or (g > 0 and (round() > 1700 or me().health_frac < 0.5))

@gene("s2", "where to walk: an altar when the bag is ready, else a living camp group")
def goals() -> Tiles:
    if bag_ready():
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
