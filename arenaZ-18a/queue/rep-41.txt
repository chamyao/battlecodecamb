```python
@gene("s2", "where to walk: an altar with gems, else a living camp group without monoliths")
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    mono = type_spawns(MONOLITH)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & mono) == 0:
                acc = acc | gt
    return acc
```
