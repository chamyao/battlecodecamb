```python
@gene('s1', 'value of doing this option now: draft, casts, focus the weakest foe, fight back against awake camps, heal on spawn, cash out a big bag; shop durability and jungle emblems; recall only out of danger')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return shop_value(o)
    if o.kind == RECALL:
        if under_threat():
            return 0
        if me().health_frac < 0.35:
            return 76
        return 76 if gems() >= 10 and me().health_frac < 0.5 and count_units(enemies()) == 0 else 0
    if (o.kind == MOVE or o.kind == CAST or o.kind == ATTACK) and fleeing():
        return retreat_value(o)
    if o.kind == CAST:
        return cast_v2(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 10 * (0.6312 - o.unit.health_frac)
        if gems() == 0:
            if big_camp(o.unit) and not is_awake(o.unit):
                return 0
            return 60 + 5 * (1 - o.unit.health_frac)
        if is_awake(o.unit):
            return 55
        return 54 if path_blocked() else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().pos in my_spawn() and me().health_frac < 0.9 and count_units(enemies()) == 0:
            return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "while retreating: step or dash home first, shield or heal, otherwise hit only what is in reach", parents=["s1"])
def retreat_value(o: Option) -> Num:
    if o.kind == MOVE:
        return 95 if progress(o.tile) > 0 else 0
    if o.kind == CAST:
        k = slot_kind(o.slot)
        if k == AB_DASH:
            return 96 if progress(o.tile) >= 2 else 0
        if k == AB_SHIELD or k == AB_HEAL:
            return 97 if o.unit == me() else 0
        return cast_v2(o) / 2
    return 40

@gene("new", "badly hurt with a foe or awake camp near: walk home", parents=["s1"])
def fleeing() -> Bool:
    return me().health_frac < 0.35 and under_threat()
```
