@gene("s1", "value of doing this option now: draft, casts at foes and camps, fight, bank, heal on spawn, recall to shop with spare gold")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        return 55 if gems() == 0 and gold() >= 120 and count_units(enemies()) == 0 and not any(is_awake(c) for c in camps()) else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 10 * (1 - o.unit.health_frac)
        return 60 if gems() == 0 or is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().pos in my_spawn() and me().health_frac < 0.9 and count_units(enemies()) == 0:
            return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("s2", "where to walk: an altar with gems, else a living camp group without monoliths")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g) and not any(spawn_type(t) == MONOLITH for t in group_tiles(g)):
            acc = acc | group_tiles(g)
    return acc
