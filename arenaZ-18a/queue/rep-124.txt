```python
@gene('s4', 'how much closer a step brings me to the goals, minus a penalty for stepping next to enemy champions when hurt (only foes at least as healthy)')
def progress(t: Tile) -> Num:
    gain = at(goal_field(), me().pos) - at(goal_field(), t)
    if me().health_frac < 0.5 and any(dist(u.pos, t) <= u.attack_range + 1 and u.health >= me().health for u in enemies()):
        return gain - 2
    return gain
```
