```python
@gene("s1", "value of doing this option now: draft, casts, focus the weakest foe, fight back against awake camps, heal on spawn")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 10 * (1 - o.unit.health_frac)
        if gems() == 0:
            return 60 + 5 * (1 - o.unit.health_frac)
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().pos in my_spawn() and me().health_frac < 0.9 and count_units(enemies()) == 0:
            return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "draft preference: Vendetta, Vesper Vale, Tavi, Ryth, Ren, then anything", parents=["s1"])
def pick_value(c: Num) -> Num:
    if c == VENDETTA:
        return 6
    if c == VESPER_VALE:
        return 5
    if c == TAVI:
        return 4
    if c == RYTH:
        return 3
    if c == REN:
        return 2
    return 1

@gene("Z6a", "value of a cast: damage or stun on an enemy champion or a camp I fight, shield or heal on myself when hurt")
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_SHIELD or k == AB_HEAL:
        return 85 if o.unit == me() and me().health_frac < 0.7 else 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit) and o.unit.is_enemy and o.unit.is_champion:
            return 86
        if exists(o.unit) and o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 65
        if any(u.pos == o.tile for u in enemies()):
            return 84
    return 0
```
