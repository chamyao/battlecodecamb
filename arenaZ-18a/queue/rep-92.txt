@gene("Z92a", "start of turn: refresh the goal field only when the goal set changed; home field once; retreat flag (also when hurt with a big bag); stuck counter")
def update() -> None:
    if not HOMED:
        HF = field(my_spawn())
        HOMED = True
    k = goal_key()
    if k != GKEY:
        GF = field(goals())
        GKEY = k
    hf = me().health_frac
    RETREAT = (hf < 0.35 or (hf < 0.5 and gems() >= 10)) and under_threat()
    if me().pos == LAST and not at_home():
        STUCK += 1
    else:
        STUCK = 0
    LAST = me().pos
