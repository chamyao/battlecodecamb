```python
GF: Field = None      # cached walking field to the goals
GKEY: Num = -1000     # goal set the cached field was computed for

@gene("s3", "walking field: home when badly hurt with a foe or awake camp near, else the cached field to the goals")
def goal_field() -> Field:
    if me().health_frac < 0.35 and under_threat():
        return field(my_spawn())
    return GF

@gene("new", "start of turn: refresh the cached goal field when the goal set changed", parents=["s3"])
def update() -> None:
    k = goal_key()
    if k != GKEY:
        GF = field(goals())
        GKEY = k

@gene("new", "a number that changes whenever the goal set changes: bag state, locked altars, living groups", parents=["s3"])
def goal_key() -> Num:
    if gems() > 0:
        return -1 - size(locked_altars())
    acc = 0
    p = 1
    for g in range(num_groups()):
        if not group_respawning(g):
            acc += p
        p *= 2
    return acc
```
