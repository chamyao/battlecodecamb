```python
@gene('s1', 'value of doing this option now: draft, casts, focus the weakest foe, heal on spawn, cash out a big bag, recall only out of danger; shop durability and jungle emblems')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return shop_value(o)
    if o.kind == RECALL:
        if under_threat():
            return 0
        if me().health_frac < (0.45 if gems() > 0 else 0.3):
            return 90
        if gems() >= 10 and me().health_frac < 0.5:
            return 90
        return 45 if gems() == 0 and gold() >= 210 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 0 if o.unit.pos in enemy_spawn() else 80 + 10 * (1 - o.unit.health_frac)
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().pos in my_spawn() and me().health_frac < 0.9 and count_units(enemies()) == 0:
            return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "shop: a durability health emblem, then a jungle health emblem, then power strength; bigger buys first", parents=["s1"])
def shop_value(o: Option) -> Num:
    if items_owned(17) == 0:
        return 100 + o.count / 100 if o.item == 16 and o.count <= 5 - items_owned(16) else 0
    if items_owned(29) == 0:
        return 100 + o.count / 100 if o.item == 28 and o.count <= 5 - items_owned(28) else 0
    return 100 + o.count / 100 if o.item == 6 else 0
```
