```python
@gene("s1", "value of doing this option now: draft, casts, heal on spawn, recall early with a big bag")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        return 90 if gems() >= 10 and me().health_frac < 0.5 and count_units(enemies()) == 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().pos in my_spawn() and me().health_frac < 0.9 and count_units(enemies()) == 0:
            return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "draft preference: Vendetta, Vesper Vale, Tavi, Ryth, Ren, then anything", parents=["s1"])
def pick_value(c: Num) -> Num:
    if c == VENDETTA:
        return 6
    if c == VESPER_VALE:
        return 5
    if c == TAVI:
        return 4
    if c == RYTH:
        return 3
    if c == REN:
        return 2
    return 1
```
