```python
@gene("s1", "value of doing this option now: draft, casts, heal on spawn, cash out a big bag by recall, recall to shop with lots of gold")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        if gems() >= 10 and me().health_frac < 0.5 and count_units(enemies()) == 0:
            return 90
        return 45 if gems() == 0 and gold() >= 210 and count_units(enemies()) == 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().pos in my_spawn() and me().health_frac < 0.9 and count_units(enemies()) == 0:
            return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
