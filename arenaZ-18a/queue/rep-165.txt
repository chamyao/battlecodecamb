@gene('s1', 'value of doing this option now: draft, casts, focus the weakest foe, fight back against awake camps, heal on spawn, cash out a big bag; shop durability and jungle emblems; recall only out of danger')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return shop_value(o)
    if o.kind == RECALL:
        if under_threat():
            return 0
        if me().health_frac < 0.45:
            return 76
        return 76 if gems() >= 10 and me().health_frac < 0.5 and count_units(enemies()) == 0 else 0
    if (o.kind == MOVE or o.kind == CAST or o.kind == ATTACK) and RETREAT:
        return retreat_value(o)
    if o.kind == CAST:
        return cast_v2(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 10 * (0.6312 - o.unit.health_frac)
        if gems() == 0:
            return 60 + 5 * (1 - o.unit.health_frac)
        if is_awake(o.unit):
            return 55
        return 54 if path_blocked() else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().pos in my_spawn() and me().health_frac < 0.9 and count_units(enemies()) == 0:
            return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "option kinds worth listing this pass: skip kinds never used, casts with no slot ready, shopping off spawn, tributes without gems", parents=["s1"])
def consider(k: Num) -> Bool:
    if k == PICK or k == MOVE or k == ATTACK:
        return True
    if k == CAST:
        return slot_ready(0) or slot_ready(1)
    if k == BUY:
        return can_shop()
    if k == TRIBUTE:
        return gems() > 0
    if k == RECALL:
        return not (me().pos in my_spawn())
    return False

ALT: Tiles = None     # tiles next to altars we can tribute at (cached)
LIVEG: Tiles = None   # spawn tiles of live groups not yet cleared, no monolith or golem groups (cached)
REWG: Tiles = None    # spawn tiles of cleared groups waiting for capture (cached)
BIGS: Tiles = None    # monolith and golem spawns
GOAL: Tile = None     # this turn's walking goal
GTURN: Num = 0        # round to re-pick the walking goal
GSKEY: Num = -1000    # bag and group state the cached sets were built for

@gene("Z121a", "start of turn: dead camp spawns, cached goal sets (rebuilt when bag or group state changes), damage and retreat flags, then a walking goal by walk distance (re-picked every 8 rounds) and a field to it around units")
def update() -> None:
    if not HOMED:
        BIGS = type_spawns(MONOLITH) | type_spawns(GOLEM)
        HOMED = True
    live = camp_tiles()
    DEAD = (DEAD | ((camp_spawns() & vision()) - live)) - live
    if size(DEAD) > 0:
        for g in range(num_groups()):
            if group_respawning(g):
                DEAD = DEAD - group_tiles(g)
    k = -1 - size(locked_altars())
    if gems() == 0:
        k = 0
        p = 1
        for g in range(num_groups()):
            if not group_respawning(g):
                k += p * (2 if group_reward(g) else 1)
            p *= 3
    force = k != GSKEY
    if force:
        GSKEY = k
        if gems() > 0:
            ALT = near(tile(-5, -5), 0)
            for a in altars() - locked_altars():
                ALT = ALT | near(a, 1)
        else:
            LIVEG = near(tile(-5, -5), 0)
            REWG = LIVEG
            for g in range(num_groups()):
                if not group_respawning(g):
                    gt = group_tiles(g)
                    if group_reward(g):
                        REWG = REWG | gt
                    elif size(gt & BIGS) == 0:
                        LIVEG = LIVEG | gt
    HIT = me().health < LASTHP - 0.5
    LASTHP = me().health
    RETREAT = me().health_frac < 0.35 and (HIT or under_threat())
    here = me().pos
    unseen = camp_spawns() - vision() - DEAD
    blk = (walls() | holes() | altars() | enemy_spawn() | ally_tiles() | enemy_tiles() | live | unseen) - near(here, 0)
    tgt = my_spawn() if RETREAT else goals()
    if here in tgt:
        GOAL = here
    elif force or not valid(GOAL) or not (GOAL in tgt) or GOAL == here or round() >= GTURN:
        cand = tgt - (blk - live - unseen)
        GOAL = walk_nearest(cand, here, blk - cand)
        GTURN = round() + 8
    gt2 = near(GOAL, 0)
    WF = field_avoid(gt2, blk - gt2, near(here, 0))

@gene("s2", "where to walk: an altar with gems; else cleared groups to capture and live groups minus spawns seen empty")
def goals() -> Tiles:
    if gems() > 0:
        return ALT
    return REWG | (LIVEG - DEAD)
