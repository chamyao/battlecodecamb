@gene("s1", "value of doing this option now: draft, casts, heal on spawn, no fights on the enemy spawn, farm camps until the bag is ready")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 0 if o.unit.pos in enemy_spawn() else 80
        return 60 if not bag_ready() else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().pos in my_spawn() and me().health_frac < 0.9 and count_units(enemies()) == 0:
            return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "bag ready to bank: 8+ gems, or any gems late, hurt, or with an enemy champion in sight", parents=["s2"])
def bag_ready() -> Bool:
    g = gems()
    return g >= 8 or (g > 0 and (round() > 1700 or me().health_frac < 0.5 or count_units(enemies()) > 0))

@gene("s2", "where to walk: an altar when the bag is ready, else a living camp group")
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if bag_ready():
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
