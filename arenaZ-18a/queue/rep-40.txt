```python
@gene("s3", "distance field to the goals; home when badly hurt with a foe or awake camp near")
def goal_field() -> Field:
    if me().health_frac < 0.35 and under_threat():
        return field(my_spawn())
    return field(goals())

@gene("new", "an enemy champion within 5 or an awake camp within its range + 1", parents=["s3"])
def under_threat() -> Bool:
    p = me().pos
    return any(dist(p, u.pos) <= 5 for u in enemies()) or any(is_awake(c) and dist(p, c.pos) <= c.attack_range + 1 for c in camps())
```
