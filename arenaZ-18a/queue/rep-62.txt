```python
@gene("s4", "how much closer a step brings me to the goals; avoid enemy reach when hurt, and healthier foes when carrying gems")
def progress(t: Tile) -> Num:
    gain = at(goal_field(), me().pos) - at(goal_field(), t)
    if me().health_frac < 0.5 and any(dist(u.pos, t) <= u.attack_range + 1 for u in enemies()):
        return gain - 2
    if gems() > 0 and any(dist(u.pos, t) <= u.attack_range + 1 and u.health > me().health for u in enemies()):
        return gain - 2
    return gain
```
