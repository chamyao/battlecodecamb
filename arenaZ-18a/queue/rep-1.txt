@gene("s2", "where to walk: an altar with gems, else a living camp group without monoliths or golems")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g) and safe_group(g):
            acc = acc | group_tiles(g)
    return acc

@gene("new", "a camp group with no monolith or golem in it", parents=["s2"])
def safe_group(g: Num) -> Bool:
    return not any(spawn_type(t) == MONOLITH or spawn_type(t) == GOLEM for t in group_tiles(g))
