```python
@gene('s1', 'value of doing this option now: draft, casts, heal on spawn, no fights on the enemy spawn, farm camps until the bag is ready; shop durability and jungle emblems; recall only out of danger')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return shop_value(o)
    if o.kind == RECALL:
        if under_threat():
            return 0
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 0 if o.unit.pos in enemy_spawn() else 80
        return 60 if not bag_ready() else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().pos in my_spawn() and me().health_frac < 0.9 and count_units(enemies()) == 0:
            return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "shop: a durability health emblem, then a jungle health emblem, then power strength; bigger buys first", parents=["s1"])
def shop_value(o: Option) -> Num:
    if items_owned(17) == 0:
        return 100 + o.count / 100 if o.item == 16 and o.count <= 5 - items_owned(16) else 0
    if items_owned(29) == 0:
        return 100 + o.count / 100 if o.item == 28 and o.count <= 5 - items_owned(28) else 0
    return 100 + o.count / 100 if o.item == 6 else 0

@gene("new", "an enemy champion within 5 or an awake camp within its range + 1", parents=["s1"])
def under_threat() -> Bool:
    p = me().pos
    return any(dist(p, u.pos) <= 5 for u in enemies()) or any(is_awake(c) and dist(p, c.pos) <= c.attack_range + 1 for c in camps())
```
