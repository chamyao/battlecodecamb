```python
@gene('s1', 'value of doing this option now: draft, casts, focus the weakest foe, heal on spawn, cash out a big bag, recall only out of danger; shop durability and jungle emblems')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return shop_value(o)
    if o.kind == RECALL:
        if under_threat():
            return 0
        if me().health_frac < (0.45 if gems() > 0 else 0.3):
            return 90
        if gems() >= 10 and me().health_frac < 0.5:
            return 90
        return 45 if gems() == 0 and gold() >= 210 else 0
    if (o.kind == MOVE or o.kind == CAST or o.kind == ATTACK) and fleeing():
        return retreat_value(o)
    if o.kind == CAST:
        return cast_v2(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 0 if o.unit.pos in enemy_spawn() else 80 + 10 * (1 - o.unit.health_frac)
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().pos in my_spawn() and me().health_frac < 0.9 and count_units(enemies()) == 0:
            return 0
        return 50 if step_gain(o.tile) > 0 else 0
    return 0

@gene("new", "while retreating: step or dash home first, shield or heal, otherwise hit only what is in reach", parents=["s1"])
def retreat_value(o: Option) -> Num:
    if o.kind == MOVE:
        return 95 if step_gain(o.tile) > 0 else 0
    if o.kind == CAST:
        k = slot_kind(o.slot)
        if k == AB_DASH:
            return 96 if step_gain(o.tile) >= 2 else 0
        if k == AB_SHIELD or k == AB_HEAL:
            return 97 if o.unit == me() else 0
        return cast_v2(o) / 2
    return 40

@gene("new", "badly hurt with a foe or awake camp near: walk home", parents=["s1"])
def fleeing() -> Bool:
    return me().health_frac < 0.35 and under_threat()
```
