@gene('Z223a', 'start of turn: refresh the goal field only when the goal set changed; home field once; retreat flag')
def update() -> None:
    if not MAPPED:
        BIGS = type_spawns(MONOLITH) | type_spawns(GOLEM)
        MAPPED = True
    live = camp_tiles()
    DEAD = (DEAD | ((camp_spawns() & vision()) - live)) - live
    if size(DEAD) > 0:
        for g in range(num_groups()):
            if group_respawning(g):
                DEAD = DEAD - group_tiles(g)
    BAGR = bag_ready()
    k = goal_key()
    force = k != GKEY
    if force:
        GKEY = k
        if BAGR:
            ALT = near(tile(-5, -5), 0)
            for a in altars() - locked_altars():
                ALT = ALT | near(a, 1)
        else:
            LIVEG = near(tile(-5, -5), 0)
            REWG = LIVEG
            for gi in range(num_groups()):
                if not group_respawning(gi):
                    gt = group_tiles(gi)
                    if group_reward(gi):
                        REWG = REWG | gt
                    elif size(gt & BIGS) == 0:
                        LIVEG = LIVEG | gt
    HIT = me().health < LASTHP - 0.5
    LASTHP = me().health
    RETREAT = me().health_frac < 0.3 and (HIT or under_threat())
    here = me().pos
    unseen = camp_spawns() - vision() - DEAD
    blk = (walls() | holes() | altars() | enemy_spawn() | ally_tiles() | enemy_tiles() | live | unseen) - near(here, 0)
    tgt = my_spawn() if RETREAT else goals()
    if here in tgt:
        GOAL = here
    elif force or not valid(GOAL) or not (GOAL in tgt) or GOAL == here or round() >= GTURN:
        cand = tgt - (blk - live - unseen)
        GOAL = walk_nearest(cand, here, blk - cand)
        GTURN = round() + 5
    g = near(GOAL, 0)
    WF = field_avoid(g, blk - g, near(here, 0))


@gene('Z223d', 'value of a cast: damage or stun on foes and camps, shield or heal when hurt, attack resets once the attack is spent, dash two steps toward the goal')
def cast_v2(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_SHIELD or k == AB_HEAL:
        return 85 if o.unit == me() and (me().health_frac < 0.7 or any(dist(u.pos, me().pos) <= 2 for u in enemies())) else 0
    if k == AB_RESET:
        if can_attack():
            return 0
        r = me().attack_range
        if any(dist(u.pos, me().pos) <= r for u in enemies()):
            return 88
        return 66 if not BAGR and any(dist(c.pos, me().pos) <= r for c in camps()) else 0
    if k == AB_DASH:
        return 52 if progress(o.tile) >= 2 else 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit) and o.unit.is_enemy and o.unit.is_champion:
            return 86
        if exists(o.unit) and o.unit.is_camp and not BAGR:
            return 65
        if any(u.pos == o.tile for u in enemies()):
            return 84
    return 0
