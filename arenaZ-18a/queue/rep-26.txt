```python
@gene("s1", "value of doing this option now: draft, casts, heal fully on spawn before leaving")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().pos in my_spawn() and me().health_frac < 0.9 and count_units(enemies()) == 0:
            return 0
        return 61 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "value of a cast: damage or stun on an enemy champion or camp, shield or heal on myself when hurt", parents=["s1"])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_SHIELD or k == AB_HEAL:
        return 85 if o.unit == me() and me().health_frac < 0.7 else 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit) and o.unit.is_enemy and o.unit.is_champion:
            return 86
        if exists(o.unit) and o.unit.is_camp and gems() == 0:
            return 65
        if any(u.pos == o.tile for u in enemies()):
            return 84
    return 0

@gene("s4", "how much closer a step brings me to the goals, minus a penalty for stepping next to enemy champions when hurt")
def progress(t: Tile) -> Num:
    gain = at(goal_field(), me().pos) - at(goal_field(), t)
    if me().health_frac < 0.5 and any(dist(u.pos, t) <= u.attack_range + 1 for u in enemies()):
        return gain - 2
    return gain
```
