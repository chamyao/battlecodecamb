```python
GOALF: Field = None   # cached walking field to the goals
GSET: Tiles = None    # cached goal set

@gene("Z113a", "start of turn: cached goal field (refreshed when the goal set changes), walk around units when it is blocked; home field once; retreat flag")
def update() -> None:
    if not HOMED:
        HF = field(my_spawn())
        HOMED = True
    live = camp_tiles()
    DEAD = (DEAD | ((camp_spawns() & vision()) - live)) - live
    if size(DEAD) > 0:
        for g in range(num_groups()):
            if group_respawning(g):
                DEAD = DEAD - group_tiles(g)
    k = goal_key()
    if k != GKEY:
        GSET = goals()
        GOALF = field(GSET)
        GKEY = k
    RETREAT = me().health_frac < 0.35 and under_threat()
    GF = GOALF
    if not RETREAT:
        here = me().pos
        blk = (walls() | holes() | altars() | enemy_spawn() | ally_tiles() | enemy_tiles() | live | (camp_spawns() - vision() - DEAD)) - near(here, 0)
        mine = at(GF, here)
        if mine > 0 and mine < 100000 and not any(at(GF, t) < mine for t in near(here, 1) - blk):
            GF = field_avoid(GSET, blk - GSET, near(here, 0))
```
