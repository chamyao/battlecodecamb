@gene("s2", "where to walk: an altar with no enemy champion near it when carrying gems (any altar if none), else a living camp group")
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        safe = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
            if not any(dist(u.pos, a) <= 4 for u in enemies()):
                safe = safe | near(a, 1)
        if size(safe) > 0:
            return safe
        if size(acc) > 0:
            return acc
        for a in altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
