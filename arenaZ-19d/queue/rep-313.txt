@gene('Z6a', 'value of casting an ability: damage and stuns on fighting targets, shields and resets in fights, dash toward goals or away from danger, area spells aimed once per pass', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit) and (o.unit.is_champion or (o.unit.is_camp and (gems() < 10 or is_awake(o.unit)))):
            return 85
        return 0
    if k == AB_AOE:
        t = aoe_center()
        return 84 if valid(t) and dist(o.tile, t) == 0 else 0
    if k == AB_SHIELD:
        threat = count(dist(u.pos, me().pos) <= 4 for u in enemies()) + count(is_awake(u) and dist(u.pos, me().pos) <= 4 for u in camps())
        return 83 if threat > 0 and me().health_frac < 0.9 else 0
    if k == AB_RESET:
        return 82 if (not can_attack()) and count(dist(u.pos, me().pos) <= me().attack_range for u in enemies()) + count(is_awake(u) and dist(u.pos, me().pos) <= me().attack_range for u in camps()) > 0 else 0
    if k == AB_DASH:
        if me().health_frac < 0.3 and threat_at(me().pos) > 0:
            return 105 if threat_at(o.tile) < threat_at(me().pos) else 0
        return 52 if progress(o.tile) >= 2 else 0
    if k == AB_HEAL:
        return 86 if exists(o.unit) and o.unit.health_frac < 0.75 else 0
    return 0

@gene("new", "where to drop an area spell this pass: the enemy champion, else the awake camp, within reach", parents=["Z6a"])
def aoe_center() -> Tile:
    e = best(enemies(), lambda u: count(dist(v.pos, u.pos) <= 1 for v in enemies()) if dist(u.pos, me().pos) <= 4 else -1)
    if exists(e) and dist(e.pos, me().pos) <= 4:
        return e.pos
    c = best(camps(), lambda u: count(dist(v.pos, u.pos) <= 1 and is_awake(v) for v in camps()) if is_awake(u) and dist(u.pos, me().pos) <= 4 else -1)
    if exists(c) and is_awake(c) and dist(c.pos, me().pos) <= 4:
        return c.pos
    return tile(-5, -5)
