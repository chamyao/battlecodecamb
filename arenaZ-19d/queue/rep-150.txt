@gene('Z105a', 'value of doing this option now: draft a ranged team, cast abilities, farm camps until 10 gems, recall when safe, retreat from threats when hurt, earlier recall and retreat, finish awake camps, wake new ones only when healthy, focus hurt and gem-carrying champions, later recall and retreat', parents=['s1'])
def score(o: Option) -> Num:
    if o.kind == PICK:
        if o.champ == VESPER_VALE:
            return 10
        if o.champ == STELLAR:
            return 9
        if o.champ == TAVI:
            return 8
        if o.champ == CASSI:
            return 7
        if o.champ == WILLOW:
            return 6
        return 1
    if o.kind == BUY:
        return 100 + item_strength(o.item) if item_cost(o.item) >= 22 and item_strength(o.item) > 0 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.25 and threat_at(me().pos) == 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac) + (3 if unit_gems(o.unit) > 0 else 0)
        return 60 if (gems() < 11 or is_awake(o.unit)) and (is_awake(o.unit) or me().health_frac > 0.5) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.3 and threat_at(me().pos) > 0:
            return 104 if threat_at(o.tile) < threat_at(me().pos) else 0
        if chase_gain(o.tile) > 0:
            return 55
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('s2', 'where to walk: a free altar once the bag holds 10 gems, else a living camp group other than golems and monoliths, gem clusters first when close, bank early when hurt, wider gem margin')
def goals() -> Tiles:
    if gems() >= 10 or (gems() > 0 and (round() > 1900 or me().health_frac < 0.4)):
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    rest = acc - type_spawns(MONOLITH) - type_spawns(GOLEM)
    gem = acc & type_spawns(GEM_CLUSTER)
    if size(gem) > 0:
        if dist(nearest(gem, me().pos), me().pos) <= dist(nearest(rest, me().pos), me().pos) + 14:
            return gem
    return rest
