@gene('Z105a', 'value of doing this option now: draft a ranged team, cast abilities, farm camps until 10 gems, recall when safe, retreat from threats when hurt, earlier recall and retreat, finish awake camps, wake new ones only when healthy, later recall and retreat, focus hurt and gem-carrying champions, earlier recall and retreat, later recall and retreat, hunt gem carriers, step away from hitters even when cornered', parents=['s1'])
def score(o: Option) -> Num:
    if o.kind == PICK:
        if o.champ == VESPER_VALE:
            return 10
        if o.champ == STELLAR:
            return 9
        if o.champ == TAVI:
            return 8
        if o.champ == CASSI:
            return 7
        if o.champ == WILLOW:
            return 6
        return 1
    if o.kind == BUY:
        return 100 + item_strength(o.item) if item_cost(o.item) >= 22 and item_strength(o.item) > 0 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.25 and threat_at(me().pos) == 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac) + (3 if unit_gems(o.unit) > 0 else 0)
        return 60 if (gems() < 10 or is_awake(o.unit)) and (is_awake(o.unit) or me().health_frac > 0.5) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.3 and threat_at(me().pos) > 0:
            return 104 if threat_at(o.tile) < threat_at(me().pos) or escape_gain(o.tile) > 0 else 0
        if hunt_gain(o.tile) > 0:
            return 62
        if chase_gain(o.tile) > 0:
            return 55
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "how much farther a step takes me from the awake camps and enemies that can hit me", parents=["s1"])
def escape_gain(t: Tile) -> Num:
    return sum(dist(u.pos, t) - dist(u.pos, me().pos) for u in camps() if is_awake(u) and dist(u.pos, me().pos) <= u.attack_range + 1) + sum(dist(u.pos, t) - dist(u.pos, me().pos) for u in enemies() if dist(u.pos, me().pos) <= u.attack_range + 2)
