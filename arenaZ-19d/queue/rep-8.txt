```python
@gene("s1", "value of doing this option now: cast abilities, recall at a quarter health")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.25 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "value of casting an ability: damage and stuns on fighting targets, shields and resets in fights", parents=["s1"])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit) and (o.unit.is_champion or (o.unit.is_camp and gems() == 0)):
            return 85
        return 0
    if k == AB_AOE:
        hits = count(dist(u.pos, o.tile) <= 1 for u in enemies()) + count(dist(u.pos, o.tile) <= 1 and is_awake(u) for u in camps())
        return 84 if hits > 0 else 0
    if k == AB_SHIELD:
        threat = count(dist(u.pos, me().pos) <= 4 for u in enemies()) + count(is_awake(u) and dist(u.pos, me().pos) <= 4 for u in camps())
        return 83 if threat > 0 and me().health_frac < 0.9 else 0
    if k == AB_RESET:
        return 82 if (not can_attack()) and count(dist(u.pos, me().pos) <= me().attack_range for u in enemies()) + count(is_awake(u) and dist(u.pos, me().pos) <= me().attack_range for u in camps()) > 0 else 0
    if k == AB_HEAL:
        return 86 if exists(o.unit) and o.unit.health_frac < 0.6 else 0
    return 0
```
