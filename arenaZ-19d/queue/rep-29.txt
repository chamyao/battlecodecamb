@gene("s2", "where to walk: an altar with gems, else a living camp group other than monoliths (golems only after round 1200)")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    if round() > 1200:
        return acc - type_spawns(MONOLITH)
    return acc - type_spawns(MONOLITH) - type_spawns(GOLEM)
