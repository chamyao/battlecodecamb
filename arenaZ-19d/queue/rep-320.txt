@gene('Z51a', 'how many awake camps and enemy champions could hit a tile, enemies only when actually in reach', parents=['s4'])
def threat_at(t: Tile) -> Num:
    return count(is_awake(u) and dist(u.pos, t) <= u.attack_range for u in camps()) + count(dist(u.pos, t) <= u.attack_range for u in enemies())
