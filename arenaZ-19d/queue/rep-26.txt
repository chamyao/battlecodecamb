@gene('s1', 'value of doing this option now: draft a ranged team, cast abilities, farm camps until 10 gems, recall late')
def score(o: Option) -> Num:
    if o.kind == PICK:
        if o.champ == VESPER_VALE:
            return 10
        if o.champ == STELLAR:
            return 9
        if o.champ == TAVI:
            return 8
        if o.champ == CASSI:
            return 7
        if o.champ == WILLOW:
            return 6
        return 1
    if o.kind == BUY:
        return 100 + item_strength(o.item) if item_cost(o.item) >= 30 and item_strength(o.item) > 0 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.25 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() < 10 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z6a', 'value of casting an ability: damage and stuns on fighting targets, shields and resets in fights', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit) and (o.unit.is_champion or (o.unit.is_camp and gems() < 10)):
            return 85
        return 0
    if k == AB_AOE:
        hits = count(dist(u.pos, o.tile) <= 1 for u in enemies()) + count(dist(u.pos, o.tile) <= 1 and is_awake(u) for u in camps())
        return 84 if hits > 0 else 0
    if k == AB_SHIELD:
        threat = count(dist(u.pos, me().pos) <= 4 for u in enemies()) + count(is_awake(u) and dist(u.pos, me().pos) <= 4 for u in camps())
        return 83 if threat > 0 and me().health_frac < 0.9 else 0
    if k == AB_RESET:
        return 82 if (not can_attack()) and count(dist(u.pos, me().pos) <= me().attack_range for u in enemies()) + count(is_awake(u) and dist(u.pos, me().pos) <= me().attack_range for u in camps()) > 0 else 0
    if k == AB_HEAL:
        return 86 if exists(o.unit) and o.unit.health_frac < 0.6 else 0
    return 0

@gene("new", "which option kinds to list at all: skip wards, locks and cancels to save fuel", parents=["s1"])
def consider(kind: Num) -> Bool:
    return kind != WARD and kind != LOCK and kind != BUY_LOCK and kind != CANCEL
