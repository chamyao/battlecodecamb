@gene('s3', 'distance field to the goals, skipping golem and monolith camps; home when hurt and empty-handed')
def goal_field() -> Field:
    if me().health_frac < 0.4 and gems() == 0:
        return field(my_spawn())
    return field(goals() - type_spawns(MONOLITH) - type_spawns(GOLEM))

@gene('s2', 'where to walk: an altar with gems, else a living camp group that carries gems, gem clusters first when close, narrow gem margin, finish fights before banking a small bag')
def goals() -> Tiles:
    if gems() > 0 and not (gems() < 7 and count(is_awake(u) for u in camps()) > 0):
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    rest = acc - type_spawns(GOLD_CLUSTER)
    gem = acc & type_spawns(GEM_CLUSTER)
    if size(gem) > 0:
        if dist(nearest(gem, me().pos), me().pos) <= dist(nearest(rest, me().pos), me().pos) + 4:
            return gem
    return rest
