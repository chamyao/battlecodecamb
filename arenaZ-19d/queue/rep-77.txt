@gene('s2', 'where to walk: an altar with gems, else a living camp group that carries gems, gem clusters first when close')
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    rest = acc - type_spawns(GOLD_CLUSTER)
    gem = acc & type_spawns(GEM_CLUSTER)
    if size(gem) > 0:
        if dist(nearest(gem, me().pos), me().pos) <= dist(nearest(rest, me().pos), me().pos) + 8:
            return gem
    return rest
