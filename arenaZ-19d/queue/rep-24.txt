```python
@gene('s1', 'value of doing this option now: draft a ranged team, cast abilities, farm camps until 10 gems, recall late, strength shards')
def score(o: Option) -> Num:
    if o.kind == PICK:
        if o.champ == VESPER_VALE:
            return 10
        if o.champ == TAVI:
            return 9
        if o.champ == STELLAR:
            return 8
        if o.champ == CASSI:
            return 7
        if o.champ == WILLOW:
            return 6
        return 1
    if o.kind == BUY:
        return 100 + item_strength(o.item) if item_cost(o.item) >= 30 and item_strength(o.item) > 0 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.25 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() < 10 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('s2', 'where to walk: a free altar once the bag holds 10 gems, else a living camp group other than golems and monoliths')
def goals() -> Tiles:
    if gems() >= 10 or (gems() > 0 and round() > 1900):
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc - type_spawns(MONOLITH) - type_spawns(GOLEM)

@gene("s3", "distance field to the goals")
def goal_field() -> Field:
    return field(goals())

@gene("new", "which option kinds to list at all: skip wards, locks and cancels to save fuel", parents=["s1"])
def consider(kind: Num) -> Bool:
    return kind != WARD and kind != LOCK and kind != BUY_LOCK and kind != CANCEL
```
