```python
@gene('s2', 'where to walk: a free altar once the bag holds 10 gems, else a living camp group other than golems and monoliths, gem clusters first when close, wider gem margin')
def goals() -> Tiles:
    if gems() >= 10 or (gems() > 0 and round() > 1900):
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    rest = acc - type_spawns(MONOLITH) - type_spawns(GOLEM)
    gem = acc & type_spawns(GEM_CLUSTER)
    if size(gem) > 0:
        if dist(nearest(gem, me().pos), me().pos) <= dist(nearest(rest, me().pos), me().pos) + 14:
            return gem
    return rest

@gene("s4", "how much closer a step brings me to the goals; when hurt under fire, how much safer it is")
def progress(t: Tile) -> Num:
    if me().health_frac < 0.3 and threat_at(me().pos) > 0:
        return threat_at(me().pos) - threat_at(t)
    return at(goal_field(), me().pos) - at(goal_field(), t)
```
