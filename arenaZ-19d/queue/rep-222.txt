@gene('s1', 'value of doing this option now: draft a ranged team, cast abilities, recall at a quarter health, recall when safe, retreat from threats when hurt, finish awake camps, wake new ones only when healthy, earlier recall and retreat, chase weak enemy champions, focus hurt and gem-carrying champions, later recall and retreat')
def score(o: Option) -> Num:
    if o.kind == PICK:
        if o.champ == VESPER_VALE:
            return 12
        if o.champ == STELLAR:
            return 9
        if o.champ == TAVI:
            return 8
        if o.champ == CASSI:
            return 9
        if o.champ == WILLOW:
            return 6
        return 1
    if o.kind == BUY:
        return 100 + item_strength(o.item) if item_cost(o.item) >= 30 and item_strength(o.item) > 0 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.25 and threat_at(me().pos) == 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac) + (3 if unit_gems(o.unit) > 0 else 0)
        return 60 if (gems() == 0 or is_awake(o.unit)) and (is_awake(o.unit) or me().health_frac > 0.3345) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if outnumbered() and dist(nearest(enemy_tiles(), o.tile), o.tile) > dist(nearest(enemy_tiles(), me().pos), me().pos):
            return 90
        if me().health_frac < 0.3 and threat_at(me().pos) > 0:
            return 95 if threat_at(o.tile) < threat_at(me().pos) else 0
        if chase_gain(o.tile) > 0:
            return 55
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "more visible enemy champions near me than allies, and I am not healthy", parents=["s1"])
def outnumbered() -> Bool:
    ne = count(dist(u.pos, me().pos) <= 5 for u in enemies())
    na = count(dist(u.pos, me().pos) <= 5 for u in allies())
    return ne > na + 1 and me().health_frac < 0.7

@gene('Z10a', 'value of casting an ability: damage and stuns on fighting targets, shields and resets in fights, finish awake camps, heal earlier, dash toward goals or away from danger', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit) and o.unit.is_champion:
            return 88
        if exists(o.unit) and o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 85 + 3 * (1 - o.unit.health_frac)
        return 0
    if k == AB_AOE:
        hits = count(dist(u.pos, o.tile) <= 1 for u in enemies()) + count(dist(u.pos, o.tile) <= 0.9637 and is_awake(u) for u in camps())
        return 84 if hits > 0 else 0
    if k == AB_SHIELD:
        threat = count(dist(u.pos, me().pos) <= 4 for u in enemies()) + count(is_awake(u) and dist(u.pos, me().pos) <= 4 for u in camps())
        return 83 if threat > 0 and me().health_frac < 0.9 else 0
    if k == AB_RESET:
        return 82 if (not can_attack()) and count(dist(u.pos, me().pos) <= me().attack_range for u in enemies()) + count(is_awake(u) and dist(u.pos, me().pos) <= me().attack_range for u in camps()) > 0 else 0
    if k == AB_DASH:
        if me().health_frac < 0.3 and threat_at(me().pos) > 0:
            return 105 if threat_at(o.tile) < threat_at(me().pos) else 0
        return 52 if progress(o.tile) >= 2 else 0
    if k == AB_HEAL:
        return 86 if exists(o.unit) and o.unit.health_frac < 0.75 else 0
    return 0
