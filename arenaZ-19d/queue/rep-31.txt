@gene('s2', 'where to walk: a free altar once the bag holds 10 gems, else a living camp group carrying gems, other than golems and monoliths')
def goals() -> Tiles:
    if gems() >= 10 or (gems() > 0 and round() > 1900):
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc - type_spawns(MONOLITH) - type_spawns(GOLEM) - type_spawns(GOLD_CLUSTER)
