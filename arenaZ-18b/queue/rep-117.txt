```python
@gene('s1', 'value of doing this option now: cast spells; with a big bag, skip healthy enemy champions and keep walking to bank; recall when out of reach; walk home when hurt with enemies near')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 + o.count if o.item == 40 else 0
    if o.kind == RECALL:
        return 54 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return spell_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            if gems() >= 20 and o.unit.health_frac > 0.5:
                return 0
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.35 and count_units(enemies()) > 0:
            return 95 if progress(o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("s4", "how much closer a step brings me to the goals, or to a nearby enemy champion carrying many gems when I am healthy")
def progress(t: Tile) -> Num:
    if me().health_frac > 0.6:
        prey = best(enemies(), lambda u: unit_gems(u) - 2 * dist(u.pos, me().pos) if u.is_champion else -99)
        if exists(prey) and prey.is_champion and unit_gems(prey) >= 10 and dist(prey.pos, me().pos) <= 6:
            return dist(prey.pos, me().pos) - dist(prey.pos, t)
    f = goal_field()
    return at(f, me().pos) - at(f, t)
```
