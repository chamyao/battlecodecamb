```python

@gene("new", "which option kinds to list each pass: skip wards, locks and cancels, and spells unless a fight or camp is close or I am hurt", parents=["s1"])
def consider(kind: Num) -> Bool:
    if kind == WARD or kind == LOCK or kind == BUY_LOCK or kind == CANCEL:
        return False
    if kind == CAST:
        if me().health_frac < 0.77:
            return True
        return any(dist(u.pos, me().pos) <= 4 for u in enemies()) or any(dist(c.pos, me().pos) <= 3 for c in camps())
    return True
```
