```python
GF: Field = None

@gene("s1", "value of doing this option now: cast spells and dash toward the goal, recall only when nothing is in reach")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 54 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_DASH:
            return 55 if valid(o.tile) and progress(o.tile) >= 2 else 0
        return spell_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("s2", "where to walk: home when hurt, an altar with gems, else a living camp group that is not a monolith or golem")
def goals() -> Tiles:
    if me().health_frac < 0.35:
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            k = spawn_type(nearest(ts, me().pos))
            if k != MONOLITH and k != GOLEM:
                acc = acc | ts
    return acc

@gene("s3", "distance field to the goals (cached per turn)")
def goal_field() -> Field:
    return GF

@gene("new", "compute the goal distance field once per turn", parents=["s3"])
def update() -> None:
    GF = field(goals())
```
