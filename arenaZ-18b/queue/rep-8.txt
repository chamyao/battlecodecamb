```python
@gene("s1", "value of doing this option now: walking home when hurt beats a recall that damage would cancel")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 and count_units(enemies()) == 0 else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.35 and count_units(enemies()) > 0:
            return 95 if retreat(o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "how much closer a step brings me to my spawn", parents=["s4"])
def retreat(t: Tile) -> Num:
    f = field(my_spawn())
    return at(f, me().pos) - at(f, t)
```
