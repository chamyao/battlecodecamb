@gene("s2", "where to walk: home when hurt, an altar with gems, a cleared camp waiting for capture, else a living camp group that is not a monolith or golem")
def goals() -> Tiles:
    if me().health_frac < 0.35:
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    loot = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g):
                if dist(nearest(ts, me().pos), me().pos) <= 12:
                    loot = loot | ts
            k = spawn_type(nearest(ts, me().pos))
            if k != MONOLITH and k != GOLEM:
                acc = acc | ts
    if size(loot) > 0:
        return loot
    return acc
