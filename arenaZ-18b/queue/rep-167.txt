@gene('s1', 'value of doing this option now: draft Vesper Vale first, cast spells, focus fire the weakest camp, recall when out of reach, walk home when hurt with enemies near')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return (10 if o.champ == VESPER_VALE else (9 if o.champ == STELLAR else (8 if o.champ == PRISM else (7 if o.champ == REN else (6 if o.champ == TAVI else 1)))))
    if o.kind == BUY:
        return 100 + o.count if o.item == 40 else 0
    if o.kind == RECALL:
        return 54 if me().health_frac < 0.35 or (gold() >= 150 and count_units(enemies()) == 0 and gems() == 0) else 0
    if o.kind == CAST:
        return spell_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + min(unit_gems(o.unit), 50) / 5
        return 60 + 5 * (1 - o.unit.health_frac) if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.35 and count_units(enemies()) > 0:
            return 95 if progress(o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("Z2a", "set the goal field from cached fields: the static home field built once, the altar and camp fields every 8 rounds")
def update() -> None:
    if not INIT:
        HF = field(my_spawn())
        INIT = True
    if me().health_frac < 0.35:
        GF = HF
    elif gems() > 0:
        if round() - AT >= 8:
            AF = field(goals())
            AT = round()
        GF = AF
    else:
        if round() - CT >= 8:
            CF = field(goals())
            CT = round()
        GF = CF

AF: Field = None
HF: Field = None
CF: Field = None
INIT: Bool = False
CT: Num = -99
AT: Num = -99
