```python
@gene("s1", "value of doing this option now: meta draft, strength shards, cast damage spells, fight, bank, walk")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 + o.count if o.item == 10 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "draft preference: the top ladder picks first", parents=["s1"])
def pick_value(c: Num) -> Num:
    if c == VESPER_VALE:
        return 10
    if c == HEART_OF_RUIN:
        return 9
    if c == MARCELLA:
        return 8
    if c == VENDETTA:
        return 7
    if c == RYTH:
        return 6
    return 1

@gene("new", "value of casting an ability: damage and aoe on enemies or camps when farming", parents=["s1"])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit) and o.unit.is_champion and o.unit.is_enemy:
            return 85
        if exists(o.unit) and o.unit.is_camp and gems() == 0:
            return 65
        return 0
    if k == AB_SHIELD or k == AB_HEAL:
        return 75 if me().health_frac < 0.6 and count_units(enemies()) > 0 else 0
    return 0
```
