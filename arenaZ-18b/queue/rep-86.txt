@gene("Z58b", "value of a spell: damage, aoe and cc on enemy champions or adjacent camps, ground aoe where it hits several targets, shield and heal early, dash toward the goal")
def spell_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if not exists(o.unit):
            if not valid(o.tile):
                return 0
            r = slot_radius(o.slot)
            hits = count(dist(u.pos, o.tile) <= r for u in enemies() if u.is_champion) * 3 + count(dist(c.pos, o.tile) <= r for c in camps())
            if hits == 0:
                return 0
            return 60 + 5 * hits if gems() == 0 or hits >= 3 else 0
        if o.unit.is_champion and o.unit.is_enemy:
            return 85
        if o.unit.is_camp and gems() == 0:
            return 65
        return 0
    if k == AB_SHIELD or k == AB_HEAL:
        return 88 if me().health_frac < 0.77 else 0
    if k == AB_DASH:
        if valid(o.tile) and progress(o.tile) >= 2:
            return 55
        return 0
    return 0

@gene("s2", "where to walk: home when hurt, an altar with gems, an ally fight announced nearby, else a living camp group worth at least 10 gems that is not a monolith or golem")
def goals() -> Tiles:
    if me().health_frac < 0.35:
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    b = beacon()
    if valid(b) and dist(b, me().pos) <= 12:
        return near(b, 1)
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g) and group_gems(g) >= 10:
            ts = group_tiles(g)
            k = spawn_type(nearest(ts, me().pos))
            if k != MONOLITH and k != GOLEM:
                acc = acc | ts
    return acc
