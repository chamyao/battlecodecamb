```python
GF: Field = None
HF: Field = None
HFOK: Bool = False

@gene("s1", "value of doing this option now: cast spells; walking home when hurt beats a recall that damage would cancel")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 and count_units(enemies()) == 0 else 0
    if o.kind == CAST:
        return spell_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.35 and count_units(enemies()) > 0:
            return 95 if retreat(o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("s2", "where to walk: an altar with gems, else a living camp group")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc

@gene("s3", "distance field to the goals (cached per turn)")
def goal_field() -> Field:
    return GF

@gene("s4", "how much closer a step brings me to the goals, minus danger from strong enemies when carrying gems")
def progress(t: Tile) -> Num:
    gain = at(goal_field(), me().pos) - at(goal_field(), t)
    if gems() == 0:
        return gain
    threat = count(dist(u.pos, t) <= 2 for u in enemies() if u.is_champion)
    return gain - 0.6 * threat

@gene("Z10a", "how much closer a step brings me to my spawn (field cached on first use)", parents=["s4"])
def retreat(t: Tile) -> Num:
    return at(HF, me().pos) - at(HF, t)

@gene("Z8a", "compute the goal distance field once per turn, and the home field once per game", parents=["s3"])
def update() -> None:
    GF = field(goals())
    if not HFOK:
        HF = field(my_spawn())
        HFOK = True

@gene("new", "value of a spell: damage, aoe and cc on enemy champions or camps, shield and heal when hurt", parents=["s1"])
def spell_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if not exists(o.unit):
            return 0
        if o.unit.is_champion and o.unit.is_enemy:
            return 85
        if o.unit.is_camp and gems() == 0:
            return 65
        return 0
    if k == AB_SHIELD or k == AB_HEAL:
        return 88 if me().health_frac < 0.5 else 0
    return 0
```
