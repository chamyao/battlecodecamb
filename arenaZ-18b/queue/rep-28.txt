@gene("s2", "where to walk: home when hurt, an altar with gems, else a living camp group that is not a monolith or golem and not contested by enemy champions")
def goals() -> Tiles:
    if me().health_frac < 0.35:
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    spare = near(tile(-5, -5), 0)
    foes = enemy_tiles()
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            k = spawn_type(nearest(ts, me().pos))
            if k != MONOLITH and k != GOLEM:
                spare = spare | ts
                if size(foes) == 0 or dist(nearest(foes, nearest(ts, me().pos)), nearest(ts, me().pos)) > 4:
                    acc = acc | ts
    if size(acc) == 0:
        return spare
    return acc
