@gene("Z300d", "set the goal field from cached fields: the static home field built once, the altar field every 8 rounds and the camp field every 5 or once reached")
def update() -> None:
    if not INIT:
        HF = field(my_spawn())
        INIT = True
    if me().health_frac < 0.35:
        GF = HF
    elif gems() > 0:
        if round() - AT >= 8:
            AF = field(goals())
            AT = round()
        GF = AF
    else:
        if round() - CT >= 5 or at(CF, me().pos) <= 0:
            CF = field(goals())
            CT = round()
        GF = CF

@gene("s2", "where to walk: home when hurt, an altar with gems (one without enemy champions next to it if possible), a nearby enemy champion carrying many gems when healthy, else a living camp group worth at least 10 gems (any group if none) that is not a monolith or golem")
def goals() -> Tiles:
    if me().health_frac < 0.35:
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        safe = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
            if not any(u.is_champion and dist(u.pos, a) <= 3 for u in enemies()):
                safe = safe | near(a, 1)
        if size(safe) > 0:
            return safe
        return acc
    if me().health_frac > 0.6:
        prey = best(enemies(), lambda u: unit_gems(u) - 2 * dist(u.pos, me().pos) if u.is_champion else -99)
        if exists(prey) and prey.is_champion and unit_gems(prey) >= 10 and dist(prey.pos, me().pos) <= 7:
            return near(prey.pos, 1)
    acc = near(tile(-5, -5), 0)
    spare = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            k = spawn_type(nearest(ts, me().pos))
            if k != MONOLITH and k != GOLEM:
                spare = spare | ts
                if group_gems(g) >= 10:
                    acc = acc | ts
    if size(acc) == 0:
        return spare
    return acc

@gene('Z300c', 'value of a spell: attack resets once the basic attack is spent, damage, aoe and cc on enemy champions or adjacent camps, ground aoe where it hits several targets, shield and heal early, dash toward the goal', parents=['s1'])
def spell_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if not exists(o.unit):
            if not valid(o.tile):
                return 0
            r = slot_radius(o.slot)
            hits = count(dist(u.pos, o.tile) <= r for u in enemies() if u.is_champion) * 3 + count(dist(c.pos, o.tile) <= r for c in camps())
            if hits == 0:
                return 0
            return 60 + 5 * hits if gems() == 0 or hits >= 3 else 0
        if o.unit.is_champion and o.unit.is_enemy:
            return 85
        if o.unit.is_camp and gems() == 0:
            return 65
        return 0
    if k == AB_RESET:
        if can_attack():
            return 0
        reach = me().attack_range
        if any(dist(u.pos, me().pos) <= reach for u in enemies() if u.is_champion):
            return 90
        if gems() == 0 and any(dist(c.pos, me().pos) <= reach for c in camps()):
            return 62
        return 0
    if k == AB_SHIELD or k == AB_HEAL:
        return 88 if me().health_frac < 0.77 else 0
    if k == AB_DASH:
        if valid(o.tile) and progress(o.tile) >= 2:
            return 55
        return 0
    return 0

@gene("s4", "how much closer a step brings me to the goals, or to an enemy champion within 7 tiles carrying 6 or more gems when above half health")
def progress(t: Tile) -> Num:
    if me().health_frac > 0.5:
        prey = best(enemies(), lambda u: unit_gems(u) - 2 * dist(u.pos, me().pos) if u.is_champion else -99)
        if exists(prey) and prey.is_champion and unit_gems(prey) >= 6 and dist(prey.pos, me().pos) <= 7:
            return dist(prey.pos, me().pos) - dist(prey.pos, t)
    f = goal_field()
    return at(f, me().pos) - at(f, t)
