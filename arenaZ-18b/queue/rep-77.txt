```python
@gene('s2', 'where to walk: home when hurt, an altar with gems, else a living camp group worth at least 10 gems that is not a monolith or golem')
def goals() -> Tiles:
    if me().health_frac < 0.35:
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            k = spawn_type(nearest(ts, me().pos))
            if k != MONOLITH and k != GOLEM and group_gems(g) >= 10:
                acc = acc | ts
    return acc
```
