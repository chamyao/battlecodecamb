```python
@gene("s1", "value of doing this option now: cast spells, focus weak champions, leave camps when hurt, walk home when hurt with enemies near")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 and count_units(enemies()) == 0 else 0
    if o.kind == CAST:
        return spell_value2(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        return 60 if gems() == 0 and me().health_frac >= 0.3 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.3 and count_units(enemies()) > 0:
            return 95 if retreat(o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
