```python
@gene('s1', 'value of doing this option now: use attack resets, draft Vesper Vale first, cast spells, focus fire the weakest camp, recall when out of reach, walk home when hurt with enemies near')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return (10 if o.champ == VESPER_VALE else (9 if o.champ == STELLAR else (8 if o.champ == PRISM else (7 if o.champ == REN else (6 if o.champ == TAVI else 1)))))
    if o.kind == BUY:
        return 100 + o.count if o.item == 40 else 0
    if o.kind == RECALL:
        return 54 if me().health_frac < 0.35 or (gold() >= 150 and count_units(enemies()) == 0 and gems() == 0) else 0
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_RESET:
            return reset_value()
        return spell_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + min(unit_gems(o.unit), 50) / 5
        return 60 + 5 * (1 - o.unit.health_frac) if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.35 and count_units(enemies()) > 0:
            return 95 if progress(o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "value of an attack reset: worth it once my basic attack is spent and an enemy champion or (when farming) a camp is still in reach", parents=["s1"])
def reset_value() -> Num:
    if can_attack():
        return 0
    reach = me().attack_range
    if any(dist(u.pos, me().pos) <= reach for u in enemies() if u.is_champion):
        return 90
    if gems() == 0 and any(dist(c.pos, me().pos) <= reach for c in camps()):
        return 62
    return 0

@gene("s2", "where to walk: home when hurt, an altar with gems, a nearby enemy champion carrying many gems when healthy, else a living camp group worth at least 10 gems (any group if none) that is not a monolith or golem")
def goals() -> Tiles:
    if me().health_frac < 0.35:
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    if me().health_frac > 0.6:
        prey = best(enemies(), lambda u: unit_gems(u) - 2 * dist(u.pos, me().pos) if u.is_champion else -99)
        if exists(prey) and prey.is_champion and unit_gems(prey) >= 15 and dist(prey.pos, me().pos) <= 7:
            return near(prey.pos, 1)
    acc = near(tile(-5, -5), 0)
    spare = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            k = spawn_type(nearest(ts, me().pos))
            if k != MONOLITH and k != GOLEM:
                spare = spare | ts
                if group_gems(g) >= 10:
                    acc = acc | ts
    if size(acc) == 0:
        return spare
    return acc
```
