@gene("s2", "where to walk: bank at an altar once the bag is worth it (or when hurt or late), else a living camp group")
def goals() -> Tiles:
    if gems() >= 15 or (gems() > 0 and (me().health_frac < 0.5 or round() > 1850)):
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    if size(acc) == 0 and gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
    return acc
