```python
@gene("s4", "how much closer a step brings me to the goals; carrying gems, steps next to enemy champions count less")
def progress(t: Tile) -> Num:
    d = at(goal_field(), me().pos) - at(goal_field(), t)
    if gems() > 0 and d > 0:
        if any(dist(e.pos, t) <= 2 for e in enemies() if e.is_champion):
            return d * 0.5
    return d
```
