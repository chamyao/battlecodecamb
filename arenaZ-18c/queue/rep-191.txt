@gene("Z125a", "where to walk: an altar with gems, else a living camp group, a nearby enemy champion carrying 10+ gems, or a nearly dead enemy champion when I am healthy; variant: hunt within 8 tiles for carriers of 6+ gems")
def goals_kw() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for e in enemies():
        if e.is_champion and e.health_frac < 0.3 and dist(e.pos, me().pos) <= 5 and me().health_frac > 0.6:
            acc = acc | near(e.pos, 0)
    for e in enemies():
        if e.is_champion and unit_gems(e) >= 6 and dist(e.pos, me().pos) <= 8:
            acc = acc | near(e.pos, 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
