```python
@gene("new", "where to walk: home to heal below 40 percent health when empty-handed, an altar with gems, else a living camp group", parents=["s2"])
def goals_low_heal() -> Tiles:
    if gems() == 0 and me().health_frac < 0.4:
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc

@gene("s3", "distance field to the goals, healing only below 40 percent health")
def goal_field() -> Field:
    return field(goals_low_heal())
```
