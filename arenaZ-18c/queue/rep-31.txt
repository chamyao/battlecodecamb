@gene("s1", "value of doing this option now; casts on targets, dashes toward goals, recall only when nearly dead, treasure then bruiser shards")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if items_owned(19) == 0:
            return 100 + o.count if o.item == 18 else 0
        return 100 + o.count if o.item == 36 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.15 else 0
    if o.kind == CAST:
        k = slot_kind(o.slot)
        if exists(o.unit) and o.unit.is_enemy:
            if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
                return 78 if o.unit.is_champion else 62
        if k == AB_DASH:
            return 40 if progress(o.tile) > 1 else 0
        return 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("s4", "how much closer a step brings me to the goals, or to my spawn when hurt and carrying no gems")
def progress(t: Tile) -> Num:
    if gems() == 0 and me().health_frac < 0.5:
        return at(home_field(), me().pos) - at(home_field(), t)
    return at(goal_field(), me().pos) - at(goal_field(), t)

@gene("new", "distance field to my spawn (healing and shopping)", parents=["s4"])
def home_field() -> Field:
    return field(my_spawn())
