```python
@gene("Z91a", "value of doing this option now: casts on targets, dashes toward goals, drafts Ren Tavi Marcella, never recalls, treasure then bruiser health shards; tuned", parents=['s1'])
def score(o: Option) -> Num:
    if o.kind == PICK:
        if o.champ == REN:
            return 10
        if o.champ == TAVI:
            return 9
        if o.champ == MARCELLA:
            return 8
        return 1
    if o.kind == BUY:
        if items_owned(19) == 0:
            return 100 + o.count if o.item == 18 else 0
        return 100 + o.count if o.item == 40 else 0
    if o.kind == RECALL:
        return 0
    if o.kind == CAST:
        k = slot_kind(o.slot)
        if exists(o.unit) and o.unit.is_enemy:
            if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
                return 78 if o.unit.is_champion else 62
        if k == AB_RESET:
            if can_attack():
                return 0
            if any(dist(u.pos, me().pos) <= me().attack_range for u in enemies()):
                return 75
            return 75 if gems() < 5 and any(dist(u.pos, me().pos) <= me().attack_range for u in camps()) else 0
        if k == AB_DASH:
            return 40 if progress(o.tile) > 1 else 0
        return 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
