@gene("s3", "distance field to the goals, or to my spawn to heal when hurt and carrying no gems; tuned: never recall, bruiser health, heal below 40 percent; variant: flee home when nearly dead with gems and an enemy champion close")
def goal_field() -> Field:
    if gems() == 0 and me().health_frac < 0.4:
        return field(my_spawn())
    if gems() > 0 and me().health_frac < 0.25 and any(e.is_champion and dist(e.pos, me().pos) <= 4 for e in enemies()):
        return field(my_spawn())
    return field(goals())
