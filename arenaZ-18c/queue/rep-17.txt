@gene("s1", "value of doing this option now; casts on targets, dashes toward goals, recall at a quarter health")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 + o.count if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.25 else 0
    if o.kind == CAST:
        k = slot_kind(o.slot)
        if exists(o.unit) and o.unit.is_enemy:
            if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
                return 80 if o.unit.is_champion else 64
        if k == AB_DASH:
            return 40 if progress(o.tile) > 1 else 0
        return 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("s2", "where to walk: an altar as soon as I carry gems, else a living camp group with gems")
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    rich = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
            if group_gems(g) > 0:
                rich = rich | group_tiles(g)
    if size(rich) > 0:
        return rich
    return acc
