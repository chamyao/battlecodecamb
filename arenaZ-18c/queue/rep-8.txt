```python
@gene("s1", "value of doing this option now; focus weak targets, tribute before fighting")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == TRIBUTE:
        return 88
    if o.kind == ATTACK:
        if o.unit.is_champion:
            if me().health_frac < 0.3 and o.unit.health_frac > 0.5:
                return 0
            return 75 + 10 * (1 - o.unit.health_frac)
        return 55 + 5 * (1 - o.unit.health_frac) if gems() < 15 else 0
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
