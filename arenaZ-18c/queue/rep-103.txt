@gene("s1", "value of doing this option now: cast abilities, dash toward goals, never recall, treasure then bruiser health shards; tuned")
def score(o: Option) -> Num:
    if o.kind == PICK:
        if o.champ == REN:
            return 10
        if o.champ == TAVI:
            return 9
        if o.champ == MARCELLA:
            return 8
        return 1
    if o.kind == BUY:
        if items_owned(19) == 0:
            return 100 + o.count if o.item == 18 else 0
        return 100 + o.count if o.item == 40 else 0
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_DASH:
            return 40 if progress(o.tile) > 1 else 0
        k = slot_kind(o.slot)
        if k == AB_RESET:
            if can_attack():
                return 0
            if any(dist(u.pos, me().pos) <= me().attack_range for u in enemies()):
                return 75
            return 75 if gems() < 5 and any(dist(u.pos, me().pos) <= me().attack_range for u in camps()) else 0
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() < 5 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("Z4a", "draft: take the first champion offered (the preference list drafted worse teams); tuned", parents=['s1'])
def pick_value(c: Num) -> Num:
    if c == REN:
        return 10
    if c == TAVI:
        return 9
    if c == MARCELLA:
        return 8
    return 1
