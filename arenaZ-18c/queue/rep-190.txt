```python
@gene("Z100a", "value of doing this option now: casts on targets, dashes toward goals, drafts Ren Tavi Marcella, never recalls, treasure then bruiser health shards; tuned; variant: dash whenever it gets me closer")
def score(o: Option) -> Num:
    if o.kind == PICK:
        if o.champ == REN:
            return 10
        if o.champ == TAVI:
            return 9
        if o.champ == MARCELLA:
            return 8
        return 1
    if o.kind == BUY:
        if items_owned(19) == 0:
            return 100 + o.count if o.item == 18 else 0
        return 100 + o.count if o.item == 40 else 0
    if o.kind == RECALL:
        return 0
    if o.kind == CAST:
        k = slot_kind(o.slot)
        if exists(o.unit) and o.unit.is_enemy:
            if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
                return 78 if o.unit.is_champion else 62
        if k == AB_RESET:
            return 30
        if k == AB_DASH:
            return 40 if progress(o.tile) > 0 else 0
        return 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 + 5 * (1 - o.unit.health_frac) if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
