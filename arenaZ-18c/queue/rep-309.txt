@gene("s1", "value of doing this option now: cast abilities, dash toward goals, never recall (walk home to heal instead), treasure then bruiser shards; tuned; variant: bruiser strength emblem before health shards")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        if items_owned(19) == 0:
            return 100 + o.count if o.item == 18 else 0
        if items_owned(37) == 0:
            return 100 + o.count if o.item == 40 else 0
        return 100 + o.count if o.item == 40 else 0
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_DASH:
            return 40 if progress(o.tile) > 0 else 0
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 71
        return 60 + 5 * (1 - o.unit.health_frac) if gems() < 5 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("Z96a", "where to walk: home below 40 percent health when empty-handed, an altar with gems, else a living camp group or a nearby enemy champion carrying 10+ gems; variant: heal below 36 percent; tuned", parents=['s3', 's4'])
def goals_v2() -> Tiles:
    if gems() == 0 and me().health_frac < 0.4:
        return my_spawn()
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for e in enemies():
        if e.is_champion and unit_gems(e) >= 10 and dist(e.pos, me().pos) <= 6:
            acc = acc | near(e.pos, 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
