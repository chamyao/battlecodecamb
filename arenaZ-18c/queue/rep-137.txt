```python
@gene("Z100b", "where to walk: an altar with gems, else a living camp group or an enemy champion within 7 carrying 8+ gems")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for e in enemies():
        if e.is_champion and unit_gems(e) >= 8 and dist(e.pos, me().pos) <= 7:
            acc = acc | near(e.pos, 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
```
