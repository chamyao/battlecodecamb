@gene("Z86a", "where to walk: home below 40 percent health when empty-handed, an altar with gems, else a living camp group or an enemy champion within 8 carrying 6+ gems", parents=["s2"])
def goals_v2() -> Tiles:
    if gems() == 0 and me().health_frac < 0.4:
        return my_spawn()
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for e in enemies():
        if e.is_champion and unit_gems(e) >= 6 and dist(e.pos, me().pos) <= 8:
            acc = acc | near(e.pos, 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
