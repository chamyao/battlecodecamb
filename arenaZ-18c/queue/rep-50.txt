@gene("new", "where to walk: home to heal below 40 percent health when carrying few gems, an altar with gems, else a living camp group", parents=["s2"])
def goals_h40() -> Tiles:
    if gems() < 5 and me().health_frac < 0.4:
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc

@gene("Z26a", "compute the goal distance field once per turn, healing below 40 percent health", parents=["s3"])
def update() -> None:
    GOALF = field(goals_h40())
