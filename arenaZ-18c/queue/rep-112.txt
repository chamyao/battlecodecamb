```python
@gene("new", "where to walk: an altar with gems, else a living camp group or an enemy champion within 8 carrying 6+ gems", parents=["Z100b"])
def goals_h68() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for e in enemies():
        if e.is_champion and unit_gems(e) >= 6 and dist(e.pos, me().pos) <= 8:
            acc = acc | near(e.pos, 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc

@gene("s3", "distance field to the goals, hunting gem carriers within 8 tiles")
def goal_field() -> Field:
    return field(goals_h68())
```
