@gene("s2", "where to walk: an altar with 6+ gems, else a living camp group with gems or gold")
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    rich = near(tile(-5, -5), 0)
    n = 0
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
            n += 1
            if group_gems(g) > 0:
                rich = rich | group_tiles(g)
    if gems() >= 6 or (gems() > 0 and n == 0):
        alt = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            alt = alt | near(a, 1)
        return alt
    if size(rich) > 0:
        return rich
    return acc
