```python
@gene("Z136c", "how much closer a step brings me to the goals, or to my spawn below 40 percent health when carrying no gems; variant: heal below 36 percent")
def progress(t: Tile) -> Num:
    if gems() == 0 and me().health_frac < 0.36:
        return at(home_field(), me().pos) - at(home_field(), t)
    return at(goal_field(), me().pos) - at(goal_field(), t)
```
