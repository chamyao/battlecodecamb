```python
@gene("s3", "distance field to the goals, or to my spawn to heal when hurt and carrying no gems; tuned: never recall, bruiser health, heal below 40 percent")
def goal_field() -> Field:
    if gems() == 0 and me().health_frac < 0.4:
        return field(my_spawn())
    return field(goals())

@gene("s1", "value of doing this option now: cast abilities, dash toward goals, never recall (walk home to heal instead), treasure then bruiser shards; tuned: never recall, bruiser health, heal below 40 percent")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if items_owned(19) == 0:
            return 100 + o.count if o.item == 18 else 0
        return 100 + o.count if o.item == 40 else 0
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_DASH:
            return 40 if progress(o.tile) > 1 else 0
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() < 5 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
