```python
@gene("s1", "value of doing this option now: no draft preference, cast abilities, recall at a quarter health")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 + o.count if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.25 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("Z4b", "value of casting an ability: damage and control on enemies, shield when hurt, dash toward goals", parents=["s1"])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_HEAL or k == AB_SHIELD:
        return 85 if me().health_frac < 0.6 else 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit) and o.unit.is_enemy:
            return 82 if o.unit.is_champion else 65
    if k == AB_DASH:
        return 40 if progress(o.tile) > 1 else 0
    return 0
```
