@gene("s2", "where to walk: home when badly hurt, a cleared group close by to capture, an altar when banking, else a living camp group (an altar if none)")
def goals() -> Tiles:
    if me().health_frac < 0.45 and not channeling():
        return my_spawn()
    acc = near(tile(-5, -5), 0)
    ripe = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            acc = acc | gt
            if group_reward(g):
                ripe = ripe | gt
    if size(ripe) > 0 and dist(nearest(ripe, me().pos), me().pos) <= 5:
        return ripe
    alt = near(tile(-5, -5), 0)
    for a in altars() - locked_altars():
        alt = alt | near(a, 1)
    if banking():
        return alt
    if size(acc) == 0 and gems() > 0:
        return alt
    return acc
