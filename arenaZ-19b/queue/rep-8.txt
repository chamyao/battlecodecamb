```python
@gene("s1", "value of doing this option now; recalls earlier when carrying many gems, never recalls when about to tribute")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < recall_at() else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 95 if gems() > 0 else 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "health fraction below which to recall: higher with a big bag to protect", parents=["s1"])
def recall_at() -> Num:
    g = gems()
    if g >= 30:
        return 0.5
    if g >= 10:
        return 0.42
    return 0.35
```
