@gene('s1', 'value of doing this option now; drafts Willow, Marcella, Vesper Vale then Tavi, buys focus bruiser shards for Marcella and Tavi and strength then health bruiser shards for the rest, casts, focus fires weak targets, gem carriers and recalling enemies, farms until banking, recalls below 45 percent when safe, flees when badly hurt, clears harmless clusters in its way while banking')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 10 if o.champ == WILLOW else (9 if o.champ == MARCELLA else (8 if o.champ == VESPER_VALE else (7 if o.champ == TAVI else 1)))
    if o.kind == BUY:
        return 100 if o.item == (38 if me().type == MARCELLA or me().type == TAVI else (40 if items_owned(37) > 0 else 36)) else 0
    hurt = me().health_frac < 0.35
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.45 and threat_at(me().pos) == 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 5 + min(unit_gems(o.unit), 50) / 10 + (8 if is_recalling(o.unit) else 0)
        return 60 + (1 - o.unit.health_frac) * 5 if not banking() else (55 if o.unit.strength == 0 else 0)
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if hurt:
            now = threat_at(me().pos)
            if now > 0:
                nxt = threat_at(o.tile)
                if nxt < now:
                    return 96 - nxt
                return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z4a', 'value of casting: hit champions and camps we fight, shield when hurt, heal any ally below 85 percent, reset only with a target in reach', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if not exists(o.unit):
            return 0
        if o.unit.is_champion and o.unit.is_enemy:
            return 85
        if o.unit.is_camp and not banking():
            return 65
        return 0
    if k == AB_AOE:
        n = count(dist(u.pos, o.tile) <= 1 for u in enemies())
        c = count(dist(u.pos, o.tile) <= 1 for u in camps())
        if n > 0:
            return 84
        if c > 0 and not banking():
            return 64
        return 0
    if k == AB_SHIELD:
        near_enemy = count(dist(u.pos, me().pos) <= 3 for u in enemies())
        near_camp = count(is_awake(u) and dist(u.pos, me().pos) <= 4 for u in camps())
        return 75 if near_enemy + near_camp > 0 and me().health_frac < 0.9 else 0
    if k == AB_HEAL:
        if exists(o.unit) and o.unit.health_frac < 0.85:
            return 86 + (1 - o.unit.health_frac) * 5
        return 0
    if k == AB_DASH:
        if me().health_frac < 0.35:
            now = threat_at(me().pos)
            if now > 0 and threat_at(o.tile) < now:
                return 97
        return 0
    if k == AB_RESET:
        r = me().attack_range
        foes = count(dist(u.pos, me().pos) <= r for u in enemies()) + count(dist(u.pos, me().pos) <= r for u in camps())
        return 55 if not can_attack() and foes > 0 else 0
    return 0

@gene("s4", "how much closer a step brings me to the goals; never stuck on a dead camp tile of an uncleared group; with a bag worth banking never step next to an enemy champion")
def progress(t: Tile) -> Num:
    f = goal_field()
    here = at(f, me().pos)
    if here == 0:
        sg = spawn_group(me().pos)
        if sg >= 0 and not group_reward(sg) and not any(dist(u.pos, me().pos) <= me().attack_range for u in camps()):
            live = field(camp_tiles())
            return at(live, me().pos) - at(live, t)
    d = here - at(f, t)
    if d > 0 and gems() >= 10:
        if any(dist(u.pos, t) <= 1 for u in enemies()):
            return 0
    return d
