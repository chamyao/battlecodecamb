@gene("s2", "where to walk: home when badly hurt, an altar with gems, else a living camp group (no monolith, golems only late)")
def goals() -> Tiles:
    if me().health_frac < 0.45 and not channeling():
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    avoid = type_spawns(MONOLITH)
    if round() < 1000:
        avoid = avoid | type_spawns(GOLEM)
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    safe = acc - avoid
    if size(safe) > 0:
        return safe
    return acc
