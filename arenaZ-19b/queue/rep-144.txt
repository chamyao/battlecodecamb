@gene('s1', 'value of doing this option now; drafts Marcella, Willow, Vesper Vale then Tavi, buys a bruiser strength emblem then bruiser health, casts, hunts gem carriers and recalling enemies, farms until banking, flees when hurt')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 9 if o.champ == WILLOW else (9 if o.champ == MARCELLA else (8 if o.champ == VESPER_VALE else (7 if o.champ == TAVI else 1)))
    if o.kind == BUY:
        return 100 if o.item == (40 if items_owned(37) > 0 else 36) else 0
    hurt = me().health_frac < 0.317
    if o.kind == RECALL:
        return 90 if hurt and threat_at(me().pos) == 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + min(unit_gems(o.unit), 50) / 10 + (8 if is_recalling(o.unit) else 0)
        return 56 if not banking() else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if hurt:
            now = threat_at(me().pos)
            if now > 0:
                nxt = threat_at(o.tile)
                if nxt < now:
                    return 96 - nxt
                return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("Z172a", "which option kinds to list: never wards or locks; casts only with a slot ready and a foe or camp close")
def consider(kind: Num) -> Bool:
    if kind == WARD or kind == LOCK or kind == BUY_LOCK:
        return False
    if kind == CAST:
        if not (slot_ready(0) or slot_ready(1)):
            return False
        return any(dist(u.pos, me().pos) <= 5 for u in enemies()) or any(dist(u.pos, me().pos) <= 5 for u in camps())
    return True
