```python
@gene("s1", "value of doing this option now; keeps farming camps until the bag holds 20 gems")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if not banking() else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "true when the bag should be taken to an altar: 20 gems, low health or the end of the game", parents=["s2"])
def banking() -> Bool:
    g = gems()
    if g >= 20:
        return True
    if g > 0 and (me().health_frac < 0.5 or round() > 1850):
        return True
    return False

@gene("s2", "where to walk: an altar when banking, else a living camp group (an altar if none and carrying gems)")
def goals() -> Tiles:
    alt = near(tile(-5, -5), 0)
    for a in altars() - locked_altars():
        alt = alt | near(a, 1)
    if banking():
        return alt
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    if size(acc) == 0 and gems() > 0:
        return alt
    return acc
```
