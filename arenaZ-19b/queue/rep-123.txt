@gene("s2", "where to walk: home when badly hurt, a cleared group close by to capture, an altar with gems, else a living camp group")
def goals() -> Tiles:
    if me().health_frac < 0.45 and not channeling():
        return my_spawn()
    ripe = near(tile(-5, -5), 0)
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            acc = acc | gt
            if group_reward(g):
                ripe = ripe | gt
    if size(ripe) > 0 and dist(nearest(ripe, me().pos), me().pos) <= 6:
        return ripe
    if gems() > 0:
        alt = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            alt = alt | near(a, 1)
        return alt
    return acc
