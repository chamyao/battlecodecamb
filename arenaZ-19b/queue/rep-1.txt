```python
@gene("s2", "where to walk: an altar with gems, else a living camp group that is not a golem or monolith")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    big = type_spawns(GOLEM) | type_spawns(MONOLITH)
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    safe = acc - big
    if size(safe) > 0:
        return safe
    return acc
```
