```python
@gene('s1', 'value of doing this option now; buys bruiser strength shards, casts, focus fires, farms until banking, flees when hurt or when a big bag meets enemies', parents=['s1'])
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 36 else 0
    hurt = me().health_frac < 0.35 or (gems() >= 20 and me().health_frac < 0.6 and count(dist(u.pos, me().pos) <= 4 for u in enemies()) > 0)
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 and threat_at(me().pos) == 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 5
        return 60 + (1 - o.unit.health_frac) * 5 if not banking() else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if hurt:
            now = threat_at(me().pos)
            if now > 0:
                nxt = threat_at(o.tile)
                if nxt < now:
                    return 96 - nxt
                return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
