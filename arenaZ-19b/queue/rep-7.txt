```python
@gene("s1", "value of doing this option now; focus fire on the weakest target and cast damage spells")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        k = slot_kind(o.slot)
        if (k == AB_DAMAGE or k == AB_CC) and exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                return 82 + (1 - o.unit.health_frac) * 5
            if o.unit.is_camp and gems() == 0:
                return 62 + (1 - o.unit.health_frac) * 5
        return 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 5
        return 60 + (1 - o.unit.health_frac) * 5 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
