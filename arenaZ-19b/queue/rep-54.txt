```python
@gene("s4", "how much closer a step brings me to the goals, or to my spawn when hurt or rich in gold with an empty bag")
def progress(t: Tile) -> Num:
    if me().health_frac < 0.45 or (gems() == 0 and gold() >= 150):
        home = field(my_spawn())
        return at(home, me().pos) - at(home, t)
    return at(goal_field(), me().pos) - at(goal_field(), t)
```
