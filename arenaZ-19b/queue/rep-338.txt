```python
@gene('s1', 'value of doing this option now; drafts Willow, Marcella, Vesper Vale then Tavi, buys focus bruiser shards for Marcella and Tavi and strength then health bruiser shards for the rest, casts, focus fires weak targets, gem carriers and recalling enemies, farms until banking, recalls below 45 percent when safe, flees when badly hurt, clears harmless clusters in its way while banking, sidesteps when blocked', parents=['s1'])
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 10 if o.champ == WILLOW else (9 if o.champ == MARCELLA else (8 if o.champ == VESPER_VALE else (7 if o.champ == TAVI else 1)))
    if o.kind == BUY:
        return 100 if o.item == (38 if me().type == MARCELLA or me().type == TAVI else (40 if items_owned(37) > 0 else 36)) else 0
    hurt = me().health_frac < 0.33 or (gems() >= 20 and me().health_frac < 0.6 and count(dist(u.pos, me().pos) <= 4 for u in enemies()) > 0)
    if o.kind == RECALL:
        return 56 if me().health_frac < 0.45 and threat_at(me().pos) == 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 5 + min(unit_gems(o.unit), 50) / 10 + (8 if is_recalling(o.unit) else 0)
        return 60 + (1 - o.unit.health_frac) * 5 if not banking() else (55 if o.unit.strength == 0 else 0)
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if hurt:
            now = threat_at(me().pos)
            if now > 0:
                nxt = threat_at(o.tile)
                if nxt < now:
                    return 96 - nxt
                return 0
        p = progress(o.tile)
        if p > 0:
            return 50
        if p == 0 and at(goal_field(), me().pos) > 0 and at(goal_field(), me().pos) < 100000:
            return 5 + rand(10)
        return 0
    return 0
```
