@gene("s1", "value of doing this option now; casts, never pulls a second camp group, steps out of reach when hurt or outnumbered and hurt")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    hurt = me().health_frac < 0.35 or (me().health_frac < 0.6 and outnumbered())
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 and threat_at(me().pos) == 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 and camp_ok(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if hurt:
            now = threat_at(me().pos)
            if now > 0:
                nxt = threat_at(o.tile)
                if nxt < now:
                    return 96 - nxt
                return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "more enemy champions than allies around me", parents=["s1"])
def outnumbered() -> Bool:
    foes = count(dist(u.pos, me().pos) <= 5 for u in enemies())
    if foes == 0:
        return False
    friends = count(dist(u.pos, me().pos) <= 5 for u in allies())
    return foes > friends
