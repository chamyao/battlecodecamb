```python
@gene('s1', 'value of doing this option now; drafts Willow, Marcella, Vesper Vale then Prism, buys a bruiser strength emblem then bruiser health, casts, hunts gem carriers, flees when hurt')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 10 if o.champ == WILLOW else (9 if o.champ == MARCELLA else (8 if o.champ == VESPER_VALE else (7 if o.champ == PRISM else 1)))
    if o.kind == BUY:
        return 100 if o.item == (40 if items_owned(37) > 0 else 36) else 0
    hurt = me().health_frac < 0.35
    if o.kind == RECALL:
        return 90 if hurt and threat_at(me().pos) == 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + min(unit_gems(o.unit), 50) / 10
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if hurt:
            now = threat_at(me().pos)
            if now > 0:
                nxt = threat_at(o.tile)
                if nxt < now:
                    return 96 - nxt
                return 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("s2", "where to walk: home when badly hurt, an altar with gems, else the camp group worth most per turn that I can survive")
def goals() -> Tiles:
    if me().health_frac < 0.45 and not channeling():
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    bestv = 0
    pick = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            acc = acc | gt
            v = group_value(g, gt)
            if v > bestv:
                bestv = v
                pick = gt
    if size(pick) > 0:
        return pick
    return acc

@gene("new", "gems and gold per turn a camp group is worth to me (walk plus kill time), 0 if fighting it alone would kill me", parents=["s2"])
def group_value(g: Num, gt: Tiles) -> Num:
    t0 = nearest(gt, me().pos)
    d = dist(t0, me().pos)
    k = spawn_type(t0)
    n = size(gt)
    ttk = camp_base_health(k) * n / max(me().strength, 1)
    if group_reward(g):
        ttk = 0
    taken = camp_base_strength(k) * n * ttk / 2
    if taken > me().health * 0.8:
        return 0
    return (group_gems(g) + group_gold(g) / 10 + 1) / (d + ttk + 5)
```
