@gene("s2", "where to walk: home when badly hurt, an altar when banking, else the camp group worth most per turn that I can survive")
def goals() -> Tiles:
    if me().health_frac < 0.45 and not channeling():
        return my_spawn()
    alt = near(tile(-5, -5), 0)
    for a in altars() - locked_altars():
        alt = alt | near(a, 1)
    if banking():
        return alt
    acc = near(tile(-5, -5), 0)
    bestv = 0
    pick = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            acc = acc | gt
            v = group_value(g, gt)
            if v > bestv:
                bestv = v
                pick = gt
    if size(pick) > 0:
        return pick
    if size(acc) == 0 and gems() > 0:
        return alt
    return acc

@gene("new", "gems and gold per turn a camp group is worth to me (walk plus kill time), 0 if fighting it alone would kill me", parents=["s2"])
def group_value(g: Num, gt: Tiles) -> Num:
    t0 = nearest(gt, me().pos)
    d = dist(t0, me().pos)
    k = spawn_type(t0)
    n = size(gt)
    ttk = camp_base_health(k) * n / max(me().strength, 1)
    if group_reward(g):
        ttk = 0
    taken = camp_base_strength(k) * n * ttk / 2
    if taken > me().health * 0.8:
        return 0
    return (group_gems(g) + group_gold(g) / 10 + 1) / (d + ttk + 5)
