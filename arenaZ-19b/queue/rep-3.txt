@gene("s2", "where to walk: an altar with gems, else a living camp group away from monoliths, golems early and the enemy base")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    avoid = type_spawns(MONOLITH)
    if round() < 1000:
        avoid = avoid | type_spawns(GOLEM)
    base = enemy_spawn()
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if dist(nearest(base, nearest(gt, me().pos)), nearest(gt, me().pos)) > 6:
                acc = acc | gt
    safe = acc - avoid
    if size(safe) > 0:
        return safe
    if size(acc) > 0:
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
