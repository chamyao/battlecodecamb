```python
@gene("s4", "how much closer a step brings me to the goals; with a bag worth banking never step next to an enemy champion")
def progress(t: Tile) -> Num:
    d = at(goal_field(), me().pos) - at(goal_field(), t)
    if d > 0 and gems() >= 10:
        if any(dist(u.pos, t) <= 1 for u in enemies()):
            return 0
    return d
```
