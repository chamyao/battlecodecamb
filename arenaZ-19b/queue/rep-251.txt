@gene("s2", "where to walk: home when badly hurt, a cleared group close by to capture, an altar when banking, else a living camp group, golems only after round 800 and the monolith only late with three allies around")
def goals() -> Tiles:
    if me().health_frac < 0.45 and not channeling():
        return my_spawn()
    acc = near(tile(-5, -5), 0)
    ripe = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            acc = acc | gt
            if group_reward(g):
                ripe = ripe | gt
    if size(ripe) > 0 and dist(nearest(ripe, me().pos), me().pos) <= 5:
        return ripe
    alt = near(tile(-5, -5), 0)
    for a in altars() - locked_altars():
        alt = alt | near(a, 1)
    if banking():
        return alt
    if round() < 1200 or count(dist(u.pos, me().pos) <= 6 for u in allies()) < 3:
        safe = acc - type_spawns(MONOLITH)
        if round() < 800:
            safe = safe - type_spawns(GOLEM)
        if size(safe) > 0:
            return safe
    if size(acc) == 0 and gems() > 0:
        return alt
    return acc

@gene("Z6a", "true when the bag should be taken to an altar: 18 gems, low health or the end of the game")
def banking() -> Bool:
    g = gems()
    if g >= 18:
        return True
    if g > 0 and (me().health_frac < 0.5 or round() > 1850):
        return True
    return False
