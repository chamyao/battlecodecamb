@gene('s2', 'where to walk: altar (or a cleared camp) with gems, else an enemy tributing or a weak enemy carrying gems, else a living non-monolith camp group')
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    prey = prey_tiles()
    if me().health_frac > 0.5:
        for u in enemies():
            if u.is_champion and is_tributing(u):
                prey = prey | near(u.pos, 1)
    if size(prey) > 0:
        return prey
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc - type_spawns(MONOLITH)
