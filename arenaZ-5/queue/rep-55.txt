@gene("s3", "distance field to the goals; while carrying gems the path avoids tiles near visible enemy champions")
def goal_field() -> Field:
    if gems() > 0:
        danger = enemy_zone()
        if size(danger) > 0:
            return field_avoid(goals(), danger, near(tile(-5, -5), 0))
    return field(goals())

@gene('new', 'tiles within 2 of a visible enemy champion, except around me', parents=['s3'])
def enemy_zone() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for u in enemies():
        if u.is_champion:
            acc = acc | near(u.pos, 2)
    return acc - near(me().pos, 1)
