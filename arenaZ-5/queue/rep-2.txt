@gene("new", "camp groups worth walking to: gem groups first, never monolith/golem", parents=["s2"])
def camp_goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    rest = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            hard = False
            for t in group_tiles(g):
                k = spawn_type(t)
                if k == MONOLITH or k == GOLEM:
                    hard = True
            if not hard:
                if group_gems(g) > 0 or group_reward(g):
                    acc = acc | group_tiles(g)
                else:
                    rest = rest | group_tiles(g)
    if size(acc) > 0:
        return acc
    return rest

@gene("s2", "where to walk: an altar with gems, else a gem-bearing camp group")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    return camp_goals()

@gene("s3", "distance field to the goals")
def goal_field() -> Field:
    return field(goals())

@gene("s4", "how much closer a step brings me to the goals")
def progress(t: Tile) -> Num:
    f = goal_field()
    return at(f, me().pos) - at(f, t)
