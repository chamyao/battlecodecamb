```python
@gene('s3', 'distance field to the goals: to my spawn when nearly dead, around visible enemy champions while hurt')
def goal_field() -> Field:
    hf = me().health_frac
    if hf < 0.3:
        return field(my_spawn())
    if hf < 0.5:
        danger = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, 2)
        danger = danger - near(me().pos, 1)
        if size(danger) > 0:
            return field_avoid(goals(), danger, near(tile(-5, -5), 0))
    return field(goals())
```
