```python
@gene('Z18a', 'value of casting: attack resets, ground AoE on enemy champions or awake camp packs, damage/cc on enemy champions (cc first when hurt) or camps we fight, heal the most hurt target, shield when hurt, dash toward goals (urgently when fleeing hurt)', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    hf = me().health_frac
    if k == AB_HEAL:
        if exists(o.unit) and not o.unit.is_enemy and not o.unit.is_camp:
            return 88 if o.unit.health_frac < 0.6 else 0
        return 88 if hf < 0.6 else 0
    if k == AB_SHIELD:
        return 86 if hf < 0.75 and count_units(enemies()) > 0 else 0
    if k == AB_DASH:
        if valid(o.tile) and not exists(o.unit):
            if hf < 0.3 and gems() == 0 and count_units(enemies()) > 0:
                return 80 if progress(o.tile) >= 2 else 0
            return 38 if progress(o.tile) >= 2 else 0
        return 0
    if k == AB_RESET:
        if can_attack():
            return 0
        here = me().pos
        reach = me().attack_range
        if any(dist(u.pos, here) <= reach for u in enemies()):
            return 83
        if gems() == 0 and any(dist(u.pos, here) <= reach for u in camps()):
            return 63
        return 0
    if not exists(o.unit):
        if (k == AB_AOE or k == AB_DAMAGE or k == AB_CC) and valid(o.tile):
            r = slot_radius(o.slot)
            if any(u.is_champion and dist(u.pos, o.tile) <= r for u in enemies()):
                return 84
            if k == AB_AOE and gems() == 0:
                if count(dist(u.pos, o.tile) <= r and is_awake(u) for u in camps()) >= 2:
                    return 63
        return 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if o.unit.is_champion and o.unit.is_enemy:
            if k == AB_CC and hf < 0.5:
                return 89
            return 85
        if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 62
    return 0
```
