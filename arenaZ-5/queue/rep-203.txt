@gene('Z116a', 'distance field to the goals (to my spawn when nearly dead); while hurt the path avoids tiles near visible enemy champions (terrain still blocks)', parents=['s3'])
def safe_field() -> Field:
    hf = me().health_frac
    if hf < 0.5:
        danger = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, 2)
        danger = danger - near(me().pos, 1)
        dest = my_spawn() if hf < 0.3 and gems() == 0 else goals()
        if size(danger) > 0:
            return field_avoid(dest, danger | walls() | holes() | altars() | enemy_spawn(), near(tile(-5, -5), 0))
        if hf < 0.3 and gems() == 0:
            return field(dest)
    return goal_field()

@gene('Z219b', 'where to walk: altar (or a cleared camp) with gems; else cleared camps, an ally fight to join, prey (weak gem carriers, nearly dead enemies close by), then living non-monolith camps (no golems before round 600)', parents=['Z47b', 'Z86a'])
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    if size(acc) > 0:
        return acc
    hf = me().health_frac
    if hf > 0.3:
        rally = rally_tile()
        if valid(rally):
            return near(rally, 1)
    if hf > 0.5:
        here = me().pos
        for u in enemies():
            if u.is_champion:
                if unit_gems(u) >= 5 and u.health_frac < hf:
                    acc = acc | near(u.pos, 0)
                elif u.health_frac < 0.3 and dist(u.pos, here) <= 4:
                    acc = acc | near(u.pos, 0)
        if size(acc) > 0:
            return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    acc = acc - type_spawns(MONOLITH)
    if round() < 600:
        acc = acc - type_spawns(GOLEM)
    return acc
