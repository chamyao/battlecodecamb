```python
@gene('Z273b', 'tile of a fresh ally fight beacon (3 rounds) within 18 tiles that I should join: not when badly hurt or carrying gems (invalid if none)', parents=['s4'])
def rally_tile() -> Tile:
    stamp = read_shared(2)
    if stamp == 0 or gems() > 0 or me().health_frac <= 0.3:
        return tile(-5, -5)
    age = (round() % 250 + 1 - stamp + 250) % 250
    if age > 3:
        return tile(-5, -5)
    t = tile(read_shared(0) - 1, read_shared(1) - 1)
    d = dist(t, me().pos)
    if d <= 1 or d > 18:
        return tile(-5, -5)
    return t
```
