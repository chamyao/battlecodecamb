```python
@gene('s2', 'where to walk: altar (or a cleared camp) with gems; else cleared camps, a weak enemy carrying gems, then living non-monolith camps (no golems before round 600)')
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    if size(acc) > 0:
        return acc
    prey = prey_tiles()
    if size(prey) > 0:
        return prey
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    acc = acc - type_spawns(MONOLITH)
    if round() < 600:
        acc = acc - type_spawns(GOLEM)
    return acc
```
