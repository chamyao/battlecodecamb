@gene('s1', 'value of doing this option now: casts, fight-back, focus the weakest champion, treasure-then-power build, recall to shop with a full purse')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if o.item == 18 and items_owned(19) == 0:
            return 100
        return 95 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        if gems() == 0 and gold() >= 153 and count_units(enemies()) == 0:
            return 75
        return 0
    if o.kind == CAST:
        k = slot_kind(o.slot)
        if k == AB_HEAL:
            return 88 if me().health_frac < 0.6 else 0
        if k == AB_SHIELD:
            return 86 if me().health_frac < 0.75 and count_units(enemies()) > 0 else 0
        if not exists(o.unit):
            return 0
        if o.unit.is_champion and o.unit.is_enemy:
            return 85 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 45
        return 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 8 * (1 - o.unit.health_frac)
        return 60 if gems() == 0 or is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 37 if progress(o.tile) > 0 else 0
    return 0

@gene('new', 'skip listing option kinds the score never takes (wards, locks, cancels) and casts while both slots cool down', parents=['s1'])
def consider(kind: Num) -> Bool:
    if kind == CAST:
        return slot_ready(0) or slot_ready(1)
    return kind != WARD and kind != LOCK and kind != BUY_LOCK and kind != CANCEL

@gene('s3', 'distance field to the goals: to my spawn when nearly dead, around visible enemy champions while hurt')
def goal_field() -> Field:
    hf = me().health_frac
    if hf < 0.3:
        return field(my_spawn())
    if hf < 0.5:
        danger = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, 2)
        danger = danger - near(me().pos, 1)
        if size(danger) > 0:
            return field_avoid(goals(), danger, near(tile(-5, -5), 0))
    return field(goals())
