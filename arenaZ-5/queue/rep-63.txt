@gene('s3', 'distance field to the goals; below 60% health the path avoids tiles near visible enemy champions')
def goal_field() -> Field:
    if me().health_frac < 0.6:
        danger = enemy_zone()
        if size(danger) > 0:
            return field_avoid(goals(), danger, near(tile(-5, -5), 0))
    return field(goals())
