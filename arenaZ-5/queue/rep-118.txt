@gene('Z18a', 'value of casting: damage/cc on enemy champions or camps we fight (ground AoE onto awake camp packs too), heal the most hurt target (myself first), shield when hurt, dash toward goals', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    hf = me().health_frac
    if k == AB_HEAL:
        if exists(o.unit) and not o.unit.is_enemy and not o.unit.is_camp:
            return 88 if o.unit.health_frac < 0.6 else 0
        return 88 if hf < 0.6 else 0
    if k == AB_SHIELD:
        return 86 if hf < 0.75 and count_units(enemies()) > 0 else 0
    if k == AB_DASH:
        if valid(o.tile) and not exists(o.unit):
            return 38 if progress(o.tile) >= 2 else 0
        return 0
    if not exists(o.unit):
        if k == AB_AOE and valid(o.tile) and gems() == 0:
            r = slot_radius(o.slot)
            if count(dist(u.pos, o.tile) <= r and is_awake(u) for u in camps()) >= 2:
                return 63
        return 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if o.unit.is_champion and o.unit.is_enemy:
            return 85
        if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 62
    return 0

@gene('Z47b', 'where to walk: altar (or a cleared camp) with gems; else cleared camps, prey (weak gem carriers, nearly dead enemies close by), then living non-monolith camps (no golems before round 624)', parents=['s2'])
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    if size(acc) > 0:
        return acc
    hf = me().health_frac
    if hf > 0.5:
        here = me().pos
        for u in enemies():
            if u.is_champion:
                if unit_gems(u) >= 4 and u.health_frac < hf:
                    acc = acc | near(u.pos, 0)
                elif u.health_frac < 0.3 and dist(u.pos, here) <= 4:
                    acc = acc | near(u.pos, 0)
        if size(acc) > 0:
            return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    acc = acc - type_spawns(MONOLITH)
    if round() < 624:
        acc = acc - type_spawns(GOLEM)
    return acc
