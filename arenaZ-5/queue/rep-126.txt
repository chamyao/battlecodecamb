```python
@gene('s1', 'value of doing this option now: casts (incl. attack resets and ground AoE), fight-back, focus the weakest champion, treasure then home then power build, recall to shop with a full purse')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if o.item == 18 and items_owned(19) == 0:
            return 100
        if o.item == 30 and items_owned(31) == 0:
            return 98
        return 95 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.2944:
            return 90
        if gems() == 0 and gold() >= 153 and count_units(enemies()) == 0:
            return 75
        return 0
    if o.kind == CAST:
        extra = extra_cast(o)
        if extra > 0:
            return extra
        k = slot_kind(o.slot)
        if k == AB_HEAL:
            return 88 if me().health_frac < 0.6 else 0
        if k == AB_SHIELD:
            return 86 if me().health_frac < 0.75 and count_units(enemies()) > 0 else 0
        if not exists(o.unit):
            return 0
        if o.unit.is_champion and o.unit.is_enemy:
            return 85 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 45
        return 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 8 * (1 - o.unit.health_frac)
        return 60 if gems() == 0 or is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 37 if progress(o.tile) > 0 else 0
    return 0
```
