@gene('s3', 'distance field to the goals; while hurt the path avoids tiles near visible enemy champions')
def goal_field() -> Field:
    if me().health_frac < 0.5:
        danger = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, 2)
        danger = danger - near(me().pos, 1)
        if size(danger) > 0:
            return field_avoid(goals(), danger, near(tile(-5, -5), 0))
    return field(goals())

@gene('s2', 'where to walk: altar (or a cleared camp) with gems; else cleared camps, a weak enemy carrying gems, then living non-monolith camps away from enemy champions (no golems before round 600)')
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    if size(acc) > 0:
        return acc
    hf = me().health_frac
    contested = near(tile(-5, -5), 0)
    for u in enemies():
        if u.is_champion:
            if hf > 0.5 and unit_gems(u) >= 5 and u.health_frac < hf:
                acc = acc | near(u.pos, 0)
            contested = contested | near(u.pos, 3)
    if size(acc) > 0:
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    acc = acc - type_spawns(MONOLITH)
    if round() < 600:
        acc = acc - type_spawns(GOLEM)
    quiet = acc - contested
    if size(quiet) > 0:
        return quiet
    return acc
