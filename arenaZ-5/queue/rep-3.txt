@gene("s2", "where to walk: altar (or a cleared camp to capture) with gems, else a living non-monolith camp group")
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc - type_spawns(MONOLITH)

@gene("s3", "distance field to the goals")
def goal_field() -> Field:
    return field(goals())
