@gene('s3', 'distance field to the goals: to my spawn when nearly dead, around visible enemy champions while hurt')
def goal_field() -> Field:
    hf = me().health_frac
    if hf < 0.3:
        return field(my_spawn())
    if hf < 0.5:
        danger = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, 2)
        danger = danger - near(me().pos, 1)
        if size(danger) > 0:
            return field_avoid(goals(), danger, near(tile(-5, -5), 0))
    return field(goals())

@gene('s2', 'where to walk: altar (or a cleared camp) with gems; else cleared camps, prey (weak gem carriers, nearly dead enemies close by), then living non-monolith camps, gem camps only after round 300 (no golems before round 624)')
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    if size(acc) > 0:
        return acc
    hf = me().health_frac
    if hf > 0.5:
        here = me().pos
        for u in enemies():
            if u.is_champion:
                if unit_gems(u) >= 4 and u.health_frac < hf:
                    acc = acc | near(u.pos, 0)
                elif u.health_frac < 0.3 and dist(u.pos, here) <= 4:
                    acc = acc | near(u.pos, 0)
        if size(acc) > 0:
            return acc
    rich = near(tile(-5, -5), 0)
    late = round() >= 300
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
            if late and group_gems(g) > 0:
                rich = rich | group_tiles(g)
    hard = type_spawns(MONOLITH)
    if round() < 624:
        hard = hard | type_spawns(GOLEM)
    rich = rich - hard
    if size(rich) > 0:
        return rich
    return acc - hard
