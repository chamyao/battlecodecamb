@gene('new', 'after acting: announce a fight with an enemy champion (tile + round) in shared bytes 0-2 so allies can join', parents=['Z47b'])
def after() -> None:
    if acted(ATTACK):
        t = acted_tile(ATTACK)
        if valid(t) and t in enemy_tiles():
            write_shared(0, t.x + 1)
            write_shared(1, t.y + 1)
            write_shared(2, round() % 250 + 1)

@gene('new', 'tile of a fresh ally fight beacon within 12 tiles (invalid if none)', parents=['Z47b'])
def rally_tile() -> Tile:
    stamp = read_shared(2)
    if stamp == 0:
        return tile(-5, -5)
    age = (round() % 250 + 1 - stamp + 250) % 250
    if age > 2:
        return tile(-5, -5)
    t = tile(read_shared(0) - 1, read_shared(1) - 1)
    d = dist(t, me().pos)
    if d <= 1 or d > 12:
        return tile(-5, -5)
    return t

@gene('Z47b', 'where to walk: altar (or a cleared camp) with gems; else cleared camps, an ally fight to join, prey, then living non-monolith camps (no golems before round 600)', parents=['s2'])
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    if size(acc) > 0:
        return acc
    hf = me().health_frac
    if hf > 0.6:
        rally = rally_tile()
        if valid(rally):
            return near(rally, 1)
    if hf > 0.5:
        here = me().pos
        for u in enemies():
            if u.is_champion:
                if unit_gems(u) >= 4 and u.health_frac < hf:
                    acc = acc | near(u.pos, 0)
                elif u.health_frac < 0.3 and dist(u.pos, here) <= 4:
                    acc = acc | near(u.pos, 0)
        if size(acc) > 0:
            return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    acc = acc - type_spawns(MONOLITH)
    if round() < 600:
        acc = acc - type_spawns(GOLEM)
    return acc
