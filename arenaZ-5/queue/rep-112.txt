@gene('s3', 'distance field to the goals: to my spawn when nearly dead, around visible enemy champions while hurt, to weak gem carriers or nearly dead enemies close by when healthy and empty-handed')
def goal_field() -> Field:
    hf = me().health_frac
    if hf < 0.3:
        return field(my_spawn())
    if hf < 0.5:
        danger = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, 2)
        danger = danger - near(me().pos, 1)
        if size(danger) > 0:
            return field_avoid(goals(), danger, near(tile(-5, -5), 0))
    if gems() == 0 and hf > 0.5:
        prey = prey_tiles()
        if size(prey) > 0:
            return field(prey)
    return field(goals())

@gene('new', 'visible enemy champions worth chasing: carrying 5+ gems and weaker than me, or nearly dead within 4 tiles', parents=['s3'])
def prey_tiles() -> Tiles:
    acc = near(tile(-5, -5), 0)
    hf = me().health_frac
    here = me().pos
    for u in enemies():
        if u.is_champion:
            if unit_gems(u) >= 5 and u.health_frac < hf:
                acc = acc | near(u.pos, 0)
            elif u.health_frac < 0.3 and dist(u.pos, here) <= 4:
                acc = acc | near(u.pos, 0)
    return acc
