```python
"""Cross of Z9 and Z8: Z9's casts and fight-back with a treasure-then-power build, on Z8's cleared-camps-first goals."""

@gene('new', 'value of doing this option now: casts, fight-back on awake camps, treasure-then-power build', parents=['s1'])
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if o.item == 18 and items_owned(19) == 0:
            return 100
        return 95 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        if not exists(o.unit):
            return 0
        k = slot_kind(o.slot)
        if k == AB_HEAL or k == AB_SHIELD:
            return 0
        if o.unit.is_champion and o.unit.is_enemy:
            return 85
        if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 61
        return 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 or is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('s2', 'where to walk: an altar with gems, else a camp group (cleared ones first)')
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    return camp_goals()

@gene('s3', 'distance field to the goals')
def goal_field() -> Field:
    return field(goals())

@gene('s4', 'how much closer a step brings me to the goals')
def progress(t: Tile) -> Num:
    f = goal_field()
    return at(f, me().pos) - at(f, t)

@gene('Z4a', 'camp groups worth walking to: cleared groups to capture, then gem groups, never monolith/golem', parents=['s2'])
def camp_goals() -> Tiles:
    free = near(tile(-5, -5), 0)
    acc = near(tile(-5, -5), 0)
    rest = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            free = free | group_tiles(g)
        elif not group_respawning(g):
            hard = False
            for t in group_tiles(g):
                k = spawn_type(t)
                if k == MONOLITH or k == GOLEM:
                    hard = True
            if not hard:
                if group_gems(g) > 0:
                    acc = acc | group_tiles(g)
                else:
                    rest = rest | group_tiles(g)
    if size(free) > 0:
        return free
    if size(acc) > 0:
        return acc
    return rest
```
