@gene('Z135a', 'value of doing this option now: casts (incl. attack resets and ground AoE), eager fight-back on awake camps, focus the weakest champion, treasure-then-power build, recall at 15% or to shop with a full purse', parents=['s1'])
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if o.item == 18 and items_owned(19) == 0:
            return 100
        return 95 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.15:
            return 90
        if gems() == 0 and gold() >= 150 and count_units(enemies()) == 0:
            return 75
        return 0
    if o.kind == CAST:
        extra = extra_cast(o)
        if extra > 0:
            return extra
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 8 * (1 - o.unit.health_frac)
        return 79 if gems() == 0 or is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z47a', 'skip listing option kinds the score never takes (wards, locks, cancels), casts while both slots cool down, shopping off spawn and tributes without gems', parents=['s1'])
def consider(kind: Num) -> Bool:
    if kind == CAST:
        return slot_ready(0) or slot_ready(1)
    if kind == BUY:
        return can_shop()
    if kind == TRIBUTE:
        return gems() > 0
    return kind != WARD and kind != LOCK and kind != BUY_LOCK and kind != CANCEL
