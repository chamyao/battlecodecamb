@gene('Z201b', 'skip listing option kinds the score never takes (wards, locks, cancels), casts while both slots cool down, shopping off spawn and tributes without gems', parents=['Z47a'])
def consider(kind: Num) -> Bool:
    if kind == CAST:
        return slot_ready(0) or slot_ready(1)
    if kind == BUY:
        return can_shop()
    if kind == TRIBUTE:
        return gems() > 0
    return kind != WARD and kind != LOCK and kind != BUY_LOCK and kind != CANCEL

@gene('Z219b', 'where to walk: altar (or a cleared camp) with gems; else cleared camps, an ally fight to join, prey (weak gem carriers, nearly dead enemies close by), then living non-monolith camps (no golems before round 400)', parents=['Z47b', 'Z86a'])
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    if size(acc) > 0:
        return acc
    hf = me().health_frac
    if hf > 0.4:
        rally = rally_tile()
        if valid(rally):
            return near(rally, 1)
    if hf > 0.5:
        here = me().pos
        for u in enemies():
            if u.is_champion:
                if unit_gems(u) >= 5 and u.health_frac < hf:
                    acc = acc | near(u.pos, 0)
                elif u.health_frac < 0.3 and dist(u.pos, here) <= 4:
                    acc = acc | near(u.pos, 0)
        if size(acc) > 0:
            return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    acc = acc - type_spawns(MONOLITH)
    if round() < 400:
        acc = acc - type_spawns(GOLEM)
    return acc
