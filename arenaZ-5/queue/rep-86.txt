"""Cross of Z84 and Z69: Z84 whole (hurt-avoiding paths, prey hunting, recall to shop from 120 gold, consider()), with Z69's stun on enemies who are tributing."""

@gene('Z99a', 'value of doing this option now: casts, fight-back on awake camps, treasure-then-power build, recall at 35% or to shop from 120 gold', parents=['s1'])
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if o.item == 18 and items_owned(19) == 0:
            return 100
        return 95 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        if gems() == 0 and gold() >= 120 and count_units(enemies()) == 0:
            return 75
        return 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 or is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z99b', 'where to walk: altar (or a cleared camp) with gems; else cleared camps, prey (weak gem carriers, nearly dead enemies close by), then living non-monolith camps (no golems before round 600)', parents=['s2'])
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    if size(acc) > 0:
        return acc
    hf = me().health_frac
    if hf > 0.5:
        here = me().pos
        for u in enemies():
            if u.is_champion:
                if unit_gems(u) >= 5 and u.health_frac < hf:
                    acc = acc | near(u.pos, 0)
                elif u.health_frac < 0.3 and dist(u.pos, here) <= 4:
                    acc = acc | near(u.pos, 0)
        if size(acc) > 0:
            return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    acc = acc - type_spawns(MONOLITH)
    if round() < 600:
        acc = acc - type_spawns(GOLEM)
    return acc

@gene('s3', 'distance field to the goals; while hurt the path avoids tiles near visible enemy champions')
def goal_field() -> Field:
    if me().health_frac < 0.5:
        danger = enemy_zone()
        if size(danger) > 0:
            return field_avoid(goals(), danger, near(tile(-5, -5), 0))
    return field(goals())

@gene('Z74a', 'tiles within 2 of a visible enemy champion, except around me', parents=['s3'])
def enemy_zone() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for u in enemies():
        if u.is_champion:
            acc = acc | near(u.pos, 2)
    return acc - near(me().pos, 1)

@gene('s4', 'how much closer a step brings me to the goals')
def progress(t: Tile) -> Num:
    f = goal_field()
    return at(f, me().pos) - at(f, t)

@gene('Z78a', 'skip listing option kinds the score never takes (wards, locks, cancels) and casts while both slots cool down', parents=['s1'])
def consider(kind: Num) -> Bool:
    if kind == CAST:
        return slot_ready(0) or slot_ready(1)
    return kind != WARD and kind != LOCK and kind != BUY_LOCK and kind != CANCEL

@gene('new', 'value of casting: stun tributing enemies, damage/cc on enemy champions or camps we fight, heal the most hurt target, shield when hurt, dash toward goals', parents=['Z18a', 'Z52a'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    hf = me().health_frac
    if k == AB_HEAL:
        if exists(o.unit) and not o.unit.is_enemy and not o.unit.is_camp:
            return 88 if o.unit.health_frac < 0.6 else 0
        return 88 if hf < 0.6 else 0
    if k == AB_SHIELD:
        return 86 if hf < 0.75 and count_units(enemies()) > 0 else 0
    if k == AB_DASH:
        if valid(o.tile) and not exists(o.unit):
            return 38 if progress(o.tile) >= 2 else 0
        return 0
    if not exists(o.unit):
        return 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if o.unit.is_champion and o.unit.is_enemy:
            if k == AB_CC and is_tributing(o.unit):
                return 96
            return 85
        if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 62
    return 0
