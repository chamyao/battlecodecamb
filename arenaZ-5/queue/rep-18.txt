```python
BANK: Num = 15

@gene('s1', 'value of doing this option now: casts, fight-back, treasure-then-power build; keep farming camps until the gem bag is full')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if o.item == 18 and items_owned(19) == 0:
            return 100
        return 95 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() < BANK or is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('new', 'where to walk: an altar once the gem bag is full (or late, or nothing to farm), else cleared camps then living non-monolith camps', parents=['s2'])
def bank_goals() -> Tiles:
    camps = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g) or not group_respawning(g):
            camps = camps | group_tiles(g)
    camps = camps - type_spawns(MONOLITH)
    if gems() > 0 and (gems() >= BANK or round() > 1850 or size(camps) == 0):
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    return camps

@gene('s4', 'how much closer a step brings me to the banking goals')
def progress(t: Tile) -> Num:
    f = field(bank_goals())
    return at(f, me().pos) - at(f, t)

@gene('Z18a', 'value of casting: damage/cc on enemy champions or camps we fight, heal/shield when hurt', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    hf = me().health_frac
    if k == AB_HEAL:
        return 88 if hf < 0.6 else 0
    if k == AB_SHIELD:
        return 86 if hf < 0.75 and count_units(enemies()) > 0 else 0
    if not exists(o.unit):
        return 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if o.unit.is_champion and o.unit.is_enemy:
            return 85
        if o.unit.is_camp and (gems() < BANK or is_awake(o.unit)):
            return 62
    return 0
```
