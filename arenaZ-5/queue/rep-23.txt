```python
@gene('s1', 'value of doing this option now: ladder draft, casts, fight-back on awake camps, treasure-then-power build')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        if o.item == 18 and items_owned(19) == 0:
            return 100
        return 95 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 or is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('new', 'draft preference: the top ladder picks first', parents=['s1'])
def pick_value(c: Num) -> Num:
    if c == VESPER_VALE:
        return 10
    if c == HEART_OF_RUIN:
        return 9
    if c == MARCELLA:
        return 8
    if c == VENDETTA:
        return 7
    if c == RYTH:
        return 6
    return 1
```
