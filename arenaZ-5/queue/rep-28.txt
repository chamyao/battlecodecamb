```python
"""Cross of Z16 and Z17: Z16's casts, build and earlier recall with gems on the shared goals, plus fuel savings:
the distance field is computed once per turn in update(), and consider() drops option kinds we never use."""

GOAL_FIELD: Field = None

@gene('Z19a', 'value of doing this option now: casts, fight-back on awake camps, treasure-then-power build, earlier recall with gems', parents=['s1'])
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if o.item == 18 and items_owned(19) == 0:
            return 100
        return 95 if o.item == 6 else 0
    if o.kind == RECALL:
        limit = 0.45 if gems() > 0 else 0.35
        return 90 if me().health_frac < limit else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 or is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('new', 'skip listing option kinds the score never takes (wards, locks, cancels)', parents=['Z19a'])
def consider(kind: Num) -> Bool:
    return kind != WARD and kind != LOCK and kind != BUY_LOCK and kind != CANCEL

@gene('new', 'once per turn while alive: the distance field to this turn goals', parents=['s3'])
def update() -> None:
    if alive() and not channeling():
        GOAL_FIELD = field(goals())

@gene('s2', 'where to walk: altar (or a cleared camp to capture) with gems, else a living non-monolith camp group')
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc - type_spawns(MONOLITH)

@gene('new', 'distance field to the goals, cached once per turn', parents=['s3'])
def goal_field() -> Field:
    return GOAL_FIELD

@gene('s4', 'how much closer a step brings me to the goals')
def progress(t: Tile) -> Num:
    f = goal_field()
    return at(f, me().pos) - at(f, t)

@gene('Z18a', 'value of casting: damage/cc on enemy champions or camps we fight, heal/shield when hurt', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    hf = me().health_frac
    if k == AB_HEAL:
        return 88 if hf < 0.6 else 0
    if k == AB_SHIELD:
        return 86 if hf < 0.75 and count_units(enemies()) > 0 else 0
    if not exists(o.unit):
        return 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if o.unit.is_champion and o.unit.is_enemy:
            return 85
        if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 62
    return 0
```
