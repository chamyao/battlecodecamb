```python
@gene('s2', 'where to walk: an altar with gems, else a camp group (cleared ones first)')
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    return camp_goals()

@gene('Z4a', 'camp groups worth walking to: cleared groups to capture, then gem groups, never monolith/golem', parents=['s2'])
def camp_goals() -> Tiles:
    free = near(tile(-5, -5), 0)
    acc = near(tile(-5, -5), 0)
    rest = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            free = free | group_tiles(g)
        elif not group_respawning(g):
            hard = False
            for t in group_tiles(g):
                k = spawn_type(t)
                if k == MONOLITH or k == GOLEM:
                    hard = True
            if not hard:
                if group_gems(g) > 0:
                    acc = acc | group_tiles(g)
                else:
                    rest = rest | group_tiles(g)
    if size(free) > 0:
        return free
    if size(acc) > 0:
        return acc
    return rest
```
