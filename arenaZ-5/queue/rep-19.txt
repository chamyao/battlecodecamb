@gene('s2', 'where to walk: altar (or a cleared camp) with gems, else a weak enemy carrying gems, else a living non-monolith camp group')
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    prey = prey_tiles()
    if size(prey) > 0:
        return prey
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc - type_spawns(MONOLITH)

@gene('new', 'tiles of visible enemy champions carrying gems that are weaker than me', parents=['s2'])
def prey_tiles() -> Tiles:
    acc = near(tile(-5, -5), 0)
    hf = me().health_frac
    for u in enemies():
        if u.is_champion and unit_gems(u) >= 5 and u.health_frac < hf and hf > 0.5:
            acc = acc | near(u.pos, 0)
    return acc
