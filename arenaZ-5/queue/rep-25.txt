```python
@gene('s1', 'value of doing this option now: casts (heal/shield too), fight-back on awake camps, focus the weakest champion, treasure-then-power build')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if o.item == 18 and items_owned(19) == 0:
            return 100
        return 95 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        k = slot_kind(o.slot)
        if k == AB_HEAL:
            return 88 if me().health_frac < 0.6 else 0
        if k == AB_SHIELD:
            return 86 if me().health_frac < 0.75 and count_units(enemies()) > 0 else 0
        if not exists(o.unit):
            return 0
        if o.unit.is_champion and o.unit.is_enemy:
            return 85 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 61
        return 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 8 * (1 - o.unit.health_frac)
        return 60 if gems() == 0 or is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
