```python
"""Cross of Z110 and Z98: Z110 (recall to shop, hurt-avoiding paths, prey) with attack resets and ground AoE casts on enemy champions and awake camp packs."""

@gene('new', 'value of doing this option now: casts (incl. attack resets and ground AoE), eager fight-back on awake camps, treasure-then-power build, recall to shop with a full purse', parents=['s1'])
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if o.item == 18 and items_owned(19) == 0:
            return 100
        return 95 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        if gems() == 0 and gold() >= 150 and count_units(enemies()) == 0:
            return 75
        return 0
    if o.kind == CAST:
        extra = extra_cast(o)
        if extra > 0:
            return extra
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 79 if gems() == 0 or is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z47a', 'skip listing option kinds the score never takes (wards, locks, cancels) and casts while both slots cool down', parents=['s1'])
def consider(kind: Num) -> Bool:
    if kind == CAST:
        return slot_ready(0) or slot_ready(1)
    return kind != WARD and kind != LOCK and kind != BUY_LOCK and kind != CANCEL

@gene('Z47b', 'where to walk: altar (or a cleared camp) with gems; else cleared camps, prey (weak gem carriers, nearly dead enemies close by), then living non-monolith camps (no golems before round 600)', parents=['s2'])
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    if size(acc) > 0:
        return acc
    hf = me().health_frac
    if hf > 0.5:
        here = me().pos
        for u in enemies():
            if u.is_champion:
                if unit_gems(u) >= 4 and u.health_frac < hf:
                    acc = acc | near(u.pos, 0)
                elif u.health_frac < 0.3 and dist(u.pos, here) <= 4:
                    acc = acc | near(u.pos, 0)
        if size(acc) > 0:
            return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    acc = acc - type_spawns(MONOLITH)
    if round() < 624:
        acc = acc - type_spawns(GOLEM)
    return acc

@gene('s3', 'distance field to the goals')
def goal_field() -> Field:
    return field(goals())

@gene('s4', 'how much closer a step brings me to the goals (by a path that avoids enemy champions while hurt)')
def progress(t: Tile) -> Num:
    f = safe_field()
    return at(f, me().pos) - at(f, t)

@gene('Z18a', 'value of casting: damage/cc on enemy champions or camps we fight, heal the most hurt target, shield when hurt, dash toward goals', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    hf = me().health_frac
    if k == AB_HEAL:
        if exists(o.unit) and not o.unit.is_enemy and not o.unit.is_camp:
            return 75 if o.unit.health_frac < 0.6 else 0
        return 84 if hf < 0.6 else 0
    if k == AB_SHIELD:
        return 86 if hf < 0.75 and count_units(enemies()) > 0 else 0
    if k == AB_DASH:
        if valid(o.tile) and not exists(o.unit):
            return 38 if progress(o.tile) >= 2 else 0
        return 0
    if not exists(o.unit):
        return 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if o.unit.is_champion and o.unit.is_enemy:
            return 85
        if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 45
    return 0

@gene('Z73a', 'distance field to the goals; while hurt the path avoids tiles near visible enemy champions', parents=['s3'])
def safe_field() -> Field:
    if me().health_frac < 0.5:
        danger = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, 2)
        danger = danger - near(me().pos, 1)
        if size(danger) > 0:
            return field_avoid(goals(), danger, near(tile(-5, -5), 0))
    return goal_field()

@gene('new', 'value of casts cast_value ignores: an attack reset after attacking with a target in range, a ground AoE that hits an enemy champion or an awake camp pack', parents=['Z18a'])
def extra_cast(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_RESET:
        if can_attack():
            return 0
        here = me().pos
        reach = me().attack_range
        if any(dist(u.pos, here) <= reach for u in enemies()):
            return 83
        if gems() == 0 and any(dist(u.pos, here) <= reach for u in camps()):
            return 63
        return 0
    if (k == AB_AOE or k == AB_DAMAGE or k == AB_CC) and not exists(o.unit) and valid(o.tile):
        r = slot_radius(o.slot)
        if any(u.is_champion and dist(u.pos, o.tile) <= r for u in enemies()):
            return 84
        if k == AB_AOE and gems() == 0:
            if count(dist(u.pos, o.tile) <= r and is_awake(u) for u in camps()) >= 2:
                return 63
    return 0
```
