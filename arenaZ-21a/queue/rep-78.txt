```python
@gene('s1', 'value of doing this option now: default draft, casts in fights, focus the weakest target, recall to shop when rich and idle, recall when low only where it will not be cancelled')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90 if safe() else 0
        return 55 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            return 80 + 5 * w
        if is_awake(o.unit):
            return 61 + w
        return 60 + w if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "worth recalling to spend gold: at least 150 gold, carrying nothing, no enemy or awake camp near", parents=["s1"])
def shop_trip() -> Bool:
    if gold() < 150 or gems() > 0:
        return False
    if any(dist(u.pos, me().pos) <= 8 for u in enemies()):
        return False
    return not any(is_awake(c) and dist(c.pos, me().pos) <= 5 for c in camps())

@gene("new", "no awake camp can reach me next turn and no enemy champion within 5: a recall here will not be cancelled", parents=["s1"])
def safe() -> Bool:
    if any(dist(u.pos, me().pos) <= 5 for u in enemies()):
        return False
    return not any(is_awake(c) and dist(c.pos, me().pos) <= c.attack_range + 1 for c in camps())
```
