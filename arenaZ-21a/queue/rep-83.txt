```python
@gene('s1', 'value of doing this option now: casts, fight back against awake camps; recall when low only where it will not be cancelled')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 and safe() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        if gems() == 0:
            return 60
        return 58 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "am I the visible ally closest to these tiles (ties count as closest)", parents=["s2"])
def closest_ally_to(ts: Tiles) -> Bool:
    d = dist(nearest(ts, me().pos), me().pos)
    return not any(dist(nearest(ts, a.pos), a.pos) < d for a in allies())

@gene('s2', 'where to walk: an altar with gems, else a living camp group; a cleared group waiting for capture only if I am the closest ally, so the others move on and the reward is not split')
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if not group_reward(g) or closest_ally_to(ts):
                acc = acc | ts
    return acc

@gene('s3', 'distance field to the goals, or home to heal when hurt, empty-handed and not in a fight')
def goal_field() -> Field:
    if gems() == 0 and me().health_frac < 0.5 and not any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()):
        return field(my_spawn())
    return field(goals())
```
