@gene('s1', 'value of doing this option now: default draft, casts in fights, focus the weakest target, recall to shop when rich and idle, recall when low only where it will not be cancelled, else walk away and only finish off nearly dead champions')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90 if safe() else 0
        return 55 if shop_trip() else 0
    if o.kind == CAST:
        if fleeing() and exists(o.unit) and o.unit.is_champion and o.unit.health_frac > 0.25:
            return 0
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            if fleeing() and o.unit.health_frac > 0.25:
                return 0
            return 80 + 5 * w
        if is_awake(o.unit):
            return 61 + w
        return 60 + w if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if fleeing():
            return 75 if at(flee_field(), me().pos) - at(flee_field(), o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "low on health with an awake camp or enemy champion close: walk home instead of fighting", parents=["s1"])
def fleeing() -> Bool:
    if me().health_frac >= 0.3:
        return False
    if any(dist(u.pos, me().pos) <= 4 for u in enemies()):
        return True
    return any(is_awake(c) and dist(c.pos, me().pos) <= c.attack_range + 1 for c in camps())

@gene("new", "distance field to our spawn, for fleeing", parents=["s1"])
def flee_field() -> Field:
    return field(my_spawn())
