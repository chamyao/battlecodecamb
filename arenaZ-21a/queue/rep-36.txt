```python
@gene('s1', 'value of doing this option now, including ability casts and fighting back against awake camps')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        if gems() == 0:
            return 60
        return 58 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "walk home to heal: hurt, carrying nothing and not in a fight", parents=["s4"])
def go_home() -> Bool:
    return gems() == 0 and me().health_frac < 0.5 and not any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps())

@gene("new", "distance field home", parents=["s4"])
def home_field() -> Field:
    return field(my_spawn())

@gene("s4", "how much closer a step brings me to the goals, or home when hurt and empty-handed")
def progress(t: Tile) -> Num:
    if go_home():
        return at(home_field(), me().pos) - at(home_field(), t)
    return at(goal_field(), me().pos) - at(goal_field(), t)
```
