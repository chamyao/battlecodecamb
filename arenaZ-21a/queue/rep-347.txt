@gene("new", "a cleared group waiting for capture is within 2 tiles (or under me): stay and finish the 5-round capture before walking home", parents=["s3"])
def capturing() -> Bool:
    for g in range(num_groups()):
        if group_reward(g):
            ts = group_tiles(g)
            if dist(nearest(ts, me().pos), me().pos) <= 2:
                return True
    return False

@gene('s3', 'distance field to the goals plus enemy champions worth hunting, or home to heal when hurt (above a quarter health only once no capture is pending nearby), empty-handed and not in a fight')
def goal_field() -> Field:
    if gems() == 0 and me().health_frac < 0.5 and not any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()):
        if me().health_frac < 0.25 or not capturing():
            return field(my_spawn())
    return field(open_goals() | hunt_tiles())

@gene('Z43a', 'where to walk: an altar with gems, else any living camp group (a cleared group only if I am the closest visible ally), plus enemy gem carriers (15 or more) within 8 when I am healthy and we are not outnumbered', parents=['s3'])
def open_goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g):
                d = dist(nearest(ts, me().pos), me().pos)
                if not any(dist(nearest(ts, a.pos), a.pos) < d for a in allies()):
                    acc = acc | ts
            else:
                acc = acc | ts
    if me().health_frac >= 0.6:
        foes = count(dist(u.pos, me().pos) <= 8 for u in enemies())
        friends = 1 + count(dist(a.pos, me().pos) <= 8 for a in allies())
        if foes > 0 and friends >= foes:
            for u in enemies():
                if unit_gems(u) >= 15 and dist(u.pos, me().pos) <= 8:
                    acc = acc | near(u.pos, 1)
    return acc
