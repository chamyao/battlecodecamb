```python
@gene("new", "visible enemy champions worth hunting: badly hurt ones within 6 while I am healthy, or all within 8 when my visible allies (me included) outnumber them", parents=["s3"])
def hunt_tiles() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if me().health_frac < 0.5:
        return acc
    foes = count(dist(u.pos, me().pos) <= 8 for u in enemies())
    if foes == 0:
        return acc
    friends = 1 + count(dist(a.pos, me().pos) <= 8 for a in allies())
    for u in enemies():
        if dist(u.pos, me().pos) <= 8:
            if friends > foes and me().health_frac >= 0.6:
                acc = acc | near(u.pos, 1)
            elif u.health_frac < 0.35 and dist(u.pos, me().pos) <= 6:
                acc = acc | near(u.pos, 1)
    return acc

@gene('s3', 'distance field to the goals plus enemy champions worth hunting, or home to heal when hurt, empty-handed and not in a fight')
def goal_field() -> Field:
    if gems() == 0 and me().health_frac < 0.5 and not any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()):
        return field(my_spawn())
    return field(open_goals() | hunt_tiles())

@gene('Z43a', 'where to walk: an altar with gems, else any living camp group; a cleared group waiting for capture only if I am the closest visible ally', parents=['s3'])
def open_goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g):
                d = dist(nearest(ts, me().pos), me().pos)
                if not any(dist(nearest(ts, a.pos), a.pos) < d for a in allies()):
                    acc = acc | ts
            else:
                acc = acc | ts
    return acc
```
