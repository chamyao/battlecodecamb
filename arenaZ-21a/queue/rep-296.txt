```python
@gene("new", "low on health with an awake camp or enemy champion close: walk or dash home instead of fighting", parents=["s4"])
def fleeing() -> Bool:
    if me().health_frac >= 0.3:
        return False
    if any(dist(u.pos, me().pos) <= 4 for u in enemies()):
        return True
    return any(is_awake(c) and dist(c.pos, me().pos) <= c.attack_range + 1 for c in camps())

@gene("new", "distance field to our spawn", parents=["s4"])
def spawn_field() -> Field:
    return field(my_spawn())

@gene("s4", "how much closer a step brings me to the goals, or to spawn when fleeing")
def progress(t: Tile) -> Num:
    if fleeing():
        return at(spawn_field(), me().pos) - at(spawn_field(), t)
    return at(goal_field(), me().pos) - at(goal_field(), t)
```
