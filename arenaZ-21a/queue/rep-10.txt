@gene("s2", "where to walk: an altar with gems, else a living camp group that carries gems")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    rest = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            if group_gems(g) > 0:
                acc = acc | group_tiles(g)
            else:
                rest = rest | group_tiles(g)
    if size(acc) == 0:
        return rest
    return acc
