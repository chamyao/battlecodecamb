@gene('Z284b', 'an ally fight announced in the last 3 rounds within 12 tiles, worth joining when I am at 60 percent health or more and empty-handed', parents=['s3'])
def rally_tiles() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0 or me().health_frac < 0.6:
        return acc
    age = round() - 256 * floor(round() / 256) - read_shared(2)
    if age < 0:
        age += 256
    if age > 3:
        return acc
    t = tile(read_shared(0), read_shared(1))
    if valid(t) and dist(t, me().pos) <= 12 and dist(t, me().pos) > 1:
        acc = near(t, 1)
    return acc
