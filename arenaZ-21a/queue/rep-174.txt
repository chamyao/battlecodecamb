@gene('s1', 'value of doing this option now: draft Ryth, Vesper Vale, Vendetta, Tavi, Varg; casts in fights, focus the weakest target and gem carriers, recall to shop when rich and idle, recall when low only where it will not be cancelled, else walk away and only finish off nearly dead champions')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return tavi_fourth(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90 if safe() else 0
        return 55 if shop_trip() else 0
    if o.kind == CAST:
        if fleeing() and exists(o.unit) and o.unit.is_champion and o.unit.health_frac > 0.25:
            return 0
        if exists(o.unit) and o.unit.is_champion:
            return cast_value(o) + min(unit_gems(o.unit), 50) / 10
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            if fleeing() and o.unit.health_frac > 0.25:
                return 0
            return 80 + 5 * w + min(unit_gems(o.unit), 50) / 10
        if is_awake(o.unit):
            return 61 + w
        return 60 + w if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if fleeing():
            return 75 if at(flee_field(), me().pos) - at(flee_field(), o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "draft preference: Ryth, Vesper Vale, Vendetta, then Tavi (shards anywhere, stun) and Varg", parents=["s1"])
def tavi_fourth(c: Num) -> Num:
    if c == RYTH:
        return 10
    if c == VESPER_VALE:
        return 9
    if c == VENDETTA:
        return 8
    if c == TAVI:
        return 7
    if c == VARG:
        return 6
    return 1

@gene('Z74a', 'no awake camp can reach me next turn and no enemy champion within 6: a recall here will not be cancelled', parents=['s1'])
def safe() -> Bool:
    if any(dist(u.pos, me().pos) <= 6 for u in enemies()):
        return False
    return not any(is_awake(c) and dist(c.pos, me().pos) <= c.attack_range + 1 for c in camps())
