@gene('s1', 'value of doing this option now: casts, fight back against awake camps, focus the weakest target; below 35 percent health, recall only where it is safe, else walk away first and only finish off nearly dead champions; recall to shop when rich and idle')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    low = me().health_frac < 0.35
    if o.kind == RECALL:
        if low:
            return 90 if safe() else 0
        return 55 if gems() == 0 and gold() >= 150 and safe() else 0
    if o.kind == MOVE:
        if low and not safe():
            return 75 if at(retreat_field(), me().pos) - at(retreat_field(), o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    if o.kind == CAST:
        if low and exists(o.unit) and o.unit.is_champion and o.unit.health_frac > 0.25:
            return 0
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            if low and o.unit.health_frac > 0.25:
                return 0
            return 80 + 5 * w
        if gems() == 0:
            return 60 + 1.5 * w
        return 58 + 1.5 * w if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    return 0

@gene("new", "tiles holding enemy champions or camps we fight (awake, or any when carrying nothing), built once per pass", parents=["Z4b"])
def hot_tiles() -> Tiles:
    if gems() == 0:
        return enemy_tiles() | camp_tiles()
    acc = enemy_tiles()
    for c in camps():
        if is_awake(c):
            acc = acc | near(c.pos, 0)
    return acc

@gene('Z4b', 'number of enemy champions and camps we fight within radius one of a tile, from the per-pass hot tile set (cheap per option)', parents=['s1'])
def hits_at(t: Tile) -> Num:
    return size(near(t, 1) & hot_tiles())

@gene("new", "skip option kinds that cannot score: wards, locks, cancels always; casts unless a target is within 6 or a dash is ready", parents=["s1"])
def consider(kind: Num) -> Bool:
    if kind == WARD or kind == LOCK or kind == BUY_LOCK or kind == CANCEL:
        return False
    if kind == CAST:
        if (slot_kind(0) == AB_DASH and slot_ready(0)) or (slot_kind(1) == AB_DASH and slot_ready(1)):
            return True
        if any(dist(u.pos, me().pos) <= 6 for u in enemies()):
            return True
        return any(dist(c.pos, me().pos) <= 6 and (is_awake(c) or gems() == 0) for c in camps())
    return True
