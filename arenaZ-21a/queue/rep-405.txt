@gene('s3', 'distance field to the goals plus enemy champions worth hunting, or home to heal when hurt (above a quarter health only once no capture is pending nearby) or when rich (120 gold) with spawn close (no capture pending), empty-handed and not in a fight')
def goal_field() -> Field:
    if gems() == 0 and not any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()):
        if me().health_frac < 0.5 and (me().health_frac < 0.25 or not capturing()):
            return field(my_spawn())
        if gold() >= 120 and dist(nearest(my_spawn(), me().pos), me().pos) <= 12 and not capturing():
            return field(my_spawn())
    return field(open_goals() | hunt_tiles())
