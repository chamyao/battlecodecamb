```python
@gene('Z106a', 'below 35 percent health with an awake camp or enemy champion within 5: walk or dash home instead of fighting', parents=['s1'])
def fleeing() -> Bool:
    if me().health_frac >= 0.35:
        return False
    if any(dist(u.pos, me().pos) <= 5 for u in enemies()):
        return True
    return any(is_awake(c) and dist(c.pos, me().pos) <= c.attack_range + 1 for c in camps())
```
