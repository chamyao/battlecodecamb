@gene("Z73a", "no awake camp can reach me next turn and no enemy champion within 6: a recall here will not be cancelled", parents=["s1"])
def safe() -> Bool:

    return not any(is_awake(c) and dist(c.pos, me().pos) <= c.attack_range + 1 for c in camps())
