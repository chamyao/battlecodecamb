@gene('Z175a', 'visible enemy champions worth hunting: badly hurt ones within 6 while I am healthy, gem carriers (15 or more) within 8 when we are not outnumbered, or all within 8 when my visible allies (me included) outnumber them', parents=['Z18a'])
def hunt_tiles() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if me().health_frac < 0.5:
        return acc
    foes = count(dist(u.pos, me().pos) <= 8 for u in enemies())
    if foes == 0:
        return acc
    friends = 1 + count(dist(a.pos, me().pos) <= 8 for a in allies())
    for u in enemies():
        if dist(u.pos, me().pos) <= 8:
            if friends > foes and me().health_frac >= 0.6:
                acc = acc | near(u.pos, 1)
            elif friends >= foes and unit_gems(u) >= 15 and me().health_frac >= 0.6:
                acc = acc | near(u.pos, 1)
            elif u.health_frac < 0.35 and dist(u.pos, me().pos) <= 6:
                acc = acc | near(u.pos, 1)
    return acc
