```python
@gene("new", "camp groups to walk to: every living group, but a cleared group waiting for capture only if I am the closest visible ally, so the others move on and the reward is not split", parents=["s3"])
def carrier_goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g):
                d = dist(nearest(ts, me().pos), me().pos)
                if not any(dist(nearest(ts, a.pos), a.pos) < d for a in allies()):
                    acc = acc | ts
            else:
                acc = acc | ts
    return acc

@gene('s3', 'distance field to the goals (one capturer per cleared group), or home to heal when hurt, empty-handed and not in a fight')
def goal_field() -> Field:
    if gems() == 0 and me().health_frac < 0.5 and not any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()):
        return field(my_spawn())
    return field(carrier_goals())
```
