@gene('s1', 'value of doing this option now: draft Ryth, Vesper Vale, Vendetta, Varg, Tavi; casts, fight back against awake camps, focus the weakest target; below 35 percent health, recall only where it is safe, else walk away first and only finish off nearly dead champions; recall to shop when rich and idle')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    low = me().health_frac < 0.35
    if o.kind == RECALL:
        if low:
            return 90 if safe() else 0
        return 55 if gems() == 0 and gold() >= 150 and safe() else 0
    if o.kind == MOVE:
        if low and not safe():
            return 75 if at(retreat_field(), me().pos) - at(retreat_field(), o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    if o.kind == CAST:
        if low and exists(o.unit) and o.unit.is_champion and o.unit.health_frac > 0.25:
            return 0
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            if low and o.unit.health_frac > 0.25:
                return 0
            return 80 + 5 * w
        if gems() == 0:
            return 60 + 1.5 * w
        return 58 + 1.5 * w if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    return 0

@gene("new", "draft preference: the default first three, then Varg and Tavi instead of the linked pair", parents=["s1"])
def pick_value(c: Num) -> Num:
    if c == RYTH:
        return 10
    if c == VESPER_VALE:
        return 9
    if c == VENDETTA:
        return 8
    if c == VARG:
        return 7
    if c == TAVI:
        return 6
    return 1

@gene("s2", "where to walk: an altar with gems, else a living camp group, plus any nearby badly hurt enemy champion while I am healthy")
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if me().health_frac > 0.5:
        for u in enemies():
            if u.health_frac < 0.35 and dist(u.pos, me().pos) <= 6:
                acc = acc | near(u.pos, 1)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
