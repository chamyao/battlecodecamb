@gene('s1', 'value of doing this option now: buy Monster strength shards (item 6 if none), default draft, casts in fights, focus the weakest target, recall to shop when rich and idle, recall when low only where it will not be cancelled, else walk away')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if item_passive(o.item) == P_MONSTER and item_strength(o.item) > 0:
            return 101
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90 if safe() else 0
        return 55 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            return 80 + 5 * w
        if is_awake(o.unit):
            return 61 + w
        return 60 + w if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if fleeing():
            return 75 if at(flee_field(), me().pos) - at(flee_field(), o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
