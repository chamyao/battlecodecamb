@gene('s1', 'value of doing this option now, including ability casts and fighting back against awake camps')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        if gems() == 0:
            return 60
        return 58 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z7a', 'value of casting an ability now: damage on champions and camps we fight, shields in fights, resets with a target, dashes toward goals', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC or k == AB_AOE:
        if exists(o.unit):
            if o.unit.is_champion:
                return 85
            if o.unit.is_camp and (is_awake(o.unit) or gems() == 0):
                return 65
            return 0
        if valid(o.tile):
            h = count(dist(u.pos, o.tile) <= 1 for u in enemies()) + count(dist(c.pos, o.tile) <= 1 and (is_awake(c) or gems() == 0) for c in camps())
            return 64 + h if h > 0 else 0
        return 0
    if k == AB_SHIELD:
        return 66 if any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()) or any(dist(u.pos, me().pos) <= 3 for u in enemies()) else 0
    if k == AB_RESET:
        if can_attack():
            return 0
        r = me().attack_range
        return 62 if any(dist(u.pos, me().pos) <= r for u in enemies()) or any(dist(c.pos, me().pos) <= r and (is_awake(c) or gems() == 0) for c in camps()) else 0
    if k == AB_HEAL:
        return 70 if exists(o.unit) and o.unit.health_frac < 0.6 else 0
    if k == AB_DASH:
        return 51 if valid(o.tile) and progress(o.tile) >= 2 else 0
    return 0
