```python
@gene("new", "still farming camps: carrying under 15 gems, healthy, and not near the end", parents=["s1"])
def hunting() -> Bool:
    return gems() < 15 and me().health_frac >= 0.5 and round() < 1850

@gene('s1', 'value of doing this option now: bank up to 15 gems before tributing, casts in fights')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if hunting() or is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z7a', 'value of casting an ability now: damage on champions and camps we fight, shields and resets in fights', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC or k == AB_AOE:
        if exists(o.unit):
            if o.unit.is_champion:
                return 85
            if o.unit.is_camp and (is_awake(o.unit) or hunting()):
                return 65
            return 0
        if valid(o.tile):
            h = count(dist(u.pos, o.tile) <= 1 for u in enemies()) + count(dist(c.pos, o.tile) <= 1 and is_awake(c) for c in camps())
            return 64 + h if h > 0 else 0
        return 0
    if k == AB_SHIELD:
        return 66 if any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()) else 0
    if k == AB_RESET:
        return 62 if not can_attack() else 0
    if k == AB_HEAL:
        return 70 if exists(o.unit) and o.unit.health_frac < 0.6 else 0
    return 0

@gene("s2", "where to walk: an altar once the bag is full enough, else a living camp group")
def goals() -> Tiles:
    if gems() > 0 and not hunting():
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
```
