@gene("new", "where to walk: an altar with gems; else the living camp group with the best gems and gold per tile of distance, plus any group already within 3 tiles", parents=["Z18a"])
def value_goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    best = -1
    bg = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            d = dist(nearest(ts, me().pos), me().pos)
            if d <= 3:
                acc = acc | ts
            v = (group_gems(g) + group_gold(g) / 8 + 2) / (d + 6)
            if v > best:
                best = v
                bg = ts
    return acc | bg

@gene('Z18a', 'once per turn: the distance field to the best-value camp group (or altar) plus badly hurt enemy champions nearby', parents=['s3'])
def update() -> None:
    GF = field(value_goals() | prey_tiles())
