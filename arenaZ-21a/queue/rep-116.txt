```python
@gene("s2", "where to walk: an altar with gems, else a living camp group; a cleared group waiting for capture only if I am the closest visible ally, so the others move on and the reward is not split")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g):
                d = dist(nearest(ts, me().pos), me().pos)
                if not any(dist(nearest(ts, a.pos), a.pos) < d for a in allies()):
                    acc = acc | ts
            else:
                acc = acc | ts
    return acc
```
