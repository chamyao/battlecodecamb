```python
@gene('s1', 'value of doing this option now: draft Ryth, Vesper Vale, Vendetta, Varg, Tavi; casts, fight back, focus the weakest target, wake no new camp below half health, recall to shop when rich and idle; when low, recall only where it is safe, else walk away first and only finish off nearly dead champions')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    hf = me().health_frac
    low = hf < 0.35
    if o.kind == RECALL:
        if low:
            return 90 if safe() else 0
        return 55 if gems() == 0 and gold() >= 150 and safe() else 0
    if o.kind == MOVE:
        if low and not safe():
            return 75 if at(retreat_field(), me().pos) - at(retreat_field(), o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    if o.kind == CAST:
        if exists(o.unit):
            if low and o.unit.is_champion and o.unit.health_frac > 0.25:
                return 0
            if hf < 0.5 and o.unit.is_camp and not is_awake(o.unit):
                return 0
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            if low and o.unit.health_frac > 0.25:
                return 0
            return 80 + 5 * w
        if is_awake(o.unit):
            return 58 + 1.5 * w if gems() > 0 else 60 + 1.5 * w
        return 60 + 1.5 * w if gems() == 0 and hf >= 0.5 else 0
    if o.kind == TRIBUTE:
        return 70
    return 0
```
