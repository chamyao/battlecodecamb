@gene("s2", "where to walk: an altar with gems, else a living camp group, plus any nearby badly hurt enemy champion")
def goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for u in enemies():
        if u.health_frac < 0.4 and dist(u.pos, me().pos) <= 6 and me().health_frac > 0.5:
            acc = acc | near(u.pos, 1)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
