@gene('Z71a', 'no awake camp within its range plus 2 and no enemy champion within 7: a recall here will not be cancelled', parents=['s1'])
def safe() -> Bool:
    if any(dist(u.pos, me().pos) <= 7 for u in enemies()):
        return False
    return not any(is_awake(c) and dist(c.pos, me().pos) <= c.attack_range + 2 for c in camps())
