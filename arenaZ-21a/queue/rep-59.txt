```python
@gene('s1', 'value of doing this option now: draft Ryth, Vesper Vale, Vendetta then Varg and Tavi; casts, fight back, focus the weakest target, recall to shop when rich and idle')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        return 55 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            return 80 + 5 * w
        if gems() == 0:
            return 60 + 1.5 * w
        return 58 + 1.5 * w if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
