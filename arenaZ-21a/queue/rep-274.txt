@gene("new", "after my actions: announce the tile I just attacked and the round, so allies can join the fight", parents=["Z18a"])
def after() -> None:
    if acted(ATTACK):
        t = acted_tile(ATTACK)
        if valid(t):
            write_shared(0, t.x)
            write_shared(1, t.y)
            write_shared(2, round() - 256 * floor(round() / 256))

@gene("new", "an ally fight announced in the last 3 rounds within 12 tiles, worth joining when I am healthy and empty-handed", parents=["Z18a"])
def rally_tiles() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0 or me().health_frac < 0.6:
        return acc
    age = round() - 256 * floor(round() / 256) - read_shared(2)
    if age < 0:
        age += 256
    if age > 3:
        return acc
    t = tile(read_shared(0), read_shared(1))
    if valid(t) and dist(t, me().pos) <= 12 and dist(t, me().pos) > 1:
        acc = near(t, 1)
    return acc

@gene('Z18a', 'once per turn: the distance field to a fight an ally just announced nearby, else to the goals plus enemy champions worth hunting', parents=['s3'])
def update() -> None:
    r = rally_tiles()
    if size(r) > 0:
        GF = field(r)
    else:
        GF = field(goals() | hunt_tiles())

@gene("new", "a cleared group waiting for capture is within 2 tiles (or under me): stay and finish the 5-round capture before walking home", parents=["s3"])
def capturing() -> Bool:
    for g in range(num_groups()):
        if group_reward(g):
            ts = group_tiles(g)
            if dist(nearest(ts, me().pos), me().pos) <= 2:
                return True
    return False

@gene('s3', 'distance field to the goals (computed once per turn in update), or home when hurt (above a quarter health only once no capture is pending nearby) or rich (and spawn is close), empty-handed and not in a fight')
def goal_field() -> Field:
    if gems() == 0 and not any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()):
        if me().health_frac < 0.5 and (me().health_frac < 0.25 or not capturing()):
            return field(my_spawn())
        if gold() >= 150 and dist(nearest(my_spawn(), me().pos), me().pos) <= 12 and not capturing():
            return field(my_spawn())
    return GF
