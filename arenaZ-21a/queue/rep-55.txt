```python
@gene('s1', 'value of doing this option now: draft Ryth, Vesper Vale, Vendetta, Varg, Tavi; casts, fight back, focus the weakest target, recall to shop when rich and idle')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        return 55 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            return 80 + 5 * w
        if gems() == 0:
            return 60 + 1.5 * w
        return 58 + 1.5 * w if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "draft preference: the default first three, then Varg and Tavi instead of the linked pair", parents=["s1"])
def pick_value(c: Num) -> Num:
    if c == RYTH:
        return 10
    if c == VESPER_VALE:
        return 9
    if c == VENDETTA:
        return 8
    if c == VARG:
        return 7
    if c == TAVI:
        return 6
    return 1

@gene('Z7a', 'value of casting an ability now: damage on the weakest champion or camp we fight, shields in fights, resets with a target, Heart of Ruin dash onto targets, dashes toward goals', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC or k == AB_AOE:
        if exists(o.unit):
            if o.unit.is_champion:
                return 85 + 3 * (1 - o.unit.health_frac)
            if o.unit.is_camp and (is_awake(o.unit) or gems() == 0):
                return 64 + (1 - o.unit.health_frac)
            return 0
        if valid(o.tile):
            h = count(dist(u.pos, o.tile) <= 1 for u in enemies()) + count(dist(c.pos, o.tile) <= 1 and (is_awake(c) or gems() == 0) for c in camps())
            return 64 + h if h > 0 else 0
        return 0
    if k == AB_SHIELD:
        return 66 if any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()) or any(dist(u.pos, me().pos) <= 3 for u in enemies()) else 0
    if k == AB_RESET:
        if can_attack():
            return 0
        r = me().attack_range
        return 62 if any(dist(u.pos, me().pos) <= r for u in enemies()) or any(dist(c.pos, me().pos) <= r and (is_awake(c) or gems() == 0) for c in camps()) else 0
    if k == AB_HEAL:
        return 70 if exists(o.unit) and o.unit.health_frac < 0.6 else 0
    if k == AB_DASH:
        if not valid(o.tile):
            return 0
        if me().type == HEART_OF_RUIN:
            h = count(dist(u.pos, o.tile) <= 1 for u in enemies()) + count(dist(c.pos, o.tile) <= 1 and (is_awake(c) or gems() == 0) for c in camps())
            if h > 0:
                return 66 + h
        return 51 if progress(o.tile) >= 2 else 0
    return 0
```
