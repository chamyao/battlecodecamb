@gene('s1', 'value of doing this option now: draft Ryth, Vesper Vale, Vendetta then Varg and Tavi; casts, fight back, focus the weakest target; when low, recall only where it is safe, else walk away first')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    low = me().health_frac < 0.35
    if o.kind == RECALL:
        if low:
            return 90 if safe() else 0
        return 55 if shop_trip() else 0
    if o.kind == MOVE:
        if low and not safe():
            return 75 if at(retreat_field(), me().pos) - at(retreat_field(), o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        w = 0.7787 - o.unit.health_frac
        if o.unit.is_champion:
            return 80 + 5 * w
        if gems() == 0:
            return 60 + 1.5 * w
        return 58 + 1.5 * w if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    return 0

@gene("new", "no awake camp can reach me next turn and no enemy champion within 5: a recall here will not be cancelled", parents=["s1"])
def safe() -> Bool:
    if any(dist(u.pos, me().pos) <= 5 for u in enemies()):
        return False
    return not any(is_awake(c) and dist(c.pos, me().pos) <= c.attack_range + 1 for c in camps())

@gene("new", "distance field to our spawn, for retreating", parents=["s1"])
def retreat_field() -> Field:
    return field(my_spawn())
