@gene("new", "tiles of cleared groups waiting for capture within 2 tiles of me", parents=["s4"])
def capture_tiles() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g):
            ts = group_tiles(g)
            if dist(nearest(ts, me().pos), me().pos) <= 2:
                acc = acc | ts
    return acc

@gene("new", "distance field to a capture pending right next to me", parents=["s4"])
def capture_field() -> Field:
    return field(capture_tiles())

@gene('s4', 'how much closer a step brings me to a fight an ally just announced nearby; else onto a capture pending right next to me (unless nearly dead or carrying gems); else to the goals')
def progress(t: Tile) -> Num:
    if size(rally_tiles()) > 0:
        return at(rally_field(), me().pos) - at(rally_field(), t)
    if gems() == 0 and me().health_frac >= 0.25 and size(capture_tiles()) > 0:
        return at(capture_field(), me().pos) - at(capture_field(), t)
    return at(goal_field(), me().pos) - at(goal_field(), t)
