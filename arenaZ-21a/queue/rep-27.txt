@gene("new", "where to walk without filtering: an altar with gems, else any living camp group", parents=["s3"])
def open_goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc

@gene("s3", "distance field to the unfiltered goals")
def goal_field() -> Field:
    return field(open_goals())
