```python
@gene('s1', 'value of doing this option now: draft Ryth, Vesper Vale, Vendetta, Varg, Tavi; casts, fight back, focus the weakest target, recall to shop when rich and idle; when low, recall only where it is safe, else walk away first and only finish off nearly dead champions')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    low = me().health_frac < 0.35
    if o.kind == RECALL:
        if low:
            return 90 if safe() else 0
        return 55 if shop_trip() else 0
    if o.kind == MOVE:
        if low and not safe():
            return 75 if at(retreat_field(), me().pos) - at(retreat_field(), o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    if o.kind == CAST:
        if low and exists(o.unit) and o.unit.is_champion and o.unit.health_frac > 0.25:
            return 0
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            if low and o.unit.health_frac > 0.25:
                return 0
            return 80 + 5 * w
        if gems() == 0:
            return 60 + 1.5 * w
        return 58 + 1.5 * w if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    return 0

@gene("new", "tiles holding enemy champions or camps we fight (awake, or any when carrying nothing), built once per pass", parents=["Z7a"])
def hot_tiles() -> Tiles:
    if gems() == 0:
        return enemy_tiles() | camp_tiles()
    acc = enemy_tiles()
    for c in camps():
        if is_awake(c):
            acc = acc | near(c.pos, 0)
    return acc

@gene("new", "number of hot targets within radius one of a tile (cheap per option)", parents=["Z7a"])
def hot_hits(t: Tile) -> Num:
    return size(near(t, 1) & hot_tiles())

@gene('Z7a', 'value of casting an ability now: burst the weakest champion or camp we fight, dash away when fleeing, Heart of Ruin dash onto targets, shield in a fight; area counts from the per-pass hot tiles', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC or k == AB_AOE:
        if exists(o.unit):
            if o.unit.is_champion:
                return 85 + 3 * (1 - o.unit.health_frac)
            if o.unit.is_camp and (is_awake(o.unit) or gems() == 0):
                return 64 + (1 - o.unit.health_frac)
            return 0
        if valid(o.tile):
            h = hot_hits(o.tile)
            return 64 + h if h > 0 else 0
        return 0
    if k == AB_SHIELD:
        return 66 if hot_hits(me().pos) > 0 or any(dist(u.pos, me().pos) <= 3 for u in enemies()) else 0
    if k == AB_RESET:
        if can_attack():
            return 0
        r = me().attack_range
        return 62 if any(dist(u.pos, me().pos) <= r for u in enemies()) or any(dist(c.pos, me().pos) <= r and (is_awake(c) or gems() == 0) for c in camps()) else 0
    if k == AB_HEAL:
        return 70 if exists(o.unit) and o.unit.health_frac < 0.6 else 0
    if k == AB_DASH:
        if not valid(o.tile):
            return 0
        if me().health_frac < 0.3 and not safe():
            return 80 if at(retreat_field(), me().pos) - at(retreat_field(), o.tile) >= 2 else 0
        if me().type == HEART_OF_RUIN and hot_hits(o.tile) > 0:
            return 67
        return 51 if progress(o.tile) >= 2 else 0
    return 0

@gene("new", "skip option kinds that cannot score: wards, locks, cancels always; casts unless a target is within 6 or a dash is ready", parents=["s1"])
def consider(kind: Num) -> Bool:
    if kind == WARD or kind == LOCK or kind == BUY_LOCK or kind == CANCEL:
        return False
    if kind == CAST:
        if (slot_kind(0) == AB_DASH and slot_ready(0)) or (slot_kind(1) == AB_DASH and slot_ready(1)):
            return True
        if any(dist(u.pos, me().pos) <= 6 for u in enemies()):
            return True
        return any(dist(c.pos, me().pos) <= 6 and (is_awake(c) or gems() == 0) for c in camps())
    return True
```
