@gene("s2", "where to walk: an altar with gems, else a living camp group that is not a monolith (golems only late)")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            bad = any(spawn_type(t) == MONOLITH for t in ts)
            if round() < 800:
                bad = bad or any(spawn_type(t) == GOLEM for t in ts)
            if not bad:
                acc = acc | ts
    return acc
