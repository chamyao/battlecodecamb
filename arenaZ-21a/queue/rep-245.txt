"""The official starter bot as a genome (sim/src/bots.cpp), for checking the runtime: it should play
about even with builtin:starter. Starter rules in priority order become scores."""

@gene('s1', 'value of doing this option now: draft Ryth, Vesper Vale, Vendetta, Tavi, Varg; casts in fights, focus the weakest target and gem carriers, recall to shop when rich and idle, recall when low only where it will not be cancelled, else walk away and only finish off nearly dead champions')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return tavi_fourth(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90 if safe() else 0
        return 55 if shop_trip() else 0
    if o.kind == CAST:
        if fleeing() and exists(o.unit) and o.unit.is_champion and o.unit.health_frac > 0.25:
            return 0
        if exists(o.unit) and o.unit.is_champion:
            return cast_value(o) + min(unit_gems(o.unit), 50) / 10
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            if fleeing() and o.unit.health_frac > 0.25:
                return 0
            return 80 + 5 * w + min(unit_gems(o.unit), 50) / 10
        if is_awake(o.unit):
            return 61 + w
        return 60 + w if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if fleeing():
            return 75 if at(flee_field(), me().pos) - at(flee_field(), o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('s3', 'distance field home to heal when hurt, empty-handed and not in a fight; else to a fight an ally just announced nearby; else to the goals plus enemy champions worth hunting')
def goal_field() -> Field:
    if gems() == 0 and me().health_frac < 0.5 and not any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()):
        return field(my_spawn())
    r = rally_tiles()
    if size(r) > 0:
        return field(r)
    return field(open_goals() | hunt_tiles())

@gene("s4", "how much closer a step brings me to the goals")
def progress(t: Tile) -> Num:
    return at(goal_field(), me().pos) - at(goal_field(), t)

@gene('Z43a', 'where to walk: an altar with gems, else any living camp group; a cleared group waiting for capture only if I am the closest visible ally', parents=['s3'])
def open_goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g):
                d = dist(nearest(ts, me().pos), me().pos)
                if not any(dist(nearest(ts, a.pos), a.pos) < d for a in allies()):
                    acc = acc | ts
            else:
                acc = acc | ts
    return acc

@gene('Z208a', 'visible enemy champions worth hunting: badly hurt ones within 6 while I am healthy, or all within 8 when we are at least as many and I am above 70 percent health', parents=['s3'])
def hunt_tiles() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if me().health_frac < 0.5:
        return acc
    foes = count(dist(u.pos, me().pos) <= 8 for u in enemies())
    if foes == 0:
        return acc
    friends = 1 + count(dist(a.pos, me().pos) <= 8 for a in allies())
    for u in enemies():
        if dist(u.pos, me().pos) <= 8:
            if friends >= foes and me().health_frac >= 0.7:
                acc = acc | near(u.pos, 1)
            elif u.health_frac < 0.35 and dist(u.pos, me().pos) <= 6:
                acc = acc | near(u.pos, 1)
    return acc

@gene('Z52a', 'worth recalling to spend gold: at least 150 gold, carrying nothing, no enemy or awake camp near', parents=['s1'])
def shop_trip() -> Bool:
    if gold() < 150 or gems() > 0:
        return False
    if any(dist(u.pos, me().pos) <= 8 for u in enemies()):
        return False
    return not any(is_awake(c) and dist(c.pos, me().pos) <= 5 for c in camps())

@gene('Z74a', 'no awake camp can reach me next turn and no enemy champion within 6: a recall here will not be cancelled', parents=['s1'])
def safe() -> Bool:
    if any(dist(u.pos, me().pos) <= 6 for u in enemies()):
        return False
    return not any(is_awake(c) and dist(c.pos, me().pos) <= c.attack_range + 1 for c in camps())

@gene('Z197a', 'draft preference: Ryth, Vesper Vale, Vendetta, then Tavi (shards anywhere, stun) and Varg', parents=['s1'])
def tavi_fourth(c: Num) -> Num:
    if c == RYTH:
        return 10
    if c == VESPER_VALE:
        return 9
    if c == VENDETTA:
        return 8
    if c == TAVI:
        return 7
    if c == VARG:
        return 6
    return 1

@gene('Z28a', 'value of casting an ability now: damage on the weakest champion or camp we fight, shields in fights, resets with a target, dash away when fleeing, Heart of Ruin dash onto targets, dashes toward goals', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC or k == AB_AOE:
        if exists(o.unit):
            if o.unit.is_champion:
                return 85 + 3 * (1 - o.unit.health_frac)
            if o.unit.is_camp and (is_awake(o.unit) or gems() == 0):
                return 64 + (1 - o.unit.health_frac)
            return 0
        if valid(o.tile):
            h = count(dist(u.pos, o.tile) <= 1 for u in enemies()) + count(dist(c.pos, o.tile) <= 1 and (is_awake(c) or gems() == 0) for c in camps())
            return 64 + h if h > 0 else 0
        return 0
    if k == AB_SHIELD:
        return 66 if any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()) or any(dist(u.pos, me().pos) <= 3 for u in enemies()) else 0
    if k == AB_RESET:
        if can_attack():
            return 0
        r = me().attack_range
        return 62 if any(dist(u.pos, me().pos) <= r for u in enemies()) or any(dist(c.pos, me().pos) <= r and (is_awake(c) or gems() == 0) for c in camps()) else 0
    if k == AB_HEAL:
        return 70 if exists(o.unit) and o.unit.health_frac < 0.6 else 0
    if k == AB_DASH:
        if not valid(o.tile):
            return 0
        if fleeing():
            return 80 if at(flee_field(), me().pos) - at(flee_field(), o.tile) >= 2 else 0
        if me().type == HEART_OF_RUIN:
            h = count(dist(u.pos, o.tile) <= 1 for u in enemies()) + count(dist(c.pos, o.tile) <= 1 and (is_awake(c) or gems() == 0) for c in camps())
            if h > 0:
                return 66 + h
        return 51 if progress(o.tile) >= 2 else 0
    return 0

@gene('Z112a', 'low on health with an awake camp or enemy champion close: walk or dash home instead of fighting', parents=['s1'])
def fleeing() -> Bool:
    if me().health_frac >= 0.3:
        return False
    if any(dist(u.pos, me().pos) <= 4 for u in enemies()):
        return True
    return any(is_awake(c) and dist(c.pos, me().pos) <= c.attack_range + 1 for c in camps())

@gene('Z112b', 'distance field to our spawn, for fleeing', parents=['s1'])
def flee_field() -> Field:
    return field(my_spawn())

@gene("new", "after my actions: announce the tile I just attacked and the round, so allies can join the fight", parents=["s3"])
def after() -> None:
    if acted(ATTACK):
        t = acted_tile(ATTACK)
        if valid(t):
            write_shared(0, t.x)
            write_shared(1, t.y)
            write_shared(2, round() - 256 * floor(round() / 256))

@gene("new", "an ally fight announced in the last 3 rounds within 12 tiles, worth joining when I am healthy and empty-handed", parents=["s3"])
def rally_tiles() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0 or me().health_frac < 0.6:
        return acc
    age = round() - 256 * floor(round() / 256) - read_shared(2)
    if age < 0:
        age += 256
    if age > 3:
        return acc
    t = tile(read_shared(0), read_shared(1))
    if valid(t) and dist(t, me().pos) <= 12 and dist(t, me().pos) > 1:
        acc = near(t, 1)
    return acc
