```python
@gene('s1', 'value of doing this option now: draft Ryth, Vesper Vale, Vendetta, Varg, Tavi; buy Durability health shards (item 6 if none); casts, focus the weakest target, recall to shop when rich and idle, recall when low only where it will not be cancelled')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        if item_passive(o.item) == P_DURABILITY and item_health(o.item) > 0:
            return 101
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90 if safe() else 0
        return 55 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            return 80 + 5 * w
        if is_awake(o.unit):
            return 61 + w
        return 60 + w if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
