@gene("s2", "where to walk: an altar with gems, else a living camp group")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc

@gene("s3", "distance field to the goals, or home to heal when hurt, empty-handed and not in a fight")
def goal_field() -> Field:
    if gems() == 0 and me().health_frac < 0.5 and not any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()):
        return field(my_spawn())
    return field(goals())
