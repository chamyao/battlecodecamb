```python
@gene("s1", "value of doing this option now: strong draft, buy strength and health items")
def score(o: Option) -> Num:
    if o.kind == PICK:
        if o.champ == VESPER_VALE:
            return 10
        if o.champ == HEART_OF_RUIN:
            return 9
        if o.champ == MARCELLA:
            return 8
        if o.champ == VENDETTA:
            return 7
        if o.champ == RYTH:
            return 6
        return 1
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "value of buying an item: combat stats per purchase, treasure early", parents=["s1"])
def buy_value(o: Option) -> Num:
    v = item_strength(o.item) * 2 + item_health(o.item) / 25
    if item_passive(o.item) == P_TREASURE and round() < 700:
        v += 6
    if item_passive(o.item) == P_JUNGLE:
        v += 3
    if v <= 0:
        return 0
    return 100 + v + o.count
```
