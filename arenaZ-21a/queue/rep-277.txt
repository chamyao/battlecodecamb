@gene("Z175a", "visible enemy champions worth hunting: badly hurt ones within 6 while I am healthy, or all within 8 when we are at least as many and I am above 70 percent health", parents=["Z18a"])
def hunt_tiles() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if me().health_frac < 0.5:
        return acc
    foes = count(dist(u.pos, me().pos) <= 8 for u in enemies())
    if foes == 0:
        return acc
    friends = 1 + count(dist(a.pos, me().pos) <= 8 for a in allies())
    for u in enemies():
        if dist(u.pos, me().pos) <= 8:
            if friends >= foes and me().health_frac >= 0.7:
                acc = acc | near(u.pos, 1)
            elif u.health_frac < 0.35 and dist(u.pos, me().pos) <= 6:
                acc = acc | near(u.pos, 1)
    return acc

@gene("new", "after my actions: announce the tile I just attacked and the round, so allies can join the fight", parents=["Z18a"])
def after() -> None:
    if acted(ATTACK):
        t = acted_tile(ATTACK)
        if valid(t):
            write_shared(0, t.x)
            write_shared(1, t.y)
            write_shared(2, round() - 256 * floor(round() / 256))

@gene("new", "an ally fight announced in the last 3 rounds within 12 tiles, worth joining when I am healthy and empty-handed", parents=["Z18a"])
def rally_tiles() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0 or me().health_frac < 0.6:
        return acc
    age = round() - 256 * floor(round() / 256) - read_shared(2)
    if age < 0:
        age += 256
    if age > 3:
        return acc
    t = tile(read_shared(0), read_shared(1))
    if valid(t) and dist(t, me().pos) <= 12 and dist(t, me().pos) > 1:
        acc = near(t, 1)
    return acc

@gene('Z18a', 'once per turn: the distance field to a fight an ally just announced nearby, else to the goals plus enemy champions worth hunting', parents=['s3'])
def update() -> None:
    r = rally_tiles()
    if size(r) > 0:
        GF = field(r)
    else:
        GF = field(goals() | hunt_tiles())
