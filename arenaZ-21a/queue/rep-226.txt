```python
@gene('Z72a', 'walk or dash home instead of fighting: below 30 percent health with an awake camp or enemy champion close, or below 60 percent when the enemies within 5 outnumber us by two', parents=['s4'])
def fleeing() -> Bool:
    hf = me().health_frac
    if hf >= 0.6:
        return False
    foes = count(dist(u.pos, me().pos) <= 5 for u in enemies())
    if foes >= 2 and foes >= 3 + count(dist(a.pos, me().pos) <= 5 for a in allies()):
        return True
    if hf >= 0.3:
        return False
    if any(dist(u.pos, me().pos) <= 4 for u in enemies()):
        return True
    return any(is_awake(c) and dist(c.pos, me().pos) <= c.attack_range + 1 for c in camps())
```
