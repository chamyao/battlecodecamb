@gene('s1', 'value of doing this option now: draft Ryth, Vesper Vale, Vendetta, Tavi, Varg; casts in fights, focus the weakest target and gem carriers, recall to shop when rich and idle (not during a capture), recall below 40 percent health only where it will not be cancelled, else walk away (also when outnumbered while hurt) and only finish off nearly dead champions')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return tavi_fourth(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.4:
            return 90 if safe() else 0
        return 55 if shop_trip() and not capturing() else 0
    esc = fleeing() or (me().health_frac < 0.7 and outnumbered())
    if o.kind == CAST:
        if esc and exists(o.unit) and o.unit.is_champion and o.unit.health_frac > 0.25:
            return 0
        if exists(o.unit) and o.unit.is_champion:
            return cast_value(o) + min(unit_gems(o.unit), 50) / 10
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            if esc and o.unit.health_frac > 0.25:
                return 0
            return 80 + 5 * w + min(unit_gems(o.unit), 50) / 10
        if is_awake(o.unit):
            return 61 + w
        return 60 + w if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if esc:
            return 75 if at(flee_field(), me().pos) - at(flee_field(), o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0
