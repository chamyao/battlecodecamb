```python
@gene('s1', 'value of doing this option now: default draft, casts in fights, focus the weakest target, recall to shop when rich and idle, recall when low only where it will not be cancelled, else walk away')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90 if safe() else 0
        return 55 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            return 80 + 5 * w
        if is_awake(o.unit):
            return 61 + w
        return 60 + w if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if fleeing():
            return 75 if at(flee_field(), me().pos) - at(flee_field(), o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "low on health with an awake camp or enemy champion close: walk or dash home instead of fighting", parents=["s1"])
def fleeing() -> Bool:
    if me().health_frac >= 0.3:
        return False
    if any(dist(u.pos, me().pos) <= 4 for u in enemies()):
        return True
    return any(is_awake(c) and dist(c.pos, me().pos) <= c.attack_range + 1 for c in camps())

@gene("new", "distance field to our spawn, for fleeing", parents=["s1"])
def flee_field() -> Field:
    return field(my_spawn())

@gene('Z4c', 'value of casting an ability now: burst the weakest champion or camp we fight, dash away when fleeing, Heart of Ruin dash onto clusters, shield in a fight', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC or k == AB_AOE:
        if exists(o.unit):
            if o.unit.is_champion:
                return 85 + 3 * (1 - o.unit.health_frac)
            if o.unit.is_camp and (is_awake(o.unit) or gems() == 0):
                return 64 + (1 - o.unit.health_frac)
            return 0
        if valid(o.tile):
            h = hits_at(o.tile)
            return 64 + h if h > 0 else 0
        return 0
    if k == AB_SHIELD:
        return 66 if hits_at(me().pos) > 0 or count(dist(u.pos, me().pos) <= 3 for u in enemies()) > 0 else 0
    if k == AB_RESET:
        if can_attack():
            return 0
        r = me().attack_range
        if any(dist(u.pos, me().pos) <= r for u in enemies()):
            return 62
        return 62 if any(dist(c.pos, me().pos) <= r and (is_awake(c) or gems() == 0) for c in camps()) else 0
    if k == AB_HEAL:
        return 70 if exists(o.unit) and o.unit.health_frac < 0.6 else 0
    if k == AB_DASH:
        if not valid(o.tile):
            return 0
        if fleeing():
            return 80 if at(flee_field(), me().pos) - at(flee_field(), o.tile) >= 2 else 0
        if me().type == HEART_OF_RUIN and hits_at(o.tile) > 0:
            return 67
        return 52 if progress(o.tile) >= 2 else 0
    return 0
```
