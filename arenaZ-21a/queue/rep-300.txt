```python
@gene('Z231a', 'a cleared group waiting for capture is within 2 tiles (or under me) and no enemy champion is within 6: stay and finish the 5-round capture before walking home', parents=['s3'])
def capturing() -> Bool:
    if any(dist(u.pos, me().pos) <= 6 for u in enemies()):
        return False
    for g in range(num_groups()):
        if group_reward(g):
            ts = group_tiles(g)
            if dist(nearest(ts, me().pos), me().pos) <= 2:
                return True
    return False
```
