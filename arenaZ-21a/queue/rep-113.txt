@gene("new", "camp groups to walk to: every living group, but a cleared group waiting for capture only if I am the closest visible ally, so the others move on and the reward is not split", parents=["Z18a"])
def carrier_goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g):
                d = dist(nearest(ts, me().pos), me().pos)
                if not any(dist(nearest(ts, a.pos), a.pos) < d for a in allies()):
                    acc = acc | ts
            else:
                acc = acc | ts
    return acc

@gene('Z18a', 'once per turn: the distance field to the goals (one capturer per cleared group), or home when hurt or rich with spawn close, empty-handed and not in a fight', parents=['s3'])
def update() -> None:
    home = False
    if gems() == 0 and not any(is_awake(c) and dist(c.pos, me().pos) <= 4 for c in camps()):
        if me().health_frac < 0.5:
            home = True
        elif gold() >= 150 and dist(nearest(my_spawn(), me().pos), me().pos) <= 12:
            home = True
    if home:
        GF = field(my_spawn())
    else:
        GF = field(carrier_goals())

@gene('s1', 'value of doing this option now: draft Ryth, Vesper Vale, Vendetta, Varg, Tavi; casts, fight back against awake camps, focus the weakest target; below 35 percent health, recall only where it is safe, else walk away first; recall to shop when rich and idle')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    low = me().health_frac < 0.35
    if o.kind == RECALL:
        if low:
            return 90 if safe() else 0
        return 55 if gems() == 0 and gold() >= 150 and safe() else 0
    if o.kind == MOVE:
        if low and not safe():
            return 75 if at(retreat_field(), me().pos) - at(retreat_field(), o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        w = 1 - o.unit.health_frac
        if o.unit.is_champion:
            return 80 + 5 * w
        if gems() == 0:
            return 60 + 1.5 * w
        return 58 + 1.5 * w if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    return 0

@gene("new", "draft preference: the default first three, then Varg and Tavi instead of the linked pair", parents=["s1"])
def pick_value(c: Num) -> Num:
    if c == RYTH:
        return 10
    if c == VESPER_VALE:
        return 9
    if c == VENDETTA:
        return 8
    if c == VARG:
        return 7
    if c == TAVI:
        return 6
    return 1
