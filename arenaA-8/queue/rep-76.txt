@gene("r7", "fight the weakest enemy champion in reach (not on its own spawn), unless I am below 35% and it is above half health")
def r_fight() -> Bool:
    f = fight_foe()
    if not exists(f) or dist(here(), f.pos) > me().attack_range or f.pos in enemy_spawn():
        return False
    return not (hurt_below(0.35) and f.health_frac > 0.5)
