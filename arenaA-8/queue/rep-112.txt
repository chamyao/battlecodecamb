@gene('v2', 'start of turn: choose the rule, its goal and target; shared-memory and priority writes')
def update() -> None:
    if not MAPPED:
        ALTAR_RING = altar_ring()
        SAFE_RING = safe_altar_ring()
        MONO_GROUPS = mono_groups()
        TERRAIN = walls() | holes() | altars() | enemy_spawn()
        BIG_CAMPS = type_spawns(MONOLITH) | type_spawns(GOLEM)
        CAMP_BASE = camp_spawns() - BIG_CAMPS - MONO_GROUPS
        SK0 = slot_kind(0)
        SK1 = slot_kind(1)
        MAPPED = True
    RULE = rule()
    GOAL = rule_goal()
    TGT = rule_target()
    OPEN = False
    if walks() and valid(GOAL) and GOAL != here():
        BLOCK = blocked()
        OPEN = walk_open()
        FIELD = walk_field()
    LOCK_BUY = RULE >= 2 and at_spawn() and wants_lock()
    if LOCK_BUY:
        write_shared(7, read_shared(7) + 2)
    SHOP_GOLD = gold() - (10 if LOCK_BUY else 0)
    if RULE == 8:
        write_shared(7, read_shared(7) + 1)
    if RULE == 20:
        set_priority(1 + rand(100))

@gene('s15', 'the nearest visible camp, big camps (monoliths, golems) ranked last; wounded camps preferred among near-equal distances')
def live_camp() -> Unit:
    return best(camps(), lambda c: -dist(here(), c.pos) - 2 * c.health_frac - (1000 if c.pos in BIG_CAMPS else 0))
