@gene('r10', 'bank now: carrying gems and the bag is ready, or next to an altar with 3+ gems, or the bag alone would reach 1000 tributed (the win)', parents=['r10'])
def bank_on() -> Bool:
    g = gems()
    return g > 0 and (bag_ready() or (g >= 3 and valid(adj_altar())) or team_gems() + g >= 1000)
