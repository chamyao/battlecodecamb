@gene('w4', 'nearest tile of s by walking; if none is walk-reachable, fall back to straight-line nearest; invalid if s empty', parents=['w4'])
def closest(s: Tiles) -> Tile:
    h = here()
    if h in s:
        return h
    cand = s - taken()
    w = walk_nearest(cand, h, blocked() - cand)
    if valid(w):
        return w
    return nearest(s, h)
