@gene('s2', 'where to walk: an altar once the bag is worth banking, else a nearby enemy champion that is tributing or carrying a big bag while hurt (when I am healthy), a living camp group, skipping monoliths and golems before round 800')
def goals() -> Tiles:
    if banking():
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    me_hf = me().health_frac
    if me_hf > 0.6:
        prey = near(tile(-5, -5), 0)
        p = me().pos
        for u in enemies():
            if dist(u.pos, p) <= 10 and (is_tributing(u) or (unit_gems(u) >= 10 and u.health_frac < 0.6)):
                prey = prey | near(u.pos, 1)
        if size(prey) > 0:
            return prey
    acc = near(tile(-5, -5), 0)
    bad = type_spawns(MONOLITH)
    if round() < 800:
        bad = bad | type_spawns(GOLEM)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & bad) == 0:
                acc = acc | gt
    if size(acc) == 0:
        for g in range(num_groups()):
            if not group_respawning(g):
                acc = acc | group_tiles(g)
    return acc

@gene('Z41b', 'bank time: carry gems to an altar once the bag holds 15 (8 when hurt), or any gems near the end', parents=['s2'])
def banking() -> Bool:
    g = gems()
    return g >= 15 or (g >= 8 and me().health_frac < 0.5) or (g > 0 and round() > 2722)
