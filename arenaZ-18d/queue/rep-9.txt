```python
GF: Field = None

@gene("new", "once per turn: cache the distance field to the goals", parents=["s3"])
def update() -> None:
    GF = field(goals())

@gene("s3", "distance field to the goals (cached once per turn in update)")
def goal_field() -> Field:
    return GF

@gene("s2", "where to walk: an altar with gems, else a living camp group, skipping monoliths always and golems early")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    bad = type_spawns(MONOLITH)
    if round() < 600:
        bad = bad | type_spawns(GOLEM)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & bad) == 0:
                acc = acc | gt
    return acc
```
