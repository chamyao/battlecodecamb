@gene("s2", "where to walk: an altar once the bag is worth banking, else a cleared group waiting for capture within 12 tiles, else a living camp group, skipping monoliths and golems before round 800")
def goals() -> Tiles:
    if banking():
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    p = me().pos
    for g in range(num_groups()):
        if group_reward(g):
            gt = group_tiles(g)
            if dist(nearest(gt, p), p) <= 12:
                acc = acc | gt
    if size(acc) > 0:
        return acc
    bad = type_spawns(MONOLITH)
    if round() < 800:
        bad = bad | type_spawns(GOLEM)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & bad) == 0:
                acc = acc | gt
    if size(acc) == 0:
        for g in range(num_groups()):
            if not group_respawning(g):
                acc = acc | group_tiles(g)
    return acc
