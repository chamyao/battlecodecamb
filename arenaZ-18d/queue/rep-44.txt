```python
@gene('Z13b', 'value of casting an ability: damage and stuns on fight targets, resets after an attack, shields when hurt, heals on hurt allies, dashes toward goals', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC or k == AB_AOE:
        if exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                return 57
            if o.unit.is_camp:
                return 75 if is_awake(o.unit) or (gems() == 0 and me().health_frac > 0.4) else 0
            return 0
        if valid(o.tile):
            return 78 if o.tile in fight_tiles() else 0
        return 0
    if k == AB_RESET:
        return 40 if not can_attack() and target_in_reach() else 0
    if k == AB_SHIELD:
        return 70 if me().health_frac < 0.85 and target_in_reach() else 0
    if k == AB_HEAL:
        if exists(o.unit) and not o.unit.is_enemy:
            return 72 if o.unit.health_frac < 0.6 else 0
        return 0
    if k == AB_DASH:
        if valid(o.tile):
            return 45 if progress(o.tile) >= 2 else 0
        return 0
    return 0
```
