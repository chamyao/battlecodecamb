```python
@gene("new", "skip listing wards and locks, shop items when broke, and abilities when no fight target is within 6 tiles and I am not in danger (saves listing fuel)", parents=["s1"])
def consider(kind: Num) -> Bool:
    if kind == WARD or kind == LOCK or kind == BUY_LOCK:
        return False
    if kind == BUY:
        return gold() >= 30
    if kind == CAST:
        return me().health_frac < 0.35 or size(fight_tiles() & near(me().pos, 6)) > 0
    return True
```
