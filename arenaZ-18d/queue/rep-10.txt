@gene("s2", "where to walk: an altar with gems, else a living camp group that is not a monolith")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    mono = type_spawns(MONOLITH)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & mono) == 0:
                acc = acc | gt
    return acc

GF: Field = None

@gene("new", "once per turn: cache the distance field to the goals", parents=["s3"])
def update() -> None:
    GF = field(goals())

@gene("new", "skip listing option kinds the policy never uses (wards, locks)", parents=["s3"])
def consider(kind: Num) -> Bool:
    return kind != WARD and kind != LOCK and kind != BUY_LOCK

@gene("s3", "distance field to the goals (cached once per turn)")
def goal_field() -> Field:
    return GF
