@gene("s2", "where to walk: an altar with gems, else a living camp group, skipping monoliths and golems before round 800")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    bad = type_spawns(MONOLITH)
    if round() < 800:
        bad = bad | type_spawns(GOLEM)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & bad) == 0:
                acc = acc | gt
    if size(acc) == 0:
        for g in range(num_groups()):
            if not group_respawning(g):
                acc = acc | group_tiles(g)
    return acc

@gene('Z13b', 'value of casting an ability: damage and stuns focus the weakest fight target, area casts value every target hit, resets after an attack, shields when hurt, heals hurt allies, dashes toward goals or away from danger when low', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC or k == AB_AOE:
        if exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                return 85 + 5 * (1 - o.unit.health_frac)
            if o.unit.is_camp:
                if gems() == 0 or is_awake(o.unit):
                    return 75 + 5 * (1 - o.unit.health_frac)
                return 0
            return 0
        if valid(o.tile):
            if o.tile in fight_tiles():
                return 76 + size(near(o.tile, slot_radius(o.slot)) & fight_tiles())
            return 0
        return 0
    if k == AB_RESET:
        return 55 if not can_attack() and target_in_reach() else 0
    if k == AB_SHIELD:
        return 70 if me().health_frac < 0.85 and target_in_reach() else 0
    if k == AB_HEAL:
        if exists(o.unit) and not o.unit.is_enemy:
            return 72 if o.unit.health_frac < 0.6 else 0
        return 0
    if k == AB_DASH:
        if valid(o.tile):
            if me().health_frac < 0.35:
                return 88 if threat_at(o.tile) < threat_at(me().pos) else 0
            return 45 if progress(o.tile) >= 2 else 0
        return 0
    return 0

@gene("new", "how many enemy champions and awake camps are within 2 tiles of a tile", parents=["Z13b"])
def threat_at(t: Tile) -> Num:
    return count(dist(u.pos, t) <= 2 for u in enemies()) + count(dist(u.pos, t) <= 2 and is_awake(u) for u in camps())
