@gene('Z118c', 'bank time: carry gems to an altar once the bag holds 15, or any gems near the end', parents=['s2'])
def banking() -> Bool:
    g = gems()
    return g >= 15 or (g >= 8 and me().health_frac < 0.5) or (g > 0 and round() > 1850)

@gene('Z118a', 'once per turn: cache the distance field to the goals; when hurt with almost no gems and no fight in reach, walk home to heal and shop instead', parents=['s3'])
def update() -> None:
    if me().health_frac < 0.5 and gems() < 5 and not target_in_reach():
        GF = field(my_spawn())
    else:
        GF = field(goals())
