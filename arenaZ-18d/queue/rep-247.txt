```python
@gene('s2', 'where to walk: an altar once the bag is worth banking, else a nearby enemy champion that is tributing or carrying a big bag while hurt (when I am healthy), a living camp group, skipping golems before round 800 and monoliths unless it is after round 1100 and I am healthy with two allies close')
def goals() -> Tiles:
    if banking():
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    me_hf = me().health_frac
    if me_hf > 0.6:
        prey = near(tile(-5, -5), 0)
        p = me().pos
        for u in enemies():
            if dist(u.pos, p) <= 8 and (is_tributing(u) or (unit_gems(u) >= 10 and u.health_frac < 0.6)):
                prey = prey | near(u.pos, 1)
        if size(prey) > 0:
            return prey
    acc = near(tile(-5, -5), 0)
    bad = near(tile(-5, -5), 0)
    p = me().pos
    if round() < 1100 or me().health_frac < 0.7 or count(dist(a.pos, p) <= 6 for a in allies()) < 2:
        bad = type_spawns(MONOLITH)
    if round() < 800:
        bad = bad | type_spawns(GOLEM)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & bad) == 0:
                acc = acc | gt
    if size(acc) == 0:
        for g in range(num_groups()):
            if not group_respawning(g):
                acc = acc | group_tiles(g)
    return acc
```
