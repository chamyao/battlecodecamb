@gene("s4", "how much closer a step brings me to the goals, minus a penalty for stepping next to enemy champions while hurt")
def progress(t: Tile) -> Num:
    d = at(goal_field(), me().pos) - at(goal_field(), t)
    if me().health_frac < 0.5:
        d -= 2 * count(dist(u.pos, t) <= 2 for u in enemies())
    return d

@gene('s1', 'value of doing this option now; default draft, abilities, Treasure then Jungle health (less camp damage) then a Bruiser strength emblem, then a Durability health emblem, then more Bruiser, hunts until the bag is worth banking, recalls earlier with a bag, avoids starting camp fights when hurt, basic attacks focus the weakest target, carrying a bank-worthy bag it walks on instead of trading hits with a healthy champion')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if items_owned(19) == 0:
            return 100 if o.item == 18 else 0
        if items_owned(29) == 0:
            return 100 if o.item == 28 else 0
        if items_owned(37) == 0:
            return 100 if o.item == 36 else 0
        if items_owned(17) == 0:
            return 100 if o.item == 16 else 0
        return 100 if o.item == 36 else 0
    if o.kind == RECALL:
        hf = me().health_frac
        return 58 if hf < 0.35 or (gems() >= 15 and hf < 0.5) else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            if banking() and o.unit.health_frac > 0.3:
                return 40
            return 80 + 5 * (1 - o.unit.health_frac)
        if is_awake(o.unit):
            return 60 + 5 * (1 - o.unit.health_frac)
        return 60 if not banking() and me().health_frac > 0.4 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 34 if progress(o.tile) > 0 else 0
    return 0
