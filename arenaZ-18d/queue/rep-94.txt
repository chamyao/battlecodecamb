@gene('Z6a', 'value of casting an ability: damage and stuns on fight targets (pure stuns only on awake camps) (no new camp fights when hurt), resets after an attack, shields when hurt, heals on hurt allies, dashes toward goals; dash saved for retreat when low, used toward goals only out of danger', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC or k == AB_AOE:
        if exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                return 85
            if o.unit.is_camp:
                if k == AB_CC and not is_awake(o.unit):
                    return 0
                return 75 if is_awake(o.unit) or (gems() == 0 and me().health_frac > 0.4) else 0
            return 0
        if valid(o.tile):
            return 78 if o.tile in fight_tiles() else 0
        return 0
    if k == AB_RESET:
        return 55 if not can_attack() and target_in_reach() else 0
    if k == AB_SHIELD:
        return 70 if me().health_frac < 0.85 and target_in_reach() else 0
    if k == AB_HEAL:
        if exists(o.unit) and not o.unit.is_enemy:
            return 72 if o.unit.health_frac < 0.6 else 0
        return 0
    if k == AB_DASH:
        if valid(o.tile):
            if me().health_frac < 0.35:
                return 88 if threat_at(o.tile) < threat_at(me().pos) else 0
            return 45 if progress(o.tile) >= 1 and threat_at(me().pos) == 0 else 0
        return 0
    return 0

@gene("new", "how many enemy champions and awake camps are within 3 tiles of a tile", parents=["Z6a"])
def threat_at(t: Tile) -> Num:
    return count(dist(u.pos, t) <= 3 for u in enemies()) + count(dist(u.pos, t) <= 3 and is_awake(u) for u in camps())
