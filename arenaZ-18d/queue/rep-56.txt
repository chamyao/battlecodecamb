@gene("s2", "where to walk: an altar once the bag is worth banking, else a living camp group, skipping monoliths and golems before round 800")
def goals() -> Tiles:
    if banking():
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    bad = type_spawns(MONOLITH)
    if round() < 800:
        bad = bad | type_spawns(GOLEM)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & bad) == 0:
                acc = acc | gt
    if size(acc) == 0:
        for g in range(num_groups()):
            if not group_respawning(g):
                acc = acc | group_tiles(g)
    return acc

@gene('s1', 'value of doing this option now; default draft, abilities, Treasure then Jungle health then Power strength, hunts until the bag is worth banking, recalls earlier with a bag, avoids starting camp fights when hurt')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if items_owned(19) == 0:
            return 100 if o.item == 18 else 0
        if items_owned(29) == 0:
            return 100 if o.item == 28 else 0
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        hf = me().health_frac
        return 90 if hf < 0.35 or (gems() >= 20 and hf < 0.5) else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        if is_awake(o.unit):
            return 60
        return 60 if not banking() and me().health_frac > 0.4 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z9a', 'value of casting an ability: damage and stuns on fight targets, resets after an attack, shields when hurt, heals on hurt allies, dashes toward goals', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC or k == AB_AOE:
        if exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                return 85
            if o.unit.is_camp:
                return 75 if is_awake(o.unit) or (not banking() and me().health_frac > 0.4) else 0
            return 0
        if valid(o.tile):
            return 78 if o.tile in fight_tiles() else 0
        return 0
    if k == AB_RESET:
        return 55 if not can_attack() and target_in_reach() else 0
    if k == AB_SHIELD:
        return 70 if me().health_frac < 0.85 and target_in_reach() else 0
    if k == AB_HEAL:
        if exists(o.unit) and not o.unit.is_enemy:
            return 72 if o.unit.health_frac < 0.6 else 0
        return 0
    if k == AB_DASH:
        if valid(o.tile):
            return 45 if progress(o.tile) >= 1 else 0
        return 0
    return 0
