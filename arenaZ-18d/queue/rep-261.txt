@gene("s2", "where to walk: an altar once the bag is worth banking, else a living camp group, skipping golems before round 800 and monoliths unless it is after round 1100 and I am healthy with two allies close")
def goals() -> Tiles:
    if banking():
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    me_hf = me().health_frac
    if me_hf > 0.6:
        prey = near(tile(-5, -5), 0)
        p = me().pos
        for u in enemies():
            if dist(u.pos, p) <= 8 and (is_tributing(u) or (unit_gems(u) >= 10 and u.health_frac < 0.6)):
                prey = prey | near(u.pos, 1)
        if size(prey) > 0:
            return prey
    acc = near(tile(-5, -5), 0)
    bad = near(tile(-5, -5), 0)
    p = me().pos
    if round() < 1100 or me().health_frac < 0.7 or count(dist(a.pos, p) <= 6 for a in allies()) < 2:
        bad = near(tile(-5, -5), 0)
    p = me().pos
    if round() < 1100 or me().health_frac < 0.7 or count(dist(a.pos, p) <= 6 for a in allies()) < 2:
        bad = type_spawns(MONOLITH)
    if round() < 800:
        bad = bad | type_spawns(GOLEM)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & bad) == 0:
                acc = acc | gt
    if size(acc) == 0:
        for g in range(num_groups()):
            if not group_respawning(g):
                acc = acc | group_tiles(g)
    return acc

@gene('Z144d', 'value of casting an ability: damage and stuns focus the weakest fight target, area casts value every target hit, resets after an attack, shields when hurt, heals hurt allies, dashes toward goals or away from danger when low', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC or k == AB_AOE:
        if exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                return 85 + 5 * (1 - o.unit.health_frac)
            if o.unit.is_camp:
                if is_awake(o.unit) or (not banking() and me().health_frac > 0.4):
                    return 75 + 5 * (1 - o.unit.health_frac)
                return 0
            return 0
        if valid(o.tile):
            n = size(near(o.tile, slot_radius(o.slot)) & fight_tiles())
            if o.tile in fight_tiles() or n >= 2:
                return 76 + n
            return 0
        return 0
    if k == AB_RESET:
        return 55 if not can_attack() and target_in_reach() else 0
    if k == AB_SHIELD:
        return 70 if me().health_frac < 0.85 and target_in_reach() else 0
    if k == AB_HEAL:
        if exists(o.unit) and not o.unit.is_enemy:
            return 72 if o.unit.health_frac < 0.6 else 0
        return 0
    if k == AB_DASH:
        if valid(o.tile):
            if me().health_frac < 0.35:
                return 88 if threat_at(o.tile) < threat_at(me().pos) else 0
            return 45 if progress(o.tile) >= 1 and threat_at(me().pos) == 0 else 0
        return 0
    return 0
