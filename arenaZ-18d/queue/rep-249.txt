@gene('s1', 'value of doing this option now; default draft, abilities, Treasure then Jungle health (less camp damage) then a Bruiser strength emblem, then a Jungle health emblem, then more Bruiser, hunts until the bag is worth banking, recalls earlier with a bag, avoids starting camp fights when hurt, basic attacks focus the weakest target, carrying a bank-worthy bag it walks on instead of trading hits with a healthy champion')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if items_owned(19) == 0:
            return 100 if o.item == 18 else 0
        if items_owned(29) == 0:
            return 100 if o.item == 28 else 0
        if items_owned(37) == 0:
            return 100 if o.item == 36 else 0
        if items_owned(29) == 0:
            return 100 if o.item == 28 else 0
        return 100 if o.item == 36 else 0
    if o.kind == RECALL:
        hf = me().health_frac
        return 90 if hf < 0.35 or (gems() >= 15 and hf < 0.5) else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            if banking() and o.unit.health_frac > 0.3:
                return 40
            return 80 + 5 * (1 - o.unit.health_frac)
        if is_awake(o.unit):
            return 60 + 5 * (1 - o.unit.health_frac)
        return 60 if not banking() and me().health_frac > 0.4 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 34 if progress(o.tile) > 0 else 0
    return 0

@gene("s2", "where to walk: an altar once the bag is worth banking, else a living camp group, skipping golems before round 800 and monoliths unless it is after round 1100 and I am healthy with two allies close")
def goals() -> Tiles:
    if banking():
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    me_hf = me().health_frac
    if me_hf > 0.6:
        prey = near(tile(-5, -5), 0)
        p = me().pos
        for u in enemies():
            if dist(u.pos, p) <= 8 and (is_tributing(u) or (unit_gems(u) >= 10 and u.health_frac < 0.8091)):
                prey = prey | near(u.pos, 1)
        if size(prey) > 0:
            return prey
    acc = near(tile(-5, -5), 0)
    bad = near(tile(-5, -5), 0)
    p = me().pos
    if round() < 1100 or me().health_frac < 0.7 or count(dist(a.pos, p) <= 6 for a in allies()) < 2:
        bad = type_spawns(MONOLITH)
    if round() < 800:
        bad = bad | type_spawns(GOLEM)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & bad) == 0:
                acc = acc | gt
    if size(acc) == 0:
        for g in range(num_groups()):
            if not group_respawning(g):
                acc = acc | group_tiles(g)
    return acc
