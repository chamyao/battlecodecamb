@gene("s2", "where to walk: an altar with gems, else a living camp group")
def goals() -> Tiles:
    if has_lock():
        hunt = near(tile(-5, -5), 0)
        for u in enemies():
            if is_tributing(u) or unit_gems(u) >= 20:
                hunt = hunt | (near(u.pos, 2) & (near(nearest(altars(), u.pos), 1)))
        if size(hunt) > 0:
            return hunt
    if gems() >= 8 or (gems() > 0 and round() > 1900):
        acc = near(tile(-5, -5), 0)
        for a in altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
