```python
@gene('s1', 'value of doing this option now: when hurt, walk home out of camp range and recall only when safe')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if o.item == 36:
            return 100 if items_owned(36) <= items_owned(40) else 0
        return 100 if o.item == 40 else 0
    hurt = me().health_frac < 0.4
    if o.kind == RECALL:
        return 90 if (hurt or (gold() >= 300 and gems() == 0)) and not threatened() else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() < 8 and not hurt else 0
    if o.kind == BUY_LOCK:
        return 96 if not has_lock() and round() > 0 else 0
    if o.kind == LOCK:
        if any((is_tributing(u) or unit_gems(u) >= 10) and dist(u.pos, o.tile) <= 2 for u in enemies()):
            return 95
        return 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == MOVE:
        if hurt:
            return 85 if home_step(o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
