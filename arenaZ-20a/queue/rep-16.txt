```python
@gene("s4", "how much closer a step brings me to the goals")
def progress(t: Tile) -> Num:
    d = at(goal_field(), me().pos) - at(goal_field(), t)
    if gems() > 0 and d > 0 and any(u.is_champion and u.health > me().health and dist(u.pos, t) <= 2 for u in enemies()):
        return 0
    return d
```
