@gene('s1', 'value of doing this option now: when hurt, walk home out of camp range and recall only when safe')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if o.item == 36:
            return 100 if items_owned(36) <= items_owned(40) else 0
        return 100 if o.item == 40 else 0
    hurt = me().health_frac < 0.3
    if o.kind == RECALL:
        return 90 if (hurt or (gold() >= 300 and gems() == 0)) and not threatened() else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        return 60 if gems() < 8 and not hurt else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_AOE:
            if o.tile in foe_hit():
                return 81
            return 64 if o.tile in camp_hit() and gems() < 8 and not hurt else 0
        if slot_kind(o.slot) == AB_DASH:
            if hurt:
                return 86 if home_step(o.tile) >= 2 else 0
            return 52 if progress(o.tile) >= 2 else 0
        return cast_value(o)
    if o.kind == MOVE:
        if hurt:
            return 85 if home_step(o.tile) > 0 else 0
        return 51 if progress(o.tile) > 0 else 0
    return 0

@gene('Z36d', 'value of casting an ability now', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    hurt = me().health_frac < 0.4
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion:
                return 74 + 5 * (1 - o.unit.health_frac)
            return 65 if gems() < 8 and not hurt else 0
        return 0
    if k == AB_AOE:
        hits = count(dist(c.pos, o.tile) <= 1 for c in camps()) + count(dist(u.pos, o.tile) <= 1 for u in enemies())
        return 64 if hits > 0 and gems() < 7 and not hurt else 0
    if k == AB_RESET:
        return 75 if not can_attack() and (threatened() or count_units(enemies()) > 0) else 0
    if k == AB_SHIELD:
        return 75 if threatened() or count_units(enemies()) > 0 else 0
    if k == AB_HEAL:
        return 75 if exists(o.unit) and o.unit.health_frac < 0.7 else 0
    return 0

@gene("new", "tiles where an area spell would hit an enemy champion (once per pass)", parents=["s1"])
def foe_hit() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for u in enemies():
        acc = acc | near(u.pos, 1)
    return acc

@gene("new", "tiles where an area spell would hit a camp (once per pass)", parents=["s1"])
def camp_hit() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for c in camps():
        acc = acc | near(c.pos, 1)
    return acc
