```python
@gene('s1', 'value of doing this option now: when hurt, walk home out of camp range and recall only when safe')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    hurt = me().health_frac < 0.4
    if o.kind == RECALL:
        return 90 if hurt and not threatened() else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if (gems() == 0 or is_awake(o.unit)) and not hurt else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == MOVE:
        if hurt:
            return 85 if home_step(o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "which option kinds are worth listing: skip wards, locks and cancels, and casts while both slots cool down", parents=["s1"])
def consider(kind: Num) -> Bool:
    if kind == CAST:
        return slot_ready(0) or slot_ready(1)
    if kind == WARD or kind == LOCK or kind == BUY_LOCK or kind == CANCEL:
        return False
    return True
```
