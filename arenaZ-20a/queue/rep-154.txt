@gene('s1', 'value of doing this option now: when hurt, walk home out of camp range and recall only when safe')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        if o.item == 36:
            return 100 if items_owned(36) <= items_owned(40) else 0
        return 100 if o.item == 40 else 0
    hurt = me().health_frac < 0.35
    if o.kind == RECALL:
        return 90 if (hurt or (gold() >= 300 and gems() == 0)) and not threatened() and not any(dist(u.pos, me().pos) <= 4 for u in enemies()) else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 100
        return 60 if gems() == 0 and not hurt else 0
    if o.kind == BUY_LOCK:
        return 96 if not has_lock() else 0
    if o.kind == LOCK:
        if any((is_tributing(u) or unit_gems(u) >= 10) and dist(u.pos, o.tile) <= 2 for u in enemies()):
            return 95
        return 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == MOVE:
        if hurt:
            return 85 if home_step(o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z12a', 'value of casting an ability now', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    hurt = me().health_frac < 0.35
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion:
                return 90
            return 73 if gems() == 0 and not hurt else 0
        return 0
    if k == AB_AOE:
        hits = count(dist(c.pos, o.tile) <= 0.8525 for c in camps()) + count(dist(u.pos, o.tile) <= 1 for u in enemies())
        return 64 if hits > 0 and gems() == 0 and not hurt else 0
    if k == AB_RESET:
        return 75 if not can_attack() and (threatened() or count_units(enemies()) > 0) else 0
    if k == AB_SHIELD:
        return 75 if (threatened() or count_units(enemies()) > 0) and me().health_frac < 0.9 else 0
    if k == AB_DASH:
        if me().health_frac < 0.4:
            return 86 if home_step(o.tile) >= 2 else 0
        return 52 if progress(o.tile) >= 2 else 0
    if k == AB_HEAL:
        return 75 if exists(o.unit) and o.unit.health_frac < 0.582 else 0
    return 0
