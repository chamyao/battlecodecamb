GF: Field = None

@gene("s4", "how much closer a step brings me to the goals")
def progress(t: Tile) -> Num:
    return at(GF, me().pos) - at(GF, t)

@gene('Z12a', 'value of casting an ability now', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    hurt = me().health_frac < 0.35
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion:
                return 82
            return 65 if gems() == 0 and not hurt else 0
        return 0
    if k == AB_AOE:
        hits = count(dist(c.pos, o.tile) <= 1 for c in camps()) + count(dist(u.pos, o.tile) <= 1 for u in enemies())
        return 64 if hits > 0 and gems() == 0 and not hurt else 0
    if k == AB_RESET:
        return 75 if not can_attack() and (threatened() or count_units(enemies()) > 0) else 0
    if k == AB_SHIELD:
        return 75 if threatened() or count_units(enemies()) > 0 else 0
    if k == AB_DASH:
        if me().health_frac < 0.4:
            return 86 if home_step(o.tile) >= 2 else 0
        return 52 if progress(o.tile) >= 2 else 0
    if k == AB_HEAL:
        return 75 if exists(o.unit) and o.unit.health_frac < 0.7 else 0
    return 0

@gene('Z9a', 'an awake camp could hit me here', parents=['s1'])
def threatened() -> Bool:
    return any(is_awake(c) and dist(c.pos, me().pos) <= 5 for c in camps()) or any(dist(u.pos, me().pos) <= 4 for u in enemies())

@gene("new", "once per turn: compute the goal distance field", parents=["s3"])
def update() -> None:
    GF = goal_field()
