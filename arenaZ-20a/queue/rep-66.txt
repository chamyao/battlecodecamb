```python
GF: Field = None

@gene('s1', 'value of doing this option now: when hurt, walk home out of camp range and recall only when safe')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        if o.item == 36:
            return 100 if items_owned(36) <= items_owned(40) else 0
        return 100 if o.item == 40 else 0
    hurt = me().health_frac < 0.35
    if o.kind == RECALL:
        return 90 if hurt and not threatened() and not any(dist(u.pos, me().pos) <= 4 for u in enemies()) else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 and not hurt else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_DASH:
            if hurt:
                return 86 if home_step(o.tile) >= 2 else 0
            return 52 if progress(o.tile) >= 2 else 0
        return cast_value(o)
    if o.kind == MOVE:
        if hurt:
            return 85 if home_step(o.tile) > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("s4", "how much closer a step brings me to the goals")
def progress(t: Tile) -> Num:
    return at(GF, me().pos) - at(GF, t)

@gene('Z9c', 'how much closer a step brings me to my spawn', parents=['s4'])
def home_step(t: Tile) -> Num:
    d = at(home_field(), me().pos) - at(home_field(), t)
    here = min((dist(c.pos, me().pos) for c in camps() if is_awake(c)), default=99)
    there = min((dist(c.pos, t) for c in camps() if is_awake(c)), default=99)
    if here <= 5:
        return (there - here) + 0.1 * d
    return d

@gene("new", "once per turn: compute the goal distance field", parents=["s3"])
def update() -> None:
    GF = goal_field()
```
