@gene('Z12a', 'value of casting an ability now', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    hurt = me().health_frac < 0.35
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion:
                return 82
            return 65 if gems() == 0 and not hurt else 0
        return 0
    if k == AB_AOE:
        if o.tile in foe_hit():
            return 81
        return 64 if o.tile in camp_hit() and gems() == 0 and not hurt else 0
    if k == AB_RESET:
        return 75 if not can_attack() and (threatened() or count_units(enemies()) > 0) else 0
    if k == AB_SHIELD:
        return 75 if threatened() or count_units(enemies()) > 0 else 0
    if k == AB_HEAL:
        return 75 if exists(o.unit) and o.unit.health_frac < 0.7 else 0
    return 0

@gene("new", "tiles where an area spell would hit an enemy champion (once per pass)", parents=["Z12a"])
def foe_hit() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for u in enemies():
        acc = acc | near(u.pos, 1)
    return acc

@gene("new", "tiles where an area spell would hit a camp (once per pass)", parents=["Z12a"])
def camp_hit() -> Tiles:
    acc = near(tile(-5, -5), 0)
    for c in camps():
        acc = acc | near(c.pos, 1)
    return acc
