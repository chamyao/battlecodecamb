@gene('Z86a', 'an awake camp could hit me here', parents=['s1'])
def threatened() -> Bool:
    return any(is_awake(c) and dist(c.pos, me().pos) <= 5 for c in camps()) or any(dist(u.pos, me().pos) <= 5 for u in enemies())

@gene('Z86d', 'value of casting an ability now', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    hurt = me().health_frac < 0.4
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion:
                return 87
            return 65 if gems() < 8 and not hurt else 0
        return 0
    if k == AB_AOE:
        if o.tile in FOEHIT:
            return 81
        return 32 if o.tile in CAMPHIT and gems() < 8 and not hurt else 0
    if k == AB_RESET:
        return 75 if not can_attack() and (threatened() or count_units(enemies()) > 0) else 0
    if k == AB_SHIELD:
        return 102 if threatened() or count_units(enemies()) > 0 else 0
    if k == AB_DASH:
        if me().health_frac < 0.4:
            return 77 if home_step(o.tile) >= 2 else 0
        return 52 if progress(o.tile) >= 2 else 0
    if k == AB_HEAL:
        return 75 if exists(o.unit) and o.unit.health_frac < 0.7 else 0
    return 0
