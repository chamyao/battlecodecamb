```python
@gene('s1', 'value of doing this option now: when hurt, walk home out of camp range and recall only when safe')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        if o.item == 36:
            return 100 if items_owned(36) <= items_owned(40) else 0
        return 100 if o.item == 40 else 0
    hurt = me().health_frac < 0.4
    if o.kind == RECALL:
        return 90 if (hurt or (gold() >= 300 and gems() == 0)) and not threatened() else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        return 56 if gems() < 8 and not hurt else 0
    if o.kind == BUY_LOCK:
        return 106 if not has_lock() else 0
    if o.kind == LOCK:
        if any((is_tributing(u) or unit_gems(u) >= 10) and dist(u.pos, o.tile) <= 1 for u in enemies()):
            return 95
        return 69 if gems() == 0 and not (o.tile in locked_altars()) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_AOE:
            if o.tile in foe_hit():
                return 81
            return 64 if o.tile in camp_hit() and gems() == 0 and not hurt else 0
        if slot_kind(o.slot) == AB_DASH:
            if hurt:
                return 86 if home_step(o.tile) >= 2 else 0
            return 52 if progress(o.tile) >= 2 else 0
        return cast_value(o)
    if o.kind == MOVE:
        if hurt:
            return 94 if home_step(o.tile) > 0 else 0
        return 51 if progress(o.tile) > 0 else 0
    return 0
```
