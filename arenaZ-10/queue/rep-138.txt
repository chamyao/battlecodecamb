@gene('Z53a', 'worth recalling to shop: 120+ gold banked, no gems carried, not home, nothing hostile near, not about to reach a camp (finish it first), not in the last 150 rounds')
def shop_trip() -> Bool:
    if gold() < 120 or gems() > 0 or can_shop() or round() > 1850:
        return False
    p = me().pos
    if any(dist(u.pos, p) <= 6 for u in enemies()):
        return False
    if any(is_awake(u) and dist(u.pos, p) <= 4 for u in camps()):
        return False
    return at(goal_field(), p) > 3
