@gene('Z47a', 'is something about to hit me: an awake camp within 2, or enemy champions within their attack range + 3, unless the only one is a lone champion below 20% health (finish it instead of recalling)', parents=['Z11a'])
def threatened() -> Bool:
    p = me().pos
    n = count(u.is_champion and dist(u.pos, p) <= max(u.attack_range, 1) + 3 for u in enemies())
    if n >= 2:
        return True
    if n == 1 and any(u.is_champion and u.health_frac >= 0.2 and dist(u.pos, p) <= max(u.attack_range, 1) + 3 for u in enemies()):
        return True
    return any(is_awake(u) and dist(u.pos, p) <= 2 for u in camps())
