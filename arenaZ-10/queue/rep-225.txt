@gene("s2", "where to walk: a cleared group awaiting capture within 5, else an altar with gems, else a living camp group (monoliths only when nothing else is left)")
def goals() -> Tiles:
    p = me().pos
    ready = near(tile(-5, -5), 0)
    acc = near(tile(-5, -5), 0)
    mono = type_spawns(MONOLITH)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g) and dist(nearest(ts, p), p) <= 5:
                ready = ready | ts
            acc = acc | ts
    if size(ready) > 0:
        return ready
    if gems() > 0:
        alt = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            alt = alt | near(a, 1)
        return alt
    easy = acc - mono
    if size(easy) > 0:
        return easy
    return acc

@gene("Z138a", "once per turn: cache the distance field to the goals not occupied by or next to another ally (spread allies over camps; altars only exclude occupied tiles)", parents=["s3"])
def update() -> None:
    g = goals()
    others = ally_tiles() - near(me().pos, 0)
    taken = others
    if gems() == 0:
        for a in others:
            taken = taken | near(a, 1)
    free = g - taken
    if size(free) == 0:
        free = g - others
    if size(free) == 0:
        free = g
    GF = field(free)
