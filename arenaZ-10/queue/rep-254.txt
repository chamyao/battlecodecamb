```python
GF: Field = None

@gene("s4", "how much closer a step brings me to the goals (field cached once per turn); hurt and carrying gems, a step into an enemy champion attack reach counts as no progress")
def progress(t: Tile) -> Num:
    d = at(GF, me().pos) - at(GF, t)
    if d > 0 and gems() > 0 and me().health_frac < 0.5:
        if any(u.is_champion and dist(u.pos, t) <= max(u.attack_range, 1) for u in enemies()):
            return 0
    return d

@gene("new", "once per turn: cache the distance field to the goals", parents=["s4"])
def update() -> None:
    GF = goal_field()
```
