@gene('s2', 'where to walk: a cleared group awaiting capture within 5, else an altar with gems (unless carrying under 15 and a living non-monolith camp is within 6: farm it first), else a living camp group (monoliths only when nothing else is left)')
def goals() -> Tiles:
    p = me().pos
    ready = near(tile(-5, -5), 0)
    acc = near(tile(-5, -5), 0)
    mono = type_spawns(MONOLITH)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g) and dist(nearest(ts, p), p) <= 5:
                ready = ready | ts
            acc = acc | ts
    if size(ready) > 0:
        return ready
    easy = acc - mono
    if gems() > 0:
        if gems() < 15 and size(easy) > 0 and dist(nearest(easy, p), p) <= 6:
            return easy & near(p, 8)
        alt = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            alt = alt | near(a, 1)
        return alt
    if size(easy) > 0:
        return easy
    return acc
