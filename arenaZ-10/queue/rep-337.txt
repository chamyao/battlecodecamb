@gene('s1', 'value of doing this option now; casts abilities, shop plan, recall at 35% (50% when carrying 20+ gems near an enemy champion, 65% with 50+) or to spend banked gold, fights back against awake camps while carrying gems; when hurt with gems, steps away from enemy champions within 4')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 0.7507
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        if gems() >= 20 and me().health_frac < (0.65 if gems() >= 50 else 0.5) and any(u.is_champion and dist(u.pos, me().pos) <= max(u.attack_range, 1) + 3 for u in enemies()):
            return 88
        return 52 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (0.7996 - o.unit.health_frac)
        if gems() == 0:
            return 60 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and is_awake(o.unit) and me().health_frac > 0.5:
            return 45
        return 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        pr = progress(o.tile)
        if gems() > 0 and me().health_frac < 0.5 and pr >= 0:
            p = me().pos
            if any(u.is_champion and dist(u.pos, p) <= 4 for u in enemies()):
                if all(dist(u.pos, o.tile) > dist(u.pos, p) for u in enemies() if u.is_champion and dist(u.pos, p) <= 4):
                    return 56
        return 50 if pr > 0 else 0
    return 0

@gene('Z157a', 'once per turn: cache the distance field to the goals not occupied by or next to another ally (spread allies over camps; altars only exclude occupied tiles); carrying 10+ gems, route around the reach (+1) of visible enemy champions when a detour exists', parents=['s3'])
def update() -> None:
    g = goals()
    p = me().pos
    others = ally_tiles() - near(p, 0)
    taken = others
    if gems() == 0:
        for a in others:
            taken = taken | near(a, 1)
    free = g - taken
    if size(free) == 0:
        free = g - others
    if size(free) == 0:
        free = g
    if gems() >= 10:
        danger = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, max(u.attack_range, 1) + 1)
        danger = danger - near(p, 0)
        if size(danger) > 0:
            safe = field_avoid(free - danger, danger, near(p, 0))
            if at(safe, p) < 100000:
                GF = safe
                return
    GF = field(free)
