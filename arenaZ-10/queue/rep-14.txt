```python
@gene('s4', 'how much closer a step brings me to the goals; a hurt champion will not step next to an enemy champion')
def progress(t: Tile) -> Num:
    f = goal_field()
    d = at(f, me().pos) - at(f, t)
    if me().health_frac < 0.5 and any(u.is_champion and dist(u.pos, t) <= 2 for u in enemies()):
        return d - 10
    return d
```
