@gene('Z5a', 'once per turn: cache the distance field to the goals', parents=['s3'])
def update() -> None:
    GF = field(goals())

@gene("s2", "where to walk: home to heal when hurt and no enemy in sight, an altar with gems, else a living camp group")
def goals() -> Tiles:
    if me().health_frac < 0.5 and count_units(enemies()) == 0 and gems() == 0:
        return my_spawn()
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
