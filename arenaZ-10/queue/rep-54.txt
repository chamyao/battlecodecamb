@gene("s2", "where to walk: a cleared group awaiting capture (also while carrying gems if it is close), else an altar with gems, else a living camp group")
def goals() -> Tiles:
    p = me().pos
    ready = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if group_reward(g) and not group_respawning(g):
            ts = group_tiles(g)
            if gems() == 0 or dist(nearest(ts, p), p) <= 6:
                ready = ready | ts
    if size(ready) > 0:
        return ready
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
