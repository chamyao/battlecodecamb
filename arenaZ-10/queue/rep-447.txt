```python
GF: Field = None

@gene('s1', 'value of doing this option now; casts abilities, shop plan, recall at 35% (at once with 10+ gems, else only when no step makes progress; 50% when carrying 20+ gems near an enemy champion, 65% with 50+) or to spend banked gold, keeps clearing camps until banking, fights back against awake camps while banking; when hurt with gems (below 50%, 70% with 20+), steps away from enemy champions within 4; finishing blows on enemy champions below 30% beat a low-health recall')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 0.7507
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90 if gems() >= 10 else 48
        if gems() >= 20 and me().health_frac < (0.65 if gems() >= 50 else 0.5) and any(u.is_champion and dist(u.pos, me().pos) <= max(u.attack_range, 1) + 3 for u in enemies()):
            return 88
        return 52 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (0.7996 - o.unit.health_frac) + (10 if o.unit.health_frac < 0.3 else 0)
        if not banking():
            return 60 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and is_awake(o.unit) and me().health_frac > 0.5098:
            return 45
        return 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        pr = progress(o.tile)
        if gems() > 0 and me().health_frac < (0.7 if gems() >= 20 else 0.5) and pr >= 0:
            p = me().pos
            if any(u.is_champion and dist(u.pos, p) <= 4 for u in enemies()):
                if all(dist(u.pos, o.tile) > dist(u.pos, p) for u in enemies() if u.is_champion and dist(u.pos, p) <= 4):
                    return 56
        return 50 if pr > 0 else 0
    return 0

@gene('s2', 'where to walk: a cleared group awaiting capture within 5, else an altar when banking gems, else a living camp group (monoliths only when nothing else is left)')
def goals() -> Tiles:
    p = me().pos
    ready = near(tile(-5, -5), 0)
    acc = near(tile(-5, -5), 0)
    mono = type_spawns(MONOLITH)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g) and dist(nearest(ts, p), p) <= 5:
                ready = ready | ts
            acc = acc | ts
    if size(ready) > 0:
        return ready
    if banking():
        alt = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            alt = alt | near(a, 1)
        return alt
    easy = acc - mono
    if size(easy) > 0:
        return easy
    return acc

@gene('s4', 'how much closer a step brings me to the goals (cached field); a step into the attack reach of an enemy champion counts as no progress when hurt and carrying gems (below 50%, or below 70% with 20+ gems) or below 35% health at all (so a low-health recall starts instead of walking into a fight)')
def progress(t: Tile) -> Num:
    d = at(GF, me().pos) - at(GF, t)
    if d > 0:
        hf = me().health_frac
        if hf < 0.35 or (gems() > 0 and hf < (0.7 if gems() >= 20 else 0.5)):
            if any(u.is_champion and dist(u.pos, t) <= max(u.attack_range, 1) for u in enemies()):
                return 0
    return d

@gene('Z23a', 'shop plan: treasure strength shards before round 500 (they compound longer), then bruiser strength', parents=['s1'])
def buy_value(o: Option) -> Num:
    if round() < 500:
        return 100 if o.item == 18 else 0
    return 100 if o.item == 36 else 0

@gene('Z11a', 'value of casting: damage/cc on enemy champions above everything else, even a low-health recall (more for gem carriers, then the most wounded, area casts that catch a champion also above a recall, by number hit) or camps (awake ones even with gems), heal below 65% (75% with 20+ gems), shield in fights with champions or awake camps, dash toward goals, or dash home when below 35% (45% with 20+ gems) with an enemy champion close', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    rich = gems() >= 20
    if k == AB_HEAL:
        return 85 if me().health_frac < (0.75 if rich else 0.65) else 0
    if k == AB_SHIELD:
        if me().health_frac < 0.7 or any(u.is_champion and dist(u.pos, me().pos) <= 4 for u in enemies()):
            return 78
        if any(is_awake(u) and dist(u.pos, me().pos) <= 3 for u in camps()):
            return 78
        return 0
    if k == AB_DASH:
        if not valid(o.tile):
            return 0
        if me().health_frac < (0.45 if rich else 0.35) and any(u.is_champion and dist(u.pos, me().pos) <= max(u.attack_range, 1) + 3 for u in enemies()):
            home = field(my_spawn())
            return 92 if at(home, me().pos) - at(home, o.tile) >= 2 else 0
        if progress(o.tile) >= 1 and (gems() > 0 or me().health_frac > 0.5):
            return 55
        return 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                return (93 if unit_gems(o.unit) > 0 else 92) + 0.7823 * (1 - o.unit.health_frac)
            if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
                return 62
            return 0
        if valid(o.tile):
            r = max(slot_radius(o.slot), 1)
            n = count(dist(u.pos, o.tile) <= r for u in enemies())
            if n > 0:
                ch = count(u.is_champion and dist(u.pos, o.tile) <= r for u in enemies())
                return (90 if ch > 0 else 80) + min(n, 3)
            if any(dist(u.pos, o.tile) <= r and (gems() == 0 or is_awake(u)) for u in camps()):
                return 61
        return 0
    return 0

@gene('Z160a', 'worth recalling to shop: 60+ gold before round 300, 90+ before round 600 (treasure shards compound), 120+ before round 1200, else 150+ (fewer late trips), or 60+ when already below 55% health (heal and shop in one trip); no gems carried, not home, nothing hostile near, not about to reach a camp (cached field), not in the last 150 rounds', parents=['s1'])
def shop_trip() -> Bool:
    r = round()
    need = 60 if r < 300 else (90 if r < 600 else (120 if r < 1200 else 150))
    if me().health_frac < 0.55:
        need = 60
    if gold() < need or gems() > 0 or can_shop() or r > 1850:
        return False
    p = me().pos
    if any(dist(u.pos, p) <= 6 for u in enemies()):
        return False
    if any(is_awake(u) and dist(u.pos, p) <= 4 for u in camps()):
        return False
    return at(GF, p) > 3

@gene('Z157a', 'once per turn: cache the distance field to the goals not occupied by another ally, nor next to one before round 1000 (spread allies over camps early, group up late; altars only exclude occupied tiles); hurt with no gems, prefer camps away from visible enemy champions; carrying 10+ gems, route around the reach (+1) of visible enemy champions when a detour exists', parents=['s3'])
def update() -> None:
    g = goals()
    p = me().pos
    others = ally_tiles() - near(p, 0)
    taken = others
    if gems() == 0 and round() < 1000:
        for a in others:
            taken = taken | near(a, 1)
    free = g - taken
    if size(free) == 0:
        free = g - others
    if size(free) == 0:
        free = g
    if gems() == 0 and me().health_frac < 0.6:
        hot = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                hot = hot | near(u.pos, max(u.attack_range, 1) + 2)
        calm = free - hot
        if size(calm) > 0:
            free = calm
    if gems() >= 10:
        danger = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, max(u.attack_range, 1) + 1)
        danger = danger - near(p, 0)
        if size(danger) > 0:
            safe = field_avoid(free - danger, danger, near(p, 0))
            if at(safe, p) < 100000:
                GF = safe
                return
    GF = field(free)

@gene('Z519a', 'time to bank carried gems at an altar: 20+ carried, or any when below 50% health or in the last 200 rounds', parents=['s2'])
def banking() -> Bool:
    n = gems()
    if n >= 20:
        return True
    return n > 0 and (me().health_frac < 0.5 or round() > 1800)
```
