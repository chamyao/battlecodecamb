```python
@gene('s4', 'how much closer a step brings me to the goals (cached field); a step into the attack reach of an enemy champion counts as no progress when hurt and carrying gems (below 50%, or below 70% with 20+ gems) or below 35% health at all (so a low-health recall starts instead of walking into a fight)')
def progress(t: Tile) -> Num:
    d = at(GF, me().pos) - at(GF, t)
    if d > 0:
        hf = me().health_frac
        if hf < 0.35 or (gems() > 0 and hf < (0.7 if gems() >= 20 else 0.5)):
            if any(u.is_champion and dist(u.pos, t) <= max(u.attack_range, 1) for u in enemies()):
                return 0
    return d
```
