```python
@gene('s1', 'value of doing this option now; casts abilities; treasure shards early then bruiser strength; recall at 42%')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.42 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        return 60 + 5 * (1 - o.unit.health_frac) if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "shop plan: treasure strength shards before round 500, then bruiser strength", parents=["s1"])
def buy_value(o: Option) -> Num:
    if round() < 500:
        return 100 if o.item == 18 else 0
    return 100 if o.item == 36 else 0

@gene('Z11a', 'value of casting: damage/cc on enemies or camps, heal when hurt, shield in fights, dash toward goals', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_HEAL:
        return 85 if me().health_frac < 0.6 else 0
    if k == AB_SHIELD:
        if me().health_frac < 0.7 or any(u.is_champion and dist(u.pos, me().pos) <= 4 for u in enemies()):
            return 78
        return 0
    if k == AB_DASH:
        if valid(o.tile) and progress(o.tile) >= 2 and (gems() > 0 or me().health_frac > 0.5):
            return 55
        return 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                return 82
            if o.unit.is_camp and gems() == 0:
                return 62
            return 0
        if valid(o.tile):
            r = max(slot_radius(o.slot), 1)
            if any(dist(u.pos, o.tile) <= r for u in enemies()):
                return 81
            if gems() == 0 and any(dist(u.pos, o.tile) <= r for u in camps()):
                return 61
        return 0
    return 0
```
