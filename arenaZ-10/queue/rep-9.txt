```python
@gene("s1", "value of doing this option now; treasure shards early then bruiser strength, recall at 42% health")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.42 else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "shop plan: treasure strength shards early (economy), then bruiser strength, else power strength", parents=["s1"])
def buy_value(o: Option) -> Num:
    if round() < 600:
        return 100 if o.item == 18 else 0
    if o.item == 36:
        return 100
    return 0
```
