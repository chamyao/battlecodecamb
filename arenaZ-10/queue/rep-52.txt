```python
GF: Field = None

@gene("new", "once per turn: cache the distance field to the goals", parents=["s3"])
def update() -> None:
    GF = field(goals())

@gene("s3", "distance field to the goals (cached once per turn by update)")
def goal_field() -> Field:
    return GF

@gene("s2", "where to walk: an altar with gems, else a cleared group awaiting capture, else a living non-monolith camp group with loot")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    ready = near(tile(-5, -5), 0)
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g):
                ready = ready | ts
            elif group_gems(g) + group_gold(g) > 0:
                if spawn_type(nearest(ts, me().pos)) != MONOLITH:
                    acc = acc | ts
    if size(ready) > 0:
        return ready
    if size(acc) == 0:
        for g in range(num_groups()):
            if not group_respawning(g):
                acc = acc | group_tiles(g)
    return acc
```
