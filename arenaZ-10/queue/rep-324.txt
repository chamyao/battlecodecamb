@gene("s2", "where to walk: an altar with gems, else a living camp group (monoliths only when nothing else is left)")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    easy = acc - type_spawns(MONOLITH)
    if size(easy) > 0:
        return easy
    return acc
