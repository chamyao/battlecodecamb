@gene('Z23a', 'shop plan: treasure strength shards before round 500, then bruiser strength', parents=['s1'])
def buy_value(o: Option) -> Num:
    if round() < 500:
        return 100 if o.item == 18 else 0
    return 100 if o.item == 36 else 0

@gene('Z157a', 'once per turn: cache the distance field to the goals not occupied by another ally, nor next to one before round 1000 (spread allies over camps early, group up late; altars only exclude occupied tiles); hurt with no gems, prefer camps away from visible enemy champions; carrying 10+ gems, route around the reach (+1) of visible enemy champions when a detour exists', parents=['s3'])
def update() -> None:
    g = goals()
    p = me().pos
    others = ally_tiles() - near(p, 0)
    taken = others
    if gems() == 0 and round() < 1000:
        for a in others:
            taken = taken | near(a, 1)
    free = g - taken
    if size(free) == 0:
        free = g - others
    if size(free) == 0:
        free = g
    if gems() == 0 and me().health_frac < 0.6:
        hot = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                hot = hot | near(u.pos, max(u.attack_range, 1) + 2)
        calm = free - hot
        if size(calm) > 0:
            free = calm
    if gems() >= 10:
        danger = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, max(u.attack_range, 1) + 1)
        danger = danger - near(p, 0)
        if size(danger) > 0:
            safe = field_avoid(free - danger, danger, near(p, 0))
            if at(safe, p) < 100000:
                GF = safe
                return
    GF = field(free)
