```python
@gene('s4', 'how much closer a step brings me to the goals (cached field); carrying gems while hurt (below 50%, or below 80% with a 40+ gem carry), a step into the attack reach of an enemy champion counts as no progress')
def progress(t: Tile) -> Num:
    d = at(GF, me().pos) - at(GF, t)
    if d > 0 and gems() > 0:
        hf = me().health_frac
        if hf < 0.5 or (gems() >= 40 and hf < 0.8):
            if any(u.is_champion and dist(u.pos, t) <= max(u.attack_range, 1) for u in enemies()):
                return 0
    return d
```
