@gene('s1', 'value of doing this option now; casts abilities, shop plan, recall at 35% only when no step makes progress (50% when carrying 20+ gems near an enemy champion, 65% with 50+) or to spend banked gold, keeps clearing camps while carrying only a few gems, fights back against awake camps while banking; when hurt with gems (below 50%, 70% with 20+), steps away from enemy champions within 4; finishing blows on enemy champions below 30% beat a low-health recall')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 0.7507
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 48
        if gems() >= 20 and me().health_frac < (0.65 if gems() >= 50 else 0.5) and any(u.is_champion and dist(u.pos, me().pos) <= max(u.attack_range, 1) + 3 for u in enemies()):
            return 88
        return 52 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (0.7996 - o.unit.health_frac) + (10 if o.unit.health_frac < 0.3 else 0)
        if not banking():
            return 60 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and is_awake(o.unit) and me().health_frac > 0.5:
            return 45
        return 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        pr = progress(o.tile)
        if gems() > 0 and me().health_frac < (0.7 if gems() >= 20 else 0.5) and pr >= 0:
            p = me().pos
            if any(u.is_champion and dist(u.pos, p) <= 4 for u in enemies()):
                if all(dist(u.pos, o.tile) > dist(u.pos, p) for u in enemies() if u.is_champion and dist(u.pos, p) <= 4):
                    return 56
        return 50 if pr > 0 else 0
    return 0

@gene('s2', 'where to walk: an altar when banking gems, else a living camp group (monoliths only when nothing else is left)')
def goals() -> Tiles:
    if banking():
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    easy = acc - type_spawns(MONOLITH)
    if size(easy) > 0:
        return easy
    return acc

@gene('new', 'time to bank carried gems at an altar: 20+ carried, or any when below 50% health or in the last 200 rounds', parents=['s2'])
def banking() -> Bool:
    n = gems()
    if n >= 20:
        return True
    return n > 0 and (me().health_frac < 0.5 or round() > 1800)
