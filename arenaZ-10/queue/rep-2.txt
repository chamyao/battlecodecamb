@gene("s2", "where to walk: an altar with gems, else a rewarding non-monolith camp group")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g) and (group_gems(g) + group_gold(g) > 0 or group_reward(g)):
            ts = group_tiles(g)
            if spawn_type(nearest(ts, me().pos)) != MONOLITH:
                acc = acc | ts
    if size(acc) == 0:
        for g in range(num_groups()):
            if not group_respawning(g):
                acc = acc | group_tiles(g)
    return acc
