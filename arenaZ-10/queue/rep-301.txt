```python
@gene('s1', 'value of doing this option now; casts abilities, shop plan, recall at 35% (50% when carrying 20+ gems near an enemy champion) or to spend banked gold, fights back against awake camps while carrying gems; finishing blows on enemy champions below 30% beat a low-health recall')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 0.7507
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        if gems() >= 20 and me().health_frac < 0.5 and any(u.is_champion and dist(u.pos, me().pos) <= max(u.attack_range, 1) + 3 for u in enemies()):
            return 88
        return 52 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (0.7996 - o.unit.health_frac) + (10 if o.unit.health_frac < 0.3 else 0)
        if gems() == 0:
            return 60 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and is_awake(o.unit) and me().health_frac > 0.5:
            return 45
        return 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z23a', 'shop plan: treasure strength shards before round 400, then bruiser strength', parents=['s1'])
def buy_value(o: Option) -> Num:
    if round() < 400:
        return 100 if o.item == 18 else 0
    return 100 if o.item == 36 else 0
```
