```python
@gene('s1', 'value of doing this option now; casts abilities, treasure shards early then bruiser strength, recall at 35% (50% when carrying 20+ gems under threat) or to spend banked gold, fights back against awake camps while carrying gems; hurt, does not start a tribute next to an enemy champion')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        if gems() >= 20 and me().health_frac < 0.5 and threatened():
            return 88
        return 52 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        if gems() == 0:
            return 60 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and is_awake(o.unit) and me().health_frac > 0.5:
            return 45
        return 0
    if o.kind == TRIBUTE:
        if me().health_frac < 0.5 and any(u.is_champion and dist(u.pos, o.tile) <= 4 for u in enemies()):
            return 52
        return 71
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
