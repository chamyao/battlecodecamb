@gene('s1', 'value of doing this option now; casts abilities, treasure shards early then bruiser strength, recall at 35% (50% when carrying 20+ gems under threat) or to spend banked gold, fights back against awake camps while carrying gems; when hurt with gems, steps away from nearby enemy champions toward the altar')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        if gems() >= 20 and me().health_frac < 0.5 and threatened():
            return 88
        return 52 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        if gems() == 0:
            return 60 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and is_awake(o.unit) and me().health_frac > 0.5:
            return 45
        return 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        pr = progress(o.tile)
        if gems() > 0 and me().health_frac < 0.5 and pr >= 0 and threatened():
            p = me().pos
            foes = count(u.is_champion and dist(u.pos, p) <= 4 for u in enemies())
            if foes > 0 and all(dist(u.pos, o.tile) > dist(u.pos, p) for u in enemies() if u.is_champion and dist(u.pos, p) <= 4):
                return 56
        return 50 if pr > 0 else 0
    return 0

@gene('Z11a', 'value of casting: damage/cc on enemy champions above everything else, even a low-health recall (gem carriers first, area casts by number hit) or camps, heal below 65% (75% with 20+ gems), shield in any fight (champions or awake camps), dash toward goals or away home when low (below 45% with 20+ gems) and threatened', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    rich = gems() >= 20
    if k == AB_HEAL:
        return 85 if me().health_frac < (0.75 if rich else 0.65) else 0
    if k == AB_SHIELD:
        if me().health_frac < 0.7 or threatened() or any(is_awake(u) and dist(u.pos, me().pos) <= 3 for u in camps()):
            return 78
        return 0
    if k == AB_DASH:
        if not valid(o.tile):
            return 0
        if me().health_frac < (0.45 if rich else 0.35) and threatened():
            home = field(my_spawn())
            return 92 if at(home, me().pos) - at(home, o.tile) >= 2 else 0
        if progress(o.tile) >= 2 and (gems() > 0 or me().health_frac > 0.5):
            return 55
        return 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                return 93 if unit_gems(o.unit) > 0 else 92
            if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
                return 62
            return 0
        if valid(o.tile):
            r = max(slot_radius(o.slot), 1)
            n = count(dist(u.pos, o.tile) <= r for u in enemies())
            if n > 0:
                return 80 + min(n, 3)
            if any(dist(u.pos, o.tile) <= r and (gems() == 0 or is_awake(u)) for u in camps()):
                return 61
        return 0
    return 0
