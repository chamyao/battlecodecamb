@gene('s4', 'how much closer a step brings me to the goals (cached field); hurt and carrying gems, a step into the attack reach of an enemy champion counts as no progress')
def progress(t: Tile) -> Num:
    d = at(GF, me().pos) - at(GF, t)
    if d > 0 and gems() > 0 and me().health_frac < 0.5:
        if any(u.is_champion and dist(u.pos, t) <= max(u.attack_range, 1) for u in enemies()):
            return 0
    return d

@gene('Z23a', 'shop plan: treasure strength shards before round 450, then bruiser strength; biggest batch first', parents=['s1'])
def buy_value(o: Option) -> Num:
    want = 18 if round() < 450 else 36
    return 100 + o.count if o.item == want else 0
