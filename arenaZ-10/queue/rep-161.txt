@gene('Z127b', 'once per turn: cache the distance field to the goals not occupied by another ally; while carrying 10+ gems, path around visible enemy champions (their attack reach when I am below 60%, else adjacent tiles)', parents=['s4'])
def update() -> None:
    g = goals()
    free = g - (ally_tiles() - near(me().pos, 0))
    if size(free) == 0:
        free = g
    danger = near(tile(-5, -5), 0)
    if gems() >= 10:
        hurt = me().health_frac < 0.6
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, max(u.attack_range, 1) if hurt else 1)
    danger = danger - near(me().pos, 0) - free
    if size(danger) == 0:
        GF = field(free)
    else:
        GF = field_avoid(free, walls() | danger, near(tile(-5, -5), 0))
