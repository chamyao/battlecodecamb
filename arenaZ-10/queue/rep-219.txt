```python
@gene('Z232a', 'where to walk: a cleared group awaiting capture within 5, else an altar with gems (but with under 15 gems and above 60% health, first a living non-monolith group within 6), else a living camp group (monoliths only when nothing else is left)', parents=['s2'])
def goals() -> Tiles:
    p = me().pos
    ready = near(tile(-5, -5), 0)
    acc = near(tile(-5, -5), 0)
    close = near(tile(-5, -5), 0)
    mono = type_spawns(MONOLITH)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            d = dist(nearest(ts, p), p)
            if group_reward(g) and d <= 5:
                ready = ready | ts
            if d <= 6:
                close = close | ts
            acc = acc | ts
    if size(ready) > 0:
        return ready
    if gems() > 0:
        near_camp = close - mono
        if gems() < 15 and me().health_frac > 0.6 and size(near_camp) > 0:
            return near_camp
        alt = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            alt = alt | near(a, 1)
        return alt
    easy = acc - mono
    if size(easy) > 0:
        return easy
    return acc
```
