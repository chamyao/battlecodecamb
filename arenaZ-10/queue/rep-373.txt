```python
@gene("s3", "distance field to the goals not occupied by or next to another ally (spread allies over camps; altars only exclude occupied tiles)")
def goal_field() -> Field:
    g = goals()
    others = ally_tiles() - near(me().pos, 0)
    taken = others
    if gems() == 0:
        for a in others:
            taken = taken | near(a, 1)
    free = g - taken
    if size(free) == 0:
        free = g - others
    if size(free) == 0:
        free = g
    return field(free)
```
