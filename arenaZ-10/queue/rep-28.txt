```python
@gene('s1', 'value of doing this option now; casts abilities; drafts ladder favourites; treasure shards early then bruiser strength; recall at 42%')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.42 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        return 41 + 5 * (1 - o.unit.health_frac) if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 71
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "shop plan: treasure strength shards before round 500, then bruiser strength", parents=["s1"])
def buy_value(o: Option) -> Num:
    if round() < 500:
        return 100 if o.item == 18 else 0
    return 100 if o.item == 36 else 0

@gene("new", "draft preference: Vesper Vale, Heart of Ruin, Marcella, Vendetta, Ryth, then anything", parents=["s1"])
def pick_value(c: Num) -> Num:
    if c == VESPER_VALE:
        return 10
    if c == HEART_OF_RUIN:
        return 9
    if c == MARCELLA:
        return 8
    if c == VENDETTA:
        return 7
    if c == RYTH:
        return 6
    return 1
```
