@gene('Z138a', 'once per turn: cache the distance field to the goals not occupied by or next to another ally (spread allies over camps; altars only exclude occupied tiles); while carrying gems the path routes around the attack reach (+1) of visible enemy champions when a detour exists', parents=['s3'])
def update() -> None:
    g = goals()
    others = ally_tiles() - near(me().pos, 0)
    taken = others
    if gems() == 0:
        for a in others:
            taken = taken | near(a, 1)
    free = g - taken
    if size(free) == 0:
        free = g - others
    if size(free) == 0:
        free = g
    if gems() > 0:
        danger = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, max(u.attack_range, 1) + 1)
        danger = danger - near(me().pos, 0)
        if size(danger) > 0:
            safe = field_avoid(free - danger, danger, near(me().pos, 0))
            if at(safe, me().pos) < 100000:
                GF = safe
                return
    GF = field(free)
