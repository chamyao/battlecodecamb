@gene('s1', 'value of doing this option now; casts abilities, shop plan, recall at 38% or to spend banked gold, fights back against awake camps while carrying gems')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.38:
            return 90
        return 52 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        if gems() == 0:
            return 60 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and is_awake(o.unit) and me().health_frac > 0.5:
            return 45
        return 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z23a', 'shop plan: treasure strength shards before round 500, then bruiser strength; biggest affordable batch first', parents=['s1'])
def buy_value(o: Option) -> Num:
    if round() < 500:
        return 100 + o.count if o.item == 18 else 0
    return 100 + o.count if o.item == 36 else 0

PF: Field = None

@gene("new", "once per turn: cache the goal field for progress", parents=["s4"])
def update() -> None:
    PF = goal_field()

@gene('s4', 'how much closer a step brings me to the goals (field cached once per turn)')
def progress(t: Tile) -> Num:
    return at(PF, me().pos) - at(PF, t)
