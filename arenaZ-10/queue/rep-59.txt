```python
GF: Field = None

@gene("new", "once per turn: cache the distance field to the goals", parents=["s3"])
def update() -> None:
    GF = field(goals())

@gene("s3", "distance field to the goals (cached once per turn in update)")
def goal_field() -> Field:
    return GF

@gene("s4", "how much closer a step brings me to the goals")
def progress(t: Tile) -> Num:
    f = goal_field()
    return at(f, me().pos) - at(f, t)

@gene('s1', 'value of doing this option now; casts abilities; treasure shards early then bruiser strength; recall at 42% or to spend 150+ banked gold when safe')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.42:
            return 90
        return 52 if safe_to_shop() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        return 60 + 5 * (1 - o.unit.health_frac) if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "worth recalling to shop: 150+ gold, no gems carried, not home, nothing hostile within 6", parents=["s1"])
def safe_to_shop() -> Bool:
    if gold() < 150 or gems() > 0 or can_shop():
        return False
    p = me().pos
    if any(dist(u.pos, p) <= 6 for u in enemies()):
        return False
    return not any(is_awake(u) and dist(u.pos, p) <= 4 for u in camps())

@gene('Z52a', 'shop plan: treasure strength shards before round 500, then bruiser strength; biggest affordable batch first', parents=['s1'])
def buy_value(o: Option) -> Num:
    if round() < 500:
        return 100 + o.count if o.item == 18 else 0
    return 100 + o.count if o.item == 36 else 0
```
