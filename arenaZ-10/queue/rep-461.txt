@gene('s4', 'how much closer a step brings me to the goals (cached field); a step into the attack reach of an enemy champion counts as no progress when hurt and carrying gems (below 50%, or below 70% with 20+ gems) or below 35% health at all (so a low-health recall starts instead of walking into a fight)')
def progress(t: Tile) -> Num:
    d = at(GF, me().pos) - at(GF, t)
    if d > 0:
        hf = me().health_frac
        if hf < 0.35 or (gems() > 0 and hf < (0.7 if gems() >= 20 else 0.5)):
            if any(u.is_champion and dist(u.pos, t) <= max(u.attack_range, 1) for u in enemies()):
                return 0
    return d

@gene('Z160a', 'worth recalling to shop: 60+ gold before round 300, 90+ before round 600 (treasure shards compound), 120+ before round 1200, else 150+ (fewer late trips), or 60+ when already below 55% health (heal and shop in one trip); no gems carried, not home, nothing hostile near, not about to reach a camp (cached field), not in the last 150 rounds', parents=['s1'])
def shop_trip() -> Bool:
    r = round()
    need = 60 if r < 300 else (90 if r < 600 else (120 if r < 1200 else 150))
    if me().health_frac < 0.55:
        need = 60
    if gold() < need or gems() > 0 or can_shop() or r > 1850:
        return False
    p = me().pos
    if any(dist(u.pos, p) <= 6 for u in enemies()):
        return False
    if any(is_awake(u) and dist(u.pos, p) <= 4 for u in camps()):
        return False
    return at(GF, p) > 3
