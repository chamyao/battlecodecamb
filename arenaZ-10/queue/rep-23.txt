@gene('s2', 'where to walk: an altar with gems, else a living camp group (golems allowed from round 800, never monoliths)')
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    hard = type_spawns(GOLEM) | type_spawns(MONOLITH)
    if round() >= 800:
        hard = type_spawns(MONOLITH)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | (group_tiles(g) - hard)
    if size(acc) == 0:
        for g in range(num_groups()):
            if not group_respawning(g):
                acc = acc | group_tiles(g)
    return acc

@gene('s1', 'value of doing this option now; casts abilities; treasure shards early then bruiser strength')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.4 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        return 60 + 5 * (1 - o.unit.health_frac) if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "shop plan: treasure strength shards before round 500, then bruiser strength", parents=["s1"])
def buy_value(o: Option) -> Num:
    if round() < 500:
        return 100 if o.item == 18 else 0
    return 100 if o.item == 36 else 0
