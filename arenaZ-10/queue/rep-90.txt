GF: Field = None

@gene("new", "once per turn: cache the distance field to the goals not occupied by another ally", parents=["s3"])
def update() -> None:
    g = goals()
    free = g - (ally_tiles() - near(me().pos, 0))
    if size(free) == 0:
        free = g
    GF = field(free)

@gene("s3", "distance field to the goals (cached once per turn in update)")
def goal_field() -> Field:
    return GF


@gene("s2", "where to walk: a cleared group awaiting capture within 5, else an altar with gems, else a living camp group")
def goals() -> Tiles:
    p = me().pos
    ready = near(tile(-5, -5), 0)
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g) and dist(nearest(ts, p), p) <= 5:
                ready = ready | ts
            acc = acc | ts
    if size(ready) > 0:
        return ready
    if gems() > 0:
        alt = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            alt = alt | near(a, 1)
        return alt
    return acc
