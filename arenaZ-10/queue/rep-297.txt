@gene('Z53a', 'worth recalling to shop: 60+ gold before round 300, 90+ before round 600 (treasure shards compound), else 120+, or 60+ when already below 55% health (heal and shop in one trip); no gems carried, not home, nothing hostile near, not about to reach a camp (finish it first), not in the last 150 rounds', parents=['s1'])
def shop_trip() -> Bool:
    r = round()
    need = 60 if r < 300 else (90 if r < 600 else 120)
    if me().health_frac < 0.55:
        need = 60
    if gold() < need or gems() > 0 or can_shop() or round() > 1850:
        return False
    p = me().pos
    if any(dist(u.pos, p) <= 6 for u in enemies()):
        return False
    if any(is_awake(u) and dist(u.pos, p) <= 4 for u in camps()):
        return False
    return at(GF, p) > 3
