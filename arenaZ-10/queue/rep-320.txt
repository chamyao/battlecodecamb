```python
@gene('s1', 'value of doing this option now; casts abilities, shop plan, recall at 35% (50% when carrying 20+ gems near an enemy champion, 65% with 50+) or to spend banked gold, fights back against awake camps while carrying gems; when hurt with gems, steps away from enemy champions within 4 toward the altar')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 0.7507
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        if gems() >= 20 and me().health_frac < (0.65 if gems() >= 50 else 0.5) and any(u.is_champion and dist(u.pos, me().pos) <= max(u.attack_range, 1) + 3 for u in enemies()):
            return 88
        return 52 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (0.7996 - o.unit.health_frac)
        if gems() == 0:
            return 60 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and is_awake(o.unit) and me().health_frac > 0.5:
            return 45
        return 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        pr = progress(o.tile)
        if gems() > 0 and me().health_frac < 0.5 and pr >= 0:
            p = me().pos
            foes = count(u.is_champion and dist(u.pos, p) <= 4 for u in enemies())
            if foes > 0 and all(dist(u.pos, o.tile) > dist(u.pos, p) for u in enemies() if u.is_champion and dist(u.pos, p) <= 4):
                return 56
        return 50 if pr > 0 else 0
    return 0
```
