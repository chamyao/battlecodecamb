```python
@gene('Z519a', 'time to bank carried gems at an altar: 20+ carried, 8+ when a free altar is within 6 tiles (a short detour), or any when below 50% health or in the last 200 rounds', parents=['s2'])
def banking() -> Bool:
    n = gems()
    if n >= 20:
        return True
    if n == 0:
        return False
    if me().health_frac < 0.5 or round() > 1800:
        return True
    if n >= 8:
        p = me().pos
        free = altars() - locked_altars()
        if size(free) > 0 and dist(nearest(free, p), p) <= 6:
            return True
    return False
```
