@gene("s2", "where to walk: a nearby cleared group to capture, an altar with gems, else a living camp group (golems from round 800, never monoliths)")
def goals() -> Tiles:
    ready = near(tile(-5, -5), 0)
    acc = near(tile(-5, -5), 0)
    hard = type_spawns(GOLEM) | type_spawns(MONOLITH)
    if round() >= 800:
        hard = type_spawns(MONOLITH)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g) and dist(nearest(ts, me().pos), me().pos) <= 5:
                ready = ready | ts
            acc = acc | (ts - hard)
    if size(ready) > 0:
        return ready
    if gems() > 0:
        alt = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            alt = alt | near(a, 1)
        return alt
    if size(acc) == 0:
        for g in range(num_groups()):
            if not group_respawning(g):
                acc = acc | group_tiles(g)
    return acc
