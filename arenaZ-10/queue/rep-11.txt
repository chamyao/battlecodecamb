@gene('s2', 'where to walk: an altar with gems, else a cleared group awaiting capture, else a living camp group (no golem/monolith before round 1000)')
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    ready = near(tile(-5, -5), 0)
    acc = near(tile(-5, -5), 0)
    hard = type_spawns(GOLEM) | type_spawns(MONOLITH)
    if round() >= 1000:
        hard = type_spawns(MONOLITH)
    for g in range(num_groups()):
        if not group_respawning(g):
            if group_reward(g):
                ready = ready | group_tiles(g)
            acc = acc | (group_tiles(g) - hard)
    if size(ready) > 0:
        return ready
    if size(acc) == 0:
        for g in range(num_groups()):
            if not group_respawning(g):
                acc = acc | group_tiles(g)
    return acc
