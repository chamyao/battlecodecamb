GF: Field = None

@gene("new", "once per turn: cache the distance field to the goals not occupied by another ally", parents=["s3"])
def update() -> None:
    g = goals()
    free = g - (ally_tiles() - near(me().pos, 0))
    if size(free) == 0:
        free = g
    GF = field(free)

@gene("s3", "distance field to the goals (cached once per turn in update)")
def goal_field() -> Field:
    return GF

@gene('Z53a', 'worth recalling to shop: 150+ gold banked, no gems carried, nothing hostile near, not already home, not in the last 150 rounds', parents=['s1'])
def shop_trip() -> Bool:
    if gold() < 150 or gems() > 0 or can_shop() or round() > 1850:
        return False
    p = me().pos
    if any(dist(u.pos, p) <= 6 for u in enemies()):
        return False
    return not any(is_awake(u) and dist(u.pos, p) <= 4 for u in camps())
