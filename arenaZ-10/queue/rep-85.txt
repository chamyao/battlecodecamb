```python
@gene('Z11a', 'value of casting: damage/cc on enemies (stun tributing or recalling enemies first, then gem carriers, area casts by number hit) or camps (awake camps even with gems), heal when hurt, shield when threatened, dash toward goals (never next to an enemy champion while carrying gems) or home when below 40% and threatened', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_HEAL:
        return 85 if me().health_frac < 0.7 else 0
    if k == AB_SHIELD:
        if me().health_frac < 0.7 or threatened():
            return 78
        return 0
    if k == AB_DASH:
        if not valid(o.tile):
            return 0
        if me().health_frac < 0.4 and threatened():
            home = field(my_spawn())
            return 92 if at(home, me().pos) - at(home, o.tile) >= 2 else 0
        if progress(o.tile) >= 2 and (gems() > 0 or me().health_frac > 0.3737):
            if gems() > 0 and any(u.is_champion and dist(u.pos, o.tile) <= 3 for u in enemies()):
                return 0
            return 55
        return 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                if k == AB_CC and (is_tributing(o.unit) or is_recalling(o.unit)):
                    return 88
                return 83 if unit_gems(o.unit) > 0 else 82
            if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
                return 62
            return 0
        if valid(o.tile):
            r = max(slot_radius(o.slot), 1)
            n = count(dist(u.pos, o.tile) <= r for u in enemies())
            if n > 0:
                return 80 + min(n, 3)
            if any(dist(u.pos, o.tile) <= r and (gems() == 0 or is_awake(u)) for u in camps()):
                return 61
        return 0
    return 0
```
