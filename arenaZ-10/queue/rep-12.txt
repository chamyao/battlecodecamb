```python
GF: Field = None

@gene("new", "once per turn: cache the goal distance field, routing around enemy champions while carrying gems", parents=["s3"])
def update() -> None:
    if gems() > 0 and count_units(enemies()) > 0:
        block = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                block = block | near(u.pos, 1)
        GF = field_avoid(goals(), walls() | (block - near(me().pos, 0)), near(tile(-5, -5), 0))
    else:
        GF = field(goals())

@gene("s3", "distance field to the goals (cached once per turn in update)")
def goal_field() -> Field:
    return GF

@gene("s4", "how much closer a step brings me to the goals")
def progress(t: Tile) -> Num:
    f = goal_field()
    return at(f, me().pos) - at(f, t)
```
