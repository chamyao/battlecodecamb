```python
@gene('s1', 'value of doing this option now; casts abilities, treasure shards early then bruiser strength, recall at 35% or to spend banked gold, fights back against awake camps while carrying gems')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.35:
            return 90
        return 52 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        if gems() == 0:
            return 60 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and is_awake(o.unit) and me().health_frac > 0.5:
            return 45
        return 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('s2', 'where to walk: a cleared group awaiting capture within 5, else an altar with gems, else a living camp group (monoliths only when nothing else is left)')
def goals() -> Tiles:
    p = me().pos
    ready = near(tile(-5, -5), 0)
    acc = near(tile(-5, -5), 0)
    mono = type_spawns(MONOLITH)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if group_reward(g) and dist(nearest(ts, p), p) <= 5:
                ready = ready | ts
            acc = acc | ts
    if size(ready) > 0:
        return ready
    if gems() > 0:
        alt = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            alt = alt | near(a, 1)
        return alt
    easy = acc - mono
    if size(easy) > 0:
        return easy
    return acc

@gene("new", "worth recalling to shop: 120+ gold banked, no gems carried, not home, nothing hostile near, not about to reach a camp, not in the last 150 rounds", parents=["s1"])
def shop_trip() -> Bool:
    if gold() < 120 or gems() > 0 or can_shop() or round() > 1850:
        return False
    p = me().pos
    if any(dist(u.pos, p) <= 6 for u in enemies()):
        return False
    if any(is_awake(u) and dist(u.pos, p) <= 4 for u in camps()):
        return False
    return at(goal_field(), p) > 3
```
