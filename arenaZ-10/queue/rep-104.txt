```python
GF: Field = None

@gene("new", "once per turn: cache the distance field to the goals not occupied by another ally", parents=["s3"])
def update() -> None:
    g = goals()
    free = g - (ally_tiles() - near(me().pos, 0))
    if size(free) == 0:
        free = g
    GF = field(free)

@gene("s3", "distance field to the goals (cached once per turn in update)")
def goal_field() -> Field:
    return GF

@gene('s1', 'value of doing this option now; casts abilities, drafts ladder favourites, shop plan, recall at 38% or to spend banked gold, fights back against awake camps while carrying gems')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.38:
            return 90
        return 52 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        if gems() == 0:
            return 60 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and is_awake(o.unit) and me().health_frac > 0.5:
            return 45
        return 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
