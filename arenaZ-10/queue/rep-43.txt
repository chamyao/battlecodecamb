```python
@gene('s1', 'value of doing this option now; casts abilities, treasure-then-bruiser shop plan, recall at 42% or to spend banked gold')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.42:
            return 90
        return 52 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        return 60 + 5 * (1 - o.unit.health_frac) if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "shop plan: treasure strength shards before round 500, then bruiser strength", parents=["s1"])
def buy_value(o: Option) -> Num:
    if round() < 500:
        return 100 if o.item == 18 else 0
    return 100 if o.item == 36 else 0

@gene("new", "worth recalling to shop: 150+ gold banked, no gems carried, nothing hostile near, not already home", parents=["s1"])
def shop_trip() -> Bool:
    if gold() < 150 or gems() > 0 or can_shop():
        return False
    p = me().pos
    if any(dist(u.pos, p) <= 6 for u in enemies()):
        return False
    return not any(is_awake(u) and dist(u.pos, p) <= 4 for u in camps())
```
