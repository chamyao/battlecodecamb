```python
@gene('s1', 'value of doing this option now; casts abilities, drafts ladder favourites, shop plan, recall at 42% or to spend banked gold, fights back against awake camps while carrying gems')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return buy_value(o)
    if o.kind == RECALL:
        if me().health_frac < 0.42:
            return 90
        return 52 if shop_trip() else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        if gems() == 0:
            return 60 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and is_awake(o.unit) and me().health_frac > 0.5:
            return 45
        return 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "worth recalling to shop: 150+ gold banked, no gems carried, nothing hostile near, not already home", parents=["s1"])
def shop_trip() -> Bool:
    if gold() < 150 or gems() > 0 or can_shop():
        return False
    p = me().pos
    if any(dist(u.pos, p) <= 6 for u in enemies()):
        return False
    return not any(is_awake(u) and dist(u.pos, p) <= 4 for u in camps())

@gene('Z11a', 'value of casting: damage/cc on enemies or camps, heal when hurt, shield in fights, dash toward goals or away home when low and threatened', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_HEAL:
        return 85 if me().health_frac < 0.6 else 0
    if k == AB_SHIELD:
        if me().health_frac < 0.7 or any(u.is_champion and dist(u.pos, me().pos) <= 4 for u in enemies()):
            return 78
        return 0
    if k == AB_DASH:
        if not valid(o.tile):
            return 0
        if me().health_frac < 0.35 and threatened():
            home = field(my_spawn())
            return 92 if at(home, me().pos) - at(home, o.tile) >= 2 else 0
        if progress(o.tile) >= 2 and (gems() > 0 or me().health_frac > 0.5):
            return 55
        return 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                return 82
            if o.unit.is_camp and gems() == 0:
                return 62
            return 0
        if valid(o.tile):
            r = max(slot_radius(o.slot), 1)
            if any(dist(u.pos, o.tile) <= r for u in enemies()):
                return 81
            if gems() == 0 and any(dist(u.pos, o.tile) <= r for u in camps()):
                return 61
        return 0
    return 0

@gene("new", "is something about to hit me: an enemy champion within 4 or an awake camp within 2", parents=["Z11a"])
def threatened() -> Bool:
    p = me().pos
    if any(u.is_champion and dist(u.pos, p) <= 4 for u in enemies()):
        return True
    return any(is_awake(u) and dist(u.pos, p) <= 2 for u in camps())
```
