```python
@gene('s1', 'value of doing this option now; casts abilities; keeps fighting an awake camp even while carrying gems')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + 5 * (1 - o.unit.health_frac)
        if gems() == 0:
            return 41 + 5 * (1 - o.unit.health_frac)
        if o.unit.is_camp and is_awake(o.unit) and me().health_frac > 0.5:
            return 40 + 5 * (1 - o.unit.health_frac)
        return 0
    if o.kind == TRIBUTE:
        return 71
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z11a', 'value of casting an ability: damage/cc on enemies or camps (awake camps even with gems), heal when hurt, shield in fights, dash toward goals', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_HEAL:
        return 85 if me().health_frac < 0.6 else 0
    if k == AB_SHIELD:
        if me().health_frac < 0.7 or any(u.is_champion and dist(u.pos, me().pos) <= 4 for u in enemies()):
            return 78
        return 0
    if k == AB_DASH:
        if valid(o.tile) and gems() == 0 and me().health_frac > 0.5 and progress(o.tile) >= 2:
            return 55
        if valid(o.tile) and gems() > 0 and progress(o.tile) >= 2:
            return 55
        return 0
    if k == AB_DAMAGE or k == AB_AOE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                return 82
            if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
                return 62
            return 0
        if valid(o.tile):
            r = max(slot_radius(o.slot), 1.694)
            if any(dist(u.pos, o.tile) <= r for u in enemies()):
                return 81
            if any(dist(u.pos, o.tile) <= r and (gems() == 0 or is_awake(u)) for u in camps()):
                return 61
        return 0
    return 0
```
