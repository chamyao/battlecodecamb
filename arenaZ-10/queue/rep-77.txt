@gene('Z85a', 'once per turn: cache the distance field to the goals not occupied by another ally; with gems, route around awake camps', parents=['s3'])
def update() -> None:
    g = goals()
    free = g - (ally_tiles() - near(me().pos, 0))
    if size(free) == 0:
        free = g
    if gems() > 0:
        block = near(tile(-5, -5), 0)
        for u in camps():
            if is_awake(u):
                block = block | near(u.pos, 1)
        block = block - near(me().pos, 0)
        if size(block) > 0:
            f = field_avoid(free, walls() | block, near(tile(-5, -5), 0))
            if at(f, me().pos) < 100000:
                GF = f
                return
    GF = field(free)
