```python
@gene('Z53a', 'worth recalling to shop: 90+ gold banked before round 600 (treasure shards compound), else 120+; no gems carried, not home, nothing hostile near, not about to reach a camp (finish it first), not in the last 150 rounds', parents=['s1'])
def shop_trip() -> Bool:
    need = 90 if round() < 600 else 120
    if gold() < need or gems() > 0 or can_shop() or round() > 1850:
        return False
    p = me().pos
    if any(dist(u.pos, p) <= 6 for u in enemies()):
        return False
    if any(is_awake(u) and dist(u.pos, p) <= 4 for u in camps()):
        return False
    return at(GF, p) > 3

@gene('s4', 'how much closer a step brings me to the goals (cached field); hurt and carrying gems, a step into an enemy champion\'s attack reach counts as no progress')
def progress(t: Tile) -> Num:
    d = at(GF, me().pos) - at(GF, t)
    if d > 0 and gems() > 0 and me().health_frac < 0.5:
        if any(u.is_champion and dist(u.pos, t) <= max(u.attack_range, 1) for u in enemies()):
            return 0
    return d
```
