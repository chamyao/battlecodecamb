@gene('Z157a', 'once per turn: cache the distance field to the goals not occupied by or next to another ally (spread allies over camps; altars only exclude occupied tiles); carrying 10+ gems, route around the reach (+1) of visible enemy champions when a detour exists', parents=['s3'])
def update() -> None:
    g = goals()
    p = me().pos
    others = ally_tiles() - near(p, 0)
    taken = others
    if gems() == 0:
        for a in others:
            taken = taken | near(a, 1)
    free = g - taken
    if size(free) == 0:
        free = g - others
    if size(free) == 0:
        free = g
    if gems() >= 10:
        danger = near(tile(-5, -5), 0)
        for u in enemies():
            if u.is_champion:
                danger = danger | near(u.pos, max(u.attack_range, 1) + 1)
        danger = danger - near(p, 0)
        if size(danger) > 0:
            safe = field_avoid(free - danger, danger, near(p, 0))
            if at(safe, p) < 100000:
                GF = safe
                return
    GF = field(free)

@gene('Z53a', 'worth recalling to shop: 60+ gold before round 300, 90+ before round 600 (treasure shards compound), else 120+, or 60+ when already below 55% health (heal and shop in one trip); no gems carried, not home, nothing hostile near, not about to reach a camp (finish it first), not in the last 150 rounds', parents=['s1'])
def shop_trip() -> Bool:
    r = round()
    need = 60 if r < 300 else (90 if r < 600 else 120)
    if me().health_frac < 0.55:
        need = 60
    if gold() < need or gems() > 0 or can_shop() or r > 1850:
        return False
    p = me().pos
    if any(dist(u.pos, p) <= 6 for u in enemies()):
        return False
    if any(is_awake(u) and dist(u.pos, p) <= 4 for u in camps()):
        return False
    return at(GF, p) > 3
