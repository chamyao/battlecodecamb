```python
@gene('a5', 'cast-ahead applies: nearest enemy within 5 (6 when it carries 10+ gems: worth the reach) and not on its spawn or recalling home')
def ahead_on() -> Bool:
    f = ahead_foe()
    if not exists(f) or f.pos in enemy_spawn() or is_recalling(f):
        return False
    d = dist(here(), f.pos)
    return d <= 5 or (d <= 6 and unit_gems(f) >= 10)
```
