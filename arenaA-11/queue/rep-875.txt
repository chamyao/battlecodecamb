```python
@gene('a9', "shopping on spawn: the team's altar lock, then shards in the policy's split, with the durability health emblem (10% less damage, for champion fights) before the jungle emblem; below 60% health a health shard comes first (spawn health purchases heal)")
def shop_value(o: Option) -> Num:
    if o.kind == BUY_LOCK:
        return 200 if LOCK_BUY and RULE >= 2 and at_spawn() else 0
    want = next_shard()
    if (want == 54 or want == 18) and me().health_frac < 0.6 and items_owned(17) == 0:
        want = 16
    elif (want == 28 or want == 36) and items_owned(17) == 0:
        want = 16
    if o.item != want:
        return 0
    if o.count > shard_left() or RULE < 2 or not at_spawn():
        return 0
    return 190 + o.count / 100
```
