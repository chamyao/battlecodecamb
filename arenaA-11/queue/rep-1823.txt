@gene('c1', "where this turn's rule walks; banking under threat, and banking before a dangerous group, go to the bank walk's altar (safe, unlocked)")
def rule_goal() -> Tile:
    r = RULE
    if r == 3:
        return closest(my_spawn())
    if r == 11 or r == 12 or r == 18:
        return bank_goal()
    if r == 15:
        return closest(camp_spawns() - respawning() - type_spawns(GOLEM))
    if r == 19:
        return live_camp().pos
    if r == 20:
        return camp_goal()
    return tile(-1, -1)
