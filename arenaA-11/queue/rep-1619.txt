```python
@gene('r3', 'retreat on foot: below 35% health and under fire or with a threat near, or below 50% under fire while carrying 15+ gems (a death hands the bag to the killer)')
def r_run() -> Bool:
    if at_spawn():
        return False
    if hurt_below(0.35) and (under_fire() or threat_near()):
        return True
    return gems() >= 15 and hurt_below(0.5) and under_fire()
```
