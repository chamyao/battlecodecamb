@gene('s11', 'turns to stand next to an altar we can tribute at (1000 if none); only enemy locks block a tribute, so our own locked altar counts')
def altar_eta() -> Num:
    la = locked_altars()
    d = min((max(0, dist(here(), a) - 1) for a in altars() if not (a in la and lock_enemy(a))), default=-1)
    if d < 0:
        return 1000
    return eta(d, me().move_speed)
