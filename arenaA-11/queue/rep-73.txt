```python
@gene('r4', 'recall: below 30% health (45% carrying 10+ gems) and nothing near (no enemy champion able to hit us within 2 turns; damage cancels the channel, so under pressure the retreat on foot runs instead)')
def r_recall() -> Bool:
    if at_spawn():
        return False
    return (hurt_below(0.3) or (gems() >= 10 and hurt_below(0.45))) and threat_eta() > 2
```
