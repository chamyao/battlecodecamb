def shop_value(o: Option) -> Num:
    if o.kind == BUY_LOCK:
        return 200 if LOCK_BUY and RULE >= 2 and at_spawn() and round() < 1200 else 0
    want = next_shard()
    if items_owned(55) == 0 and items_owned(19) == 0:
        want = 54

        want = 16

        return 0
    if o.count > shard_left() or RULE < 2 or not at_spawn():
        return 0
    return 190 + o.count / 100
