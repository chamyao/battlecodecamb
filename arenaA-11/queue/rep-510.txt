@gene('m11', "the camp walk's goal: weak gem-paying groups first once I have bought 4 times, else the nearest non-big spawn of a live group that no ally is already capturing, falling back to any non-big camp spawn, then any camp spawn")
def camp_goal() -> Tile:
    if BUYS >= 4 and size(GEM_BASE) > 0:
        t = closest(GEM_BASE - respawning() - held_groups())
        if valid(t):
            return t
    t = closest(CAMP_BASE - respawning() - BIG_CAMPS - held_groups())
    if valid(t):
        return t
    t = closest(CAMP_BASE - BIG_CAMPS)
    if valid(t):
        return t
    return closest(CAMP_BASE)
