@gene('s5', 'a visible enemy champion can hit me now or after one step (move + attack); nearly dead ones (15% or less) count only if they carry gems')
def foe_in_reach() -> Bool:
    h = here()
    r = me().attack_range
    return any(dist(h, u.pos) <= max(r, u.attack_range + u.move_speed) for u in enemies() if u.health_frac > 0.15 or unit_gems(u) > 0)
