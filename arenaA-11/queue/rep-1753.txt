@gene('a4', 'cast-ahead target: enemies off their spawn and not recalling home, carrying the most gems, then nearest, with a strong pull toward wounded ones (finishable)')
def ahead_foe() -> Unit:
    es = enemy_spawn()
    h = here()
    return best(enemies(), lambda u: -INF if (u.pos in es or is_recalling(u)) else unit_gems(u) * 10 - dist(h, u.pos) - 16 * u.health_frac)
