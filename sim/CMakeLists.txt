cmake_minimum_required(VERSION 3.20)
project(cbs CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -g")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options(-Wall -Wextra -Wno-unused-parameter)

file(GLOB CBS_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(cbs STATIC ${CBS_SOURCES})
target_include_directories(cbs PUBLIC include)
# Integer bit-manipulation instructions only (std::popcount / countr_zero would otherwise be libgcc
# calls). No FMA or other float-changing ISA: results must stay bit-identical (-ffp-contract=off
# keeps it so even if a later flag enables FMA). Every x86-64 CPU since ~2013 has these.
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  set(CBS_ISA_FLAGS -mpopcnt -mbmi -mbmi2 -mlzcnt -ffp-contract=off)
  target_compile_options(cbs PUBLIC ${CBS_ISA_FLAGS})
endif()

enable_testing()
add_executable(geomtest tests/geomtest.cpp)
target_link_libraries(geomtest cbs)
set(REPO ${CMAKE_CURRENT_SOURCE_DIR}/..)
foreach(m 00elkring 00great-wall 01islands 00lanes)
  add_test(NAME geom_${m} COMMAND sh -c "zcat ${REPO}/oracle/fixtures/geom/${m}.txt.gz | $<TARGET_FILE:geomtest> ${REPO}/maps/${m}.map26a")
endforeach()
add_executable(maptest tests/maptest.cpp)
target_link_libraries(maptest cbs)
add_test(NAME maps COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_maps.py $<TARGET_FILE:maptest> ${REPO})
add_executable(replay tools/replay.cpp)
target_link_libraries(replay cbs)
# Full traced games from cambc (oracle/run_trace.py): replay must match line for line.
foreach(t 00elkring-100 01islands-12 00contest-13)
  string(REGEX REPLACE "-[0-9]+$" "" tmap ${t})
  add_test(NAME replay_${t} COMMAND sh -c "xz -dc ${REPO}/oracle/fixtures/traces/${t}.txt.xz | $<TARGET_FILE:replay> ${REPO}/maps/${tmap}.map26a /dev/stdin --quiet")
endforeach()
add_library(cbs_cov STATIC ${CBS_SOURCES})
target_include_directories(cbs_cov PUBLIC include)
target_compile_definitions(cbs_cov PUBLIC CBS_RULE_COVERAGE)
add_executable(replay_cov tools/replay.cpp)
target_link_libraries(replay_cov cbs_cov)
add_executable(bench tools/bench.cpp)
target_link_libraries(bench cbs pthread)

# Python module (optional): pip install pybind11
execute_process(COMMAND python3 -c "import pybind11; print(pybind11.get_cmake_dir())" OUTPUT_VARIABLE PYBIND11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(PYBIND11_DIR)
  set(pybind11_DIR ${PYBIND11_DIR})
  find_package(pybind11 CONFIG)
endif()
if(pybind11_FOUND)
  pybind11_add_module(cambsim python/cambsim.cpp)
  target_link_libraries(cambsim PRIVATE cbs)
endif()
