@gene('a9', "shopping on spawn: the team's altar lock, then shards in the policy's split; cheap checks first")
def shop_value(o: Option) -> Num:
    if RULE < 2 or not at_spawn():
        return 0
    if o.kind == BUY_LOCK:
        return 200 if LOCK_BUY else 0
    if o.item != next_shard() or o.count > shard_left():
        return 0
    return 190 + o.count / 100
@gene('s13', 'in-reach enemy champion off its own spawn: one our next attack kills first, then carried gems, then lowest health')
def fight_foe() -> Unit:
    m = me()
    es = enemy_spawn()
    return best(enemies(), lambda u: unit_gems(u) * 10 - u.health + (500 if attack_damage(m, u) >= u.health + shield(u) else 0) if in_reach(u) and not (u.pos in es) else -INF)
