@gene('a8', 'a step downhill on the walking field to GOAL; among equal steps prefer tiles farther from enemies, nearer allies (stay grouped) and in bushes')
def move_value(o: Option) -> Num:
    g = step_gain(o.tile)
    if g <= 0:
        return 0
    v = 100 + g
    ft = foe_tiles()
    if size(ft) > 0:
        v -= 0.3 * count(dist(t, o.tile) <= 2 for t in ft)
        v -= 0.1 * count(dist(t, o.tile) <= 3 for t in ft)
        if o.tile in bushes():
            v += 0.05
    v += 0.02 * size(ally_tiles() & near(o.tile, 3))
    return v
