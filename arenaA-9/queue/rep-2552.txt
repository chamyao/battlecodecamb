@gene('w10', 'GOAL can be walked to: not terrain and not held by an enemy champion unless it is over 3 away or recalling (it will likely have moved; allies move, so an ally on GOAL does not block the walk)', parents=['w10'])
def walk_open() -> Bool:
    if not valid(GOAL) or GOAL == here():
        return False
    if GOAL in TERRAIN:
        return False
    if dist(here(), GOAL) > 3 or not (GOAL in enemy_tiles()):
        return True
    return any(u.pos == GOAL and is_recalling(u) for u in enemies())
