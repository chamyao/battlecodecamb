@gene('v3', "option kinds worth listing this pass: only those this turn's rule (or shopping, cast-ahead) can use; shop kinds only when we can actually shop on spawn", parents=['v3'])
def consider(k: Num) -> Bool:
    r = RULE
    if k == PICK:
        return True
    if k == BUY or k == BUY_LOCK:
        return r >= 2 and can_shop() and at_spawn()
    if shop_pending():
        return False
    if k == MOVE:
        return walks()
    if k == CAST:
        return r == 7 or r == 14 or ((SK0 == AB_DASH or SK1 == AB_DASH) and (r == 3 or r == 20 or r == 5 or walks()) and dash_gate()) or (r >= 6 and ahead_on())
    if k == ATTACK:
        return r == 7 or r == 13 or r == 14 or r == 20
    if k == TRIBUTE:
        return r == 6 or r == 9 or r == 10 or r == 17
    if k == RECALL:
        return r == 4 or r == 5
    if k == LOCK:
        return r == 8
    if k == WARD:
        return r == 20 and count_units(enemies()) == 0 and here() in bushes()
    return False


@gene('a16', "units of the next shard still to buy: what the turn's budget still pays for, capped at the shards that finish the current emblem (no surplus treasure or jungle shards); nothing under 30 gold")
def shard_left() -> Num:
    if gold() < 30:
        return 0
    n = floor(SHOP_GOLD / 30) - shards_done()
    if n <= 0:
        return 0
    ns = next_shard()
    need = 5 - items_owned(ns)
    if ns != 36 and need > 0:
        return min(n, need)
    return n
