```python
@gene('a9', "shopping on spawn: the team's altar lock, then shards: treasure emblem, then the durability health emblem (10% less damage) before the jungle emblem, then the rest of the split")
def shop_value(o: Option) -> Num:
    if o.kind == BUY_LOCK:
        return 200 if LOCK_BUY and RULE >= 2 and at_spawn() else 0
    want = next_shard()
    if (want == 36 or want == 28) and items_owned(17) == 0:
        want = 16
    if o.item != want:
        return 0
    if o.count > shard_left() or RULE < 2 or not at_spawn():
        return 0
    return 190 + o.count / 100
```
