@gene("a9", "shopping on spawn: the team's altar lock, then shards in the policy's split; cheap checks first")
def shop_value(o: Option) -> Num:
    if RULE < 2 or not at_spawn():
        return 0
    if o.kind == BUY_LOCK:
        return 200 if LOCK_BUY else 0
    if o.item != next_shard() or o.count > shard_left():
        return 0
    return 190 + o.count / 100

@gene("a16", "units of the next shard still to buy: everything SHOP_GOLD still pays for, in one purchase (fewer passes)")
def shard_left() -> Num:
    return max(0, shard_until(4) - shards_done())
