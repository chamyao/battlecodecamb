@gene('s5', 'a visible enemy champion can hit me now or after one step (move + attack); nearly dead ones count only if they carry gems (my unit read once)')
def foe_in_reach() -> Bool:
    m = me()
    h = m.pos
    r = m.attack_range
    return any(dist(h, u.pos) <= max(r, u.attack_range + u.move_speed) for u in enemies() if u.health_frac > 0.2 or unit_gems(u) > 0)
@gene('a3', 'cast-ahead: damage, area or stun (slot kind k) on a close enemy champion before it reaches us; big bonus when the cast kills; cheap checks before the target search', parents=['a3'])
def ahead_value(o: Option, k: Num) -> Num:
    if RULE < 6 or k == AB_SHIELD or k == AB_HEAL or k == AB_RESET or k == AB_DASH:
        return 0
    f = ahead_foe()
    if not exists(f) or not ahead_on():
        return 0
    near_foe = dist(o.tile, f.pos) <= 1
    if k == AB_AOE:
        if not near_foe:
            return 0
    elif o.tile != f.pos:
        return 0
    base = 180 - o.slot
    if k == AB_CC and threat_eta() <= 2:
        base += 20
    if k == AB_DAMAGE and f.health_frac < 0.3:
        base += 30
    if (k == AB_DAMAGE or k == AB_AOE) and cast_damage(me(), o.slot, f) >= f.health + shield(f):
        base += 40
    return base
