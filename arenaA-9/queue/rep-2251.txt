@gene("r7", "fight the weakest enemy champion in reach (not on its own spawn), unless I am below 40% health with two or more enemy champions within 4")
def r_fight() -> Bool:
    f = fight_foe()
    if not exists(f) or dist(here(), f.pos) > me().attack_range or f.pos in enemy_spawn():
        return False
    return not (hurt_below(0.4) and enemies_within(4) >= 2)
