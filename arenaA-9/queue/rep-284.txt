@gene('m2', 'an altar next to us: one free of enemy locks first (our own lock does not block our tribute), else an enemy-locked one (a tribute there hits the lock)')
def adj_altar() -> Tile:
    adj = altars() & near(here(), 1)
    if size(adj) == 0:
        return tile(-1, -1)
    la = adj & locked_altars()
    el = la - la
    for t in la:
        if lock_enemy(t):
            el = el | near(t, 0)
    open_adj = adj - el
    if size(open_adj) > 0:
        return best(open_adj, lambda t: 0 if t in la else 1)
    return first_tile(adj)
@gene("a10", "the team's lock is neither bought nor placed and I have none and can pay; not before round 150 so early gold goes to shards")
def wants_lock() -> Bool:
    v = read_shared(7)
    return floor(v) - 4 * floor(v / 4) == 0 and not has_lock() and gold() >= 10 and round() >= 150
