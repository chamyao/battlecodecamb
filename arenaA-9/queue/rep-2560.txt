```python
@gene('m1', 'first tile of a set, preferring altars not enemy-locked, then the one nearest the enemy spawn (the altar they are most likely to use), invalid if empty')
def first_tile(s: Tiles) -> Tile:
    open_s = s - locked_altars()
    if size(open_s) == 0:
        open_s = s
    es = enemy_spawn()
    return best(open_s, lambda t: -min((dist(t, e) for e in es), default=0))
```
