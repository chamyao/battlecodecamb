@gene('a4', 'cast-ahead target: enemies off their spawn and within 7 (the cast-ahead range), carrying the most gems, then nearest, with a strong pull toward wounded ones (finishable) and toward tributing ones (a stun or hit there breaks the channel)')
def ahead_foe() -> Unit:
    es = enemy_spawn()
    h = here()
    return best(enemies(), lambda u: -INF if (u.pos in es or dist(h, u.pos) > 7) else unit_gems(u) * 10 - dist(h, u.pos) - 15 * u.health_frac + (40 if is_tributing(u) else 0))
