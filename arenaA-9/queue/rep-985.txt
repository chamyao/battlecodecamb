```python
@gene("s11", "turns to stand next to an altar we can tribute at (not enemy-locked: our own lock does not block our tribute; 1000 if none)")
def altar_eta() -> Num:
    h = here()
    d = min((max(0, dist(h, a) - 1) for a in altars() if not (a in locked_altars() and lock_enemy(a))), default=-1)
    if d < 0:
        return 1000
    return eta(d, me().move_speed)
```
