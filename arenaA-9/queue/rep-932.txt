@gene("w5", "walking field to GOAL, searched out to my own tile; reuses the turn-start BLOCK set (update computes it just before)")
def walk_field() -> Field:
    g = near(GOAL, 0)
    return field_avoid(g, BLOCK - g, near(here(), 0))
@gene('s7', 'in danger: an enemy champion is within its range + 1, a gem carrier is within 2, or an awake camp can hit me (enemy check first, camps only if needed)')
def in_danger() -> Bool:
    p = here()
    if any(dist(p, u.pos) <= u.attack_range + 1 or (unit_gems(u) > 0 and dist(p, u.pos) <= 2) for u in enemies()):
        return True
    return any(c.attack_range > 0 and dist(p, c.pos) <= c.attack_range and is_awake(c) and dist(p, camp_spawn(c)) <= c.attack_range + 2 for c in camps())
