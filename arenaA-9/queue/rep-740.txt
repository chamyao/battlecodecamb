@gene('r13', 'every altar-adjacent tile is a camp spawn and the camp in reach stands on the nearest one: fight the blocker (cheap camp check first)')
def bank_blocked() -> Bool:
    c = reach_camp()
    if not exists(c) or not in_reach(c) or not (c.pos in camp_spawns()):
        return False
    return not valid(bank_t1()) and not valid(bank_t2()) and c.pos == bank_t3()
@gene('r5', 'recall with 5+ gems when hurt (threshold rises faster with carried gems: 0.67 at 10, 0.83 at 20) and nothing near, unless an altar is close enough to bank first; cheapest checks first')
def r_cashout() -> Bool:
    g = gems()
    if g < 5:
        return False
    f = min(0.85, 0.5 + g / 60)
    if not hurt_below(f) or at_spawn():
        return False
    if under_fire():
        return False
    return altar_eta() > 4
