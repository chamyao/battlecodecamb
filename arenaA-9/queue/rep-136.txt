@gene('m11', "the camp walk's goal: nearest non-big spawn of a live group away from the enemy spawn (over 8 tiles), else nearest non-big spawn of a live group, falling back to any non-big camp spawn, then any camp spawn")
def camp_goal() -> Tile:
    live = CAMP_BASE - respawning() - BIG_CAMPS
    es = enemy_spawn()
    if size(es) > 0:
        far = live - live
        for t in live:
            if dist(t, nearest(es, t)) > 8:
                far = far | near(t, 0)
        t = closest(far)
        if valid(t):
            return t
    t = closest(live)
    if valid(t):
        return t
    t = closest(CAMP_BASE - BIG_CAMPS)
    if valid(t):
        return t
    return closest(CAMP_BASE)
