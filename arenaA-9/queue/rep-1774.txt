@gene('m2', 'an altar next to us: one free of enemy locks first (our own lock does not block our tribute); if every adjacent altar is enemy-locked, none when an unlocked altar is within 4 (walk there instead of spending 5 attacks on the lock), else an enemy-locked one (a tribute there hits the lock)')
def adj_altar() -> Tile:
    adj = altars() & near(here(), 1)
    if size(adj) == 0:
        return tile(-1, -1)
    la = adj & locked_altars()
    el = la - la
    for t in la:
        if lock_enemy(t):
            el = el | near(t, 0)
    open_adj = adj - el
    if size(open_adj) > 0:
        return best(open_adj, lambda t: 0 if t in la else 1)
    h = here()
    if any(dist(h, a) <= 4 and not lock_enemy(a) for a in altars() - adj):
        return tile(-1, -1)
    return first_tile(adj)
