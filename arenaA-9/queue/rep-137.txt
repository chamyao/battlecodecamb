@gene('m2', 'an altar next to us: one free of enemy locks first (our own lock does not block our tribute), else an enemy-locked one (a tribute there hits the lock)')
def adj_altar() -> Tile:
    adj = altars() & near(here(), 1)
    if size(adj) == 0:
        return tile(-1, -1)
    la = adj & locked_altars()
    el = la - la
    for t in la:
        if lock_enemy(t):
            el = el | near(t, 0)
    open_adj = adj - el
    if size(open_adj) > 0:
        return best(open_adj, lambda t: 0 if t in la else 1)
    return first_tile(adj)
@gene('r12', 'threat near: enemy within 5, two or more enemies within 9, hurt with an enemy within 7, or able to hit us within 3 turns')
def threat_near() -> Bool:
    return enemies_within(5) > 0 or enemies_within(9) >= 2 or (hurt_below(0.5) and enemies_within(7) > 0) or threat_eta() <= 3
