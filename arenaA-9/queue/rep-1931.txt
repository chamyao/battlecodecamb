@gene('c1', "where this turn's rule walks; banking under threat also prefers the safe, unlocked altar; the monolith rule heads for a live monolith group first")
def rule_goal() -> Tile:
    r = rule()
    if r == 3:
        return closest(my_spawn())
    if r == 11 or r == 12:
        return bank_goal()
    if r == 15:
        m = closest(MONO_GROUPS - respawning())
        if valid(m):
            return m
        return closest(camp_spawns() - respawning() - type_spawns(GOLEM))
    if r == 18:
        return closest(SAFE_RING)
    if r == 19:
        return live_camp().pos
    if r == 20:
        return camp_goal()
    return tile(-1, -1)
