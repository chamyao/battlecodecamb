@gene("w3", "tiles I cannot step onto: terrain, altars, enemy spawn, enemy units, allies within 2 (farther allies will have moved), unseen camp spawns")
def blocked() -> Tiles:

    return (TERRAIN | (ally_tiles() & near(h, 2)) | enemy_tiles() | camp_block()) - near(h, 0)
