@gene('v2', 'start of turn: choose the rule, its goal and target; shared-memory and priority writes (fighters, retreaters, pre-fight bankers and bankers walking under threat act first next round, bankers at an altar or walking to bank next)')
def update() -> None:
    if not MAPPED:
        ALTAR_RING = altar_ring()
        SAFE_RING = safe_altar_ring()
        MONO_GROUPS = mono_groups()
        TERRAIN = walls() | holes() | altars() | enemy_spawn()
        BIG_CAMPS = type_spawns(MONOLITH) | type_spawns(GOLEM)
        CAMP_BASE = camp_spawns() - BIG_CAMPS - MONO_GROUPS
        SK0 = slot_kind(0)
        SK1 = slot_kind(1)
        MAPPED = True
    RULE = rule()
    GOAL = rule_goal()
    TGT = rule_target()
    OPEN = False
    if walks() and valid(GOAL) and GOAL != here():
        BLOCK = blocked()
        OPEN = walk_open()
        FIELD = walk_field()
    LOCK_BUY = RULE >= 2 and at_spawn() and wants_lock()
    if LOCK_BUY:
        write_shared(7, read_shared(7) + 2)
    SHOP_GOLD = gold() - (10 if LOCK_BUY else 0)
    if RULE == 8:
        write_shared(7, read_shared(7) + 1)
    if RULE == 7 or RULE == 3 or RULE == 9 or RULE == 11:
        set_priority(150 + rand(20))
    elif RULE == 10 or RULE == 6 or RULE == 17 or RULE == 12:
        set_priority(120 + rand(20))
    elif RULE == 20:
        set_priority(1 + rand(100))

@gene('a3', 'cast-ahead: damage, area or stun (slot kind k) on a close enemy champion before it reaches us; big bonus when the cast kills', parents=['a3'])
def ahead_value(o: Option, k: Num) -> Num:
    f = ahead_foe()
    if not exists(f) or RULE < 6 or k == AB_SHIELD or k == AB_HEAL or k == AB_RESET or not ahead_on():
        return 0
    near_foe = dist(o.tile, f.pos) <= 1
    if k == AB_AOE:
        if not near_foe:
            return 0
    elif o.tile != f.pos:
        return 0
    base = 180 - o.slot
    if k == AB_CC and threat_eta() <= 2:
        base += 20
    if k == AB_DAMAGE and f.health_frac < 0.3:
        base += 30
    if (k == AB_DAMAGE or k == AB_AOE) and cast_damage(me(), o.slot, f) >= f.health + shield(f):
        base += 40
    return base
