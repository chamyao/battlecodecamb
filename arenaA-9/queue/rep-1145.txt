@gene('a13', 'a purchase is still due this turn (other options wait, unscored); cheap checks first, gold read once; a lock purchase holds other options only before round 1200')
def shop_pending() -> Bool:
    if RULE < 2:
        return False
    g = gold()
    if g < 10 or not can_shop() or not at_spawn():
        return False
    return (LOCK_BUY and not has_lock() and round() < 1200) or (g >= 30 and next_shard() >= 0)

@gene("a6", "fight casts: shield/heal on self below 90% health or the slot at TGT first, attack resets once the attack is spent")
def strike_cast(o: Option, k: Num) -> Num:
    if k == AB_SHIELD or k == AB_HEAL:
        return 170 - o.slot if o.unit == me() and me().health_frac < 0.9 else 0
    if k == AB_RESET:
        return 150 - o.slot if not can_attack() else 0
    return 170 - o.slot if o.tile == TGT else 0
