@gene('w7', 'a dash may land on p: a free middle tile next to both ends that no visible enemy champion is adjacent to, unless p is adjacent to me (enemy tiles read once)', parents=['w7'])
def dash_path_ok(p: Tile) -> Bool:
    if dist(here(), p) <= 1:
        return True
    ft = foe_tiles()
    return any(dist(q, p) <= 1 and not (q in BLOCK) and not any(dist(e, q) <= 1 for e in ft) for q in dash_mids())
@gene('r13', 'every altar-adjacent tile is a camp spawn and the camp in reach stands on the nearest one: fight the blocker (cheap camp check first)')
def bank_blocked() -> Bool:
    c = reach_camp()
    if not exists(c) or not in_reach(c) or not (c.pos in camp_spawns()):
        return False
    return not valid(bank_t1()) and not valid(bank_t2()) and c.pos == bank_t3()
