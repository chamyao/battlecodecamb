```python
@gene("r17", "bank before a dangerous group: 10+ gems, the next group's strength is 240+ and an altar is within 12 turns (else just go)")
def r_bankgroup() -> Bool:
    g = camp_goal()
    return valid(g) and gems() >= 10 and fuel() > 65000 and altar_eta() <= 12 and group_threat(g) >= 240
```
