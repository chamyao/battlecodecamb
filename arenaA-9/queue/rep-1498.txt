@gene('m15', 'nearest open tile next to any altar (not blocked), preferring tiles no visible camp stands on, else nearest tile next to any altar', parents=['m15'])
def bank_t3() -> Tile:
    open_ring = ALTAR_RING - taken()
    if size(open_ring) > 0:
        free = open_ring - camp_tiles()
        if size(free) > 0:
            t = closest(free)
            if valid(t):
                return t
        return closest(open_ring)
    return closest(ALTAR_RING)
