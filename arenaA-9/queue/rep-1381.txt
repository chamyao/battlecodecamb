```python
@gene("c2", "the altar or unit tile this turn's rule acts on; a camp fight hits the in-reach camp my hit kills, else the one with least health")
def rule_target() -> Tile:
    r = rule()
    if r == 6 or r == 9 or r == 10 or r == 17:
        return adj_altar()
    if r == 7:
        return fight_foe().pos
    if r == 8:
        return lock_altar()
    if r == 13:
        return reach_camp().pos
    if r == 14:
        m = me()
        return best(camps(), lambda c: ((1000 if attack_damage(m, c) >= c.health else 0) - c.health) if in_reach(c) else -INF).pos
    return tile(-1, -1)
```
