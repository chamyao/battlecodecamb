@gene("r6", "pass-through bank: gems, an altar next to us and nothing able to hit us; a bag of 10 or fewer (a one-turn channel) is banked even with a foe in reach")
def r_pass() -> Bool:
    g = gems()
    return g > 0 and valid(adj_altar()) and (g <= 10 or not foe_in_reach())

@gene('s15', 'the nearest visible camp, big camps (monoliths, golems) ranked last; wounded camps preferred among near-equal distances (my tile read once)')
def live_camp() -> Unit:
    h = here()
    return best(camps(), lambda c: -dist(h, c.pos) - 2 * c.health_frac - (1000 if c.pos in BIG_CAMPS else 0))
