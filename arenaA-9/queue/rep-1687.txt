@gene("a10", "the team's lock is neither bought nor placed and I have none and can pay, before round 1200 (a late lock is rarely worth its gold and turn)")
def wants_lock() -> Bool:
    if round() >= 1200:
        return False
    v = read_shared(7)
    return floor(v) - 4 * floor(v / 4) == 0 and not has_lock() and gold() >= 10
