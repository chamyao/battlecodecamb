@gene('w4', 'nearest tile of s by walking; if none is walk-reachable, fall back to the straight-line nearest free tile of s (any tile of s if all are held); invalid if s empty', parents=['w4'])
def closest(s: Tiles) -> Tile:
    h = here()
    if h in s:
        return h
    if size(s) == 0:
        return tile(-1, -1)
    cand = s - taken()
    w = walk_nearest(cand, h, blocked() - cand)
    if valid(w):
        return w
    if size(cand) > 0:
        return nearest(cand, h)
    return nearest(s, h)
