```python
@gene('s5', 'a visible enemy champion can hit me now or after one step (move + attack); nearly dead ones count only if they carry gems; while I am above 80% health, ones below 40% without gems do not count (they will not engage)')
def foe_in_reach() -> Bool:
    h = here()
    r = me().attack_range
    floor_h = 0.4 if me().health_frac > 0.8 else 0.2
    return any(dist(h, u.pos) <= max(r, u.attack_range + u.move_speed) for u in enemies() if u.health_frac > floor_h or unit_gems(u) > 0)
```
