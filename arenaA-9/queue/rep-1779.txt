@gene('s15', 'the nearest visible camp, big camps (monoliths, golems) ranked last; camps of the group I stand on strongly preferred (finish the group before waking another); awake camps slightly preferred (they come to us anyway); wounded camps preferred among near-equal distances')
def live_camp() -> Unit:
    g = spawn_group(here())
    return best(camps(), lambda c: -dist(here(), c.pos) - 2 * c.health_frac - (1000 if c.pos in BIG_CAMPS else 0) + (8 if g >= 0 and camp_group(c) == g else 0) + (3 if is_awake(c) else 0))
