```python
@gene('r5', 'recall with 5+ gems when hurt (threshold rises faster with carried gems: 0.6 at 5, 0.7 at 10, 0.85 from 18) and nothing near (no enemy able to hit us within the 5-turn channel), unless an altar is close enough to bank first')
def r_cashout() -> Bool:
    if at_spawn():
        return False
    g = gems()
    if g < 5:
        return False
    f = min(0.85, 0.5 + g / 50)
    if not hurt_below(f):
        return False
    if under_fire() or threat_eta() <= 5:
        return False
    return altar_eta() > 4
```
