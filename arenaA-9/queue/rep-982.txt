```python
@gene('a3', 'cast-ahead: damage, area or stun (slot kind k) on a close enemy champion before it reaches us; an area cast lands within its own radius of the target and gains per extra enemy champion caught', parents=['a3'])
def ahead_value(o: Option, k: Num) -> Num:
    f = ahead_foe()
    if not exists(f) or RULE < 6 or k == AB_SHIELD or k == AB_HEAL or k == AB_RESET or not ahead_on():
        return 0
    base = 180 - o.slot
    if k == AB_AOE:
        rad = max(1, slot_radius(o.slot))
        if dist(o.tile, f.pos) > rad:
            return 0
        base += 8 * count(dist(o.tile, u.pos) <= rad for u in enemies() if u.pos != f.pos)
    elif o.tile != f.pos:
        return 0
    if k == AB_CC and threat_eta() <= 2:
        base += 20
    if k == AB_DAMAGE and f.health_frac < 0.3:
        base += 30
    return base
```
