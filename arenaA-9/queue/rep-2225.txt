@gene('v3', "option kinds worth listing this pass: only those this turn's rule (or shopping, cast-ahead) can use; casts only with a ready slot; shop kinds only when we can actually shop on spawn; wards only with a charge in hand, in a bush on the camp walk or while holding a capture", parents=['v3'])
def consider(k: Num) -> Bool:
    r = RULE
    if k == PICK:
        return True
    if k == BUY or k == BUY_LOCK:
        return r >= 2 and can_shop() and at_spawn()
    if shop_pending():
        return False
    if k == MOVE:
        return walks()
    if k == CAST:
        if not slot_ready(0) and not slot_ready(1):
            return False
        return r == 7 or r == 14 or ((r == 3 or r == 20 or r == 5 or walks()) and dash_gate() and (SK0 == AB_DASH or SK1 == AB_DASH)) or (r >= 6 and ahead_on())
    if k == ATTACK:
        return r == 7 or r == 13 or r == 14 or r == 20
    if k == TRIBUTE:
        return r == 6 or r == 9 or r == 10 or r == 17
    if k == RECALL:
        return r == 4 or r == 5
    if k == LOCK:
        return r == 8
    if k == WARD:
        return (r == 20 or r == 16) and ward_charges() > 0 and here() in bushes()
    return False

@gene('a12', 'dash this turn: retreat with fuel > 40k; bank walks with GOAL over 1 away, other walking rules over 2 away, fuel > 45k; not on the camp walk while an enemy champion is within 5 (keep the dash to escape)')
def dash_gate() -> Bool:
    if RULE == 3:
        return fuel() > 40000
    if not walks() or not valid(GOAL) or fuel() <= 45000:
        return False
    if RULE == 20 and enemies_within(5) > 0:
        return False
    return dist(here(), GOAL) > (1 if (RULE == 11 or RULE == 12) else 2)
