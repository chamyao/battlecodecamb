```python
@gene('w10', 'GOAL can be walked to: not terrain and not held by an enemy champion (allies move, so an ally on GOAL does not block the walk); enemy tiles read once per pass', parents=['w10'])
def walk_open() -> Bool:
    if not valid(GOAL) or GOAL == here() or GOAL in TERRAIN:
        return False
    return not (GOAL in foe_tiles())
@gene('w7', 'a dash may land on p: a free middle tile next to both ends that no visible enemy champion is adjacent to, unless p is adjacent to me (enemy tiles read once)', parents=['w7'])
def dash_path_ok(p: Tile) -> Bool:
    if dist(here(), p) <= 1:
        return True
    ft = foe_tiles()
    return any(dist(q, p) <= 1 and not (q in BLOCK) and not any(dist(e, q) <= 1 for e in ft) for q in dash_mids())
```
