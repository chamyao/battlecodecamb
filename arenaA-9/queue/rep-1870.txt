def dash_gate() -> Bool:
    if RULE == 3:
        return fuel() > 40000
    if not walks() or not valid(GOAL) or fuel() <= 45000:
        return False
    if RULE == 11 or RULE == 12:
        return dist(here(), GOAL) > 1
    if RULE == 20:
        return dist(here(), GOAL) > 4
    return dist(here(), GOAL) > 2
