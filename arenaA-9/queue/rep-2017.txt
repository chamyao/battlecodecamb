@gene('r13', 'the camp in reach stands on the nearest altar-adjacent tile and every safe altar tile is unreachable or over 12 away: fight the blocker (cheap camp check first)')
def bank_blocked() -> Bool:
    c = reach_camp()
    if not exists(c) or not in_reach(c) or not (c.pos in camp_spawns()):
        return False
    if c.pos != bank_t3():
        return False
    a = bank_t1()
    b = bank_t2()
    return (not valid(a) or dist(here(), a) > 12) and (not valid(b) or dist(here(), b) > 12)
