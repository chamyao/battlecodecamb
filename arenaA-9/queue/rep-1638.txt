@gene('m11', "the camp walk's goal: nearest non-big spawn of a live group that no ally is already capturing, falling back to any non-big camp spawn, then any camp spawn")
def camp_goal() -> Tile:
    t = closest(CAMP_BASE - respawning() - BIG_CAMPS - held_groups())
    if valid(t):
        return t
    t = closest(CAMP_BASE - BIG_CAMPS)
    if valid(t):
        return t
    return closest(CAMP_BASE)

@gene("new", "spawn tiles of cleared groups an ally already stands on (its capture needs no second champion)", parents=["m11"])
def held_groups() -> Tiles:
    acc = camp_spawns() - camp_spawns()
    others = ally_tiles() - near(here(), 0)
    for g in range(num_groups()):
        if group_reward(g):
            gt = group_tiles(g)
            if size(others & gt) > 0:
                acc = acc | gt
    return acc
