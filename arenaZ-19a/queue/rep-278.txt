def after() -> None:
    t = acted_tile(ATTACK)
    if acted(CAST) and not acted(ATTACK):
        t = acted_tile(CAST)
    if acted(ATTACK) or acted(CAST):
        if any(dist(u.pos, t) == 0 for u in enemies()):
            write_shared(0, t.x)
            write_shared(1, t.y)
            write_shared(2, round() % 200)
    elif any(unit_gems(u) >= 20 and dist(u.pos, me().pos) <= 4 for u in enemies()):
        p = best(enemies(), lambda u: unit_gems(u))
        write_shared(0, p.pos.x)
        write_shared(1, p.pos.y)
        write_shared(2, round() % 200)
