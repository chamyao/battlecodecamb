```python
GF: Field = None

@gene("new", "once per turn: cache the distance field to the goals", parents=["s3"])
def update() -> None:
    GF = field(goals())

@gene("s3", "distance field to the goals (cached in update)")
def goal_field() -> Field:
    return GF

@gene("new", "true if a camp group holds a monolith (too strong for us)", parents=["s2"])
def deadly_group(g: Num) -> Bool:
    return any(spawn_type(t) == MONOLITH for t in group_tiles(g))

@gene("s2", "where to walk: an altar with gems, else a living camp group that is not a monolith")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g) and not deadly_group(g):
            acc = acc | group_tiles(g)
    return acc
```
