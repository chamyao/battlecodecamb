@gene("Z258g", "an ally called this tile (or next to it) as the focus target in the last three rounds")
def focus_called(t: Tile) -> Bool:
    age = (round() - read_shared(2)) % 200
    return age <= 3 and dist(t, tile(read_shared(0), read_shared(1))) <= 1

@gene("Z258c", "hurt and threatened: walk out of reach before recalling (damage cancels a recall)")
def retreating() -> Bool:
    return me().health_frac < 0.3 and danger_at(me().pos) > 0
