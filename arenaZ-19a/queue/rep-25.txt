```python
@gene("Z20a", "once per turn: remember every tile of monolith groups (first turn) and cache the distance field to the goals", parents=["s3"])
def update() -> None:
    if not MONO_DONE:
        mono = type_spawns(MONOLITH)
        acc = near(tile(-5, -5), 0)
        for g in range(num_groups()):
            gt = group_tiles(g)
            if size(gt & mono) > 0:
                acc = acc | gt
        MONO = acc
        MONO_DONE = True
    GF = field(goals())

@gene("s2", "where to walk: an altar with gems, else a living camp group that is not a monolith (one set subtraction)")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc - MONO
```
