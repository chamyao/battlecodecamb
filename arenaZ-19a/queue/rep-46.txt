```python
GOLEMS: Tiles = None

@gene("Z20a", "once per turn: remember every tile of monolith and golem groups (first turn) and cache the distance field to the goals, skipping golems before round 500", parents=["s3"])
def update() -> None:
    if not MONO_DONE:
        mono = type_spawns(MONOLITH)
        gol = type_spawns(GOLEM)
        acc = near(tile(-5, -5), 0)
        acc2 = near(tile(-5, -5), 0)
        for g in range(num_groups()):
            gt = group_tiles(g)
            if size(gt & mono) > 0:
                acc = acc | gt
            if size(gt & gol) > 0:
                acc2 = acc2 | gt
        MONO = acc
        GOLEMS = acc2
        MONO_DONE = True
    if round() < 500 and gems() == 0:
        GF = field(goals() - GOLEMS)
    else:
        GF = field(goals())

@gene("s2", "where to walk: an altar with gems, else a living camp group that is not a monolith (one set subtraction)")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc - MONO
```
