```python
GF: Field = None

@gene("new", "once per turn: cache the distance field to the goals", parents=["s3"])
def update() -> None:
    GF = field(goals())

@gene("s3", "distance field to the goals (cached once per turn in update)")
def goal_field() -> Field:
    return GF

@gene("s4", "how much closer a step brings me to the goals; among equal steps prefer bushes (hidden)")
def progress(t: Tile) -> Num:
    d = at(goal_field(), me().pos) - at(goal_field(), t)
    if d > 0 and t in bushes():
        return d + 0.5
    return d

@gene("new", "value of casting an ability option: damage and stuns on champions or camps, resets after an attack, shields and heals when hurt, dashes toward the goal", parents=["s1"])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit) and o.unit.is_champion and o.unit.is_enemy:
            return 85
        if exists(o.unit) and o.unit.is_camp and gems() == 0 and o.unit.type != MONOLITH:
            return 65
        return 0
    if k == AB_AOE:
        n = count(dist(u.pos, o.tile) <= 1 for u in enemies()) * 2 + count(dist(u.pos, o.tile) <= 1 and is_awake(u) for u in camps())
        return 60 + 5 * n if n > 0 else 0
    if k == AB_RESET:
        return 62 if not can_attack() and (count(dist(u.pos, me().pos) <= me().attack_range for u in enemies()) + count(is_awake(u) and dist(u.pos, me().pos) <= me().attack_range for u in camps())) > 0 else 0
    if k == AB_SHIELD:
        return 88 if me().health_frac < 0.8 and (count(dist(u.pos, me().pos) <= 4 for u in enemies()) + count(is_awake(u) for u in camps())) > 0 else 0
    if k == AB_HEAL:
        return 87 if exists(o.unit) and o.unit.health_frac < 0.6 else 0
    if k == AB_DASH:
        return 55 if progress(o.tile) >= 2 else 0
    return 0

@gene("s1", "value of doing this option now")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return buy_value(o.item, o.count)
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0


@gene("new", "shop value: a Treasure strength emblem first, then Power strength", parents=["s1"])
def buy_value(item: Num, n: Num) -> Num:
    if item == 18 and items_owned(19) == 0:
        return 100 + n
    if item == 6:
        return 95 + n
    return 0
```
