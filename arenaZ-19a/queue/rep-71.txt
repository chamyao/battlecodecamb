```python
@gene('Z7a', 'value of casting (spells prefer weak enemy champions carrying gems) an ability option: damage and stuns on champions or camps, resets after an attack, shields and heals when hurt, dashes toward the goal (counts cached per turn)', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit) and o.unit.is_champion and o.unit.is_enemy:
            return 85 + (1 - o.unit.health_frac) * 9 + min(unit_gems(o.unit), 30) / 3
        if exists(o.unit) and o.unit.is_camp and gems() == 0 and o.unit.type != MONOLITH:
            return 65
        return 0
    if k == AB_AOE:
        if not (o.tile in HOT):
            return 0
        n = count(dist(u.pos, o.tile) <= 1 for u in enemies()) * 2 + count(dist(u.pos, o.tile) <= 1 and is_awake(u) for u in camps())
        return 60 + 5 * n if n > 0 else 0
    if k == AB_RESET:
        return 62 if not can_attack() and FOES_IN_REACH > 0 else 0
    if k == AB_SHIELD:
        return 88 if me().health_frac < 0.8 and THREATS > 0 else 0
    if k == AB_HEAL:
        return 87 + (1 - o.unit.health_frac) * 5 if exists(o.unit) and o.unit.health_frac < 0.7 else 0
    if k == AB_DASH:
        if slot_radius(o.slot) > 0 and o.tile in HOT:
            return 70
        return 55 if progress(o.tile) >= 2 else 0
    return 0
```
