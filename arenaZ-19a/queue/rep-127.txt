@gene('s1', 'value of doing this option now; focus the weakest enemy champion, never poke a monolith')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return draft_value(o.champ)
    if o.kind == BUY:
        return buy_jungle(o.item, o.count)
    if retreating():
        if o.kind == MOVE:
            dn = danger_at(o.tile)
            return 95 - dn if dn < danger_at(me().pos) else 0
        if o.kind == RECALL:
            return 0
        if o.kind == ATTACK:
            return 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.25 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 9 + (6 if focus_called(o.unit.pos) else 0)
        if o.unit.type == MONOLITH:
            return 0
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 92 if gems() >= 15 else 70
    if o.kind == MOVE:
        if exists(prey()) and hunt_ok(prey()) and dist(o.tile, prey().pos) < dist(me().pos, prey().pos):
            return 57
        if assist_on() and dist(o.tile, focus_tile()) < dist(me().pos, focus_tile()):
            return 56
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "an enemy champion worth hunting: carries 15+ gems, weaker than me, within 6 tiles but out of reach", parents=["s1"])
def hunt_ok(u: Unit) -> Bool:
    return unit_gems(u) >= 15 and u.health_frac < me().health_frac and me().health_frac > 0.5 and dist(u.pos, me().pos) <= 6 and dist(u.pos, me().pos) > me().attack_range

@gene("new", "the visible enemy champion carrying the most gems", parents=["s1"])
def prey() -> Unit:
    return best(enemies(), lambda u: unit_gems(u) - u.health_frac * 10)
