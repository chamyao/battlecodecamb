```python
GF: Field = None
MONO: Tiles = None
MONO_DONE: Bool = False

@gene("new", "once per turn: remember monolith tiles (first turn) and cache the distance field to the goals", parents=["s3"])
def update() -> None:
    if not MONO_DONE:
        MONO = type_spawns(MONOLITH)
        MONO_DONE = True
    GF = field(goals())

@gene("s3", "distance field to the goals (cached once per turn in update)")
def goal_field() -> Field:
    return GF

@gene("s2", "where to walk: an altar with gems, else a living camp group that is not a monolith")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & MONO) == 0:
                acc = acc | gt
    return acc
```
