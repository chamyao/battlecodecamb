@gene('s1', 'value of doing this option now; focus the weakest enemy champion, never poke a monolith')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return draft_value(o.champ)
    if o.kind == BUY:
        return buy_jungle(o.item, o.count)
    if retreating():
        if o.kind == MOVE:
            dn = danger_at(o.tile)
            return 95 - dn if dn < danger_at(me().pos) else 0
        if o.kind == RECALL:
            return 0
        if o.kind == ATTACK:
            return 0
    if o.kind == RECALL:
        if me().health_frac < 0.25:
            return 90
        return 88 if gold() >= 90 and gems() == 0 and me().health_frac < 0.7 and danger_at(me().pos) == 0 and not (me().pos in my_spawn()) else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 9 + (6 if focus_called(o.unit.pos) else 0) + (5 if is_tributing(o.unit) else 0) + min(unit_gems(o.unit), 30) / 5
        if o.unit.type == MONOLITH:
            return 0
        return 60 if gems() < 5 else 0
    if o.kind == TRIBUTE:
        return 92 if gems() >= 10 else 70
    if o.kind == MOVE:
        if exists(prey()) and hunt_ok(prey()) and dist(o.tile, prey().pos) < dist(me().pos, prey().pos):
            return 57
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("Z350f", "end of turn: announce the enemy champion I just hit or cast on so allies focus it (shared bytes 0-2: x, y, round)")
def after() -> None:
    t = acted_tile(ATTACK)
    if acted(CAST) and not acted(ATTACK):
        t = acted_tile(CAST)
    if acted(ATTACK) or acted(CAST):
        if any(dist(u.pos, t) == 0 for u in enemies()):
            write_shared(0, t.x)
            write_shared(1, t.y)
            write_shared(2, round() % 200)

@gene("Z350g", "an ally called this tile (or next to it) as the focus target in the last three rounds")
def focus_called(t: Tile) -> Bool:
    age = (round() - read_shared(2)) % 200
    return age <= 3 and dist(t, tile(read_shared(0), read_shared(1))) <= 1

@gene("Z350j", "an enemy champion worth hunting: carries 12+ gems, weaker than me, within 7 tiles but out of reach")
def hunt_ok(u: Unit) -> Bool:
    return unit_gems(u) >= 12 and u.health_frac < me().health_frac and me().health_frac > 0.5 and dist(u.pos, me().pos) <= 7 and dist(u.pos, me().pos) > me().attack_range
