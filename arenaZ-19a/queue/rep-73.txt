```python
GF: Field = None
MONO: Tiles = None
MONO_DONE: Bool = False

@gene("new", "once per turn: remember every tile of monolith groups (first turn), cache the distance field to the goals minus monolith groups", parents=["s3"])
def update() -> None:
    if not MONO_DONE:
        mono = type_spawns(MONOLITH)
        acc = near(tile(-5, -5), 0)
        for g in range(num_groups()):
            gt = group_tiles(g)
            if size(gt & mono) > 0:
                acc = acc | gt
        MONO = acc
        MONO_DONE = True
    GF = field(goals() - MONO)

@gene("s3", "distance field to the goals without monolith groups (cached once per turn in update)")
def goal_field() -> Field:
    return GF

@gene("s2", "where to walk: an altar with gems; a nearby cleared group waiting for capture; else a living camp group")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    rew = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            acc = acc | gt
            if group_reward(g) and dist(nearest(gt, me().pos), me().pos) <= 10:
                rew = rew | gt
    if size(rew) > 0:
        return rew
    return acc
```
