```python
GF: Field = None
MONO: Tiles = None
MONO_DONE: Bool = False

@gene("new", "once per turn: remember monolith tiles (first turn) and cache the distance field to the goals", parents=["s3"])
def update() -> None:
    if not MONO_DONE:
        MONO = type_spawns(MONOLITH)
        MONO_DONE = True
    GF = field(goals())

@gene("s3", "distance field to the goals (cached once per turn in update)")
def goal_field() -> Field:
    return GF

@gene("s2", "where to walk: an altar with gems, else a living camp group that is not a monolith")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            gt = group_tiles(g)
            if size(gt & MONO) == 0:
                acc = acc | gt
    return acc

@gene("s1", "value of doing this option now; focus the weakest enemy champion, never poke a monolith")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 9
        if o.unit.type == MONOLITH:
            return 0
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
