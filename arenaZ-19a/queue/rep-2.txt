```python
@gene("new", "draft value of a champion: strong ladder picks first", parents=["s1"])
def pick_value(c: Num) -> Num:
    if c == VESPER_VALE:
        return 10
    if c == VENDETTA:
        return 9
    if c == MARCELLA:
        return 8
    if c == HEART_OF_RUIN:
        return 7
    if c == RYTH:
        return 6
    if c == VARG:
        return 5
    return 1

@gene("s1", "value of doing this option now")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        if o.unit.is_camp and o.unit.type == MONOLITH:
            return 0
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
