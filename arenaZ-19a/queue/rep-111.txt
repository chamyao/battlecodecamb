@gene('s1', 'value of doing this option now; focus the weakest enemy champion, never poke a monolith')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return draft_value(o.champ)
    if o.kind == BUY:
        return buy_jungle(o.item, o.count)
    if retreating():
        if o.kind == MOVE:
            dn = danger_at(o.tile)
            return 95 - dn if dn < danger_at(me().pos) else 0
        if o.kind == RECALL:
            return 0
        if o.kind == ATTACK:
            return 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.25 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 9 + (6 if focus_called(o.unit.pos) else 0)
        if o.unit.type == MONOLITH:
            return 0
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if assist_on() and dist(o.tile, focus_tile()) < dist(me().pos, focus_tile()):
            return 56.0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "end of turn: announce the enemy champion I just hit so allies focus it (shared bytes 0-2: x, y, round)", parents=["s1"])
def after() -> None:
    if acted(ATTACK):
        t = acted_tile(ATTACK)
        if any(dist(u.pos, t) == 0 for u in enemies()):
            write_shared(0, t.x)
            write_shared(1, t.y)
            write_shared(2, round() % 200)

@gene("new", "an ally called this tile as the focus target this round or last", parents=["s1"])
def focus_called(t: Tile) -> Bool:
    age = (round() - read_shared(2)) % 200
    return age <= 1 and t.x == read_shared(0) and t.y == read_shared(1)

@gene("new", "the focus tile announced by allies", parents=["s1"])
def focus_tile() -> Tile:
    return tile(read_shared(0), read_shared(1))

@gene("new", "join an ally fight: a fresh focus call within 8 tiles, healthy, light bag", parents=["s1"])
def assist_on() -> Bool:
    age = (round() - read_shared(2)) % 200
    return age <= 2 and me().health_frac > 0.5 and gems() < 20 and dist(me().pos, focus_tile()) <= 8 and dist(me().pos, focus_tile()) > me().attack_range

@gene('Z7a', 'value of casting an ability option: damage and stuns on champions or camps, resets after an attack, shields and heals when hurt, dashes toward the goal', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit) and o.unit.is_champion and o.unit.is_enemy:
            return 85 + (1 - o.unit.health_frac) * 9 + min(unit_gems(o.unit), 30) / 3
        if exists(o.unit) and o.unit.is_camp and gems() == 0 and o.unit.type != MONOLITH:
            return 65
        return 0
    if k == AB_AOE:
        n = count(dist(u.pos, o.tile) <= 1 for u in enemies()) * 2 + count(dist(u.pos, o.tile) <= 1 and is_awake(u) for u in camps())
        return 60 + 5 * n if n > 0 else 0
    if k == AB_RESET:
        return 62 if not can_attack() and (count(dist(u.pos, me().pos) <= me().attack_range for u in enemies()) + count(is_awake(u) and dist(u.pos, me().pos) <= me().attack_range for u in camps())) > 0 else 0
    if k == AB_SHIELD:
        return 88 if me().health_frac < 0.8 and (count(dist(u.pos, me().pos) <= 4 for u in enemies()) + count(is_awake(u) for u in camps())) > 0 else 0
    if k == AB_HEAL:
        return 87 if exists(o.unit) and o.unit.health_frac < 0.6 else 0
    if k == AB_DASH:
        if slot_radius(o.slot) > 0 and dist(o.tile, me().pos) <= 2 and any(dist(u.pos, o.tile) <= 1 for u in enemies()):
            return 72
        return 55 if progress(o.tile) >= 2 else 0
    return 0
