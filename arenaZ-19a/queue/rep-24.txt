@gene("Z8a", "once per turn: remember every tile of monolith groups (first turn), cache the distance field to the safe goals", parents=["s3"])
def update() -> None:
    if not MONO_DONE:
        mono = type_spawns(MONOLITH)
        acc = near(tile(-5, -5), 0)
        for g in range(num_groups()):
            gt = group_tiles(g)
            if size(gt & mono) > 0:
                acc = acc | gt
        MONO = acc
        MONO_DONE = True
    GF = field(safe_goals())

@gene("Z17a", "goals without monolith groups: an altar with gems, else a living camp group we can beat (one set subtraction)", parents=["s2"])
def safe_goals() -> Tiles:
    if gems() > 0:
        return goals()
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc - MONO
