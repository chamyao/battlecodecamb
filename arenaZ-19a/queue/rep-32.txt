```python
MONO: Tiles = None
MONO_DONE: Bool = False

@gene("Z10a", "once per turn: remember every tile of monolith groups (first turn), cache the distance field to the goals minus monolith groups", parents=["s3"])
def update() -> None:
    if not MONO_DONE:
        mono = type_spawns(MONOLITH)
        acc = near(tile(-5, -5), 0)
        for g in range(num_groups()):
            gt = group_tiles(g)
            if size(gt & mono) > 0:
                acc = acc | gt
        MONO = acc
        MONO_DONE = True
    GF = field(goals() - MONO)

@gene("s3", "distance field to the goals without monolith groups (cached once per turn in update)")
def goal_field() -> Field:
    return GF

@gene("s1", "value of doing this option now; focus the weakest enemy champion, never poke a monolith")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return buy_value(o.item, o.count)
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 9
        if o.unit.type == MONOLITH:
            return 0
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
