```python
@gene("Z179a", "end of turn: announce the enemy champion I just hit or cast on so allies focus it (shared bytes 0-2: x, y, round)")
def after() -> None:
    t = acted_tile(ATTACK)
    if acted(CAST) and not acted(ATTACK):
        t = acted_tile(CAST)
    if acted(ATTACK) or acted(CAST):
        if any(dist(u.pos, t) == 0 for u in enemies()):
            write_shared(0, t.x)
            write_shared(1, t.y)
            write_shared(2, round() % 200)

@gene("Z179d", "join an ally fight: a fresh focus call within 10 tiles, healthy, bag under 30")
def assist_on() -> Bool:
    age = (round() - read_shared(2)) % 200
    return age <= 3 and me().health_frac > 0.45 and gems() < 30 and dist(me().pos, focus_tile()) <= 10 and dist(me().pos, focus_tile()) > me().attack_range
```
