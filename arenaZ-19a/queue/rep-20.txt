```python
HOT: Tiles = None
THREATS: Num = 0
FOES_IN_REACH: Num = 0

@gene('Z8a', 'once per turn: cache the distance field to the goals and the nearby threat counts', parents=['s3'])
def update() -> None:
    GF = field(goals())
    hot = near(tile(-5, -5), 0)
    reach = 0
    threats = 0
    for u in enemies():
        hot = hot | near(u.pos, 1)
        threats += 1 if dist(u.pos, me().pos) <= 4 else 0
        reach += 1 if dist(u.pos, me().pos) <= me().attack_range else 0
    for u in camps():
        if is_awake(u):
            hot = hot | near(u.pos, 1)
            threats += 1
            reach += 1 if dist(u.pos, me().pos) <= me().attack_range else 0
    HOT = hot
    THREATS = threats
    FOES_IN_REACH = reach

@gene("new", 'value of casting an ability option: damage and stuns on champions or camps, resets after an attack, shields and heals when hurt, dashes toward the goal (counts cached per turn)', parents=["s1"])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit) and o.unit.is_champion and o.unit.is_enemy:
            return 85
        if exists(o.unit) and o.unit.is_camp and gems() == 0 and o.unit.type != MONOLITH:
            return 65
        return 0
    if k == AB_AOE:
        if not (o.tile in HOT):
            return 0
        n = count(dist(u.pos, o.tile) <= 1 for u in enemies()) * 2 + count(dist(u.pos, o.tile) <= 1 and is_awake(u) for u in camps())
        return 60 + 5 * n if n > 0 else 0
    if k == AB_RESET:
        return 62 if not can_attack() and FOES_IN_REACH > 0 else 0
    if k == AB_SHIELD:
        return 88 if me().health_frac < 0.8 and THREATS > 0 else 0
    if k == AB_HEAL:
        return 87 if exists(o.unit) and o.unit.health_frac < 0.6 else 0
    if k == AB_DASH:
        return 55 if progress(o.tile) >= 2 else 0
    return 0

@gene("s1", "value of doing this option now; casts abilities, focuses the weakest enemy champion, never pokes a monolith")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 9
        if o.unit.type == MONOLITH:
            return 0
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
