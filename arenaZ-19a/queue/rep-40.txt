```python
@gene("s1", "value of doing this option now; focus the weakest enemy champion, finish fights with awake camps even with a bag, never poke a monolith")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 9
        if o.unit.type == MONOLITH:
            return 0
        if gems() == 0:
            return 60 + (1 - o.unit.health_frac) * 5
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
