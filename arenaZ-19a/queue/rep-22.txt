```python
@gene("s1", "value of doing this option now; focus the weakest enemy champion, never poke a monolith")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return buy_value(o.item, o.count)
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 9
        if o.unit.type == MONOLITH:
            return 0
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("Z10c", "shop value: Treasure strength emblem, then Treasure health emblem, then Power strength")
def buy_value(item: Num, n: Num) -> Num:
    if item == 18 and items_owned(19) == 0:
        return 100 + n
    if item == 22 and items_owned(23) == 0:
        return 98 + n
    if item == 6:
        return 95 + n
    return 0
```
