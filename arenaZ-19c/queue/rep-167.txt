@gene('Z147b', 'once per turn: cache the goal field and the tiles of enemy champions and awake camps', parents=['s3'])
def update() -> None:
    g = goals()
    sig = size(g) + 10000 * size(g & near(me().pos, 0))
    if sig != GSIG or round() % 25 == 0:
        GF = field(g)
        GSIG = sig
    acc = near(tile(-5, -5), 0)
    for u in enemies():
        acc = acc | near(u.pos, 0)
    for c in camps():
        if is_awake(c):
            acc = acc | near(c.pos, 0)
    TH = acc

@gene('Z6a', 'draft preference for each champion', parents=['s1'])
def draft_value(c: Num) -> Num:
    if c == VESPER_VALE:
        return 10
    if c == VENDETTA:
        return 9
    if c == RYTH:
        return 8
    if c == MARCELLA:
        return 7
    if c == WILLOW:
        return 6
    if c == TAVI:
        return 5
    if c == HEART_OF_RUIN:
        return 4
    if c == VARG:
        return 3
    return 1
