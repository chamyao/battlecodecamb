```python
TH: Tiles = None

@gene('s1', 'value of doing this option now; drafts strong champions, uses abilities, backs off from threats before recalling when low')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return draft_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    low = me().health_frac < 0.35
    if o.kind == RECALL:
        return 90 if low and threat_gap(me().pos) > 4 else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() < 8 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if low and threat_gap(me().pos) <= 4:
            return 86 if threat_gap(o.tile) > threat_gap(me().pos) else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('s2', 'where to walk: an altar once carrying 8+ gems, else a living camp group that is not a monolith')
def goals() -> Tiles:
    if gems() >= 8 or (gems() > 0 and me().health_frac < 0.5):
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if spawn_type(nearest(ts, me().pos)) != MONOLITH:
                acc = acc | ts
    return acc

@gene("Z10a", "once per turn: cache the goal field and the tiles of enemy champions and awake camps")
def update() -> None:
    GF = field(goals())
    acc = near(tile(-5, -5), 0)
    for u in enemies():
        acc = acc | near(u.pos, 0)
    for c in camps():
        if is_awake(c):
            acc = acc | near(c.pos, 0)
    TH = acc

@gene("new", "distance from tile t to the nearest enemy champion or awake camp (99 if none seen)", parents=["s1"])
def threat_gap(t: Tile) -> Num:
    if size(TH) == 0:
        return 99
    return dist(t, nearest(TH, t))
```
