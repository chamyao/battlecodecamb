@gene('s1', 'value of doing this option now; drafts strong champions, uses abilities, backs off from threats before recalling when low')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return draft_value(o.champ)
    if o.kind == BUY:
        return 100 if o.item == next_shard() else 0
    low = me().health_frac < (0.2339 if me().attack_range > 1 else 0.25)
    if o.kind == RECALL:
        if low and threat_gap(me().pos) > 4:
            return 90
        return 45 if gold() >= 300 and threat_gap(me().pos) > 4 else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 95 if o.unit.health_frac < 0.25 else 80
        if gems() >= 8:
            return 0
        if o.unit.type == MONOLITH and not is_awake(o.unit) and count(dist(u.pos, o.unit.pos) <= 6 for u in allies()) < 3:
            return 0
        return 61 if is_awake(o.unit) else 60
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if low and threat_gap(me().pos) <= 4:
            return 86 if threat_gap(o.tile) > threat_gap(me().pos) else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('s2', 'where to walk: an altar once carrying 8+ gems, else a living camp group that is not a monolith')
def goals() -> Tiles:
    if gems() >= 8 or (gems() > 0 and me().health_frac < 0.5):
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    if round() > 1000 and me().health_frac > 0.6:
        for g in range(num_groups()):
            if not group_respawning(g):
                ts = group_tiles(g)
                if spawn_type(nearest(ts, me().pos)) == MONOLITH:
                    acc = acc | ts
        if size(acc) > 0:
            return acc
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            if spawn_type(nearest(ts, me().pos)) != MONOLITH:
                acc = acc | ts
    if size(acc) == 0:
        for g in range(num_groups()):
            ts = group_tiles(g)
            if spawn_type(nearest(ts, me().pos)) != MONOLITH:
                acc = acc | ts
    return acc
