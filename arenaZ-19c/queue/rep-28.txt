@gene('Z10b', 'draft preference for each champion', parents=['s1'])
def draft_value(c: Num) -> Num:
    if c == VESPER_VALE:
        return 10
    if c == VENDETTA:
        return 9
    if c == RYTH:
        return 8
    if c == MARCELLA:
        return 5.5
    if c == HEART_OF_RUIN:
        return 6
    if c == VARG:
        return 5
    if c == TAVI:
        return 4
    return 1

@gene('Z10c', 'value of casting an ability: damage and stuns on fights, resets after an attack, shields and heals when hurt', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_RESET:
        return 55 if not can_attack() and fighting() else 0
    if k == AB_SHIELD:
        return 75 if fighting() else 0
    if k == AB_HEAL:
        if exists(o.unit):
            return 85 if o.unit.health_frac < 0.6 else 0
        return 0
    if k == AB_DAMAGE or k == AB_CC:
        if exists(o.unit):
            if o.unit.is_champion and o.unit.is_enemy:
                return 82
            if o.unit.is_camp and gems() == 0:
                return 65
        return 0
    if k == AB_DASH:
        if gems() == 0 and me().health_frac >= 0.35 and any(dist(x, o.tile) <= 1 for x in TH):
            return 58
        return 52 if progress(o.tile) >= 2 else 0
    return 0
