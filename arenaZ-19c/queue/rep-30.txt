TE: Tiles = None

@gene('Z24a', 'distance from tile t to the nearest enemy champion or awake camp (99 if none seen)', parents=['s1'])
def threat_gap(t: Tile) -> Num:
    d = 99
    if size(TH) > 0:
        d = dist(t, nearest(TH, t))
    if size(TE) > 0:
        d = min(d, dist(t, nearest(TE, t)) - 1)
    return d

@gene('Z24b', 'once per turn: cache the goal field and the tiles of enemy champions and awake camps', parents=['s3'])
def update() -> None:
    GF = field(goals())
    acc = near(tile(-5, -5), 0)
    TE = enemy_tiles()
    for c in camps():
        if is_awake(c):
            acc = acc | near(c.pos, 0)
    TH = acc
