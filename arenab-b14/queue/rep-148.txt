@gene("a3", "cast-ahead: damage, area or stun (slot kind k) on a close enemy champion before it reaches us; a damage cast that kills it now (exact cast damage against health plus shield) comes first, for the kill bounty", parents=['a3'])
def ahead_value(o: Option, k: Num) -> Num:
    f = ahead_foe()
    if not exists(f) or RULE < 6 or k == AB_SHIELD or k == AB_HEAL or k == AB_RESET or not ahead_on():
        return 0
    near_foe = dist(o.tile, f.pos) <= 1
    if k == AB_AOE:
        if not near_foe:
            return 0
    elif o.tile != f.pos:
        return 0
    base = 180 - o.slot
    if k == AB_CC and threat_eta() <= 2:
        base += 20
    if k == AB_DAMAGE and f.health_frac < 0.3:
        base += 30
    if (k == AB_DAMAGE or k == AB_AOE) and f.health_frac < 0.5:
        if cast_damage(me(), o.slot, f) >= f.health + shield(f):
            base += 60
    return base

@gene('a4', 'the enemy champion carrying the most gems, then nearest, weighted by health (within 4 tiles a low-health foe is strongly preferred: it is the kill); champions standing on their own spawn are skipped so they do not mask a reachable target')
def ahead_foe() -> Unit:
    es = enemy_spawn()
    h = here()
    return best(enemies(), lambda u: -1000 if u.pos in es else unit_gems(u) * 10 - dist(h, u.pos) - (10 if dist(h, u.pos) <= 4 else 3) * u.health_frac)
