@gene('a4', 'the enemy champion carrying the most gems, then nearest, lightly weighted by health; an enemy channelling a tribute or recall comes first (damage or a stun cancels the channel); champions standing on their own spawn are skipped so they do not mask a reachable target')
def ahead_foe() -> Unit:
    es = enemy_spawn()
    return best(enemies(), lambda u: -1000 if u.pos in es else unit_gems(u) * 10 - dist(here(), u.pos) - 3 * u.health_frac + (60 if is_tributing(u) or is_recalling(u) else 0))
