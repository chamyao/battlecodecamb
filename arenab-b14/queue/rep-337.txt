@gene('m11', "the camp walk's goal: nearest non-big spawn of a live group, falling back to any non-big camp spawn, then any camp spawn (intel: weak gem-paying groups first once I have bought 8 times: more early gold for shards first)", parents=["m11"])
def camp_goal() -> Tile:
    if BUYS >= 8 and size(GEM_BASE) > 0:
        t = closest(GEM_BASE - respawning())
        if valid(t):
            return t
    t = closest(CAMP_BASE - respawning() - BIG_CAMPS)
    if valid(t):
        return t
    t = closest(CAMP_BASE - BIG_CAMPS)
    if valid(t):
        return t
    return closest(CAMP_BASE)
