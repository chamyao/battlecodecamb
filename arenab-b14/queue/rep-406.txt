@gene('s15', 'the nearest visible camp, preferring a living camp of the group whose spawn I stand on (finish that group so its capture can start); big camps (monoliths, golems) ranked last')
def live_camp() -> Unit:
    g = spawn_group(here())
    return best(camps(), lambda c: -dist(here(), c.pos) + (20 if g >= 0 and camp_group(c) == g else 0) - (1000 if c.pos in BIG_CAMPS else 0))
