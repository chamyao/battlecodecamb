@gene('m15', 'nearest open tile next to any altar (not blocked), preferring tiles no enemy champion stands within 2 of, else nearest tile next to any altar', parents=['m15'])
def bank_t3() -> Tile:
    open_ring = ALTAR_RING - taken()
    if size(open_ring) > 0:
        clear = open_ring - ring(enemy_tiles(), 2)
        if size(clear) > 0:
            return closest(clear)
        return closest(open_ring)
    return closest(ALTAR_RING)
