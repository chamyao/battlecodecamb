@gene('a12', 'dash this turn: retreat with fuel > 40k; walking rules with GOAL over 2 away (the camp walk: over 4 away, the dash is kept for escapes near camps) and fuel > 45k')
def dash_gate() -> Bool:
    if RULE == 3:
        return fuel() > 40000
    if walks() and valid(GOAL) and dist(here(), GOAL) > (4 if RULE == 20 else 2) and fuel() > 45000:
        return True
    return False
