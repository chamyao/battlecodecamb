```python
@gene("r7", "fight the weakest enemy champion in reach (not on its own spawn); a hurt carrier (15+ gems, below half health) does not start a fight with a healthier foe it cannot kill now (the bag goes home first)")
def r_fight() -> Bool:
    f = fight_foe()
    if not exists(f) or dist(here(), f.pos) > me().attack_range or f.pos in enemy_spawn():
        return False
    if gems() >= 15 and hurt_below(0.5) and f.health_frac > me().health_frac and attack_damage(me(), f) < f.health + shield(f):
        return False
    return True
```
