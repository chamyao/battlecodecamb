@gene('r13', 'every altar-adjacent tile is a camp spawn and the camp in reach stands on the nearest one: fight the blocker (intel: farmable camps only); the cheap camp tests run before the three bank-tile searches', parents=["r13"])
def bank_blocked() -> Bool:
    c = reach_camp()
    if not exists(c) or not farmable(c) or not (c.pos in camp_spawns()):
        return False
    if valid(bank_t1()) or valid(bank_t2()):
        return False
    t = bank_t3()
    return t in camp_spawns() and c.pos == t
