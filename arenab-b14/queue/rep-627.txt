@gene('a13', 'a purchase is still due this turn (other options wait, unscored); fight first: no shopping while an enemy champion can hit us now or after one step; the lock only holds other options after round 400 (early turns go to shards and the walk)')
def shop_pending() -> Bool:
    if RULE < 2 or not at_spawn() or not can_shop() or gold() < 10:
        return False
    if foe_in_reach():
        return False
    return (LOCK_BUY and round() >= 400 and not has_lock()) or (next_shard() >= 0 and gold() >= 30)
