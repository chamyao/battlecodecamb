```python
@gene('r14', 'fight the chosen camp in reach; never wake a sleeping monolith unless the monolith rule is on; a sleeping camp of a strong group (240+) is not woken alone below half health (no ally within 6)')
def r_camp() -> Bool:
    c = reach_camp()
    if not exists(c):
        return False
    if not in_reach(c):
        return False
    if c.type == MONOLITH and not is_awake(c) and not r_mono():
        return False
    if not is_awake(c) and hurt_below(0.5) and group_threat(camp_spawn(c)) >= 240:
        return count(dist(a.pos, here()) <= 6 and a.pos != here() for a in allies()) > 0
    return True
```
