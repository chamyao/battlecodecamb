@gene('r5', 'recall with 5+ gems when hurt (threshold rises faster with carried gems: 0.67 at 10, 0.83 at 20) and nothing near, unless an altar is close enough to bank first; never while an enemy champion within 3 is nearly dead or channelling a tribute (finish the kill or break the channel first)')
def r_cashout() -> Bool:
    if at_spawn():
        return False
    g = gems()
    if g < 5:
        return False
    f = min(0.85, 0.5 + g / 60)
    if not hurt_below(f):
        return False
    if under_fire():
        return False
    if count(dist(e.pos, here()) <= 3 and (e.health_frac < 0.3 or is_tributing(e)) for e in enemies()) > 0:
        return False
    return altar_eta() > 4
