```python
@gene('c1', "where this turn's rule walks (reads the rule update() just chose); banking under threat also prefers the safe, unlocked altar; the mono walk skips spawns an enemy champion stands on when another is open")
def rule_goal() -> Tile:
    r = RULE
    if r == 3:
        return closest(my_spawn())
    if r == 11 or r == 12:
        return bank_goal()
    if r == 15:
        open_s = camp_spawns() - respawning() - type_spawns(GOLEM)
        t = closest(open_s - enemy_tiles())
        if valid(t):
            return t
        return closest(open_s)
    if r == 18:
        return closest(SAFE_RING)
    if r == 19:
        return live_camp().pos
    if r == 20:
        return camp_goal()
    return tile(-1, -1)
```
