```python
@gene("r14", "fight the first camp in reach (intel: farmable camps only; the camp is read once); not below 25% health with an enemy champion within 5 (it would take the camp and us)", parents=["r14"])
def r_camp() -> Bool:
    c = reach_camp()
    if not exists(c) or not farmable(c):
        return False
    return not hurt_below(0.25) or count(dist(e.pos, here()) <= 5 for e in enemies()) == 0
```
