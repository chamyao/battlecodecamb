```python
@gene("a9", "shopping on spawn: the team's altar lock (only after round 400: early gold goes to shards and the emblem), then shards in the policy's split")
def shop_value(o: Option) -> Num:
    if o.kind == BUY_LOCK:
        return 200 if LOCK_BUY and RULE >= 2 and at_spawn() and round() >= 400 else 0
    if o.item != next_shard():
        return 0
    if o.count > shard_left() or RULE < 2 or not at_spawn():
        return 0
    return 190 + o.count / 100
```
