```python
@gene('r10', 'bank now: carrying gems and the bag is ready, or next to an altar with 3+ gems, or one step from a usable altar with 6+ gems (bank on the way past)', parents=['r10'])
def bank_on() -> Bool:
    g = gems()
    if g <= 0:
        return False
    if bag_ready():
        return True
    if g >= 3 and valid(adj_altar()):
        return True
    return g >= 6 and altar_eta() <= 1
```
