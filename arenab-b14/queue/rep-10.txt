@gene('r5', 'recall with gems when below half health, with 10+ gems below 70%, or 20+ gems below 85% health, and nothing near; not when an altar is adjacent or within 4 turns (bank walk instead)', parents=['r5'])
def r_cashout() -> Bool:
    if at_spawn() or gems() <= 0 or under_fire():
        return False
    if valid(adj_altar()):
        return False
    if altar_eta() <= 4 and hurt_below(0.7):
        return False
    if gems() >= 20 and hurt_below(0.85):
        return True
    if gems() >= 10 and hurt_below(0.7):
        return True
    return hurt_below(0.5)
