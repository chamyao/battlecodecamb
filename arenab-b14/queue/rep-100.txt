@gene('v2', 'start of turn: choose the rule, its goal and target; shared-memory and priority writes; a retreating champion acts first next round, fighters before idle allies and the healthiest fighter first; fighters share their target in bytes 4-5 and focus the ally's target when it is in reach')
def update() -> None:
    if not MAPPED:
        ALTAR_RING = altar_ring()
        SAFE_RING = safe_altar_ring()
        MONO_GROUPS = mono_groups()
        TERRAIN = walls() | holes() | altars() | enemy_spawn()
        BIG_CAMPS = type_spawns(MONOLITH) | type_spawns(GOLEM)
        CAMP_BASE = camp_spawns() - BIG_CAMPS - MONO_GROUPS
        SK0 = slot_kind(0)
        SK1 = slot_kind(1)
        MAPPED = True
    RULE = rule()
    GOAL = rule_goal()
    TGT = rule_target()
    if RULE == 7:
        fx = read_shared(4)
        if fx > 0:
            ft = tile(fx - 1, read_shared(5) - 1)
            if ft != TGT and any(u.pos == ft and in_reach(u) for u in enemies()):
                TGT = ft
        write_shared(4, TGT.x + 1)
        write_shared(5, TGT.y + 1)
    OPEN = False
    if walks() and valid(GOAL) and GOAL != here():
        BLOCK = blocked()
        OPEN = walk_open()
        FIELD = walk_field()
    LOCK_BUY = RULE >= 2 and at_spawn() and wants_lock()
    if LOCK_BUY:
        write_shared(7, read_shared(7) + 2)
    SHOP_GOLD = gold() - (10 if LOCK_BUY else 0)
    if RULE == 8:
        write_shared(7, read_shared(7) + 1)
    if RULE == 20:
        set_priority(1 + rand(100))
    elif RULE == 3:
        set_priority(250)
    elif RULE == 7:
        set_priority(150 + floor(me().health_frac * 50))
