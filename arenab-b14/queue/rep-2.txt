```python
@gene('w4', 'nearest tile of s by walking; if none is walk-reachable, the nearest free tile within 2 of s (wait close by), then straight-line nearest; invalid if s empty', parents=['w4'])
def closest(s: Tiles) -> Tile:
    if here() in s:
        return here()
    cand = s - taken()
    w = walk_nearest(cand, here(), blocked() - cand)
    if valid(w):
        return w
    near2 = ring(s, 2) - blocked() - taken()
    if size(near2) > 0:
        return nearest(near2, here())
    return nearest(s, here())
```
