@gene("w5", "walking field to GOAL, searched out to my own tile; carrying 8+ gems the tiles next to visible enemy champions (within 2 at 15+ gems) count as walls so the bag routes around them, falling back to the plain field when that cuts me off")
def walk_field() -> Field:
    g = near(GOAL, 0)
    h = near(here(), 0)
    b = blocked() - g
    n = gems()
    if n >= 8:
        et = enemy_tiles()
        if size(et) > 0:
            f = field_avoid(g, (b | ring(et, 2 if n >= 15 else 1)) - g - h, h)
            if at(f, here()) < 100000:
                return f
    return field_avoid(g, b, h)
