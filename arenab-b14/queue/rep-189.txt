```python
@gene("r7", "fight the weakest enemy champion in reach (not on its own spawn), except a hurt carrier (8+ gems, below half health) facing a healthier foe it cannot kill soon: then bank or retreat rules take over so the gems are not handed over")
def r_fight() -> Bool:
    f = fight_foe()
    if not exists(f) or dist(here(), f.pos) > me().attack_range or f.pos in enemy_spawn():
        return False
    if gems() >= 8 and hurt_below(0.5) and f.health_frac > me().health_frac + 0.15:
        return attack_damage(me(), f) * 2 >= f.health + shield(f)
    return True
```
