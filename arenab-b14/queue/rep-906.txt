@gene('a8', 'a step downhill on the walking field to GOAL; among downhill steps prefer tiles farther from enemies (much more so below half health) and in bushes')
def move_value(o: Option) -> Num:
    g = step_gain(o.tile)
    if g <= 0:
        return 0
    v = 100 + g
    ft = foe_tiles()
    if size(ft) > 0:
        w = 3 if me().health_frac < 0.5 else 1
        v -= 0.3 * w * count(dist(t, o.tile) <= 2 for t in ft)
        v -= 0.1 * w * count(dist(t, o.tile) <= 3 for t in ft)
        if o.tile in bushes():
            v += 0.05
    return v
