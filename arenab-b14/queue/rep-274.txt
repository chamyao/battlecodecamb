```python
@gene('m11', "the camp walk's goal: nearest non-big spawn of a live group, falling back to any non-big camp spawn, then any camp spawn; below 70% health with no ally within 6, a dangerous nearest group (strength 240+) is passed over for the next live group (no solo deaths to a woken elk group)")
def camp_goal() -> Tile:
    s = CAMP_BASE - respawning() - BIG_CAMPS
    t = closest(s)
    if valid(t):
        if hurt_below(0.7) and group_threat(t) >= 240 and count(dist(a.pos, here()) <= 6 and a.pos != here() for a in allies()) == 0:
            g = spawn_group(t)
            if g >= 0:
                t2 = closest(s - group_tiles(g))
                if valid(t2):
                    return t2
        return t
    t = closest(CAMP_BASE - BIG_CAMPS)
    if valid(t):
        return t
    return closest(CAMP_BASE)
```
