@gene('w4', 'nearest tile of s by walking, preferring tiles no enemy champion stands next to when s has any; if none is walk-reachable, the nearest free tile within 2 of s by walking (wait close by, not behind a wall), then straight-line nearest; invalid if s empty; block and taken sets read once per call', parents=['w4'])
def closest(s: Tiles) -> Tile:
    if here() in s:
        return here()
    b = blocked()
    tk = taken()
    cand = s - tk
    safe = cand - ring(enemy_tiles(), 1)
    if size(safe) > 0:
        cand = safe
    w = walk_nearest(cand, here(), b - cand)
    if valid(w):
        return w
    near2 = ring(s, 2) - b - tk
    if size(near2) > 0:
        w2 = walk_nearest(near2, here(), b)
        if valid(w2):
            return w2
        return nearest(near2, here())
    return nearest(s, here())
