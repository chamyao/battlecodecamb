@gene('i1', "camp in reach worth hitting: not in a monolith's group (a hit wakes the group), unless the team goes for monoliths (intel GAPS #1); a sleeping camp of a strong group (240+) is not woken alone below half health (no ally within 6), nor with an enemy champion within 4 (it would finish us after the camp); below 35% health no strong-group camp is hit with an enemy champion within 4", parents=['s14'])
def farmable(c: Unit) -> Bool:
    if not in_reach(c):
        return False
    if camp_spawn(c) in MONO_GROUPS:
        return r_mono()
    if group_threat(camp_spawn(c)) >= 240:
        foe = count(dist(e.pos, here()) <= 4 for e in enemies()) > 0
        if not is_awake(c):
            if hurt_below(0.5) and count(dist(a.pos, here()) <= 6 and a.pos != here() for a in allies()) == 0:
                return False
            if foe:
                return False
        elif foe and hurt_below(0.35):
            return False
    return True
