```python
@gene("r7", "fight the weakest enemy champion in reach (not on its own spawn); a hurt carry (20+ gems, below half health) only fights a foe that is itself below 30%", parents=["r7"])
def r_fight() -> Bool:
    f = fight_foe()
    if not exists(f) or dist(here(), f.pos) > me().attack_range or f.pos in enemy_spawn():
        return False
    if gems() >= 20 and hurt_below(0.5) and f.health_frac > 0.3:
        return False
    return True
```
