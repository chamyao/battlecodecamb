```python
@gene("w1", "camp spawn tiles out of vision, not respawning and not in a cleared group waiting for capture (its camps are dead, so its tiles are walkable): assumed occupied")
def unseen_camps() -> Tiles:
    return camp_spawns() - vision() - near(here(), 0) - respawning() - cleared_spawns()

@gene("new", "spawn tiles of camp groups that are cleared and waiting for capture (no living camps)", parents=["w1"])
def cleared_spawns() -> Tiles:
    acc = camp_spawns() - camp_spawns()
    for g in range(num_groups()):
        if group_reward(g):
            acc = acc | group_tiles(g)
    return acc
```
