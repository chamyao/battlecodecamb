```python
@gene("r14", "fight the first camp in reach (intel: farmable camps only), but not while 2+ enemy champions within 6 outnumber the allies with us (a raid would third-party the camp fight)", parents=["r14"])
def r_camp() -> Bool:
    c = reach_camp()
    if not exists(c) or not farmable(c):
        return False
    h = here()
    foes = count(dist(h, u.pos) <= 6 for u in enemies())
    if foes < 2:
        return True
    friends = 1 + count(dist(h, u.pos) <= 6 and u.pos != h for u in allies())
    return friends >= foes
```
