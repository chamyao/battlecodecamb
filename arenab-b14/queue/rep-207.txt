@gene("a3", "cast-ahead: damage, area or stun (slot kind k) on a close enemy champion before it reaches us; a damage or area cast that kills it now (exact cast damage against health plus shield) comes first, for the kill bounty and its dropped gems", parents=['a3'])
def ahead_value(o: Option, k: Num) -> Num:
    f = ahead_foe()
    if not exists(f) or RULE < 6 or k == AB_SHIELD or k == AB_HEAL or k == AB_RESET or not ahead_on():
        return 0
    near_foe = dist(o.tile, f.pos) <= 1
    if k == AB_AOE:
        if not near_foe:
            return 0
    elif o.tile != f.pos:
        return 0
    base = 180 - o.slot
    if k == AB_CC and threat_eta() <= 2:
        base += 20
    if k == AB_DAMAGE and f.health_frac < 0.3:
        base += 30
    if (k == AB_DAMAGE or k == AB_AOE) and f.health_frac < 0.5:
        if cast_damage(me(), o.slot, f) >= f.health + shield(f):
            base += 60
    return base

@gene("s10", "fewest turns until a visible enemy champion could hit us (1000 if none); recalling enemies and enemies on their own spawn are ignored")
def threat_eta() -> Num:
    h = here()
    es = enemy_spawn()
    return min((eta(dist(h, u.pos) - u.attack_range, u.move_speed) for u in enemies() if not is_recalling(u) and not (u.pos in es)), default=1000)
