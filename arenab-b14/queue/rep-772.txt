@gene('s11', 'turns to stand on a bank tile: next to a safe altar (out of monolith reach) that no enemy lock blocks and no visible enemy champion stands within 2 of, else any such safe tile, else next to any open altar (1000 if none)')
def altar_eta() -> Num:
    h = here()
    open_safe = SAFE_RING - ring(locked_altars(), 1)
    clear = open_safe - ring(enemy_tiles(), 2)
    d = min((dist(h, t) for t in clear), default=-1)
    if d < 0:
        d = min((dist(h, t) for t in open_safe), default=-1)
    if d < 0:
        d = min((max(0, dist(h, a) - 1) for a in altars() - locked_altars()), default=-1)
    if d < 0:
        return 1000
    return eta(d, me().move_speed)
