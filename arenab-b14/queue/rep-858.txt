@gene("w5", "walking field to GOAL, searched out to my own tile; below half health tiles next to a visible enemy champion are walked around")
def walk_field() -> Field:
    g = near(GOAL, 0)
    b = blocked() - g
    if hurt_below(0.5):
        b = b | (ring(enemy_tiles(), 1) - g - near(here(), 0))
    return field_avoid(g, b, near(here(), 0))
