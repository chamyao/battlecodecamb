```python
@gene("s10", "fewest turns until a visible enemy champion could hit us (1000 if none); a 1-tile margin, 3 tiles of slack for enemies below 25% health (they tend to retreat), enemies on their own spawn ignored", parents=["s10"])
def threat_eta() -> Num:
    h = here()
    es = enemy_spawn()
    return min((eta(dist(h, u.pos) - u.attack_range - 1 + (3 if u.health_frac < 0.25 else 0), u.move_speed) for u in enemies() if not (u.pos in es)), default=1000)
```
