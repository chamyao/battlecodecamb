GEM_MAPPED: Bool = False
GEM_SPAWNS: Tiles = None

@gene('v2', 'start of turn: block set once, then the rule, its goal and target; shared-memory and priority writes; after round 250 the camp walk goes to the nearest live gem-paying group under 300 design strength (set once at round 3) when one exists, since gold-only groups add no score')
def update() -> None:
    if not MAPPED:
        ALTAR_RING = altar_ring()
        SAFE_RING = safe_altar_ring()
        MONO_GROUPS = mono_groups()
        TERRAIN = walls() | holes() | altars() | enemy_spawn()
        BIG_CAMPS = type_spawns(MONOLITH) | type_spawns(GOLEM)
        CAMP_BASE = camp_spawns() - BIG_CAMPS - MONO_GROUPS
        SK0 = slot_kind(0)
        SK1 = slot_kind(1)
        MAPPED = True
    if not GEM_MAPPED and round() >= 3:
        GEM_SPAWNS = CAMP_BASE - CAMP_BASE
        for g in range(num_groups()):
            if group_gems(g) > 0 and sum(camp_base_strength(spawn_type(t)) for t in group_tiles(g)) < 300:
                GEM_SPAWNS = GEM_SPAWNS | (group_tiles(g) & CAMP_BASE)
        GEM_MAPPED = True
    BLOCK = blocked_now()
    RULE = rule()
    GOAL = rule_goal()
    TGT = rule_target()
    if RULE == 20 and round() > 250 and valid(GOAL) and size(GEM_SPAWNS) > 0 and not (GOAL in GEM_SPAWNS):
        gt = closest(GEM_SPAWNS - respawning())
        if valid(gt):
            GOAL = gt
    OPEN = False
    if walks() and valid(GOAL) and GOAL != here():
        OPEN = walk_open()
        FIELD = walk_field()
    LOCK_BUY = RULE >= 1 and at_spawn() and wants_lock()
    if LOCK_BUY:
        write_shared(7, read_shared(7) + 2)
    SHOP_GOLD = gold() - (10 if LOCK_BUY else 0)
    if RULE == 8:
        write_shared(7, read_shared(7) + 1)
    if RULE == 20:
        set_priority(1 + rand(149))
