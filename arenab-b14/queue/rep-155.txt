```python
@gene("w5", "walking field to GOAL, searched out to my own tile; carrying 10+ gems the tiles next to visible enemy champions count as walls (route the bag around them), falling back to the plain field when that cuts me off")
def walk_field() -> Field:
    g = near(GOAL, 0)
    h = near(here(), 0)
    b = blocked() - g
    if gems() >= 10:
        et = enemy_tiles()
        if size(et) > 0:
            f = field_avoid(g, (b | ring(et, 1)) - g - h, h)
            if at(f, here()) < 100000:
                return f
    return field_avoid(g, b, h)
```
