@gene("v3", "option kinds worth listing this pass: only those this turn's rule (or shopping, cast-ahead) can use; dash casts also on the threatened bank walk (rule 11: get the bag to the altar before the chaser)")
def consider(k: Num) -> Bool:
    r = RULE
    if k == PICK:
        return True
    if k == BUY or k == BUY_LOCK:
        return r >= 2 and at_spawn()
    if shop_pending():
        return False
    if k == MOVE:
        return walks()
    if k == CAST:
        return r == 7 or r == 14 or ((r == 3 or r == 11 or r == 20) and dash_gate() and (SK0 == AB_DASH or SK1 == AB_DASH)) or (r >= 3 and ahead_on())
    if k == ATTACK:
        return r == 7 or r == 13 or r == 14 or r == 20
    if k == TRIBUTE:
        return r == 6 or r == 9 or r == 10 or r == 17
    if k == RECALL:
        return r == 4 or r == 5
    if k == LOCK:
        return r == 8
    if k == WARD:
        return r == 20 and here() in bushes() and count_units(enemies()) == 0
    return False

@gene("a12", "dash this turn: retreat with fuel > 40k; walks with GOAL over 2 away and fuel > 45k (the walk costs little fuel, the turns saved are worth it)")
def dash_gate() -> Bool:
    if RULE == 3:
        return fuel() > 40000
    return valid(GOAL) and dist(here(), GOAL) > 2 and fuel() > 45000
