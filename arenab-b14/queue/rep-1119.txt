@gene("r7", "fight the weakest enemy champion in reach (not on its own spawn); below 25% health only when the foe is below 30% too (a trade we win)")
def r_fight() -> Bool:
    f = fight_foe()
    if not exists(f) or dist(here(), f.pos) > me().attack_range or f.pos in enemy_spawn():
        return False
    return not hurt_below(0.25) or f.health_frac < 0.3
