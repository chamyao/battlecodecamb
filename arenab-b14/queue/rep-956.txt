```python
@gene("r7", "fight the weakest enemy champion in reach (not on its own spawn); not a losing trade: below 25% health only when the foe is weaker or dies to our next basic attack")
def r_fight() -> Bool:
    f = fight_foe()
    if not exists(f) or dist(here(), f.pos) > me().attack_range or f.pos in enemy_spawn():
        return False
    if me().health_frac < 0.25:
        return f.health_frac < me().health_frac or attack_damage(me(), f) >= f.health + shield(f)
    return True
```
