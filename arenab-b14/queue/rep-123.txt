```python
@gene("m13", "nearest tile next to a safe altar that no enemy lock blocks, skipping altar tiles within 4 of a visible enemy champion when another such tile exists (bank where we keep the gems)")
def bank_t1() -> Tile:
    s = SAFE_RING - ring(locked_altars(), 1)
    et = enemy_tiles()
    if size(et) > 0:
        calm = s - ring(et, 4)
        if size(calm) > 0:
            return closest(calm)
    return closest(s)
```
