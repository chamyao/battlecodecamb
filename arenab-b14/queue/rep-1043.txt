```python
@gene('a9', "shopping on spawn: the team's altar lock (only after round 250, or earlier once shards are done: early gold goes to shards), then shards in the policy's split")
def shop_value(o: Option) -> Num:
    if o.kind == BUY_LOCK:
        return 200 if LOCK_BUY and RULE >= 2 and at_spawn() and (round() >= 250 or next_shard() < 0) else 0
    if o.item != next_shard():
        return 0
    if o.count > shard_left() or RULE < 2 or not at_spawn():
        return 0
    return 190 + o.count / 100
```
