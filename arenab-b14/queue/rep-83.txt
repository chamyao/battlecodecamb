@gene('m15', 'nearest open tile next to any altar (not blocked), preferring tiles not within 2 of a visible enemy champion, else nearest tile next to any altar', parents=['m15'])
def bank_t3() -> Tile:
    open_ring = ALTAR_RING - taken()
    if size(open_ring) > 0:
        ft = enemy_tiles()
        if size(ft) > 0:
            safe = open_ring - ring(ft, 2)
            if size(safe) > 0:
                return closest(safe)
        return closest(open_ring)
    return closest(ALTAR_RING)
