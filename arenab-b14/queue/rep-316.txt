@gene('s13', 'in-reach enemy champion: one our basic attack kills now (health plus shield) first, for the bounty and its dropped gems; then carried gems, then lowest health')
def fight_foe() -> Unit:
    return best(enemies(), lambda u: (2000 if attack_damage(me(), u) >= u.health + shield(u) else 0) + unit_gems(u) * 10 - u.health if in_reach(u) else -INF)

@gene("c2", "the altar or unit tile this turn's rule acts on (reads the rule update() just chose instead of re-running rule())")
def rule_target() -> Tile:
    r = RULE
    if r == 6 or r == 9 or r == 10 or r == 17:
        return adj_altar()
    if r == 7:
        return fight_foe().pos
    if r == 8:
        return lock_altar()
    if r == 13 or r == 14:
        return reach_camp().pos
    return tile(-1, -1)
