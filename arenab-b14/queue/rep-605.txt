```python
@gene('r5', 'recall with 5+ gems when hurt (threshold rises with carried gems, lower than before so a contested group is held longer) and nothing near, unless an altar is close enough to bank first')
def r_cashout() -> Bool:
    if at_spawn():
        return False
    g = gems()
    if g < 5:
        return False
    f = min(0.7, 0.45 + g / 100)
    if not hurt_below(f):
        return False
    if under_fire():
        return False
    return altar_eta() > 6
```
