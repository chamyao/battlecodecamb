```python
@gene('v3', "option kinds worth listing this pass: only those this turn's rule (or shopping, cast-ahead) can use and act_value scores; shop kinds only when we can actually shop on spawn; dash casts on cash-out and walking rules; no wards (no measured effect, saves fuel)", parents=['v3'])
def consider(k: Num) -> Bool:
    r = RULE
    if k == PICK:
        return True
    if k == BUY or k == BUY_LOCK:
        return r >= 2 and at_spawn() and can_shop()
    if shop_pending():
        return False
    if k == MOVE:
        return walks()
    if k == ATTACK:
        return r == 7 or r == 13 or r == 14 or r == 20
    if k == TRIBUTE:
        return r == 6 or r == 9 or r == 10 or r == 17
    if k == RECALL:
        return r == 4 or r == 5
    if k == LOCK:
        return r == 8
    if k == CAST:
        if r == 7 or r == 14:
            return True
        if (SK0 == AB_DASH or SK1 == AB_DASH) and (r == 3 or r == 5 or r == 20 or walks()):
            if dash_gate():
                return True
        return r >= 6 and ahead_on()
    return False
```
