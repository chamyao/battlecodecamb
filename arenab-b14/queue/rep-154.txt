@gene('r5', 'recall with 5+ gems when hurt (threshold rises with carried gems) and nothing near, unless an altar is close enough to bank first (4 turns, or 8 when no enemy champion is in sight: walking the bag in beats a 5-turn recall and the walk back)')
def r_cashout() -> Bool:
    if at_spawn():
        return False
    g = gems()
    if g < 5:
        return False
    f = min(0.8, 0.55 + g / 100)
    if not hurt_below(f):
        return False
    if under_fire():
        return False
    lim = 8 if count_units(enemies()) == 0 and not hurt_below(0.4) else 4
    return altar_eta() > lim
