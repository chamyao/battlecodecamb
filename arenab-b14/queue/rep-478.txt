```python
@gene('s8', 'under fire: in danger or an enemy champion within 6 that is neither recalling nor standing on its own spawn (those cannot chase us)')
def under_fire() -> Bool:
    if in_danger():
        return True
    p = here()
    es = enemy_spawn()
    return any(dist(p, u.pos) <= 6 and not is_recalling(u) and not (u.pos in es) for u in enemies())
```
