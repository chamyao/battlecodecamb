@gene('r1', 'nothing to do: dead, channeling, cannot act, or recalling while no enemy can hit us before the recall completes (stay put so it completes)', parents=['r1'])
def r_idle() -> Bool:
    if not alive() or channeling() or (not can_move() and not can_attack()):
        return True
    if not is_recalling(me()):
        return False
    return not under_fire() or channel_left(me()) < threat_eta()
