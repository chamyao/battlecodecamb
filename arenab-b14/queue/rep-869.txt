@gene('w7', 'a dash may land on p: a free middle tile (precomputed in update) next to p unless p is adjacent to me (an adjacent landing needs no middle tile)', parents=['w7'])
def dash_path_ok(p: Tile) -> Bool:
    if dist(here(), p) <= 1:
        return True
    return any(dist(q, p) <= 1 for q in DASH_FREE)
