@gene('m11', "the camp walk's goal: keep last turn's camp goal while it is still a live non-big spawn (no flip-flopping between groups), else the nearest non-big spawn of a live group, falling back to any non-big camp spawn, then any camp spawn", parents=['m11'])
def camp_goal() -> Tile:
    live = CAMP_BASE - respawning() - BIG_CAMPS
    if RULE == 20 and valid(GOAL) and GOAL in live and GOAL != here():
        return GOAL
    t = closest(live)
    if valid(t):
        return t
    t = closest(CAMP_BASE - BIG_CAMPS)
    if valid(t):
        return t
    return closest(CAMP_BASE)
