```python
@gene("r10", "bank now: carrying gems and the bag is ready, or next to an altar with 3+ gems (any gems while an enemy champion is in sight: deny the kill bounty)", parents=['r10'])
def bank_on() -> Bool:
    g = gems()
    if g <= 0:
        return False
    if bag_ready():
        return True
    if not valid(adj_altar()):
        return False
    return g >= 3 or count_units(enemies()) > 0
```
