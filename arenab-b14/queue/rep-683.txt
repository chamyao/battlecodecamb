```python
@gene('a0', "value of a move, attack, tribute, recall, lock or ward option under this turn's rule; a basic attack that kills an enemy champion comes first outside recall rules (bounty and its gems); free hits while holding a capture or retreating, focus the weaker champion")
def act_value(o: Option) -> Num:
    k = o.kind
    r = RULE
    if k == MOVE:
        return move_value(o) if walks() else 0
    if k == ATTACK:
        u = o.unit
        foe = u.is_enemy and u.is_champion
        if foe and r != 4 and r != 5 and in_reach(u) and attack_damage(me(), u) >= u.health + shield(u):
            return 190
        if r == 7 or r == 14:
            return 178 if u.pos == TGT else 0
        if r == 13:
            return 150 if u.pos == TGT else 0
        if r == 20:
            if u.is_camp:
                return 90 if u.pos == reach_camp().pos else 0
            if foe and in_reach(u):
                return 120 + 20 * (1 - u.health_frac)
            return 0
        if r == 16:
            if foe and in_reach(u):
                return 100 + 20 * (1 - u.health_frac)
            return 0
        if r == 3:
            if foe and in_reach(u):
                return 60 + 20 * (1 - u.health_frac)
            return 0
        return 0
    if k == TRIBUTE:
        return 150 if (r == 6 or r == 9 or r == 10 or r == 17) and o.tile == TGT else 0
    if k == RECALL:
        return 150 if r == 4 or r == 5 else 0
    if k == LOCK:
        return 150 if r == 8 and o.tile == TGT else 0
    if k == WARD:
        return 130 if r == 20 and o.tile == here() and here() in bushes() and count_units(enemies()) == 0 else 0
    return 0
```
