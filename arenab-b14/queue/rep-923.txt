@gene('m11', "the camp walk's goal: nearest non-big spawn of a live group not stood on by an enemy champion, falling back to any live non-big spawn, then any non-big camp spawn, then any camp spawn")
def camp_goal() -> Tile:
    live = CAMP_BASE - respawning() - BIG_CAMPS
    t = closest(live - enemy_tiles())
    if valid(t):
        return t
    t = closest(live)
    if valid(t):
        return t
    t = closest(CAMP_BASE - BIG_CAMPS)
    if valid(t):
        return t
    return closest(CAMP_BASE)
