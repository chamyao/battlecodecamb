@gene('m11', "the camp walk's goal: nearest non-big spawn of a live group, skipping spawns within 4 of a visible enemy champion when 2+ are seen (a raid pack holds that camp), falling back to any non-big camp spawn, then any camp spawn")
def camp_goal() -> Tile:
    live = CAMP_BASE - respawning() - BIG_CAMPS
    ft = enemy_tiles()
    if size(ft) >= 2:
        t = closest(live - ring(ft, 4))
        if valid(t):
            return t
    t = closest(live)
    if valid(t):
        return t
    t = closest(CAMP_BASE - BIG_CAMPS)
    if valid(t):
        return t
    return closest(CAMP_BASE)
