@gene('B2249c', 'cast-ahead: damage or stun on a close enemy champion (aoe may land near it); stun a chaser and shield or heal self while retreating or hurt; a cast that kills the foe outright (exact damage vs health plus shield) ranks above everything', parents=['a3'])
def ahead_value(o: Option, k: Num) -> Num:
    if RULE < 3 or RULE == 4 or RULE == 5 or k == AB_RESET:
        return 0
    if not ahead_on():
        return 0
    foe = ahead_foe()
    hit = o.unit.pos if exists(o.unit) else o.tile
    if k == AB_SHIELD or k == AB_HEAL:
        if hit != here():
            return 0
        if RULE == 3 or hurt_below(0.6):
            return 175 - o.slot
        return 0
    if RULE < 6:
        if RULE == 3 and k == AB_CC and not slot_moveless(o.slot) and hit == foe.pos:
            return 185 - o.slot
        return 0
    if hit == foe.pos:
        if (k == AB_DAMAGE or k == AB_CC) and cast_damage(me(), o.slot, foe) >= foe.health + shield(foe):
            return 195 - o.slot
        return 180 - o.slot
    if k == AB_AOE and dist(hit, foe.pos) <= slot_radius(o.slot):
        return 170 - o.slot - dist(hit, foe.pos)
    return 0
