```python
@gene('s6', 'visible enemy champions within r tiles (nearly dead ones count only if carrying gems; recalling ones and ones on their own spawn do not count)')
def enemies_within(r: Num) -> Num:
    es = enemy_spawn()
    return count(
        dist(here(), u.pos) <= r and (u.health_frac >= 0.15 or unit_gems(u) > 0) and not is_recalling(u) and not (u.pos in es)
        for u in enemies()
    )
```
