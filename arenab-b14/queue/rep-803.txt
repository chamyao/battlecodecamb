@gene('a0', "value of a move, attack, tribute, recall, lock or ward option under this turn's rule; default rule also hits enemy champions in reach; a basic attack that kills an enemy champion comes first outside recall rules; free hits on champions in reach while holding a capture or retreating")
def act_value(o: Option) -> Num:
    k = o.kind
    r = RULE
    if k == MOVE:
        return move_value(o) if walks() else 0
    if k == ATTACK:
        u = o.unit
        if u.is_enemy and u.is_champion and r != 4 and r != 5 and in_reach(u) and attack_damage(me(), u) >= u.health + shield(u):
            return 190
        if (r == 16 or r == 3) and u.is_enemy and u.is_champion and in_reach(u):
            return 100 + 20 * (1 - u.health_frac)
        if r == 7 or r == 14:
            return 178 if o.unit.pos == TGT else 0
        if r == 13:
            return 150 if o.unit.pos == TGT else 0
        if r == 20:
            if o.unit.is_enemy and o.unit.is_champion:
                return 120
            return 90 if o.unit.is_camp and o.unit.pos == reach_camp().pos else 0
        return 0
    if k == TRIBUTE:
        return 181 if (r == 6 or r == 9 or r == 10 or r == 17) and o.tile == TGT else 0
    if k == RECALL:
        return 150 if r == 4 or r == 5 else 0
    if k == LOCK:
        return 150 if r == 8 and o.tile == TGT else 0
    if k == WARD:
        return 130 if r == 20 and o.tile == here() and here() in bushes() and count_units(enemies()) == 0 else 0
    return 0
