```python
@gene('s4', 'within my basic attack range; an enemy champion standing on its own spawn (where it heals) never counts, so the fight picks a foe we can actually finish')
def in_reach(u: Unit) -> Bool:
    if dist(here(), u.pos) > me().attack_range:
        return False
    return not (u.is_champion and u.is_enemy and u.pos in enemy_spawn())
```
