```python
@gene("a6", "fight casts: heal on self below 70% health, shield on self when hurt or in danger (no cooldown spent at full health out of danger), or the slot at TGT first; attack resets once the attack is spent")
def strike_cast(o: Option, k: Num) -> Num:
    if k == AB_HEAL:
        return 170 - o.slot if o.unit == me() and hurt_below(0.7) else 0
    if k == AB_SHIELD:
        return 170 - o.slot if o.unit == me() and (hurt_below(0.7) or in_danger()) else 0
    if k == AB_RESET:
        return 150 - o.slot if not can_attack() else 0
    return 170 - o.slot if o.tile == TGT else 0
```
