```python
@gene("a8", "a step downhill on the walking field to GOAL; among equal steps prefer tiles farther from enemy champions and in bushes")
def move_value(o: Option) -> Num:
    g = step_gain(o.tile)
    if g <= 0:
        return 0
    v = 100 + g
    ev = enemies()
    if count_units(ev) > 0:
        v -= 0.3 * count(dist(u.pos, o.tile) <= 2 for u in ev)
        v -= 0.1 * count(dist(u.pos, o.tile) <= 3 for u in ev)
        if o.tile in bushes():
            v += 0.05
    return v
```
