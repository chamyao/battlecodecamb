```python
@gene('s5', 'a visible enemy champion can hit me now or after one step (move + attack), ignoring nearly dead ones and ones recalling (they cannot act on us)')
def foe_in_reach() -> Bool:
    h = here()
    r = me().attack_range
    return any(dist(h, u.pos) <= max(r, u.attack_range + u.move_speed) for u in enemies() if u.health_frac > 0.2 and not is_recalling(u))
```
