```python
@gene('r12', 'threat near: hurt with an enemy within 7, or enemies within 5 (or able to hit us within 3 turns) whose summed health-weighted strength is at least 60% of mine; a lone nearly dead or recalling enemy is prey, not a threat')
def threat_near() -> Bool:
    if hurt_below(0.5) and enemies_within(7) > 0:
        return True
    h = here()
    m = me()
    w = sum(u.strength * u.health for u in enemies() if not is_recalling(u) and (dist(h, u.pos) <= 5 or eta(dist(h, u.pos) - u.attack_range, u.move_speed) <= 3))
    return w > 0 and w >= 0.6 * m.strength * m.health
```
