```python
@gene("a15", "shards of the first k kinds in the turn's split (of the shards SHOP_GOLD pays for): 60% treasure strength first, then halves of the rest")
def shard_until(k: Num) -> Num:
    n = floor(SHOP_GOLD / 30)
    t = floor(n * 0.6 + 0.5)
    j = floor((n - t) / 2)
    s = floor((n - t - j) / 2)
    if k == 1:
        return t
    if k == 2:
        return t + j
    if k == 3:
        return t + j + s
    return n
```
