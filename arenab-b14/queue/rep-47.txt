@gene('a2', 'value of a cast option: dash on walks and cash-out retreats, a stun on the nearest chaser while retreating, cast-ahead, casts and resets of a fight', parents=['a2'])
def cast_value(o: Option) -> Num:
    k = SK0 if o.slot == 0 else SK1
    if k == AB_DASH:
        return dash_value(o) if (RULE == 3 or RULE == 5 or RULE == 20 or walks()) else 0
    if RULE == 3:
        if k == AB_CC:
            f = ahead_foe()
            if exists(f) and o.tile == f.pos and dist(here(), f.pos) <= 4:
                return 175 - o.slot
        return 0
    if RULE == 7 or RULE == 14:
        return max(ahead_value(o, k), strike_cast(o, k))
    return ahead_value(o, k)
