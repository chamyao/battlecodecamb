```python
@gene('r14', 'fight the first camp in reach (intel: farmable camps only); a sleeping camp of a strong group (240+) is not woken alone below 60% health, or below 80% when the group is very strong (400+) (no ally within 6)', parents=['r14'])
def r_camp() -> Bool:
    c = reach_camp()
    if not exists(c) or not farmable(c):
        return False
    if not is_awake(c):
        th = group_threat(camp_spawn(c))
        if th >= 240 and (hurt_below(0.6) or (th >= 400 and hurt_below(0.8))):
            return count(dist(a.pos, here()) <= 6 and a.pos != here() for a in allies()) > 0
    return True
```
