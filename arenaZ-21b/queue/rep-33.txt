```python
@gene("s1", "value of doing this option now; when hurt, step out of reach before recalling; dash to travel or escape")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac >= 0.4:
            return 0
        return 0 if in_fight() else 90
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_DASH:
            return dash_value(o.tile)
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        if gems() == 0:
            return 60
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.4 and in_fight():
            g = threat_dist(o.tile) - threat_dist(me().pos)
            return 95 + g if g > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "worth of a dash: escape when hurt in a fight, else cover two tiles toward the goals", parents=["s1"])
def dash_value(t: Tile) -> Num:
    if not valid(t):
        return 0
    if in_fight():
        if me().health_frac < 0.4 and threat_dist(t) - threat_dist(me().pos) >= 2:
            return 96
        return 0
    return 51 if progress(t) >= 2 else 0
```
