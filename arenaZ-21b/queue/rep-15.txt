```python
GF: Field = None

@gene("new", "once per turn: cache the distance field to the goals", parents=["s4"])
def update() -> None:
    GF = goal_field()

@gene("new", "skip listing option kinds this policy never takes", parents=["s1"])
def consider(kind: Num) -> Bool:
    return not (kind == WARD or kind == LOCK or kind == BUY_LOCK or kind == CANCEL)

@gene("s1", "value of doing this option now")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        if gems() == 0:
            return 60
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("s4", "how much closer a step brings me to the goals (field cached per turn)")
def progress(t: Tile) -> Num:
    return at(GF, me().pos) - at(GF, t)
```
