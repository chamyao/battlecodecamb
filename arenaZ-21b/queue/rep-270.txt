```python
@gene('s1', 'value of doing this option now; join allies fighting enemy champions when healthy; draft Marcella, Ryth, Tavi, Vesper Vale; keep fighting camps until banking; flee then recall when hurt; recall to shop once 200 gold is saved; dash; chase weak enemy champions')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return twin_order(o.champ)
    if o.kind == BUY:
        return jungle_shop(o.item)
    if o.kind == RECALL:
        if in_fight():
            return 0
        if me().health_frac < 0.35:
            return 90
        if gems() == 0 and gold() >= 200 and not can_shop() and at(GF, me().pos) >= 3:
            return 52 if me().health_frac < 0.85 or gold() >= 400 else 0
        return 0
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_DASH:
            if engage_dash(o.tile):
                return 125
            return dash_value(o.tile)
        if slot_kind(o.slot) == AB_AOE and not (o.tile in HOT):
            return 0
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 10
        if not banking():
            return 60
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.35 and in_fight():
            g = threat_dist(o.tile) - threat_dist(me().pos)
            return 95 + g if g > 0 else 0
        if me().health_frac > 0.6:
            w = best(enemies(), lambda u: 0 - u.health_frac)
            if exists(w) and w.health_frac < 0.35 and dist(w.pos, me().pos) <= 5:
                return 70 if dist(o.tile, w.pos) < dist(me().pos, w.pos) else 0
        if me().health_frac > 0.7 and gems() < 10:
            p = me().pos
            e = best(enemies(), lambda u: 0 - dist(u.pos, p))
            if exists(e) and dist(e.pos, p) <= 7 and dist(e.pos, p) > 1:
                if any(dist(a.pos, e.pos) <= 4 and dist(a.pos, p) > 0 for a in allies()):
                    return 65 if dist(o.tile, e.pos) < dist(p, e.pos) else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "draft order: Marcella, then Cassi and Issac (two bodies for one pick), Ryth, Tavi, then the default order", parents=["s1"])
def twin_order(c: Num) -> Num:
    if c == MARCELLA:
        return 5
    if c == CASSI or c == ISSAC:
        return 4
    if c == RYTH:
        return 3
    if c == TAVI:
        return 2
    return 1

@gene('Z7a', 'worth of casting an ability: damage and stuns on champions and fought camps, shields and heals when in a fight', parents=['s1'])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if not exists(o.unit):
            return 0
        if o.unit.is_champion and o.unit.is_enemy:
            if k == AB_CC and is_tributing(o.unit):
                return 97
            return 85
        if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            if k == AB_CC and is_awake(o.unit) and me().health_frac < 0.7:
                return 72
            return 65 + (1 - o.unit.health_frac) * 4
        return 0
    if k == AB_AOE:
        if not valid(o.tile):
            return 0
        r = slot_radius(o.slot)
        n = count(dist(u.pos, o.tile) <= r for u in enemies())
        c = count(dist(u.pos, o.tile) <= r and (gems() == 0 or is_awake(u)) for u in camps())
        if n + c == 0:
            return 0
        return 64 + n * 3 + c
    if k == AB_SHIELD:
        return 75 if in_fight() else 0
    if k == AB_HEAL:
        if exists(o.unit) and o.unit.health_frac < 0.6:
            return 85
        return 0
    if k == AB_RESET:
        return 58 if not can_attack() and in_fight() else 0
    return 0

@gene("Z46a", "worth of a dash: escape when hurt in a fight; when healthy keep dashing two tiles toward the goals even near a fight")
def dash_value(t: Tile) -> Num:
    if not valid(t):
        return 0
    if in_fight():
        if me().health_frac < 0.4:
            return 96 if threat_dist(t) - threat_dist(me().pos) >= 2 else 0
        if me().health_frac > 0.6 and progress(t) >= 2:
            return 51
        return 0
    return 51 if progress(t) >= 2 else 0
```
