@gene("s1", "value of doing this option now")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return shop_value(o.item) + o.count * 0.01
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        if gems() == 0:
            return 60
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "draft preference: strong melee first, then the gold ranger", parents=["s1"])
def pick_value(c: Num) -> Num:
    if c == VENDETTA:
        return 10
    if c == HEART_OF_RUIN:
        return 9
    if c == RYTH:
        return 8
    if c == MARCELLA:
        return 7
    if c == VESPER_VALE:
        return 6
    return 1

@gene("new", "worth of buying a stat shard: strength first, Jungle family favoured", parents=["s1"])
def shop_value(i: Num) -> Num:
    if item_strength(i) <= 0 and item_health(i) <= 0:
        return 0
    v = 100 + item_strength(i) * 2 + item_health(i) * 0.02
    if item_passive(i) == P_JUNGLE:
        v += 15
    return v

@gene("new", "worth of casting: damage and stuns on enemies and fought camps, shields in fights", parents=["s1"])
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if not exists(o.unit):
            return 0
        if o.unit.is_champion and o.unit.is_enemy:
            return 85
        if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 65
        return 0
    if k == AB_AOE:
        if not valid(o.tile):
            return 0
        n = count(dist(u.pos, o.tile) <= slot_radius(o.slot) for u in enemies())
        c = count(dist(u.pos, o.tile) <= slot_radius(o.slot) and is_awake(u) for u in camps())
        return 64 + n * 3 + c if n + c > 0 else 0
    if k == AB_SHIELD:
        return 75 if me().health_frac < 0.9 and any(is_awake(u) and dist(u.pos, me().pos) <= 5 for u in camps()) else 0
    if k == AB_HEAL:
        return 85 if exists(o.unit) and o.unit.health_frac < 0.6 else 0
    return 0
