```python
@gene("Z80a", "keep hunting camps while the bag is small: bank once carrying 16+ gems, or 6+ with a visible enemy champion within 6, when below half health, or near the end")
def banking() -> Bool:
    if gems() <= 0:
        return False
    if gems() >= 16 or me().health_frac < 0.5:
        return True
    if gems() >= 6 and any(dist(u.pos, me().pos) <= 6 for u in enemies()):
        return True
    return round() > 1900
```
