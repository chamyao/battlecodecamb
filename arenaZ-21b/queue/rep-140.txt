```python
@gene("Z118a", "worth of a purchase: the starter item first; then strength shards, Jungle family (less camp damage) first, finishing an emblem already started")
def strength_more(i: Num) -> Num:
    if i == 6:
        return 100
    if item_strength(i) <= 0 or item_cost(i) > 30:
        return 0
    v = 90 + item_strength(i) * 0.1
    if item_passive(i) == P_JUNGLE:
        v += 4
    n = items_owned(i)
    if n > 0 and n < 5:
        v += 2
    return v
```
