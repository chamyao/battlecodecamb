```python
@gene("s1", "value of doing this option now")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return pick_value(o.champ)
    if o.kind == BUY:
        return shop_value(o.item) + o.count * 0.01
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        return 60 if gems() == 0 else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "draft preference: sturdy melee carries first", parents=["s1"])
def pick_value(c: Num) -> Num:
    if c == VENDETTA:
        return 10
    if c == HEART_OF_RUIN:
        return 9
    if c == MARCELLA:
        return 8
    if c == RYTH:
        return 7
    if c == VESPER_VALE:
        return 6
    if c == VARG:
        return 5
    return 1

@gene("new", "worth of buying a shard: Jungle (less camp damage) strength first, then any strength or health", parents=["s1"])
def shop_value(i: Num) -> Num:
    if item_strength(i) <= 0 and item_health(i) <= 0:
        return 0
    v = 100 + item_strength(i) + item_health(i) * 0.02
    if item_passive(i) == P_JUNGLE:
        v += 20
    if item_strength(i) > 0:
        v += 5
    return v
```
