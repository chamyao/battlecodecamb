@gene("s2", "where to walk: a free altar with no visible enemy champion near it when carrying gems, else a living camp group")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        safe = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
            if not any(dist(u.pos, a) <= 4 for u in enemies()):
                safe = safe | near(a, 1)
        return safe if size(safe) > 0 else acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
