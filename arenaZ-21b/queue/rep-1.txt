@gene("s2", "where to walk: an altar with gems, else the easiest living camp groups (no monoliths or golems)")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    rest = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            d = group_danger(g)
            if d <= 60:
                acc = acc | group_tiles(g)
            elif d < 150:
                rest = rest | group_tiles(g)
    if size(acc) == 0:
        return rest
    return acc

@gene("new", "summed base strength of the monsters in a camp group (999 if it holds a monolith or golem)", parents=["s2"])
def group_danger(g: Num) -> Num:
    s = 0
    for t in group_tiles(g):
        k = spawn_type(t)
        if k == MONOLITH or k == GOLEM:
            return 999
        s += camp_base_strength(k)
    return s
