@gene('Z17a', 'once per turn: note the tiles next to area-spell targets (only with an area spell ready); recompute the walk field only when the walk goals change; skip while channeling or dead', parents=['s4'])
def update() -> None:
    if channeling() or not alive():
        return
    hot = near(tile(-5, -5), 0)
    if (slot_kind(0) == AB_AOE and slot_ready(0)) or (slot_kind(1) == AB_AOE and slot_ready(1)):
        for u in enemies():
            hot = hot | near(u.pos, 1)
        for u in camps():
            if gems() == 0 or is_awake(u):
                hot = hot | near(u.pos, 1)
    HOT = hot
    gs = walk_goals()
    k = size(gs) * 100003 + sum(t.x * 131 + t.y for t in gs)
    if k != GK:
        GK = k
        GF = field(gs)
