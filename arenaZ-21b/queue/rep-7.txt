@gene("s4", "how much closer a step brings me to my chosen target")
def progress(t: Tile) -> Num:
    f = target_field()
    return at(f, me().pos) - at(f, t)

@gene("new", "distance field to the altars with gems, else to the single most rewarding safe camp group", parents=["s4"])
def target_field() -> Field:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return field(acc)
    p = me().pos
    bestv = -1
    bestg = -1
    for g in range(num_groups()):
        if not group_respawning(g):
            ts = group_tiles(g)
            d = group_threat(g)
            if d < 500 and size(ts) > 0:
                v = (group_gems(g) * 3 + group_gold(g) * 0.3 + 5) / (d + 30) / (dist(nearest(ts, p), p) + 8)
                if v > bestv:
                    bestv = v
                    bestg = g
    if bestg < 0:
        return field(my_spawn())
    return field(group_tiles(bestg))

@gene("new", "summed base strength of the group monsters, 999 for monoliths", parents=["s4"])
def group_threat(g: Num) -> Num:
    s = 0
    for t in group_tiles(g):
        k = spawn_type(t)
        if k == MONOLITH:
            return 999
        s += camp_base_strength(k)
    return s
