@gene("Z80b", "where to walk: all free altars when banking; with a small bag only altars within 6 tiles, plus living camp groups (monolith and golem spawns only as a last resort)")
def walk_goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        if banking():
            for a in altars() - locked_altars():
                acc = acc | near(a, 1)
            return acc
        p = me().pos
        for a in altars() - locked_altars():
            if dist(a, p) <= 6:
                acc = acc | near(a, 1)
    camps_t = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            camps_t = camps_t | group_tiles(g)
    rest = camps_t - type_spawns(MONOLITH) - type_spawns(GOLEM)
    if size(rest) > 0:
        return acc | rest
    if size(camps_t) > 0:
        return acc | camps_t
    for a in altars() - locked_altars():
        acc = acc | near(a, 1)
    return acc
