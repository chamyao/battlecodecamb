@gene('Z46a', 'worth of a dash: escape when hurt in a fight; when healthy keep dashing two tiles toward the goals even near a fight', parents=['s1'])
def dash_value(t: Tile) -> Num:
    if not valid(t):
        return 0
    if in_fight():
        if me().health_frac < 0.4:
            g = threat_dist(t) - threat_dist(me().pos)
            return 96 + g if g >= 2 else 0
        if me().health_frac > 0.6 and progress(t) >= 2:
            return 51
        return 0
    return 51 if progress(t) >= 2 else 0

@gene("Z7b", "an enemy champion within 4, or an awake camp that could reach me (its own range plus one)")
def in_fight() -> Bool:
    p = me().pos
    if any(dist(u.pos, p) <= 4 for u in enemies()):
        return True
    return any(is_awake(u) and dist(u.pos, p) <= u.attack_range + 1 for u in camps())
