@gene('Z17b', 'skip listing option kinds this policy never takes, and shopping away from spawn', parents=['s1'])
def consider(kind: Num) -> Bool:
    if kind == BUY:
        return can_shop()
    if kind == BUY_LOCK:
        return can_shop() and not has_lock()
    if kind == LOCK:
        return has_lock() and size(enemy_tiles()) > 0
    if kind == CAST:
        return slot_ready(0) or slot_ready(1)
    return not (kind == WARD or kind == CANCEL)

@gene("Z46a", "worth of a dash: escape when hurt in a fight (farthest first); when healthy close on a weak enemy champion, or keep dashing two tiles toward the goals even near a fight")
def dash_value(t: Tile) -> Num:
    if not valid(t):
        return 0
    if in_fight():
        if me().health_frac < 0.4:
            g = threat_dist(t) - threat_dist(me().pos)
            return 96 + g if g >= 2 else 0
        if me().health_frac > 0.6:
            p = me().pos
            if any(u.health_frac < 0.4 and dist(u.pos, t) <= 1 and dist(u.pos, p) > 1 for u in enemies()):
                return 125
            if progress(t) >= 2:
                return 51
        return 0
    return 51 if progress(t) >= 2 else 0
