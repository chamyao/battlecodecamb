```python
@gene("s4", "how much closer a step brings me to the goals; no credit for stepping next to enemy champions while hurt")
def progress(t: Tile) -> Num:
    if me().health_frac < 0.5 and risky(t):
        return 0
    return at(goal_field(), me().pos) - at(goal_field(), t)

@gene("new", "a visible enemy champion could hit this tile next turn", parents=["s4"])
def risky(t: Tile) -> Bool:
    return any(dist(u.pos, t) <= u.attack_range + 1 for u in enemies())
```
