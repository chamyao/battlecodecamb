```python
@gene("s2", "where to walk: an altar with gems, else a living camp group (monolith and golem spawns excluded, contested camps next to visible enemy champions skipped when possible)")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    rest = acc - type_spawns(MONOLITH) - type_spawns(GOLEM)
    if size(rest) == 0:
        return acc
    free = rest
    for u in enemies():
        free = free - near(u.pos, 3)
    return free if size(free) > 0 else rest
```
