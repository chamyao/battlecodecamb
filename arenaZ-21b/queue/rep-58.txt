```python
@gene("s1", "value of doing this option now; flee then recall when hurt; recall to shop once 100 gold is saved and it is safe; dash to travel or escape; chase weak enemy champions")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if in_fight():
            return 0
        if me().health_frac < 0.4:
            return 90
        if gems() == 0 and gold() >= 100 and not can_shop() and at(GF, me().pos) >= 3:
            return 52 if me().health_frac < 0.85 or gold() >= 200 else 0
        return 0
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_DASH:
            return dash_value(o.tile)
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 10 + (5 if is_tributing(o.unit) else 0)
        if gems() == 0:
            return 60
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.4 and in_fight():
            g = threat_dist(o.tile) - threat_dist(me().pos)
            return 95 + g if g > 0 else 0
        if me().health_frac > 0.6:
            w = best(enemies(), lambda u: 0 - u.health_frac)
            if exists(w) and w.health_frac < 0.4 and dist(w.pos, me().pos) <= 5:
                return 70 if dist(o.tile, w.pos) < dist(me().pos, w.pos) else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("Z7a", "worth of casting: stun enemy tributes first, damage and stuns on champions and fought camps, shields and resets in fights, heals")
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if not exists(o.unit):
            return 0
        if o.unit.is_champion and o.unit.is_enemy:
            if k == AB_CC and is_tributing(o.unit):
                return 97
            return 85
        if o.unit.is_camp and (gems() == 0 or is_awake(o.unit)):
            return 65
        return 0
    if k == AB_AOE:
        if not valid(o.tile):
            return 0
        r = slot_radius(o.slot)
        n = count(dist(u.pos, o.tile) <= r for u in enemies())
        c = count(dist(u.pos, o.tile) <= r and (gems() == 0 or is_awake(u)) for u in camps())
        if n + c == 0:
            return 0
        return 64 + n * 3 + c
    if k == AB_SHIELD:
        return 75 if in_fight() else 0
    if k == AB_HEAL:
        if exists(o.unit) and o.unit.health_frac < 0.6:
            return 85
        return 0
    if k == AB_RESET:
        return 58 if not can_attack() and in_fight() else 0
    return 0
```
