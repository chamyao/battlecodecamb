```python
@gene("s1", "value of doing this option now")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        if not banking():
            return 60 + (1 - o.unit.health_frac) * 5
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "carry gems until the bag is worth the trip, or bank early when hurt or late", parents=["s2"])
def banking() -> Bool:
    if gems() <= 0:
        return False
    if gems() >= 20:
        return True
    if me().health_frac < 0.5:
        return True
    return round() > 1900

@gene("s2", "where to walk: an altar when banking, else a living camp group")
def goals() -> Tiles:
    if banking():
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            acc = acc | group_tiles(g)
    return acc
```
