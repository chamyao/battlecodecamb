```python
GK: Num = -1

@gene("new", "keep hunting camps while the bag is small: bank once carrying 12+ gems, when below half health, or near the end", parents=["s1"])
def banking() -> Bool:
    if gems() <= 0:
        return False
    if gems() >= 12 or me().health_frac < 0.5:
        return True
    return round() > 1900

@gene("new", "where to walk: free altars while carrying gems, plus living camp groups (no monolith or golem spawns) unless banking", parents=["Z11a"])
def walk_goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        if banking():
            return acc
    camps_t = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            camps_t = camps_t | group_tiles(g)
    rest = camps_t - type_spawns(MONOLITH) - type_spawns(GOLEM)
    if size(rest) > 0:
        return acc | rest
    return acc | camps_t

@gene("Z11a", "recompute the walk field only when the walk goals change; skip while channeling or dead")
def update() -> None:
    if channeling() or not alive():
        return
    gs = walk_goals()
    k = size(gs) * 100003 + sum(t.x * 131 + t.y for t in gs)
    if k != GK:
        GK = k
        GF = field(gs)

@gene("s1", "value of doing this option now; keep fighting camps until banking; never wake a sleeping golem or monolith; flee then recall when hurt; recall to shop when safe and idle; dash")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if threat_dist(me().pos) <= 5:
            return 0
        if me().health_frac < 0.35:
            return 90
        if gems() == 0 and gold() >= 150 and not can_shop() and at(GF, me().pos) >= 3:
            return 52 if me().health_frac < 0.85 or gold() >= 300 else 0
        return 0
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_DASH:
            return dash_value(o.tile)
        if exists(o.unit) and o.unit.is_camp and not fightable(o.unit):
            return 0
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        if not fightable(o.unit):
            return 0
        if not banking():
            return 60
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.35:
            here = threat_dist(me().pos)
            if here <= 5:
                g = threat_dist(o.tile) - here
                return 95 + g if g > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
