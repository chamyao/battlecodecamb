```python
@gene("s1", "value of doing this option now; draft Marcella first; flee then recall when hurt; recall to heal or shop when safe, empty-handed and between camps; dash; chase weak enemy champions when healthy")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 3 if o.champ == MARCELLA else 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if threat_dist(me().pos) <= 5:
            return 0
        if me().health_frac < 0.35:
            return 90
        if gems() == 0 and not can_shop() and at(GF, me().pos) >= 3:
            if me().health_frac < 0.7 or gold() >= 300:
                return 52
            return 52 if gold() >= 150 and me().health_frac < 0.85 else 0
        return 0
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_DASH:
            return dash_value(o.tile)
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 10
        if gems() == 0:
            return 60
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.35:
            here = threat_dist(me().pos)
            if here <= 5:
                g = threat_dist(o.tile) - here
                return 95 + g if g > 0 else 0
        if me().health_frac > 0.6:
            w = best(enemies(), lambda u: 0 - u.health_frac)
            if exists(w) and w.health_frac < 0.4 and dist(w.pos, me().pos) <= 5:
                return 70 if dist(o.tile, w.pos) < dist(me().pos, w.pos) else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0
```
