@gene("s1", "value of doing this option now")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return health_shop(o.item) + o.count * 0.01
    if o.kind == RECALL:
        return 90 if me().health_frac < 0.35 else 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        if gems() == 0:
            return 60
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "worth of buying: the starter item first, then health shards (more health per gold), Jungle family favoured", parents=["Z10b"])
def health_shop(i: Num) -> Num:
    if i == 6:
        return 120
    if item_health(i) <= 0:
        return 0
    v = 100 + item_health(i) * 0.05
    if item_passive(i) == P_JUNGLE:
        v += 10
    return v
