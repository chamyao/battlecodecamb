@gene("s2", "where to walk: an altar with gems, else a living camp group that holds no monolith or golem")
def goals() -> Tiles:
    if gems() > 0:
        acc = near(tile(-5, -5), 0)
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        return acc
    acc = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            if not any(spawn_type(t) == MONOLITH or spawn_type(t) == GOLEM for t in group_tiles(g)):
                acc = acc | group_tiles(g)
    return acc
