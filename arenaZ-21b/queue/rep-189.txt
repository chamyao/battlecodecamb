@gene("Z80a", "keep hunting camps while the bag is small: bank once carrying 16+ gems (8+ after round 1500), when below half health, or near the end")
def banking() -> Bool:
    if gems() <= 0:
        return False
    if gems() >= 16 or me().health_frac < 0.5:
        return True
    if gems() >= 8 and round() > 1500:
        return True
    return round() > 1900
