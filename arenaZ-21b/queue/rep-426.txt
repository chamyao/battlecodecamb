@gene('s1', 'value of doing this option now; prefer enemy champions carrying gems; draft Marcella, Ryth, Tavi, Vesper Vale; keep fighting camps until banking; flee then recall when hurt; recall to shop once 200 gold is saved; dash; chase weak enemy champions')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return twin_order(o.champ)
    if o.kind == BUY:
        return jungle_shop(o.item)
    if o.kind == RECALL:
        if in_fight():
            return 0
        if me().health_frac < 0.35:
            return 90
        if gems() == 0 and gold() >= 200 and not can_shop() and at(GF, me().pos) >= 3:
            return 52 if me().health_frac < 0.85 or gold() >= 400 else 0
        return 0
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_DASH:
            if engage_dash(o.tile):
                return 125
            return dash_value(o.tile)
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 10 + min(unit_gems(o.unit), 30) * 0.2
        if not banking():
            return 60 + (1 - o.unit.health_frac) * 3
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.35 and in_fight():
            g = threat_dist(o.tile) - threat_dist(me().pos)
            return 95 + g if g > 0 else 0
        if me().health_frac > 0.6:
            w = best(enemies(), lambda u: 0 - u.health_frac)
            if exists(w) and w.health_frac < 0.35 and dist(w.pos, me().pos) <= 5:
                return 70 if dist(o.tile, w.pos) < dist(me().pos, w.pos) else 0
        if me().health_frac > 0.7 and gems() < 10:
            p = me().pos
            e = best(enemies(), lambda u: 0 - dist(u.pos, p))
            if exists(e) and dist(e.pos, p) <= 7 and dist(e.pos, p) > 1:
                if any(dist(a.pos, e.pos) <= 4 and dist(a.pos, p) > 0 for a in allies()):
                    return 65 if dist(o.tile, e.pos) < dist(p, e.pos) else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene('Z17a', 'after round 1000 living golem groups join the goals; recompute the walk field only when the walk goals change; after round 1000 living golem groups join the camp goals; skip while channeling or dead', parents=['s4'])
def update() -> None:
    if channeling() or not alive():
        return
    gs = walk_goals()
    if round() > 1000 and not banking():
        for g in range(num_groups()):
            if not group_respawning(g):
                if size(group_tiles(g) & type_spawns(GOLEM)) > 0:
                    gs = gs | group_tiles(g)
    if round() > 1000 and not banking():
        for g in range(num_groups()):
            if not group_respawning(g):
                gl = group_tiles(g) & type_spawns(GOLEM)
                if size(gl) > 0:
                    gs = gs | group_tiles(g)
    k = size(gs) * 100003 + sum(t.x * 131 + t.y for t in gs)
    if k != GK:
        GK = k
        GF = field(gs)
