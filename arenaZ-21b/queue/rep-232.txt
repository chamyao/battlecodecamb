```python
@gene("Z27a", "how far a tile sits outside the reach of the nearest awake camp or visible enemy champion (plus 5), so fleeing steps clear the longest reach first")
def threat_dist(t: Tile) -> Num:
    a = min((dist(u.pos, t) - u.attack_range for u in camps() if is_awake(u)), default=99)
    b = min((dist(u.pos, t) - u.attack_range for u in enemies()), default=99)
    return min(a, b) + 5
```
