```python
@gene("s1", "value of doing this option now; never wake a sleeping golem or monolith; leave reach before recalling; dash to travel or escape")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac >= 0.35:
            return 0
        return 90 if threat_dist(me().pos) > 5 else 0
    if o.kind == CAST:
        if slot_kind(o.slot) == AB_DASH:
            return dash_value(o.tile)
        if exists(o.unit) and o.unit.is_camp and not fightable(o.unit):
            return 0
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80
        if not fightable(o.unit):
            return 0
        if gems() == 0:
            return 60
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.35:
            here = threat_dist(me().pos)
            if here <= 5:
                g = threat_dist(o.tile) - here
                return 95 + g if g > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "a camp worth hitting: anything already awake, or a sleeping one that is not a golem or monolith", parents=["s1"])
def fightable(u: Unit) -> Bool:
    if is_awake(u):
        return True
    return not (u.type == GOLEM or u.type == MONOLITH)

@gene("new", "worth of a dash: escape when hurt near threats, else cover two tiles toward the goals", parents=["s1"])
def dash_value(t: Tile) -> Num:
    if not valid(t):
        return 0
    here = threat_dist(me().pos)
    if here <= 5:
        if me().health_frac < 0.35 and threat_dist(t) - here >= 2:
            return 96
        return 0
    return 51 if progress(t) >= 2 else 0
```
