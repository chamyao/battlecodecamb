@gene('Z17a', 'after round 1000 living golem groups join the goals; recompute the walk field only when the walk goals change; skip while channeling or dead', parents=['s4'])
def update() -> None:
    if channeling() or not alive():
        return
    gs = walk_goals()
    if round() > 1000 and not banking():
        for g in range(num_groups()):
            if not group_respawning(g):
                if size(group_tiles(g) & type_spawns(GOLEM)) > 0:
                    gs = gs | group_tiles(g)
    k = size(gs) * 100003 + sum(t.x * 131 + t.y for t in gs)
    if k != GK:
        GK = k
        GF = field(gs)

@gene("Z190a", "worth of a purchase: the starter item first; then strength shards: Jungle (less camp damage) until its emblem is done, then Monster (more camp damage), finishing an emblem already started")
def jungle_shop(i: Num) -> Num:
    if i == 6:
        return 100
    if item_strength(i) <= 0 or item_cost(i) > 30:
        return 0
    v = 90 + item_strength(i) * 0.1
    n = items_owned(i)
    if item_passive(i) == P_JUNGLE and n < 5:
        v += 6
    if item_passive(i) == P_MONSTER:
        v += 3
    if n > 0 and n < 5:
        v += 2
    return v
