@gene('s1', 'value of doing this option now; draft Marcella first; shop the starter item, then strength shards with leftover gold; keep fighting camps until banking; flee then recall when hurt; recall to shop with a full purse; chase weak enemy champions')
def score(o: Option) -> Num:
    if o.kind == PICK:
        return twin_order(o.champ)
    if o.kind == BUY:
        return jungle_shop(o.item)
    if o.kind == BUY_LOCK:
        return 101 if gold() >= 40 else 0
    if o.kind == LOCK:
        return lock_value(o.tile)
    if o.kind == RECALL:
        if in_fight():
            return 0
        if me().health_frac < 0.4:
            return 90
        if gems() == 0 and gold() >= 150 and not can_shop() and at(GF, me().pos) >= 3:
            return 52 if me().health_frac < 0.85 or gold() >= 300 else 0
        return 0
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 10
        if not banking():
            return 60 + (1 - o.unit.health_frac) * 3
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.4 and in_fight():
            g = threat_dist(o.tile) - threat_dist(me().pos)
            return 95 + g if g > 0 else 0
        if me().health_frac > 0.6:
            w = best(enemies(), lambda u: 0 - u.health_frac)
            if exists(w) and w.health_frac < 0.5 and dist(w.pos, me().pos) <= 5:
                return 70 if dist(o.tile, w.pos) < dist(me().pos, w.pos) else 0
        if me().health_frac > 0.7 and gems() < 10:
            p = me().pos
            e = best(enemies(), lambda u: 0 - dist(u.pos, p))
            if exists(e) and dist(e.pos, p) <= 7 and dist(e.pos, p) > 1:
                if any(dist(a.pos, e.pos) <= 4 and dist(a.pos, p) > 0 for a in allies()):
                    return 65 if dist(o.tile, e.pos) < dist(p, e.pos) else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "worth of locking this altar: an enemy champion is tributing at it (the lock cancels the channel), or a visible enemy carrying 10+ gems is about to bank there", parents=["s1"])
def lock_value(a: Tile) -> Num:
    if not valid(a):
        return 0
    if any(is_tributing(u) and dist(u.pos, a) <= 1 for u in enemies()):
        return 98
    if any(unit_gems(u) >= 10 and dist(u.pos, a) <= 2 for u in enemies()):
        return 75
    return 0

@gene('Z17b', 'skip listing option kinds this policy never takes, and shopping away from spawn', parents=['s1'])
def consider(kind: Num) -> Bool:
    if kind == BUY:
        return can_shop()
    if kind == CAST:
        return slot_ready(0) or slot_ready(1)
    if kind == BUY_LOCK:
        return can_shop() and not has_lock()
    if kind == LOCK:
        return has_lock() and size(enemy_tiles()) > 0
    return not (kind == WARD or kind == CANCEL)
