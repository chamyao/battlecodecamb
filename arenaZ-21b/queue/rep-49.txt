```python
@gene("s1", "value of doing this option now; never wake a sleeping golem or monolith; finish weak targets first; leave reach before recalling")
def score(o: Option) -> Num:
    if o.kind == PICK:
        return 1
    if o.kind == BUY:
        return 100 if o.item == 6 else 0
    if o.kind == RECALL:
        if me().health_frac >= 0.35:
            return 0
        return 0 if in_fight() else 90
    if o.kind == CAST:
        return cast_value(o)
    if o.kind == ATTACK:
        if o.unit.is_champion:
            return 80 + (1 - o.unit.health_frac) * 5
        if not fightable(o.unit):
            return 0
        if gems() == 0:
            return 60 + (1 - o.unit.health_frac) * 3
        return 55 if is_awake(o.unit) else 0
    if o.kind == TRIBUTE:
        return 70
    if o.kind == MOVE:
        if me().health_frac < 0.35 and in_fight():
            g = threat_dist(o.tile) - threat_dist(me().pos)
            return 95 + g if g > 0 else 0
        return 50 if progress(o.tile) > 0 else 0
    return 0

@gene("new", "a camp worth hitting: anything already awake, or a sleeping one that is not a golem or monolith", parents=["s1"])
def fightable(u: Unit) -> Bool:
    if is_awake(u):
        return True
    return not (u.type == GOLEM or u.type == MONOLITH)

@gene("Z7a", "worth of casting: damage and stuns on champions and fought camps (never waking golems or monoliths), shields and resets in fights, heals; dashes to travel or escape")
def cast_value(o: Option) -> Num:
    k = slot_kind(o.slot)
    if k == AB_DAMAGE or k == AB_CC:
        if not exists(o.unit):
            return 0
        if o.unit.is_champion and o.unit.is_enemy:
            return 85
        if o.unit.is_camp and fightable(o.unit) and (gems() == 0 or is_awake(o.unit)):
            return 65
        return 0
    if k == AB_AOE:
        if not valid(o.tile):
            return 0
        r = slot_radius(o.slot)
        n = count(dist(u.pos, o.tile) <= r for u in enemies())
        c = count(dist(u.pos, o.tile) <= r and fightable(u) and (gems() == 0 or is_awake(u)) for u in camps())
        bad = count(dist(u.pos, o.tile) <= r and not fightable(u) for u in camps())
        if n + c == 0 or bad > 0:
            return 0
        return 64 + n * 3 + c
    if k == AB_SHIELD:
        return 75 if in_fight() else 0
    if k == AB_HEAL:
        if exists(o.unit) and o.unit.health_frac < 0.6:
            return 85
        return 0
    if k == AB_RESET:
        return 58 if not can_attack() and in_fight() else 0
    if k == AB_DASH:
        if not valid(o.tile):
            return 0
        if in_fight():
            if me().health_frac < 0.35 and threat_dist(o.tile) - threat_dist(me().pos) >= 2:
                return 96
            return 0
        return 51 if progress(o.tile) >= 2 else 0
    return 0
```
