@gene("Z88b", "where to walk: free altars while carrying gems, plus living camp groups (no monolith or golem spawns, none next to a visible enemy champion) unless banking")
def walk_goals() -> Tiles:
    acc = near(tile(-5, -5), 0)
    if gems() > 0:
        for a in altars() - locked_altars():
            acc = acc | near(a, 1)
        if banking():
            return acc
    camps_t = near(tile(-5, -5), 0)
    for g in range(num_groups()):
        if not group_respawning(g):
            camps_t = camps_t | group_tiles(g)
    rest = camps_t - type_spawns(MONOLITH) - type_spawns(GOLEM)
    if size(rest) == 0:
        return acc | camps_t
    free = rest
    for u in enemies():
        free = free - near(u.pos, 3)
    if size(free) > 0:
        return acc | free
    return acc | rest
